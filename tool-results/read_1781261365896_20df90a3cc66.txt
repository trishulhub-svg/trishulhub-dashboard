    99→    99→    99→function formatRelativeTime(dateStr: string | null | undefined): string {
   100→   100→   100→  if (!dateStr) return "N/A"
   101→   101→   101→  const date = new Date(dateStr)
   102→   102→   102→  if (isNaN(date.getTime())) return "N/A"
   103→   103→   103→  const now = new Date()
   104→   104→   104→  const diffMs = now.getTime() - date.getTime()
   105→   105→   105→  const diffSec = Math.floor(diffMs / 1000)
   106→   106→   106→  if (diffSec < 60) return "just now"
   107→   107→   107→  const diffMin = Math.floor(diffSec / 60)
   108→   108→   108→  if (diffMin < 60) return `${diffMin}m ago`
   109→   109→   109→  const diffHr = Math.floor(diffMin / 60)
   110→   110→   110→  if (diffHr < 24) return `${diffHr}h ago`
   111→   111→   111→  const diffDay = Math.floor(diffHr / 24)
   112→   112→   112→  if (diffDay < 7) return `${diffDay}d ago`
   113→   113→   113→  return formatDateTime(dateStr)
   114→   114→   114→}
