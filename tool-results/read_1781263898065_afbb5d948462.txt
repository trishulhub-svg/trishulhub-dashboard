    98→function formatRelativeTime(dateStr: string | null | undefined): string {
    99→  if (!dateStr) return "N/A"
   100→  const date = new Date(dateStr)
   101→  if (isNaN(date.getTime())) return "N/A"
   102→  const now = new Date()
   103→  const diffMs = now.getTime() - date.getTime()
   104→  const diffSec = Math.floor(diffMs / 1000)
   105→  if (diffSec < 60) return "just now"
   106→  const diffMin = Math.floor(diffSec / 60)
   107→  if (diffMin < 60) return `${diffMin}m ago`
   108→  const diffHr = Math.floor(diffMin / 60)
   109→  if (diffHr < 24) return `${diffHr}h ago`
   110→  const diffDay = Math.floor(diffHr / 24)
   111→  if (diffDay < 7) return `${diffDay}d ago`
   112→  return formatDateTime(dateStr)
   113→}

   115→/** Safe helper: extract initials from a potentially null user name */
   116→function getInitials(userName: string | null | undefined): string {
   117→  if (!userName) return "?"
   118→  return userName.split(" ").filter(Boolean).map(n => n[0] || "").join("").toUpperCase().slice(0, 2) || "?"
   119→}
   120→
   121→/** Safe helper: format role for display */
   122→function formatRole(role: string | null | undefined): string {
   123→  if (!role) return "—"
   124→  return role.replace(/_/g, " ")
   125→}

   127→/**
   128→ * Local error boundary for the Audit Trail page.
   129→ * Catches render-time errors so they show a useful message
   130→ * instead of the generic DashboardError boundary.
   131→ */
   132→class AuditTrailErrorBoundary extends React.Component<
   133→  { children: React.ReactNode },
   134→  { hasError: boolean; error: Error | null }
   135→> {
   136→  constructor(props: { children: React.ReactNode }) {
   137→    super(props)
   138→    this.state = { hasError: false, error: null }
   139→  }
   140→  static getDerivedStateFromError(error: Error) {
   141→    return { hasError: true, error }
   142→  }
   143→  componentDidCatch(error: Error, info: React.ErrorInfo) {
   144→    console.error("[AuditTrailErrorBoundary]", error.message, info.componentStack)
   145→  }
   146→  render() {
   147→    if (this.state.hasError) {
   148→      return (
   149→        <div className="min-h-[40vh] flex items-center justify-center p-6">
   150→          <Card className="max-w-md w-full border-red-200 dark:border-red-900/50">
   151→            <CardContent className="p-6 space-y-4">
   152→              <div className="flex justify-center">
   153→                <div className="h-14 w-14 rounded-full bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   154→                  <AlertTriangle className="h-7 w-7 text-red-600 dark:text-red-400" />
   155→                </div>
   156→              </div>
   157→              <div className="text-center space-y-2">
   158→                <h2 className="text-lg font-bold">Audit Trail Error</h2>
   159→                <p className="text-sm text-muted-foreground">
   160→                  Something went wrong loading the audit trail. This is likely a data or rendering issue.
   161→                </p>
   162→                <details className="text-left">
   163→                  <summary className="text-xs text-muted-foreground cursor-pointer hover:text-foreground">
   164→                    Error details
   165→                  </summary>
   166→                  <pre className="mt-2 text-xs bg-muted p-3 rounded-md overflow-auto max-h-32 text-red-600 dark:text-red-400">
   167→                    {this.state.error?.message || "Unknown error"}
   168→                  </pre>
   169→                </details>
   170→              </div>
   171→              <div className="flex gap-2 justify-center">
   172→                <Button onClick={() => this.setState({ hasError: false, error: null })}>
   173→                  <RotateCcw className="h-4 w-4 mr-2" /> Try Again
   174→                </Button>
   175→              </div>
   176→            </CardContent>
   177→          </Card>
   178→        </div>
   179→      )
   180→    }
   181→    return this.props.children
   182→  }
   183→}
