    98→    98→function formatRelativeTime(dateStr: string | null | undefined): string {
    99→    99→  if (!dateStr) return "N/A"
   100→   100→  const date = new Date(dateStr)
   101→   101→  if (isNaN(date.getTime())) return "N/A"
   102→   102→  const now = new Date()
   103→   103→  const diffMs = now.getTime() - date.getTime()
   104→   104→  const diffSec = Math.floor(diffMs / 1000)
   105→   105→  if (diffSec < 60) return "just now"
   106→   106→  const diffMin = Math.floor(diffSec / 60)
   107→   107→  if (diffMin < 60) return `${diffMin}m ago`
   108→   108→  const diffHr = Math.floor(diffMin / 60)
   109→   109→  if (diffHr < 24) return `${diffHr}h ago`
   110→   110→  const diffDay = Math.floor(diffHr / 24)
   111→   111→  if (diffDay < 7) return `${diffDay}d ago`
   112→   112→  return formatDateTime(dateStr)
   113→   113→}

   115→   115→/** Safe helper: extract initials from a potentially null user name */
   116→   116→function getInitials(userName: string | null | undefined): string {
   117→   117→  if (!userName) return "?"
   118→   118→  return userName.split(" ").filter(Boolean).map(n => n[0] || "").join("").toUpperCase().slice(0, 2) || "?"
   119→   119→}
