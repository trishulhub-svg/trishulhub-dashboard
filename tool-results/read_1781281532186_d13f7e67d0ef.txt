    99→// Global Task Board Page
   100→// ═══════════════════════════════════════════════════════════════
   101→export default function GlobalTaskBoardPage() {
   102→  const { data: session, status: sessionStatus } = useSession();
   103→
   104→  // ── State ──
   105→  const [tasks, setTasks] = useState<Record<string, unknown>[]>([]);
   106→  const [projects, setProjects] = useState<Record<string, unknown>[]>([]);
   107→  const [users, setUsers] = useState<TeamUser[]>([]);
   108→  const [loading, setLoading] = useState(true);
   109→  const [search, setSearch] = useState("");
   110→  const [statusFilter, setStatusFilter] = useState("ALL");
   111→  const [priorityFilter, setPriorityFilter] = useState("ALL");
   112→  const [projectFilter, setProjectFilter] = useState("ALL");
   113→  const [addOpen, setAddOpen] = useState(false);
   114→  const [selectedTask, setSelectedTask] = useState<Record<string, unknown> | null>(null);
   115→  const [taskDetailOpen, setTaskDetailOpen] = useState(false);
   116→  const [editOpen, setEditOpen] = useState(false);
   117→  const [editForm, setEditForm] = useState<Record<string, string>>({});
   118→  const [editSaving, setEditSaving] = useState(false);
   119→  const [movingTaskId, setMovingTaskId] = useState<string | null>(null);
   120→  const [deletingTaskId, setDeletingTaskId] = useState<string | null>(null);
   121→
   122→  // ── Derived ──
   123→  const userRole = session?.user?.role || "DEVELOPER";
   124→  const userId = session?.user?.id || "";
   125→  const isAdminUser = userRole === "SUPER_ADMIN" || userRole === "ADMIN";
   126→
   127→  // ── Project name lookup map ──
   128→  const projectNameMap = new Map<string, string>(
   129→    (projects as Record<string, unknown>[]).map((p) => [
   130→      extractStr(p, "id", ""),
   131→      extractStr(p, "name", "Unknown"),
   132→    ])
   133→  );
   134→
   135→  // ── User name lookup map ──
   136→  const userNameMap = new Map<string, string>(users.map((u) => [u.id, u.name]));
   137→
   138→  // ── Fetch tasks ──
   139→  const fetchTasks = useCallback(async () => {
   140→    setLoading(true);
   141→    try {
   142→      const params = new URLSearchParams();
   143→      params.set("page", "1");
   144→      params.set("limit", "100");
   145→
   146→      // Non-admin only sees their own tasks
   147→      if (!isAdminUser) {
   148→        params.set("assignedTo", "current");
   149→      }
   150→
   151→      // Apply filters
   152→      if (statusFilter !== "ALL") params.set("status", statusFilter);
   153→      if (priorityFilter !== "ALL") params.set("priority", priorityFilter);
   154→      if (projectFilter !== "ALL") params.set("projectId", projectFilter);
   155→
   156→      const res = await fetch(`/api/tasks?${params.toString()}`, {
   157→        credentials: "include",
   158→      });
   159→      if (res.status === 401) {
   160→        window.location.href = "/login";
   161→        return;
   162→      }
   163→      if (!res.ok) throw new Error("Failed to fetch tasks");
   164→      const raw = deepSanitize(await res.json());
   165→      const list = Array.isArray((raw as Record<string, unknown>)?.tasks)
   166→        ? (raw as Record<string, unknown>).tasks as unknown[]
   167→        : Array.isArray(raw)
   168→          ? raw
   169→          : Array.isArray((raw as Record<string, unknown>)?.data)
   170→            ? (raw as Record<string, unknown>).data as unknown[]
   171→            : [];
   172→      setTasks(list as Record<string, unknown>[]);
   173→    } catch {
   174→      toast.error("Failed to load tasks");
   175→    } finally {
   176→      setLoading(false);
   177→    }
   178→  }, [isAdminUser, statusFilter, priorityFilter, projectFilter]);
   179→
   180→  // ── Fetch projects (for filter dropdown) ──
   181→  const fetchProjects = useCallback(async () => {
   182→    try {
   183→      const res = await fetch("/api/projects", { credentials: "include" });
   184→      if (res.status === 401) { window.location.href = "/login"; return; }
   185→      if (!res.ok) return;
   186→      const raw = deepSanitize(await res.json());
   187→      const list = Array.isArray(raw)
   188→        ? raw
   189→        : Array.isArray((raw as Record<string, unknown>)?.data)
   190→          ? (raw as Record<string, unknown>).data as unknown[]
   191→          : [];
   192→      setProjects(list as Record<string, unknown>[]);
   193→    } catch {
   194→      // Silent fail for projects
   195→    }
   196→  }, []);
   197→
   198→  // ── Fetch users (for assignee dropdown) ──
   199→  const fetchUsers = useCallback(async () => {
   200→    try {
   201→      const res = await fetch("/api/team?type=users", { credentials: "include" });
   202→      if (res.status === 401) { window.location.href = "/login"; return; }
   203→      if (!res.ok) return;
   204→      const raw = deepSanitize(await res.json());
   205→      const list = Array.isArray(raw) ? raw : [];
   206→      setUsers(list as TeamUser[]);
   207→    } catch {
   208→      // Silent fail
   209→    }
   210→  }, []);
   211→
   212→  useEffect(() => {
   213→    if (sessionStatus === "loading") return;
   214→    fetchTasks();
   215→    if (isAdminUser) {
   216→      fetchProjects();
   217→      fetchUsers();
   218→    }
   219→  }, [sessionStatus, fetchTasks, fetchProjects, fetchUsers, isAdminUser]);
   220→
   221→  // ── Client-side search filter ──
   222→  const filteredTasks = search.trim()
   223→    ? tasks.filter((t) =>
   224→        extractStr(t, "title", "").toLowerCase().includes(search.toLowerCase())
   225→      )
   226→    : tasks;
   227→
   228→  // ── Move task handler ──
   229→  const handleMoveTask = async (taskId: string, newStatus: string) => {
   230→    setMovingTaskId(taskId);
   231→    try {
   232→      const res = await fetch("/api/tasks", {
   233→        method: "PATCH",
   234→        headers: { "Content-Type": "application/json" },
   235→        credentials: "include",
   236→        body: JSON.stringify({ id: taskId, status: newStatus }),
   237→      });
   238→      if (res.ok) {
   239→        toast.success(`Task moved to ${newStatus.replace("_", " ")}`);
   240→        fetchTasks();
   241→      } else {
   242→        if (res.status === 401) { window.location.href = "/login"; return; }
   243→        const err = await res.json().catch(() => null);
   244→        toast.error(err?.error || "Failed to move task");
   245→      }
   246→    } catch {
   247→      toast.error("Failed to move task");
   248→    } finally {
   249→      setMovingTaskId(null);
   250→    }
   251→  };
   252→
   253→  // ── Create task handler ──
   254→  const handleCreateTask = async (e: React.FormEvent<HTMLFormElement>) => {
   255→    e.preventDefault();
   256→    const form = new FormData(e.currentTarget);
   257→    const data: Record<string, unknown> = {
   258→      title: String(form.get("title") || ""),
   259→      description: String(form.get("description") || "") || undefined,
   260→      priority: String(form.get("priority") || "MEDIUM"),
   261→      category: String(form.get("category") || "GENERAL"),
   262→      projectId: String(form.get("projectId") || "") || undefined,
   263→      assignedTo: String(form.get("assignedTo") || "") || undefined,
   264→      deadline: String(form.get("deadline") || "") || undefined,
   265→    };
   266→    if (!data.title) { toast.error("Title is required"); return; }
   267→    try {
   268→      const res = await fetch("/api/tasks", {
   269→        method: "POST",
   270→        headers: { "Content-Type": "application/json" },
   271→        credentials: "include",
   272→        body: JSON.stringify(data),
   273→      });
   274→      if (res.ok) {
   275→        toast.success("Task created");
   276→        setAddOpen(false);
   277→        fetchTasks();
   278→      } else {
   279→        if (res.status === 401) { window.location.href = "/login"; return; }
   280→        const err = await res.json().catch(() => null);
   281→        toast.error(err?.error || "Failed to create task");
   282→      }
   283→    } catch {
   284→      toast.error("Failed to create task");
   285→    }
   286→  };
   287→
   288→  // ── Edit task handlers ──
   289→  const openEditDialog = (task: Record<string, unknown>) => {
   290→    setEditForm({
   291→      id: extractStr(task, "id", ""),
   292→      title: extractStr(task, "title", ""),
   293→      description: extractStr(task, "description", ""),
   294→      priority: extractStr(task, "priority", "MEDIUM"),
   295→      category: extractStr(task, "category", "GENERAL"),
   296→      assignedTo: extractStr(task, "assignedTo", ""),
   297→      projectId: extractStr(task, "projectId", ""),
   298→      deadline: extractStr(task, "deadline", "") ? extractStr(task, "deadline", "").split("T")[0] : "",
   299→    });
   300→    setEditOpen(true);
   301→  };
   302→
   303→  const handleEditSave = async () => {
   304→    if (!editForm.id) return;
   305→    setEditSaving(true);
   306→    const data: Record<string, unknown> = {
   307→      id: editForm.id,
   308→      title: editForm.title,
   309→      description: editForm.description || undefined,
   310→      priority: editForm.priority,
   311→      category: editForm.category,
   312→      assignedTo: editForm.assignedTo || undefined,
   313→      projectId: editForm.projectId || undefined,
   314→      deadline: editForm.deadline ? new Date(editForm.deadline).toISOString() : null,
   315→    };
   316→    try {
   317→      const res = await fetch("/api/tasks", {
   318→        method: "PATCH",
   319→        headers: { "Content-Type": "application/json" },
   320→        credentials: "include",
   321→        body: JSON.stringify(data),
   322→      });
   323→      if (res.ok) {
   324→        toast.success("Task updated");
   325→        setEditOpen(false);
   326→        setTaskDetailOpen(false);
   327→        fetchTasks();
   328→      } else {
   329→        const err = await res.json().catch(() => null);
   330→        toast.error(err?.error || "Failed to update task");
   331→      }
   332→    } catch {
   333→      toast.error("Failed to update task");
   334→    } finally {
   335→      setEditSaving(false);
   336→    }
   337→  };
   338→
   339→  // ── Delete task handler ──
   340→  const handleDeleteTask = async (taskId: string) => {
   341→    if (!confirm("Are you sure you want to delete this task? This action cannot be undone.")) return;
   342→    setDeletingTaskId(taskId);
   343→    try {
   344→      const res = await fetch(`/api/tasks?id=${taskId}`, {
   345→        method: "DELETE",
   346→        credentials: "include",
   347→      });
   348→      if (res.ok) {
   349→        toast.success("Task deleted");
   350→        setTaskDetailOpen(false);
   351→        setSelectedTask(null);
   352→        fetchTasks();
   353→      } else {
   354→        const err = await res.json().catch(() => null);
   355→        toast.error(err?.error || "Failed to delete task");
   356→      }
   357→    } catch {
   358→      toast.error("Failed to delete task");
   359→    } finally {
   360→      setDeletingTaskId(null);
   361→    }
   362→  };
   363→
   364→  // ── Loading ──
   365→  if (sessionStatus === "loading" || loading) {
   366→    return (
   367→      <div className="space-y-5">
   368→        <div className="flex items-center gap-3">
   369→          <div className="h-10 w-10 rounded-xl bg-muted/50 animate-pulse" />
   370→          <div className="space-y-2 flex-1">
   371→            <div className="h-6 w-52 bg-muted/50 rounded-lg animate-pulse" />
   372→            <div className="h-3.5 w-80 bg-muted/40 rounded-lg animate-pulse" />
   373→          </div>
   374→        </div>
   375→        <div className="flex items-center gap-2">
   376→          <div className="h-9 w-64 bg-muted/40 rounded-lg animate-pulse" />
   377→          <div className="h-9 w-36 bg-muted/40 rounded-lg animate-pulse" />
   378→          <div className="h-9 w-36 bg-muted/40 rounded-lg animate-pulse" />
   379→        </div>
   380→        <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-5 gap-2">
   381→          {Array.from({ length: 5 }).map((_, i) => (
   382→            <div key={i} className="h-72 bg-muted/30 rounded-xl animate-pulse" />
   383→          ))}
   384→        </div>
   385→      </div>
   386→    );
   387→  }
   388→
   389→  return (
   390→    <div className="space-y-4" style={{ animation: "fade-in 0.35s ease-out both" }}>
   391→      {/* ═══════ Page Header ═══════ */}
   392→      <div
   393→        className="flex items-center justify-between flex-wrap gap-3"
   394→        style={{ animation: "card-enter 0.4s ease-out both", animationDelay: "50ms" }}
   395→      >
   396→        <div className="flex items-center gap-3">
   397→          <div className="h-10 w-10 rounded-xl bg-gradient-to-br from-violet-500/15 to-blue-500/10 flex items-center justify-center">
   398→            <Kanban className="h-5 w-5 text-violet-600 dark:text-violet-400" />
   399→          </div>
   400→          <div>
   401→            <h1 className="text-xl font-bold tracking-tight flex items-center gap-2">
   402→              Global Task Board
   403→              <Badge variant="secondary" className="text-[10px] font-bold h-5 px-1.5">
   404→                {String(filteredTasks.length)}
   405→              </Badge>
   406→            </h1>
   407→            <p className="text-sm text-muted-foreground/70">
   408→              {isAdminUser
   409→                ? "All tasks across every project"
   410→                : "Your personal tasks across all projects"}
   411→            </p>
   412→          </div>
   413→        </div>
   414→
   415→        {isAdminUser && (
   416→          <Dialog open={addOpen} onOpenChange={setAddOpen}>
   417→            <DialogTrigger asChild>
   418→              <Button size="sm" className="gap-1.5 h-8 text-xs px-3.5 shadow-sm">
   419→                <Plus className="h-3.5 w-3.5" /> Add Task
   420→              </Button>
   421→            </DialogTrigger>
   422→            <DialogContent className="sm:max-w-md bg-white/80 dark:bg-gray-950/80 backdrop-blur-xl border-white/20 dark:border-white/10">
   423→              <DialogHeader>
   424→                <DialogTitle className="text-base font-bold">Add Task</DialogTitle>
   425→              </DialogHeader>
   426→              <form onSubmit={handleCreateTask} className="space-y-3">
   427→                <div className="space-y-1">
   428→                  <label className="text-xs font-medium">Title *</label>
   429→                  <Input name="title" required className="h-8 text-sm" placeholder="Task title..." />
   430→                </div>
   431→                <div className="space-y-1">
   432→                  <label className="text-xs font-medium">Description</label>
   433→                  <Input name="description" className="h-8 text-sm" placeholder="Optional details..." />
   434→                </div>
   435→                <div className="grid grid-cols-2 gap-3">
   436→                  <div className="space-y-1">
   437→                    <label className="text-xs font-medium">Priority</label>
   438→                    <select
   439→                      name="priority"
   440→                      defaultValue="MEDIUM"
   441→                      className="h-8 border rounded-md px-2.5 text-xs bg-background/80 w-full"
   442→                    >
   443→                      <option value="LOW">Low</option>
   444→                      <option value="MEDIUM">Medium</option>
   445→                      <option value="HIGH">High</option>
   446→                      <option value="URGENT">Urgent</option>
   447→                    </select>
   448→                  </div>
   449→                  <div className="space-y-1">
   450→                    <label className="text-xs font-medium">Category</label>
   451→                    <select
   452→                      name="category"
   453→                      defaultValue="GENERAL"
