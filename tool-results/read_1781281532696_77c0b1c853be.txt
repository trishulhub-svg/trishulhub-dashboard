    28→// ── Helper: serialize Task dates for JSON response ──

    30→  return {
    31→    ...t,
    32→    createdAt: t.createdAt?.toISOString(),
    33→    updatedAt: t.updatedAt?.toISOString(),
    34→    deadline: t.deadline?.toISOString() ?? null,
    35→    completedAt: t.completedAt?.toISOString() ?? null,
    36→    approvedAt: t.approvedAt?.toISOString() ?? null,

    40→// ── Helper: fetch Lark task IDs for a batch of tasks ──
    41→async function getLarkTaskIds(taskIds: string[]): Promise<Record<string, string>> {
    42→  if (taskIds.length === 0) return {}
    43→  try {
    44→    const rows = await db.$queryRawUnsafe<Array<{ taskId: string; larkTaskId: string }>>(
    45→      'SELECT "taskId", "larkTaskId" FROM "LarkTaskMapping" WHERE "taskId" IN (?)',
    46→      taskIds
    47→    )
    48→    const map: Record<string, string> = {}
    49→    for (const r of rows) map[r.taskId] = r.larkTaskId
    50→    return map
    51→  } catch {
    52→    return {}
    53→  }
    54→}

    56→// ── Helper: check if assignee is on approved leave during task period ──
    57→async function checkAssigneeLeave(db: any, userId: string, deadline: Date): Promise<{ name: string; leaveType: string; startDate: Date; endDate: Date } | null> {
    58→  const leave = await db.leave.findFirst({
    59→    where: { userId, status: "APPROVED", startDate: { lte: deadline }, endDate: { gte: new Date() } },
    60→    include: { user: { select: { name: true } } },
    61→  })
    62→  if (!leave) return null
    63→  return { name: leave.user.name, leaveType: leave.leaveType, startDate: leave.startDate, endDate: leave.endDate }
    64→}

    77→export async function GET(req: NextRequest) {
    78→  try {
    79→  // Auto-migrate: ensure all tables/columns exist before querying
    80→  await ensureAllTables()
    81→
    82→  const session = await getServerSession(authOptions)
    83→  if (!session?.user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
    84→
    85→  // Rate limit
    86→  const rl = rateLimit(`tasks-get-${session.user.id}`, RATE_LIMITS.general.limit, RATE_LIMITS.general.windowMs)
    87→  if (!rl.success) {
    88→    return NextResponse.json({ error: "Rate limit exceeded" }, { status: 429 })
    89→  }
    90→
    91→  const userRole = session.user.role
    92→  const userId = session.user.id
    93→
    94→  const { searchParams } = new URL(req.url)
    95→  const projectId = searchParams.get("projectId")
    96→  const assignedToFilter = searchParams.get("assignedTo")
    97→  const createdByFilter = searchParams.get("createdBy")
    98→  const standaloneFilter = searchParams.get("standalone")
    99→  const categoryFilter = searchParams.get("category")
   100→  const page = Math.max(1, parseInt(searchParams.get("page") || "1", 10))
   101→  const limit = Math.min(100, Math.max(1, parseInt(searchParams.get("limit") || "50", 10)))
   102→
   103→  // SUPER_ADMIN and ADMIN see all tasks with unrestricted filters; others have RBAC restrictions
   104→  if (isAdmin(userRole)) {
   105→    // Build where clause with all filters — no RBAC restriction
   106→    const where: { projectId?: string | null | { in: string[] }; assignedTo?: string; createdBy?: string; category?: string } = {}
   107→
   108→    if (projectId) {
   109→      where.projectId = projectId
   110→    } else if (standaloneFilter === "true") {
   111→      where.projectId = null
   112→    }
   113→
   114→    if (assignedToFilter && assignedToFilter !== "current") {
   115→      where.assignedTo = assignedToFilter
   116→    } else if (assignedToFilter === "current") {
   117→      where.assignedTo = userId
   118→    }
   119→
   120→    if (createdByFilter === "current") {
   121→      where.createdBy = userId
   122→    } else if (createdByFilter) {
   123→      where.createdBy = createdByFilter
   124→    }
   125→
   126→    if (categoryFilter) {
   127→      where.category = categoryFilter
   128→    }
   129→
   130→    const [tasks, total] = await Promise.all([
   131→      db.task.findMany({
   132→        where,
   133→        orderBy: { createdAt: "desc" },
   134→        take: limit,
   135→        skip: (page - 1) * limit,
   136→      }),
   137→      db.task.count({ where }),
   138→    ])
   139→
   140→    // Resolve userIds to names for assignee, approver, and creator
   141→    const userIds = new Set<string>()
   142→    for (const t of tasks) {
   143→      if (t.assignedTo) userIds.add(t.assignedTo)
   144→      if (t.approvedBy) userIds.add(t.approvedBy)
   145→      if (t.createdBy) userIds.add(t.createdBy)
   146→    }
   147→    let userMap: Record<string, string> = {}
   148→    if (userIds.size > 0) {
   149→      const users = await db.user.findMany({
   150→        where: { id: { in: Array.from(userIds) } },
   151→        select: { id: true, name: true }
   152→      })
   153→      for (const u of users) userMap[u.id] = u.name
   154→    }
   155→
   156→    // Fetch Lark task IDs for all returned tasks
   157→    const larkIds = await getLarkTaskIds(tasks.map(t => t.id))
   158→
   159→    const enriched = tasks.map(t => ({
   160→      ...serializeTask(t),
   161→      assignedToName: t.assignedTo ? (userMap[t.assignedTo] || null) : null,
   162→      approvedByName: t.approvedBy ? (userMap[t.approvedBy] || null) : null,
   163→      createdByName: t.createdBy ? (userMap[t.createdBy] || null) : null,
   164→      larkTaskId: larkIds[t.id] || null,
   165→    }))
   166→    void logAudit({
   167→      userId, userName: session.user.name || "unknown", userRole,
   168→      department: "TEAM_WORK", page: "tasks", action: "READ",
   169→      description: buildDescription("READ", "tasks"),
   170→      ipAddress: getIpAddress(req), userAgent: getUserAgent(req),
   171→    })
   172→    return NextResponse.json({ tasks: enriched, total, page, totalPages: Math.ceil(total / limit) })
   173→  }
   174→
   175→  // ── Non-admin users: RBAC-filtered view ──
   176→  // C20: Visibility filter moved into Prisma where clause so total count is accurate
   177→  const assignedProjectIds = await getAssignedProjectIds(userId, userRole)
   178→
   179→  // Build filter conditions (separate from visibility)
   180→  const filterConditions: Prisma.TaskWhereInput[] = []
   181→
   182→  // If a specific projectId is requested, verify access and restrict to it
   183→  if (projectId) {
   184→    // I9: Removed unnecessary `as string[]` cast — TypeScript narrows after truthiness check
   185→    if (assignedProjectIds && !assignedProjectIds.includes(projectId)) {
   186→      return NextResponse.json({ tasks: [], total: 0, page: 1, totalPages: 0 })
   187→    }
   188→    filterConditions.push({ projectId })
   189→  } else if (standaloneFilter === "true") {
   190→    // Standalone tasks — no project filter, but RBAC still applies
   191→    filterConditions.push({ projectId: null })
   192→  }
   193→
   194→  // assignedTo filter — non-admin: only allow "current"
   195→  if (assignedToFilter === "current") {
   196→    filterConditions.push({ assignedTo: userId })
   197→  }
   198→  // else: ignore any other value for non-admin
   199→
   200→  // createdBy filter — non-admin: only allow "current"
   201→  if (createdByFilter === "current") {
   202→    filterConditions.push({ createdBy: userId })
   203→  }
   204→  // else: ignore any other value for non-admin
   205→
   206→  // category filter
   207→  if (categoryFilter) {
   208→    filterConditions.push({ category: categoryFilter })
   209→  }
   210→
   211→  // C20: Visibility filter in Prisma where clause
   212→  // Non-admin can only see tasks assigned to them, created by them,
   213→  // or in projects they're a member of
   214→  const visibilityOr: Prisma.TaskWhereInput[] = [
   215→    { assignedTo: userId },
   216→    { createdBy: userId },
   217→  ]
   218→  if (assignedProjectIds && assignedProjectIds.length > 0) {
   219→    visibilityOr.push({ projectId: { in: assignedProjectIds } })
   220→  }
   221→
   222→  const where: Prisma.TaskWhereInput = {
   223→    AND: [
   224→      ...filterConditions,
   225→      { OR: visibilityOr },
   226→    ]
   227→  }
   228→
   229→  const [tasks, total] = await Promise.all([
   230→    db.task.findMany({
   231→      where,
   232→      orderBy: { createdAt: "desc" },
   233→      take: limit,
   234→      skip: (page - 1) * limit,
   235→    }),
   236→    db.task.count({ where }),
   237→  ])
   238→
   239→  // Resolve userIds to names for assignee, approver, and creator
   240→  const userIds = new Set<string>()
   241→  for (const t of tasks) {
   242→    if (t.assignedTo) userIds.add(t.assignedTo)
   243→    if (t.approvedBy) userIds.add(t.approvedBy)
   244→    if (t.createdBy) userIds.add(t.createdBy)
   245→  }
   246→  let userMap: Record<string, string> = {}
   247→  if (userIds.size > 0) {
   248→    const users = await db.user.findMany({
   249→      where: { id: { in: Array.from(userIds) } },
   250→      select: { id: true, name: true }
   251→    })
   252→    for (const u of users) userMap[u.id] = u.name
   253→  }
   254→
   255→  // Fetch Lark task IDs for all returned tasks
   256→  const larkIds = await getLarkTaskIds(tasks.map(t => t.id))
   257→
   258→  const enriched = tasks.map(t => ({
   259→    ...serializeTask(t),
   260→    assignedToName: t.assignedTo ? (userMap[t.assignedTo] || null) : null,
   261→    approvedByName: t.approvedBy ? (userMap[t.approvedBy] || null) : null,
   262→    createdByName: t.createdBy ? (userMap[t.createdBy] || null) : null,
   263→    larkTaskId: larkIds[t.id] || null,
   264→  }))
   265→  void logAudit({
   266→    userId, userName: session.user.name || "unknown", userRole,
   267→    department: "TEAM_WORK", page: "tasks", action: "READ",
   268→    description: buildDescription("READ", "tasks"),
   269→    ipAddress: getIpAddress(req), userAgent: getUserAgent(req),
   270→  })
   271→  return NextResponse.json({ tasks: enriched, total, page, totalPages: Math.ceil(total / limit) })
   272→  } catch (error: unknown) {
   273→    console.error("[tasks] GET error:", error instanceof Error ? error.message : String(error))
   274→    return NextResponse.json({ error: "An error occurred" }, { status: 500 })
   275→  }
   276→}

   278→export async function POST(req: NextRequest) {
   279→  try {
   280→  await ensureAllTables()
   281→
   282→  const session = await getServerSession(authOptions)
   283→  if (!session?.user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
   284→
   285→  // Rate limit
   286→  const rl = rateLimit(`tasks-write-${session.user.id}`, RATE_LIMITS.crmWrite.limit, RATE_LIMITS.crmWrite.windowMs)
   287→  if (!rl.success) {
   288→    return NextResponse.json({ error: "Rate limit exceeded" }, { status: 429 })
   289→  }
   290→
   291→  const userRole = session.user.role
   292→  const userId = session.user.id
   293→  let body: Record<string, unknown>
   294→  try { body = await req.json() } catch {
   295→    return NextResponse.json({ error: "Invalid JSON body" }, { status: 400 })
   296→  }
   297→
   298→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   299→  // ROLE RESTRICTION: Only SUPER_ADMIN and ADMIN can create tasks
   300→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   301→  if (!isAdmin(userRole)) {
   302→    return NextResponse.json({ error: "Forbidden: Only admin and superadmin can create tasks" }, { status: 403 })
   303→  }
   304→
   305→  // Title is required
   306→  if (!body.title) {
   307→    return NextResponse.json({ error: "Task title is required" }, { status: 400 })
   308→  }
   309→
   310→  // C21: Title length limit
   311→  if (String(body.title).length > 500) {
   312→    return NextResponse.json({ error: "Title must be 500 characters or less" }, { status: 400 })
   313→  }
   314→
   315→  // C21: Description length cap
   316→  if (body.description && String(body.description).length > 50000) {
   317→    return NextResponse.json({ error: "Description must be 50000 characters or less" }, { status: 400 })
   318→  }
   319→
   320→  // C21: Deadline validation
   321→  if (body.deadline) {
   322→    const d = new Date(String(body.deadline))
   323→    if (isNaN(d.getTime())) {
   324→      return NextResponse.json({ error: "Invalid deadline date" }, { status: 400 })
   325→    }
   326→  }
   327→
   328→  // I5: Removed dead code — the isAdmin check above already returns 403 for non-admins
   329→  // If assignedTo is provided and is NOT the creator, verify the assignee exists
   330→  if (body.assignedTo && String(body.assignedTo) !== userId) {
   331→    // Verify the assignee exists and is an active user
   332→    const assigneeExists = await db.user.findFirst({
   333→      where: { id: String(body.assignedTo), isActive: true }
   334→    })
   335→    if (!assigneeExists) {
   336→      return NextResponse.json({ error: "Assigned user not found or inactive" }, { status: 400 })
   337→    }
   338→  }
   339→
   340→  // projectId is optional now — standalone tasks are allowed
   341→  // But if projectId IS provided, check project membership for non-admins
   342→  if (body.projectId) {
   343→    if (!isAdmin(userRole)) {
   344→      const membership = await db.projectMember.findFirst({
   345→        where: { userId, projectId: String(body.projectId) }
   346→      })
   347→      if (!membership) {
   348→        return NextResponse.json({ error: "Forbidden: You can only create tasks in your assigned projects" }, { status: 403 })
   349→      }
   350→    }
   351→  }
   352→
   353→  // Validate status
   354→  const taskStatus = body.status ? String(body.status) : "TODO"
   355→  if (!VALID_TASK_STATUSES.includes(taskStatus)) {
   356→    return NextResponse.json({ error: `Invalid status. Must be one of: ${VALID_TASK_STATUSES.join(", ")}` }, { status: 400 })
   357→  }
   358→
   359→  // Validate priority
   360→  const taskPriority = body.priority ? String(body.priority) : "MEDIUM"
   361→  if (!VALID_TASK_PRIORITIES.includes(taskPriority)) {
   362→    return NextResponse.json({ error: `Invalid priority. Must be one of: ${VALID_TASK_PRIORITIES.join(", ")}` }, { status: 400 })
   363→  }
   364→
   365→  // Validate category
   366→  const taskCategory = body.category ? String(body.category) : "GENERAL"
   367→  if (!VALID_TASK_CATEGORIES.includes(taskCategory)) {
   368→    return NextResponse.json({ error: `Invalid category. Must be one of: ${VALID_TASK_CATEGORIES.join(", ")}` }, { status: 400 })
   369→  }
   370→
   371→  // SECURITY: Whitelist allowed fields only (prevent mass assignment)
   372→  const data: Prisma.TaskUncheckedCreateInput = {
   373→    title: String(body.title),
   374→    description: body.description ? String(body.description) : null,
   375→    status: taskStatus,
   376→    priority: taskPriority,
   377→    category: taskCategory,
   378→    createdBy: userId,
   379→    projectId: body.projectId ? String(body.projectId) : null,
   380→    assignedTo: body.assignedTo ? String(body.assignedTo) : null,
   381→    assigneeType: body.assigneeType ? String(body.assigneeType) : "HUMAN",
   382→    deadline: body.deadline ? new Date(String(body.deadline)) : null,
   383→  }
   384→
   385→  // Check if assignee is on approved leave during the task period
   386→  if (data.assignedTo && data.deadline) {
   387→    const assigneeLeave = await checkAssigneeLeave(db, data.assignedTo as string, data.deadline as Date)
   388→    if (assigneeLeave) {
   389→      return NextResponse.json({
   390→        error: `Cannot assign task: ${assigneeLeave.name} is on ${assigneeLeave.leaveType.replace("_", " ").toLowerCase()} leave from ${new Date(assigneeLeave.startDate).toLocaleDateString()} to ${new Date(assigneeLeave.endDate).toLocaleDateString()}`,
   391→      }, { status: 400 })
   392→    }
   393→  }
   394→
   395→  const task = await db.task.create({ data })
   396→
   397→  // Send notification to assignee when a task is created with an assignee
   398→  if (data.assignedTo && (data.assignedTo as string) !== userId) {
   399→    const deadlineStr = data.deadline ? ` (Due: ${new Date(data.deadline as Date).toLocaleDateString()})` : ""
   400→    await sendNotification(
   401→      data.assignedTo as string,
   402→      "New Task Assigned",
   403→      `You have been assigned a new task: ${String(body.title)}${deadlineStr}`,
   404→      "TASK",
   405→      "/dashboard/tasks"
   406→    )
   407→  }
   408→
   409→  // Background: sync tasks to Git (fire-and-forget) — only when projectId exists
   410→  if (data.projectId) {
   411→    syncTasksToGit().catch(() => {})
   412→  }
   413→  // Background: sync new task to Lark (fire-and-forget)
   414→  syncTaskToLark(task.id, {
   415→    title: task.title,
   416→    description: task.description || undefined,
   417→    status: task.status,
   418→    priority: task.priority,
   419→    assignedTo: task.assignedTo || undefined,
   420→    projectId: task.projectId || undefined,
   421→    deadline: task.deadline || undefined,
   422→  }, userId).catch((err) => console.error("[tasks] Lark sync error:", err))
   423→  // I6: Use serializeTask for consistency instead of JSON.parse(JSON.stringify(task))
   424→  void logAudit({
   425→    userId, userName: session.user.name || "unknown", userRole,
   426→    department: "TEAM_WORK", page: "tasks", action: "CREATE",
   427→    entityType: "task", entityId: task.id,
   428→    description: buildDescription("CREATE", "task", task.title),
   429→    ipAddress: getIpAddress(req), userAgent: getUserAgent(req),
   430→  })
   431→  return NextResponse.json(serializeTask(task), { status: 201 })
   432→  } catch (error: unknown) {
   433→    console.error("[tasks] POST error:", error instanceof Error ? error.message : String(error))
   434→    return NextResponse.json({ error: "An error occurred" }, { status: 500 })
   435→  }
   436→}

   438→export async function PATCH(req: NextRequest) {
   439→  try {
   440→  await ensureAllTables()
   441→
   442→  const session = await getServerSession(authOptions)
   443→  if (!session?.user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
   444→
   445→  // Rate limit
   446→  const rl = rateLimit(`tasks-write-${session.user.id}`, RATE_LIMITS.crmWrite.limit, RATE_LIMITS.crmWrite.windowMs)
   447→  if (!rl.success) {
   448→    return NextResponse.json({ error: "Rate limit exceeded" }, { status: 429 })
   449→  }
   450→
   451→  const userRole = session.user.role
   452→  const userId = session.user.id
   453→  const userName = session.user.name || "User"
   454→  let body: Record<string, unknown>
   455→  try { body = await req.json() } catch {
   456→    return NextResponse.json({ error: "Invalid JSON body" }, { status: 400 })
   457→  }
   458→  const id = body.id ? String(body.id) : ""
   459→
   460→  if (!id) return NextResponse.json({ error: "Task ID is required" }, { status: 400 })
   461→
   462→  // Fetch existing task early — needed for approval logic
   463→  const existingTask = await db.task.findUnique({ where: { id } })
   464→  if (!existingTask) return NextResponse.json({ error: "Task not found" }, { status: 404 })
   465→
   466→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   467→  // AUTHORIZATION: verify user can edit this task BEFORE processing fields
   468→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   469→  if (!isAdmin(userRole) && existingTask.projectId) {
   470→    const membership = await db.projectMember.findFirst({
   471→      where: { userId, projectId: existingTask.projectId }
   472→    })
   473→    if (!membership) {
   474→      if (existingTask.createdBy !== userId && existingTask.assignedTo !== userId) {
   475→        return NextResponse.json({ error: "Forbidden: You can only update tasks in your assigned projects or tasks assigned/created by you" }, { status: 403 })
   476→      }
   477→    }
   478→  }
   479→  if (!isAdmin(userRole) && !existingTask.projectId) {
   480→    if (existingTask.createdBy !== userId && existingTask.assignedTo !== userId) {
   481→      return NextResponse.json({ error: "Forbidden: You can only update tasks assigned or created by you" }, { status: 403 })
   482→    }
   483→  }
   484→
   485→  // SECURITY: Whitelist allowed fields only (prevent mass assignment)
   486→  const data: Prisma.TaskUncheckedUpdateInput = {}
   487→
   488→  // C21: Title validation with length limit
   489→  if (body.title !== undefined) {
   490→    if (String(body.title).length > 500) {
   491→      return NextResponse.json({ error: "Title must be 500 characters or less" }, { status: 400 })
   492→    }
   493→    data.title = String(body.title)
   494→  }
   495→
   496→  // C21: Description validation with length cap
   497→  if (body.description !== undefined) {
   498→    if (body.description && String(body.description).length > 50000) {
   499→      return NextResponse.json({ error: "Description must be 50000 characters or less" }, { status: 400 })
   500→    }
   501→    data.description = body.description ? String(body.description) : null
   502→  }
   503→
   504→  if (body.priority !== undefined) {
   505→    if (!VALID_TASK_PRIORITIES.includes(String(body.priority))) {
   506→      return NextResponse.json({ error: `Invalid priority. Must be one of: ${VALID_TASK_PRIORITIES.join(", ")}` }, { status: 400 })
   507→    }
   508→    data.priority = String(body.priority)
   509→  }
   510→  if (body.assignedTo !== undefined) {
   511→    // ROLE RESTRICTION: Only SUPER_ADMIN and ADMIN can reassign tasks
   512→    if (!isAdmin(userRole)) {
   513→      return NextResponse.json({ error: "Forbidden: Only admin and superadmin can reassign tasks" }, { status: 403 })
   514→    }
   515→    const newAssignee = body.assignedTo ? String(body.assignedTo) : null
   516→    // If assigning to a specific user, verify they exist and are active
   517→    if (newAssignee && newAssignee !== existingTask.assignedTo) {
   518→      const assigneeExists = await db.user.findFirst({
   519→        where: { id: newAssignee, isActive: true }
   520→      })
   521→      if (!assigneeExists) {
   522→        return NextResponse.json({ error: "Assigned user not found or inactive" }, { status: 400 })
   523→      }
   524→    }
   525→    data.assignedTo = newAssignee
   526→  }
   527→  if (body.assigneeType !== undefined) {
   528→    if (!["HUMAN", "AI"].includes(String(body.assigneeType))) {
   529→      return NextResponse.json({ error: "Invalid assigneeType. Must be HUMAN or AI" }, { status: 400 })
   530→    }
   531→    data.assigneeType = String(body.assigneeType)
   532→  }
   533→
   534→  // C21: Deadline validation
   535→  if (body.deadline !== undefined) {
   536→    if (body.deadline) {
   537→      const d = new Date(String(body.deadline))
   538→      if (isNaN(d.getTime())) {
   539→        return NextResponse.json({ error: "Invalid deadline date" }, { status: 400 })
   540→      }
   541→    }
   542→    data.deadline = body.deadline ? new Date(String(body.deadline)) : null
   543→  }
   544→
   545→  if (body.category !== undefined) {
   546→    if (!VALID_TASK_CATEGORIES.includes(String(body.category))) {
   547→      return NextResponse.json({ error: `Invalid category. Must be one of: ${VALID_TASK_CATEGORIES.join(", ")}` }, { status: 400 })
   548→    }
   549→    data.category = String(body.category)
   550→  }
   551→
   552→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   553→  // APPROVAL FLOW — status change logic
   554→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   555→  if (body.status !== undefined) {
   556→    if (!VALID_TASK_STATUSES.includes(String(body.status))) {
   557→      return NextResponse.json({ error: `Invalid status. Must be one of: ${VALID_TASK_STATUSES.join(", ")}` }, { status: 400 })
   558→    }
   559→
   560→    const newStatus = body.status as string
   561→    const currentStatus = existingTask.status
   562→
   563→    // W1: Status transition validation (non-admin only)
   564→    if (newStatus !== currentStatus && !isAdmin(userRole)) {
   565→      const allowed = VALID_TRANSITIONS[currentStatus] || []
   566→      if (!allowed.includes(newStatus)) {
   567→        return NextResponse.json({ error: `Invalid status transition: ${currentStatus} → ${newStatus}. Allowed: ${allowed.length > 0 ? allowed.join(", ") : "none (terminal state)"}` }, { status: 409 })
   568→      }
   569→    }
   570→
   571→    // ── CASE 1: User is trying to mark task as DONE ──
   572→    if (newStatus === "DONE") {
   573→      // If task is already AWAITING_APPROVAL, this is an approval action
   574→      if (currentStatus === "AWAITING_APPROVAL") {
   575→        // Only admin/superadmin can approve
   576→        if (!isAdmin(userRole)) {
   577→          return NextResponse.json({ error: "Forbidden: Only admin or superadmin can approve tasks" }, { status: 403 })
   578→        }
   579→
   580→        // Self-approval prevention: ADMIN cannot approve tasks assigned to themselves
   581→        if (userRole === "ADMIN" && existingTask.assignedTo === userId) {
   582→          return NextResponse.json({ error: "Forbidden: You cannot approve your own task. Only superadmin can approve your tasks." }, { status: 403 })
   583→        }
   584→
   585→        // APPROVE — set status, approvedBy, approvedAt, completedAt
   586→        data.status = "DONE"
   587→        data.approvedBy = userId
   588→        data.approvedAt = new Date()
   589→        data.completedAt = new Date()
   590→
   591→      } else {
   592→        // Task is NOT currently AWAITING_APPROVAL — user wants to "complete" it
   593→        // SUPERADMIN can directly mark as DONE (no approval needed for their own actions)
   594→        if (userRole === "SUPER_ADMIN") {
   595→          data.status = "DONE"
   596→          data.approvedBy = userId
   597→          data.approvedAt = new Date()
   598→          data.completedAt = new Date()
   599→        } else {
   600→          // All other users (DEVELOPER, ADMIN) → goes to AWAITING_APPROVAL
   601→          data.status = "AWAITING_APPROVAL"
   602→        }
   603→      }
   604→    } else {
   605→      // Regular status change (not DONE)
   606→      data.status = newStatus
   607→    }
   608→  }
   609→
   610→  // Only admins can change projectId on a task
   611→  if (body.projectId !== undefined) {
   612→    if (!isAdmin(userRole)) {
   613→      return NextResponse.json({ error: "Forbidden: Only admins can move tasks between projects" }, { status: 403 })
   614→    }
   615→    data.projectId = String(body.projectId)
   616→  }
   617→
   618→  // Check if data object is empty — nothing to update
   619→  if (Object.keys(data).length === 0) {
   620→    return NextResponse.json({ error: "No fields to update" }, { status: 400 })
   621→  }
   622→
   623→  // Check if assignee is on approved leave during the task period
   624→  const assignedUserId = typeof data.assignedTo === "string" ? data.assignedTo : null
   625→  const effectiveDeadline = data.deadline instanceof Date ? data.deadline : (existingTask?.deadline || null)
   626→
   627→  if (assignedUserId && effectiveDeadline) {
   628→    const assigneeLeave = await checkAssigneeLeave(db, assignedUserId, effectiveDeadline)
   629→    if (assigneeLeave) {
   630→      return NextResponse.json({
   631→        error: `Cannot assign task: ${assigneeLeave.name} is on ${assigneeLeave.leaveType.replace("_", " ").toLowerCase()} leave from ${new Date(assigneeLeave.startDate).toLocaleDateString()} to ${new Date(assigneeLeave.endDate).toLocaleDateString()}`,
   632→      }, { status: 400 })
   633→    }
   634→  }
   635→
   636→  // Update the task
   637→  const updatedTask = await db.task.update({ where: { id }, data })
   638→
   639→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   640→  // NOTIFICATIONS
   641→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   642→
   643→  // Reassignment notification — when assignedTo changed to a different user
   644→  if (assignedUserId && existingTask.assignedTo !== assignedUserId) {
   645→    await sendNotification(
   646→      assignedUserId,
   647→      "Task Reassigned to You",
   648→      `Task "${existingTask.title}" has been reassigned to you`,
   649→      "TASK",
   650→      "/dashboard/tasks"
   651→    )
   652→  }
   653→
   654→  const finalStatus = data.status as string | undefined
   655→
   656→  // I8: Task sent for approval → notify all admin/superadmin (batch createMany)
   657→  // Fetch admins AND assignee in a single combined query to avoid separate DB lookup
   658→  if (finalStatus === "AWAITING_APPROVAL") {
   659→    const assigneeId = existingTask.assignedTo
   660→    // Batch fetch admins AND the assignee in one query
   661→    const allUsers = await db.user.findMany({
   662→      where: {
   663→        OR: [
   664→          { role: { in: ["SUPER_ADMIN", "ADMIN"] }, isActive: true },
   665→          ...(assigneeId ? [{ id: assigneeId }] : []),
   666→        ]
   667→      },
   668→      select: { id: true, name: true, role: true }
   669→    })
   670→    // Extract assignee name from the combined result
   671→    const assigneeName = assigneeId
   672→      ? (allUsers.find(u => u.id === assigneeId)?.name || "Someone")
   673→      : "Someone"
   674→    // Filter to just admins for notifications
   675→    const adminUsers = allUsers.filter(u => u.role === "SUPER_ADMIN" || u.role === "ADMIN")
   676→
   677→    if (adminUsers.length > 0) {
   678→      await db.notification.createMany({
   679→        data: adminUsers.map(admin => ({
   680→          userId: admin.id,
   681→          title: `Task approval needed: ${existingTask.title}`,
   682→          message: `${assigneeName} submitted "${existingTask.title}" for review`,
   683→          type: "TASK",
   684→          link: `/dashboard/todos`,
   685→          isRead: false,
   686→        }))
   687→      })
   688→    }
   689→  }
   690→
   691→  // Task approved → notify the assignee
   692→  if (finalStatus === "DONE" && data.approvedBy && existingTask.assignedTo && existingTask.assignedTo !== userId) {
   693→    const taskLink = `/dashboard/projects/todos`
   694→    await sendNotification(
   695→      existingTask.assignedTo,
   696→      "Task Approved",
   697→      `Your task "${existingTask.title}" has been approved by ${userName}.`,
   698→      "SUCCESS",
   699→      taskLink
   700→    )
   701→  }
   702→
   703→  // Task rejected (sent back) → notify the assignee
   704→  if (finalStatus && finalStatus !== "AWAITING_APPROVAL" && finalStatus !== "DONE" && existingTask.status === "AWAITING_APPROVAL") {
   705→    if (existingTask.assignedTo) {
   706→      const taskLink = `/dashboard/projects/todos`
   707→      await sendNotification(
   708→        existingTask.assignedTo,
   709→        "Task Revision Needed",
   710→        `Your task "${existingTask.title}" was sent back by ${userName}. Status: ${finalStatus.replace("_", " ")}.`,
   711→        "WARNING",
   712→        taskLink
   713→      )
   714→    }
   715→  }
   716→
   717→  // Background: sync tasks to Git (fire-and-forget) — only when task has a project
   718→  if (updatedTask.projectId) {
   719→    syncTasksToGit().catch(() => {})
   720→  }
   721→  // Background: sync task update to Lark (fire-and-forget)
   722→  const larkUpdateData: { title?: string; description?: string; status?: string; priority?: string; assignedTo?: string; deadline?: string | Date | null } = {}
   723→  if (data.title) larkUpdateData.title = String(data.title)
   724→  if (data.description !== undefined) larkUpdateData.description = (data.description as string | null) ?? undefined
   725→  if (data.status) larkUpdateData.status = String(data.status)
   726→  if (data.priority) larkUpdateData.priority = String(data.priority)
   727→  if (data.assignedTo !== undefined) larkUpdateData.assignedTo = (data.assignedTo as string) || undefined
   728→  if (data.deadline !== undefined) larkUpdateData.deadline = data.deadline as Date | null | undefined
   729→  if (Object.keys(larkUpdateData).length > 0) {
   730→    syncTaskUpdateToLark(id, larkUpdateData, userId).catch((err) => console.error("[tasks] Lark update sync error:", err))
   731→  }
   732→  return NextResponse.json(serializeTask(updatedTask))
   733→  } catch (error: unknown) {
   734→    console.error("[tasks] PATCH error:", error instanceof Error ? error.message : String(error))
   735→    return NextResponse.json({ error: "An error occurred" }, { status: 500 })
   736→  }
   737→}

   739→export async function DELETE(req: NextRequest) {
   740→  try {
   741→  await ensureAllTables()
   742→
   743→  const session = await getServerSession(authOptions)
   744→  if (!session?.user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
   745→
   746→  // Rate limit
   747→  const rl = rateLimit(`tasks-write-${session.user.id}`, RATE_LIMITS.crmWrite.limit, RATE_LIMITS.crmWrite.windowMs)
   748→  if (!rl.success) {
   749→    return NextResponse.json({ error: "Rate limit exceeded" }, { status: 429 })
   750→  }
   751→
   752→  const userRole = session.user.role
   753→  const userId = session.user.id
   754→
   755→  // I7: TODO: Move to /api/tasks/[id]/route.ts for proper REST
   756→  // Currently using query params for task ID instead of URL path
   757→  const { searchParams } = new URL(req.url)
   758→  const id = searchParams.get("id")
   759→
   760→  if (!id) return NextResponse.json({ error: "Task ID is required" }, { status: 400 })
   761→
   762→  // Verify task exists
   763→  const existingTask = await db.task.findUnique({ where: { id } })
   764→  if (!existingTask) return NextResponse.json({ error: "Task not found" }, { status: 404 })
   765→
   766→  // Developers can only delete tasks in their assigned projects, or standalone tasks they created/are assigned to
   767→  if (!isAdmin(userRole)) {
   768→    if (existingTask.projectId) {
   769→      // Project task — check membership
   770→      const membership = await db.projectMember.findFirst({
   771→        where: { userId, projectId: existingTask.projectId }
   772→      })
   773→      if (!membership && existingTask.createdBy !== userId && existingTask.assignedTo !== userId) {
   774→        return NextResponse.json({ error: "Forbidden: You can only delete tasks in your assigned projects or tasks assigned/created by you" }, { status: 403 })
   775→      }
   776→    } else {
   777→      // Standalone task — must be creator or assignee
   778→      if (existingTask.createdBy !== userId && existingTask.assignedTo !== userId) {
   779→        return NextResponse.json({ error: "Forbidden: You can only delete tasks assigned or created by you" }, { status: 403 })
   780→      }
   781→    }
   782→  }
   783→
   784→  await db.task.delete({ where: { id } })
   785→  // Background: sync tasks to Git (fire-and-forget) — only when task had a project
   786→  if (existingTask.projectId) {
   787→    syncTasksToGit().catch(() => {})
   788→  }
   789→  // Background: sync task deletion to Lark (fire-and-forget)
   790→  syncTaskDeleteToLark(id, userId).catch((err) => console.error("[tasks] Lark delete sync error:", err))
   791→  return NextResponse.json({ success: true })
   792→  } catch (error: unknown) {
   793→    console.error("[tasks] DELETE error:", error instanceof Error ? error.message : String(error))
   794→    return NextResponse.json({ error: "An error occurred" }, { status: 500 })
   795→  }
   796→}
