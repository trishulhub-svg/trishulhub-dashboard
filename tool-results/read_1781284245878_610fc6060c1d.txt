   236→/** Single personal task item (checkbox + title + meta) — upgraded with category & hover scale */
   237→function PersonalTaskItem({ task, togglingId, onToggleDone, index = 0 }: {
   238→  task: unknown; togglingId: string | null; onToggleDone: (id: string) => void; index?: number;
   239→}) {
   240→  const taskId = extractStr(task, "id", "");
   241→  const title = extractStr(task, "title", "Untitled");
   242→  const status = extractStr(task, "status", "TODO");
   243→  const priority = extractStr(task, "priority", "MEDIUM");
   244→  const deadline = extractStr(task, "deadline", "");
   245→  const category = extractStr(task, "category", "GENERAL");
   246→  const isAwaiting = status === "AWAITING_APPROVAL";
   247→  const isOverdue = deadline && new Date(deadline) < new Date() && !isAwaiting;
   248→  const isToggling = togglingId === taskId;
   249→  const priorityDot = priorityDotColors[priority] || "bg-gray-400";
   250→  const priorityBorder = priorityBorderColors[priority] || "border-l-gray-300";
   251→
   252→  return (
   253→    <div
   254→      className={cn(
   255→        "flex items-center gap-3 p-3 rounded-lg border border-l-[3px] transition-all duration-200",
   256→        "bg-white/70 dark:bg-white/[0.03] border-gray-200/60 dark:border-gray-700/40",
   257→        "hover:bg-white dark:hover:bg-white/[0.06] hover:shadow-sm hover:scale-[1.005]",
   258→        priorityBorder,
   259→        isAwaiting && "opacity-60",
   260→        "animate-slide-up",
   261→      )}
   262→      style={{ animationDelay: `${index * 50}ms` }}
   263→    >
   264→      <button
   265→        type="button"
   266→        onClick={(e) => { e.stopPropagation(); onToggleDone(taskId); }}
   267→        disabled={isToggling || isAwaiting}
   268→        className={cn(
   269→          "shrink-0 h-[18px] w-[18px] rounded border-2 flex items-center justify-center",
   270→          "transition-all duration-200 hover:scale-110",
   271→          isToggling && "animate-pulse",
   272→          isAwaiting && "cursor-default",
   273→          !isAwaiting && "hover:border-primary/60 hover:bg-primary/5",
   274→          "border-gray-300 dark:border-gray-600",
   275→        )}
   276→      >
   277→        {isAwaiting && <CheckCircle2 className="h-3 w-3 text-orange-400" />}
   278→      </button>
   279→      <div className="flex-1 min-w-0">
   280→        <p className={cn("text-sm leading-snug", isAwaiting && "line-through text-muted-foreground/70", !isAwaiting && "font-medium")}>
   281→          {safeText(title)}
   282→        </p>
   283→      </div>
   284→      <div className="flex items-center gap-2.5 shrink-0">
   285→        <CategoryBadge category={category} />
   286→        <span className={cn("h-2 w-2 rounded-full shrink-0", priorityDot)} title={priority} />
   287→        {deadline && (
   288→          <span className={cn("text-[11px] flex items-center gap-1", isOverdue ? "text-red-500 font-medium" : "text-muted-foreground")}>
   289→            {isOverdue ? <AlertTriangle className="h-3 w-3" /> : <Clock className="h-3 w-3" />}
   290→            {safeDate(deadline)}
   291→          </span>
   292→        )}
   293→        {isAwaiting && (
   294→          <Badge className="text-[10px] bg-orange-100 text-orange-700 dark:bg-orange-900/30 dark:text-orange-300 font-medium">
   295→            Reviewing
   296→          </Badge>
   297→        )}
   298→      </div>
   299→    </div>
   300→  );
   301→}
