   202→   202→function formatRelativeTime(dateStr: string): string {
   203→   203→  if (!dateStr) return "unknown";
   204→   204→  const date = new Date(dateStr);
   205→   205→  if (isNaN(date.getTime())) return "unknown";
   206→   206→  const now = new Date();
   207→   207→  const diffMs = now.getTime() - date.getTime();
   208→   208→  const diffMin = Math.floor(diffMs / 60000);
   209→   209→  if (diffMin < 1) return "just now";
   210→   210→  if (diffMin < 60) return `${diffMin}m ago`;
   211→   211→  const diffHr = Math.floor(diffMin / 60);
   212→   212→  if (diffHr < 24) return `${diffHr}h ago`;
   213→   213→  const diffDay = Math.floor(diffHr / 24);
   214→   214→  if (diffDay < 7) return `${diffDay}d ago`;
   215→   215→  return safeDateStr(date);
   216→   216→}
