    40→    40→// ── Helper: fetch Lark task IDs for a batch of tasks ──
    41→    41→async function getLarkTaskIds(taskIds: string[]): Promise<Record<string, string>> {
    42→    42→  if (taskIds.length === 0) return {}
    43→    43→  try {
    44→    44→    const rows = await db.$queryRawUnsafe<Array<{ taskId: string; larkTaskId: string }>>(
    45→    45→      'SELECT "taskId", "larkTaskId" FROM "LarkTaskMapping" WHERE "taskId" IN (?)',
    46→    46→      taskIds
    47→    47→    )
    48→    48→    const map: Record<string, string> = {}
    49→    49→    for (const r of rows) map[r.taskId] = r.larkTaskId
    50→    50→    return map
    51→    51→  } catch {
    52→    52→    return {}
    53→    53→  }
    54→    54→}

    57→    57→async function checkAssigneeLeave(db: any, userId: string, deadline: Date): Promise<{ name: string; leaveType: string; startDate: Date; endDate: Date } | null> {
    58→    58→  const leave = await db.leave.findFirst({
    59→    59→    where: { userId, status: "APPROVED", startDate: { lte: deadline }, endDate: { gte: new Date() } },

    61→    61→  })
    62→    62→  if (!leave) return null
    63→    63→  return { name: leave.user.name, leaveType: leave.leaveType, startDate: leave.startDate, endDate: leave.endDate }
    64→    64→}

    89→    89→  }
    90→    90→
    91→    91→  const userRole = session.user.role
    92→    92→  const userId = session.user.id
    93→    93→
    94→    94→  const { searchParams } = new URL(req.url)
    95→    95→  const projectId = searchParams.get("projectId")
    96→    96→  const assignedToFilter = searchParams.get("assignedTo")
    97→    97→  const createdByFilter = searchParams.get("createdBy")
    98→    98→  const standaloneFilter = searchParams.get("standalone")

   102→   102→
   103→   103→  // SUPER_ADMIN and ADMIN see all tasks with unrestricted filters; others have RBAC restrictions

   105→   105→    // Build where clause with all filters — no RBAC restriction

   108→   108→    if (projectId) {

   110→   110→    } else if (standaloneFilter === "true") {
   111→   111→      where.projectId = null
   112→   112→    }
   113→   113→
   114→   114→    if (assignedToFilter && assignedToFilter !== "current") {
   115→   115→      where.assignedTo = assignedToFilter

   118→   118→    }

   120→   120→    if (createdByFilter === "current") {
   121→   121→      where.createdBy = userId

   123→   123→      where.createdBy = createdByFilter

   127→   127→      where.category = categoryFilter

   131→   131→      db.task.findMany({
   132→   132→        where,
   133→   133→        orderBy: { createdAt: "desc" },
   134→   134→        take: limit,
   135→   135→        skip: (page - 1) * limit,
   136→   136→      }),
   137→   137→      db.task.count({ where }),
   138→   138→    ])
   139→   139→
   140→   140→    // Resolve userIds to names for assignee, approver, and creator
   141→   141→    const userIds = new Set<string>()
   142→   142→    for (const t of tasks) {
   143→   143→      if (t.assignedTo) userIds.add(t.assignedTo)

   145→   145→      if (t.createdBy) userIds.add(t.createdBy)

   151→   151→        select: { id: true, name: true }

   155→   155→
   156→   156→    // Fetch Lark task IDs for all returned tasks
   157→   157→    const larkIds = await getLarkTaskIds(tasks.map(t => t.id))
   158→   158→
   159→   159→    const enriched = tasks.map(t => ({

   161→   161→      assignedToName: t.assignedTo ? (userMap[t.assignedTo] || null) : null,
   162→   162→      approvedByName: t.approvedBy ? (userMap[t.approvedBy] || null) : null,
   163→   163→      createdByName: t.createdBy ? (userMap[t.createdBy] || null) : null,
   164→   164→      larkTaskId: larkIds[t.id] || null,
   165→   165→    }))
   166→   166→    void logAudit({
   167→   167→      userId, userName: session.user.name || "unknown", userRole,
   168→   168→      department: "TEAM_WORK", page: "tasks", action: "READ",

   170→   170→      ipAddress: getIpAddress(req), userAgent: getUserAgent(req),
   171→   171→    })

   173→   173→  }
   174→   174→
   175→   175→  // ── Non-admin users: RBAC-filtered view ──
   176→   176→  // C20: Visibility filter moved into Prisma where clause so total count is accurate
   177→   177→  const assignedProjectIds = await getAssignedProjectIds(userId, userRole)
   178→   178→
   179→   179→  // Build filter conditions (separate from visibility)
   180→   180→  const filterConditions: Prisma.TaskWhereInput[] = []
   181→   181→
   182→   182→  // If a specific projectId is requested, verify access and restrict to it
   183→   183→  if (projectId) {
   184→   184→    // I9: Removed unnecessary `as string[]` cast — TypeScript narrows after truthiness check
   185→   185→    if (assignedProjectIds && !assignedProjectIds.includes(projectId)) {
   186→   186→      return NextResponse.json({ tasks: [], total: 0, page: 1, totalPages: 0 })
   187→   187→    }
   188→   188→    filterConditions.push({ projectId })
   189→   189→  } else if (standaloneFilter === "true") {
   190→   190→    // Standalone tasks — no project filter, but RBAC still applies
   191→   191→    filterConditions.push({ projectId: null })
   192→   192→  }
   193→   193→
   194→   194→  // assignedTo filter — non-admin: only allow "current"
   195→   195→  if (assignedToFilter === "current") {
   196→   196→    filterConditions.push({ assignedTo: userId })
   197→   197→  }
   198→   198→  // else: ignore any other value for non-admin
   199→   199→
   200→   200→  // createdBy filter — non-admin: only allow "current"
   201→   201→  if (createdByFilter === "current") {
   202→   202→    filterConditions.push({ createdBy: userId })
   203→   203→  }
   204→   204→  // else: ignore any other value for non-admin
   205→   205→
   206→   206→  // category filter
   207→   207→  if (categoryFilter) {
   208→   208→    filterConditions.push({ category: categoryFilter })
   209→   209→  }
   210→   210→
   211→   211→  // C20: Visibility filter in Prisma where clause
   212→   212→  // Non-admin can only see tasks assigned to them, created by them,
   213→   213→  // or in projects they're a member of
   214→   214→  const visibilityOr: Prisma.TaskWhereInput[] = [
   215→   215→    { assignedTo: userId },
   216→   216→    { createdBy: userId },
   217→   217→  ]
   218→   218→  if (assignedProjectIds && assignedProjectIds.length > 0) {
   219→   219→    visibilityOr.push({ projectId: { in: assignedProjectIds } })
   220→   220→  }
   221→   221→
   222→   222→  const where: Prisma.TaskWhereInput = {
   223→   223→    AND: [
   224→   224→      ...filterConditions,
   225→   225→      { OR: visibilityOr },
   226→   226→    ]
   227→   227→  }
   228→   228→
   229→   229→  const [tasks, total] = await Promise.all([
   230→   230→    db.task.findMany({
   231→   231→      where,
   232→   232→      orderBy: { createdAt: "desc" },
   233→   233→      take: limit,
   234→   234→      skip: (page - 1) * limit,
   235→   235→    }),
   236→   236→    db.task.count({ where }),
   237→   237→  ])
   238→   238→
   239→   239→  // Resolve userIds to names for assignee, approver, and creator
   240→   240→  const userIds = new Set<string>()
   241→   241→  for (const t of tasks) {
   242→   242→    if (t.assignedTo) userIds.add(t.assignedTo)
   243→   243→    if (t.approvedBy) userIds.add(t.approvedBy)
   244→   244→    if (t.createdBy) userIds.add(t.createdBy)
   245→   245→  }
   246→   246→  let userMap: Record<string, string> = {}
   247→   247→  if (userIds.size > 0) {
   248→   248→    const users = await db.user.findMany({
   249→   249→      where: { id: { in: Array.from(userIds) } },
   250→   250→      select: { id: true, name: true }
   251→   251→    })
   252→   252→    for (const u of users) userMap[u.id] = u.name
   253→   253→  }
   254→   254→
   255→   255→  // Fetch Lark task IDs for all returned tasks
   256→   256→  const larkIds = await getLarkTaskIds(tasks.map(t => t.id))
   257→   257→
   258→   258→  const enriched = tasks.map(t => ({
   259→   259→    ...serializeTask(t),
   260→   260→    assignedToName: t.assignedTo ? (userMap[t.assignedTo] || null) : null,
   261→   261→    approvedByName: t.approvedBy ? (userMap[t.approvedBy] || null) : null,
   262→   262→    createdByName: t.createdBy ? (userMap[t.createdBy] || null) : null,
   263→   263→    larkTaskId: larkIds[t.id] || null,
   264→   264→  }))
   265→   265→  void logAudit({
   266→   266→    userId, userName: session.user.name || "unknown", userRole,
   267→   267→    department: "TEAM_WORK", page: "tasks", action: "READ",
   268→   268→    description: buildDescription("READ", "tasks"),
   269→   269→    ipAddress: getIpAddress(req), userAgent: getUserAgent(req),
   270→   270→  })
   271→   271→  return NextResponse.json({ tasks: enriched, total, page, totalPages: Math.ceil(total / limit) })
   272→   272→  } catch (error: unknown) {
   273→   273→    console.error("[tasks] GET error:", error instanceof Error ? error.message : String(error))
   274→   274→    return NextResponse.json({ error: "An error occurred" }, { status: 500 })
   275→   275→  }
   276→   276→}

   278→   278→export async function POST(req: NextRequest) {
   279→   279→  try {
   280→   280→  await ensureAllTables()
   281→   281→
   282→   282→  const session = await getServerSession(authOptions)
   283→   283→  if (!session?.user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
   284→   284→
   285→   285→  // Rate limit
   286→   286→  const rl = rateLimit(`tasks-write-${session.user.id}`, RATE_LIMITS.crmWrite.limit, RATE_LIMITS.crmWrite.windowMs)
   287→   287→  if (!rl.success) {
   288→   288→    return NextResponse.json({ error: "Rate limit exceeded" }, { status: 429 })
   289→   289→  }
   290→   290→
   291→   291→  const userRole = session.user.role
   292→   292→  const userId = session.user.id
   293→   293→  let body: Record<string, unknown>
   294→   294→  try { body = await req.json() } catch {
   295→   295→    return NextResponse.json({ error: "Invalid JSON body" }, { status: 400 })
   296→   296→  }
   297→   297→
   298→   298→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   299→   299→  // ROLE RESTRICTION: Only SUPER_ADMIN and ADMIN can create tasks
   300→   300→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   301→   301→  if (!isAdmin(userRole)) {
   302→   302→    return NextResponse.json({ error: "Forbidden: Only admin and superadmin can create tasks" }, { status: 403 })
   303→   303→  }
   304→   304→
   305→   305→  // Title is required
   306→   306→  if (!body.title) {
   307→   307→    return NextResponse.json({ error: "Task title is required" }, { status: 400 })
   308→   308→  }
   309→   309→
   310→   310→  // C21: Title length limit
   311→   311→  if (String(body.title).length > 500) {
   312→   312→    return NextResponse.json({ error: "Title must be 500 characters or less" }, { status: 400 })
   313→   313→  }
   314→   314→
   315→   315→  // C21: Description length cap
   316→   316→  if (body.description && String(body.description).length > 50000) {
   317→   317→    return NextResponse.json({ error: "Description must be 50000 characters or less" }, { status: 400 })
   318→   318→  }
   319→   319→
   320→   320→  // C21: Deadline validation
   321→   321→  if (body.deadline) {
   322→   322→    const d = new Date(String(body.deadline))
   323→   323→    if (isNaN(d.getTime())) {
   324→   324→      return NextResponse.json({ error: "Invalid deadline date" }, { status: 400 })
   325→   325→    }
   326→   326→  }
   327→   327→
   328→   328→  // I5: Removed dead code — the isAdmin check above already returns 403 for non-admins
   329→   329→  // If assignedTo is provided and is NOT the creator, verify the assignee exists
   330→   330→  if (body.assignedTo && String(body.assignedTo) !== userId) {
   331→   331→    // Verify the assignee exists and is an active user
   332→   332→    const assigneeExists = await db.user.findFirst({
   333→   333→      where: { id: String(body.assignedTo), isActive: true }
   334→   334→    })
   335→   335→    if (!assigneeExists) {
   336→   336→      return NextResponse.json({ error: "Assigned user not found or inactive" }, { status: 400 })
   337→   337→    }
   338→   338→  }
   339→   339→
   340→   340→  // projectId is optional now — standalone tasks are allowed
   341→   341→  // But if projectId IS provided, check project membership for non-admins
   342→   342→  if (body.projectId) {
   343→   343→    if (!isAdmin(userRole)) {
   344→   344→      const membership = await db.projectMember.findFirst({
   345→   345→        where: { userId, projectId: String(body.projectId) }
   346→   346→      })
   347→   347→      if (!membership) {
   348→   348→        return NextResponse.json({ error: "Forbidden: You can only create tasks in your assigned projects" }, { status: 403 })
   349→   349→      }
   350→   350→    }
   351→   351→  }
   352→   352→
   353→   353→  // Validate status
   354→   354→  const taskStatus = body.status ? String(body.status) : "TODO"
   355→   355→  if (!VALID_TASK_STATUSES.includes(taskStatus)) {
   356→   356→    return NextResponse.json({ error: `Invalid status. Must be one of: ${VALID_TASK_STATUSES.join(", ")}` }, { status: 400 })
   357→   357→  }
   358→   358→
   359→   359→  // Validate priority
   360→   360→  const taskPriority = body.priority ? String(body.priority) : "MEDIUM"
   361→   361→  if (!VALID_TASK_PRIORITIES.includes(taskPriority)) {
   362→   362→    return NextResponse.json({ error: `Invalid priority. Must be one of: ${VALID_TASK_PRIORITIES.join(", ")}` }, { status: 400 })
   363→   363→  }
   364→   364→
   365→   365→  // Validate category
   366→   366→  const taskCategory = body.category ? String(body.category) : "GENERAL"
   367→   367→  if (!VALID_TASK_CATEGORIES.includes(taskCategory)) {
   368→   368→    return NextResponse.json({ error: `Invalid category. Must be one of: ${VALID_TASK_CATEGORIES.join(", ")}` }, { status: 400 })
   369→   369→  }
   370→   370→
   371→   371→  // SECURITY: Whitelist allowed fields only (prevent mass assignment)
   372→   372→  const data: Prisma.TaskUncheckedCreateInput = {
   373→   373→    title: String(body.title),
   374→   374→    description: body.description ? String(body.description) : null,
   375→   375→    status: taskStatus,
   376→   376→    priority: taskPriority,
   377→   377→    category: taskCategory,
   378→   378→    createdBy: userId,
   379→   379→    projectId: body.projectId ? String(body.projectId) : null,
   380→   380→    assignedTo: body.assignedTo ? String(body.assignedTo) : null,
   381→   381→    assigneeType: body.assigneeType ? String(body.assigneeType) : "HUMAN",
   382→   382→    deadline: body.deadline ? new Date(String(body.deadline)) : null,
   383→   383→  }
   384→   384→
   385→   385→  // Check if assignee is on approved leave during the task period
   386→   386→  if (data.assignedTo && data.deadline) {
   387→   387→    const assigneeLeave = await checkAssigneeLeave(db, data.assignedTo as string, data.deadline as Date)
   388→   388→    if (assigneeLeave) {
   389→   389→      return NextResponse.json({
   390→   390→        error: `Cannot assign task: ${assigneeLeave.name} is on ${assigneeLeave.leaveType.replace("_", " ").toLowerCase()} leave from ${new Date(assigneeLeave.startDate).toLocaleDateString()} to ${new Date(assigneeLeave.endDate).toLocaleDateString()}`,
   391→   391→      }, { status: 400 })
   392→   392→    }
   393→   393→  }
   394→   394→
   395→   395→  const task = await db.task.create({ data })
   396→   396→
   397→   397→  // Send notification to assignee when a task is created with an assignee
   398→   398→  if (data.assignedTo && (data.assignedTo as string) !== userId) {
   399→   399→    const deadlineStr = data.deadline ? ` (Due: ${new Date(data.deadline as Date).toLocaleDateString()})` : ""
   400→   400→    await sendNotification(
   401→   401→      data.assignedTo as string,
   402→   402→      "New Task Assigned",
   403→   403→      `You have been assigned a new task: ${String(body.title)}${deadlineStr}`,
   404→   404→      "TASK",
   405→   405→      "/dashboard/tasks"
   406→   406→    )
   407→   407→  }
   408→   408→
   409→   409→  // Background: sync tasks to Git (fire-and-forget) — only when projectId exists
   410→   410→  if (data.projectId) {
   411→   411→    syncTasksToGit().catch(() => {})
   412→   412→  }
   413→   413→  // Background: sync new task to Lark (fire-and-forget)
   414→   414→  syncTaskToLark(task.id, {

   417→   417→    status: task.status,

   419→   419→    assignedTo: task.assignedTo || undefined,
   420→   420→    projectId: task.projectId || undefined,
   421→   421→    deadline: task.deadline || undefined,
   422→   422→  }, userId).catch((err) => console.error("[tasks] Lark sync error:", err))
   423→   423→  // I6: Use serializeTask for consistency instead of JSON.parse(JSON.stringify(task))
   424→   424→  void logAudit({
   425→   425→    userId, userName: session.user.name || "unknown", userRole,
   426→   426→    department: "TEAM_WORK", page: "tasks", action: "CREATE",
   427→   427→    entityType: "task", entityId: task.id,
   428→   428→    description: buildDescription("CREATE", "task", task.title),
   429→   429→    ipAddress: getIpAddress(req), userAgent: getUserAgent(req),
   430→   430→  })
   431→   431→  return NextResponse.json(serializeTask(task), { status: 201 })
   432→   432→  } catch (error: unknown) {
   433→   433→    console.error("[tasks] POST error:", error instanceof Error ? error.message : String(error))
   434→   434→    return NextResponse.json({ error: "An error occurred" }, { status: 500 })
   435→   435→  }
