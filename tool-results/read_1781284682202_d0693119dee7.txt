    99→// Global Task Board Page
   100→// ═══════════════════════════════════════════════════════════════
   101→export default function GlobalTaskBoardPage() {
   102→  const { data: session, status: sessionStatus } = useSession();
   103→
   104→  // ── State ──
   105→  const [tasks, setTasks] = useState<Record<string, unknown>[]>([]);
   106→  const [projects, setProjects] = useState<Record<string, unknown>[]>([]);
   107→  const [users, setUsers] = useState<TeamUser[]>([]);
   108→  const [loading, setLoading] = useState(true);
   109→  const [search, setSearch] = useState("");
   110→  const [statusFilter, setStatusFilter] = useState("ALL");
   111→  const [priorityFilter, setPriorityFilter] = useState("ALL");
   112→  const [projectFilter, setProjectFilter] = useState("ALL");
   113→  const [addOpen, setAddOpen] = useState(false);
   114→  const [selectedTask, setSelectedTask] = useState<Record<string, unknown> | null>(null);
   115→  const [taskDetailOpen, setTaskDetailOpen] = useState(false);
   116→  const [editOpen, setEditOpen] = useState(false);
   117→  const [editForm, setEditForm] = useState<Record<string, string>>({});
   118→  const [editSaving, setEditSaving] = useState(false);
   119→  const [movingTaskId, setMovingTaskId] = useState<string | null>(null);
   120→  const [deletingTaskId, setDeletingTaskId] = useState<string | null>(null);
   121→
   122→  // ── Derived ──
   123→  const userRole = session?.user?.role || "DEVELOPER";
   124→  const userId = session?.user?.id || "";
   125→  const isAdminUser = userRole === "SUPER_ADMIN" || userRole === "ADMIN";
   126→
   127→  // ── Project name lookup map ──
   128→  const projectNameMap = new Map<string, string>(
   129→    (projects as Record<string, unknown>[]).map((p) => [
   130→      extractStr(p, "id", ""),
   131→      extractStr(p, "name", "Unknown"),
   132→    ])
   133→  );
   134→
   135→  // ── User name lookup map ──
   136→  const userNameMap = new Map<string, string>(users.map((u) => [u.id, u.name]));
   137→
   138→  // ── Fetch tasks ──
   139→  const fetchTasks = useCallback(async () => {
   140→    setLoading(true);
   141→    try {
   142→      const params = new URLSearchParams();
   143→      params.set("page", "1");
   144→      params.set("limit", "100");
   145→
   146→      // Developers see all tasks in their assigned projects (handled server-side by RBAC)
   147→      // No forced assignedTo filter — the API uses getAssignedProjectIds
   148→
   149→      // Apply filters
   150→      if (statusFilter !== "ALL") params.set("status", statusFilter);
   151→      if (priorityFilter !== "ALL") params.set("priority", priorityFilter);
   152→      if (projectFilter !== "ALL") params.set("projectId", projectFilter);
   153→
   154→      const res = await fetch(`/api/tasks?${params.toString()}`, {
   155→        credentials: "include",
   156→      });
   157→      if (res.status === 401) {
   158→        window.location.href = "/login";
   159→        return;
   160→      }
   161→      if (!res.ok) throw new Error("Failed to fetch tasks");
   162→      const raw = deepSanitize(await res.json());
   163→      const list = Array.isArray((raw as Record<string, unknown>)?.tasks)
   164→        ? (raw as Record<string, unknown>).tasks as unknown[]
   165→        : Array.isArray(raw)
   166→          ? raw
   167→          : Array.isArray((raw as Record<string, unknown>)?.data)
   168→            ? (raw as Record<string, unknown>).data as unknown[]
   169→            : [];
   170→      setTasks(list as Record<string, unknown>[]);
   171→    } catch {
   172→      toast.error("Failed to load tasks");
   173→    } finally {
   174→      setLoading(false);
   175→    }
   176→  }, [isAdminUser, statusFilter, priorityFilter, projectFilter]);
   177→
   178→  // ── Fetch projects (for filter dropdown) ──
   179→  const fetchProjects = useCallback(async () => {
   180→    try {
   181→      const res = await fetch("/api/projects", { credentials: "include" });
   182→      if (res.status === 401) { window.location.href = "/login"; return; }
   183→      if (!res.ok) return;
   184→      const raw = deepSanitize(await res.json());
   185→      const list = Array.isArray(raw)
   186→        ? raw
   187→        : Array.isArray((raw as Record<string, unknown>)?.data)
   188→          ? (raw as Record<string, unknown>).data as unknown[]
   189→          : [];
   190→      setProjects(list as Record<string, unknown>[]);
   191→    } catch {
   192→      // Silent fail for projects
   193→    }
   194→  }, []);
   195→
   196→  // ── Fetch users (for assignee dropdown) ──
   197→  const fetchUsers = useCallback(async () => {
   198→    try {
   199→      const res = await fetch("/api/team?type=users", { credentials: "include" });
   200→      if (res.status === 401) { window.location.href = "/login"; return; }
   201→      if (!res.ok) return;
   202→      const raw = deepSanitize(await res.json());
   203→      const list = Array.isArray(raw) ? raw : [];
   204→      setUsers(list as TeamUser[]);
   205→    } catch {
   206→      // Silent fail
   207→    }
   208→  }, []);
   209→
   210→  useEffect(() => {
   211→    if (sessionStatus === "loading") return;
   212→    fetchTasks();
   213→    // All users need project list for filtering; only admins need user list
   214→    fetchProjects();
   215→    if (isAdminUser) {
   216→      fetchUsers();
   217→    }
   218→  }, [sessionStatus, fetchTasks, fetchProjects, fetchUsers, isAdminUser]);
   219→
   220→  // ── Client-side search filter ──
   221→  const filteredTasks = search.trim()
   222→    ? tasks.filter((t) =>
   223→        extractStr(t, "title", "").toLowerCase().includes(search.toLowerCase())
   224→      )
   225→    : tasks;
   226→
   227→  // ── Move task handler ──
   228→  const handleMoveTask = async (taskId: string, newStatus: string) => {
   229→    setMovingTaskId(taskId);
   230→    try {
   231→      const res = await fetch("/api/tasks", {
   232→        method: "PATCH",
   233→        headers: { "Content-Type": "application/json" },
   234→        credentials: "include",
   235→        body: JSON.stringify({ id: taskId, status: newStatus }),
   236→      });
   237→      if (res.ok) {
   238→        toast.success(`Task moved to ${newStatus.replace("_", " ")}`);
   239→        fetchTasks();
   240→      } else {
   241→        if (res.status === 401) { window.location.href = "/login"; return; }
   242→        const err = await res.json().catch(() => null);
   243→        toast.error(err?.error || "Failed to move task");
   244→      }
   245→    } catch {
   246→      toast.error("Failed to move task");
   247→    } finally {
   248→      setMovingTaskId(null);
   249→    }
   250→  };
   251→
   252→  // ── Create task handler ──
   253→  const handleCreateTask = async (e: React.FormEvent<HTMLFormElement>) => {
   254→    e.preventDefault();
   255→    const form = new FormData(e.currentTarget);
   256→    const data: Record<string, unknown> = {
   257→      title: String(form.get("title") || ""),
   258→      description: String(form.get("description") || "") || undefined,
   259→      priority: String(form.get("priority") || "MEDIUM"),
   260→      category: String(form.get("category") || "GENERAL"),
   261→      projectId: String(form.get("projectId") || "") || undefined,
   262→      assignedTo: String(form.get("assignedTo") || "") || undefined,
   263→      deadline: String(form.get("deadline") || "") || undefined,
   264→    };
   265→    if (!data.title) { toast.error("Title is required"); return; }
   266→    try {
   267→      const res = await fetch("/api/tasks", {
   268→        method: "POST",
   269→        headers: { "Content-Type": "application/json" },
   270→        credentials: "include",
   271→        body: JSON.stringify(data),
   272→      });
   273→      if (res.ok) {
   274→        toast.success("Task created");
   275→        setAddOpen(false);
   276→        fetchTasks();
   277→      } else {
   278→        if (res.status === 401) { window.location.href = "/login"; return; }
   279→        const err = await res.json().catch(() => null);
   280→        toast.error(err?.error || "Failed to create task");
   281→      }
   282→    } catch {
   283→      toast.error("Failed to create task");
   284→    }
   285→  };
   286→
   287→  // ── Edit task handlers ──
   288→  const openEditDialog = (task: Record<string, unknown>) => {
   289→    setEditForm({
   290→      id: extractStr(task, "id", ""),
   291→      title: extractStr(task, "title", ""),
   292→      description: extractStr(task, "description", ""),
   293→      priority: extractStr(task, "priority", "MEDIUM"),
   294→      category: extractStr(task, "category", "GENERAL"),
   295→      assignedTo: extractStr(task, "assignedTo", ""),
   296→      projectId: extractStr(task, "projectId", ""),
   297→      deadline: extractStr(task, "deadline", "") ? extractStr(task, "deadline", "").split("T")[0] : "",
   298→    });
   299→    setEditOpen(true);
   300→  };
   301→
   302→  const handleEditSave = async () => {
   303→    if (!editForm.id) return;
   304→    setEditSaving(true);
   305→    const data: Record<string, unknown> = {
   306→      id: editForm.id,
   307→      title: editForm.title,
   308→      description: editForm.description || undefined,
   309→      priority: editForm.priority,
   310→      category: editForm.category,
   311→      assignedTo: editForm.assignedTo || undefined,
   312→      projectId: editForm.projectId || undefined,
   313→      deadline: editForm.deadline ? new Date(editForm.deadline).toISOString() : null,
   314→    };
   315→    try {
   316→      const res = await fetch("/api/tasks", {
   317→        method: "PATCH",
   318→        headers: { "Content-Type": "application/json" },
   319→        credentials: "include",
   320→        body: JSON.stringify(data),
   321→      });
   322→      if (res.ok) {
   323→        toast.success("Task updated");
   324→        setEditOpen(false);
   325→        setTaskDetailOpen(false);
   326→        fetchTasks();
   327→      } else {
   328→        const err = await res.json().catch(() => null);
   329→        toast.error(err?.error || "Failed to update task");
   330→      }
   331→    } catch {
   332→      toast.error("Failed to update task");
   333→    } finally {
   334→      setEditSaving(false);
   335→    }
   336→  };
   337→
   338→  // ── Delete task handler ──
   339→  const handleDeleteTask = async (taskId: string) => {
   340→    if (!confirm("Are you sure you want to delete this task? This action cannot be undone.")) return;
   341→    setDeletingTaskId(taskId);
   342→    try {
   343→      const res = await fetch(`/api/tasks?id=${taskId}`, {
   344→        method: "DELETE",
   345→        credentials: "include",
   346→      });
   347→      if (res.ok) {
   348→        toast.success("Task deleted");
   349→        setTaskDetailOpen(false);
   350→        setSelectedTask(null);
   351→        fetchTasks();
   352→      } else {
   353→        const err = await res.json().catch(() => null);
   354→        toast.error(err?.error || "Failed to delete task");
   355→      }
   356→    } catch {
   357→      toast.error("Failed to delete task");
   358→    } finally {
   359→      setDeletingTaskId(null);
   360→    }
   361→  };
   362→
   363→  // ── Loading ──
   364→  if (sessionStatus === "loading" || loading) {
   365→    return (
   366→      <div className="space-y-5">
   367→        <div className="flex items-center gap-3">
   368→          <div className="h-10 w-10 rounded-xl bg-muted/50 animate-pulse" />
   369→          <div className="space-y-2 flex-1">
   370→            <div className="h-6 w-52 bg-muted/50 rounded-lg animate-pulse" />
   371→            <div className="h-3.5 w-80 bg-muted/40 rounded-lg animate-pulse" />
   372→          </div>
   373→        </div>
   374→        <div className="flex items-center gap-2">
   375→          <div className="h-9 w-64 bg-muted/40 rounded-lg animate-pulse" />
   376→          <div className="h-9 w-36 bg-muted/40 rounded-lg animate-pulse" />
   377→          <div className="h-9 w-36 bg-muted/40 rounded-lg animate-pulse" />
   378→        </div>
   379→        <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-5 gap-2">
   380→          {Array.from({ length: 5 }).map((_, i) => (
   381→            <div key={i} className="h-72 bg-muted/30 rounded-xl animate-pulse" />
   382→          ))}
   383→        </div>
   384→      </div>
   385→    );
   386→  }
   387→
   388→  return (
   389→    <div className="space-y-4" style={{ animation: "fade-in 0.35s ease-out both" }}>
   390→      {/* ═══════ Page Header ═══════ */}
   391→      <div
   392→        className="flex items-center justify-between flex-wrap gap-3"
   393→        style={{ animation: "card-enter 0.4s ease-out both", animationDelay: "50ms" }}
   394→      >
   395→        <div className="flex items-center gap-3">
   396→          <div className="h-10 w-10 rounded-xl bg-gradient-to-br from-violet-500/15 to-blue-500/10 flex items-center justify-center">
   397→            <Kanban className="h-5 w-5 text-violet-600 dark:text-violet-400" />
   398→          </div>
   399→          <div>
   400→            <h1 className="text-xl font-bold tracking-tight flex items-center gap-2">
   401→              Global Task Board
   402→              <Badge variant="secondary" className="text-[10px] font-bold h-5 px-1.5">
   403→                {String(filteredTasks.length)}
   404→              </Badge>
   405→            </h1>
   406→            <p className="text-sm text-muted-foreground/70">
   407→              {isAdminUser
   408→                ? "All tasks across every project"
   409→                : "Your personal tasks across all projects"}
   410→            </p>
   411→          </div>
   412→        </div>
   413→
   414→        {isAdminUser && (
   415→          <Dialog open={addOpen} onOpenChange={setAddOpen}>
   416→            <DialogTrigger asChild>
   417→              <Button size="sm" className="gap-1.5 h-8 text-xs px-3.5 shadow-sm">
   418→                <Plus className="h-3.5 w-3.5" /> Add Task
   419→              </Button>
   420→            </DialogTrigger>
   421→            <DialogContent className="sm:max-w-md bg-white/80 dark:bg-gray-950/80 backdrop-blur-xl border-white/20 dark:border-white/10">
   422→              <DialogHeader>
   423→                <DialogTitle className="text-base font-bold">Add Task</DialogTitle>
   424→              </DialogHeader>
   425→              <form onSubmit={handleCreateTask} className="space-y-3">
   426→                <div className="space-y-1">
   427→                  <label className="text-xs font-medium">Title *</label>
   428→                  <Input name="title" required className="h-8 text-sm" placeholder="Task title..." />
   429→                </div>
   430→                <div className="space-y-1">
   431→                  <label className="text-xs font-medium">Description</label>
   432→                  <Input name="description" className="h-8 text-sm" placeholder="Optional details..." />
   433→                </div>
   434→                <div className="grid grid-cols-2 gap-3">
   435→                  <div className="space-y-1">
   436→                    <label className="text-xs font-medium">Priority</label>
   437→                    <select
   438→                      name="priority"
   439→                      defaultValue="MEDIUM"
   440→                      className="h-8 border rounded-md px-2.5 text-xs bg-background/80 w-full"
   441→                    >
   442→                      <option value="LOW">Low</option>
   443→                      <option value="MEDIUM">Medium</option>
   444→                      <option value="HIGH">High</option>
   445→                      <option value="URGENT">Urgent</option>
   446→                    </select>
   447→                  </div>
   448→                  <div className="space-y-1">
