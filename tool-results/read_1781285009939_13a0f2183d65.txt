   390→/**
   391→ * Validates data against a schema and returns either the validated data or an error response
   392→ * @param schema - The Zod schema to validate against
   393→ * @param data - The unknown data to validate
   394→ * @returns An object with either success=true and validated data, or success=false and an error message
   395→ */
   396→export function validateRequest<T>(schema: z.ZodSchema<T>, data: unknown): { success: true; data: T } | { success: false; error: string } {
   397→  const result = schema.safeParse(data)
   398→  if (!result.success) {
   399→    const firstError = result.error.issues?.[0]
   400→    return { success: false, error: firstError?.message || "Invalid input" }
   401→  }
   402→  return { success: true, data: result.data }
   403→}
