   172→function getInitials(name: string): string {
   173→  if (!name) return "?";
   174→  return name
   175→    .split(" ")
   176→    .map((n) => n[0])
   177→    .join("")
   178→    .toUpperCase()
   179→    .slice(0, 2);
   180→}
