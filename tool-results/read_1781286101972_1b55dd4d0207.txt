  1037→  const formatRelativeTime = (dateStr: string | null) => {
  1038→    if (!dateStr) return null;
  1039→    const date = new Date(dateStr);
  1040→    const now = new Date();
  1041→    const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
  1042→    if (seconds < 60) return "Just now";
  1043→    if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`;
  1044→    if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`;
