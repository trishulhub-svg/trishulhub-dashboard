    57→// ━━ Meeting link URL sanitization — prevent javascript: XSS ━━
    58→function safeMeetingLink(link: string | null): string {
    59→  if (!link) return '#';
    60→  try {
    61→    const url = new URL(link);
    62→    return url.protocol === 'https:' || url.protocol === 'http:' ? link : '#';
    63→  } catch {
    64→    return '#';
    65→  }
    66→}

   140→// ━━ Helpers ━━
   141→function calculateMeetingDuration(start: string, end: string): string {
   142→  const [startH, startM] = start.split(":").map(Number)
   143→  const [endH, endM] = end.split(":").map(Number)
   144→  let diffMinutes = (endH * 60 + endM) - (startH * 60 + startM)
   145→  if (diffMinutes < 0) diffMinutes += 24 * 60 // overnight
   146→  const hours = Math.floor(diffMinutes / 60)
   147→  const mins = diffMinutes % 60
   148→  if (hours === 0) return `${mins}m`
   149→  if (mins === 0) return `${hours}h`
   150→  return `${hours}h ${mins}m`
   151→}
   152→
   153→function isOvernightMeeting(start: string, end: string): boolean {
   154→  if (!start || !end) return false
   155→  const [startH, startM] = start.split(":").map(Number)
   156→  const [endH, endM] = end.split(":").map(Number)
   157→  return (endH * 60 + endM) < (startH * 60 + startM) // strictly less (equal = same time = invalid, not overnight)
   158→}

   160→function formatTime(time: string): string {
   161→  if (!time) return "";
   162→  const [h, m] = time.split(":");
   163→  const hour = parseInt(h);
   164→  const ampm = hour >= 12 ? "PM" : "AM";
   165→  const displayHour = hour % 12 || 12;
   166→  return `${displayHour}:${m} ${ampm}`;
   167→}

   169→function formatDate(dateStr: string): string {
   170→  try {
   171→    const date = new Date(dateStr);
   172→    if (isNaN(date.getTime())) return "Invalid date";
   173→    return date.toLocaleDateString("en-US", { weekday: "short", month: "short", day: "numeric" });
   174→  } catch {
   175→    return "Invalid date";
   176→  }
   177→}
   178→
   179→function isToday(dateStr: string): boolean {
   180→  const date = new Date(dateStr);
   181→  const today = new Date();
   182→  return date.toDateString() === today.toDateString();
   183→}
   184→
   185→function isTomorrow(dateStr: string): boolean {
   186→  const date = new Date(dateStr);
   187→  const tomorrow = new Date();
   188→  tomorrow.setDate(tomorrow.getDate() + 1);
   189→  return date.toDateString() === tomorrow.toDateString();
   190→}
   191→
   192→function isThisWeek(dateStr: string): boolean {
   193→  const date = new Date(dateStr);
   194→  const now = new Date();
   195→  const startOfWeek = new Date(now);
   196→  startOfWeek.setDate(now.getDate() - now.getDay());
   197→  startOfWeek.setHours(0, 0, 0, 0);
   198→  const endOfWeek = new Date(startOfWeek);
   199→  endOfWeek.setDate(startOfWeek.getDate() + 7);
   200→  return date >= startOfWeek && date < endOfWeek;
   201→}
   202→
   203→function getInitials(name: string): string {
   204→  return name.split(" ").map((n) => n[0]).join("").toUpperCase().slice(0, 2);
   205→}
   206→
   207→function getDateLabel(dateStr: string): string {
   208→  if (isToday(dateStr)) return "Today";
   209→  if (isTomorrow(dateStr)) return "Tomorrow";
   210→  if (isThisWeek(dateStr)) return "This Week";
   211→  return "Later";
   212→}
