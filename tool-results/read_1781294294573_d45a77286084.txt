   141→   141→function calculateMeetingDuration(start: string, end: string): string {
   142→   142→  const [startH, startM] = start.split(":").map(Number)
   143→   143→  const [endH, endM] = end.split(":").map(Number)
   144→   144→  let diffMinutes = (endH * 60 + endM) - (startH * 60 + startM)
   145→   145→  if (diffMinutes < 0) diffMinutes += 24 * 60 // overnight
   146→   146→  const hours = Math.floor(diffMinutes / 60)
   147→   147→  const mins = diffMinutes % 60
   148→   148→  if (hours === 0) return `${mins}m`
   149→   149→  if (mins === 0) return `${hours}h`
   150→   150→  return `${hours}h ${mins}m`
   151→   151→}

   153→   153→function isOvernightMeeting(start: string, end: string): boolean {
   154→   154→  if (!start || !end) return false
   155→   155→  const [startH, startM] = start.split(":").map(Number)
   156→   156→  const [endH, endM] = end.split(":").map(Number)
   157→   157→  return (endH * 60 + endM) < (startH * 60 + startM) // strictly less (equal = same time = invalid, not overnight)
   158→   158→}

   160→   160→function formatTime(time: string): string {
   161→   161→  if (!time) return "";
   162→   162→  const [h, m] = time.split(":");
   163→   163→  const hour = parseInt(h);
   164→   164→  const ampm = hour >= 12 ? "PM" : "AM";
   165→   165→  const displayHour = hour % 12 || 12;
   166→   166→  return `${displayHour}:${m} ${ampm}`;
   167→   167→}

   169→   169→function formatDate(dateStr: string): string {
   170→   170→  try {
   171→   171→    const date = new Date(dateStr);
   172→   172→    if (isNaN(date.getTime())) return "Invalid date";
   173→   173→    return date.toLocaleDateString("en-US", { weekday: "short", month: "short", day: "numeric" });
   174→   174→  } catch {
   175→   175→    return "Invalid date";
   176→   176→  }
   177→   177→}

   179→   179→function isToday(dateStr: string): boolean {
   180→   180→  const date = new Date(dateStr);
   181→   181→  const today = new Date();
   182→   182→  return date.toDateString() === today.toDateString();
   183→   183→}
   184→   184→
   185→   185→function isTomorrow(dateStr: string): boolean {
   186→   186→  const date = new Date(dateStr);
   187→   187→  const tomorrow = new Date();
   188→   188→  tomorrow.setDate(tomorrow.getDate() + 1);
   189→   189→  return date.toDateString() === tomorrow.toDateString();
   190→   190→}
   191→   191→
   192→   192→function isThisWeek(dateStr: string): boolean {
   193→   193→  const date = new Date(dateStr);
   194→   194→  const now = new Date();
   195→   195→  const startOfWeek = new Date(now);
   196→   196→  startOfWeek.setDate(now.getDate() - now.getDay());
   197→   197→  startOfWeek.setHours(0, 0, 0, 0);
   198→   198→  const endOfWeek = new Date(startOfWeek);
   199→   199→  endOfWeek.setDate(startOfWeek.getDate() + 7);
   200→   200→  return date >= startOfWeek && date < endOfWeek;
   201→   201→}

   203→   203→function getInitials(name: string): string {
   204→   204→  return name.split(" ").map((n) => n[0]).join("").toUpperCase().slice(0, 2);
   205→   205→}

   207→   207→function getDateLabel(dateStr: string): string {
   208→   208→  if (isToday(dateStr)) return "Today";
   209→   209→  if (isTomorrow(dateStr)) return "Tomorrow";
   210→   210→  if (isThisWeek(dateStr)) return "This Week";
   211→   211→  return "Later";
   212→   212→}
