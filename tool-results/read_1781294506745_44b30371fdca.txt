   186→function formatRelativeTime(dateStr: string): string {
   187→  if (!dateStr) return "unknown";
   188→  const date = new Date(dateStr);
   189→  if (isNaN(date.getTime())) return "unknown";
   190→  const now = new Date();
   191→  const diffMs = now.getTime() - date.getTime();
   192→  const diffMin = Math.floor(diffMs / 60000);
   193→  if (diffMin < 1) return "just now";
   194→  if (diffMin < 60) return `${diffMin}m ago`;
   195→  const diffHr = Math.floor(diffMin / 60);
   196→  if (diffHr < 24) return `${diffHr}h ago`;
   197→  const diffDay = Math.floor(diffHr / 24);
   198→  if (diffDay < 7) return `${diffDay}d ago`;
   199→  return safeDateStr(date);
   200→}

   538→      if (!notif.isRead) await markAsRead(notif.id);
   539→    } catch {}
   540→    if (notif.link && notif.link.startsWith("/")) {
   541→      router.push(notif.link);
   542→      setNotifOpen(false);
   543→    }
   544→  };
   545→
   546→  const handleNavigate = (href: string) => {
   547→    router.push(href);
   548→    setMobileOpen(false);
   549→  };
   550→
   551→  // Safety timeout for session loading — show fallback after 15s
   552→  const [sessionTimedOut, setSessionTimedOut] = useState(false);
   553→  useEffect(() => {
   554→    if (status !== "loading") return;
   555→    const t = setTimeout(() => setSessionTimedOut(true), 15000);
   556→    return () => clearTimeout(t);
   557→  }, [status]);
   558→
   559→  if (status === "loading" && !sessionTimedOut) {
   560→    return <LoadingScreen />;
   561→  }
   562→
   563→  if (sessionTimedOut && status === "loading") {
   564→    return (
   565→      <div className="fixed inset-0 z-50 flex min-h-screen items-center justify-center bg-background">
   566→        <div className="flex flex-col items-center gap-4">
   567→          <p className="text-sm text-muted-foreground">Session is taking too long to load...</p>
   568→          <button
   569→            onClick={() => signOut({ callbackUrl: "/login" })}
   570→            className="rounded-lg bg-primary px-4 py-2 text-sm text-primary-foreground hover:bg-primary/90"
   571→          >
   572→            Sign In Again
   573→          </button>
   574→        </div>
   575→      </div>
   576→    );
   577→  }
   578→
   579→  if (!session) return null;
   580→  // W3: Prevent CLIENT users from seeing any dashboard content before redirect
   581→  if (userRole === "CLIENT") return null;
   582→
   583→  return (
   584→    <div className="min-h-screen flex bg-background">
   585→      {/* Desktop Sidebar - wider and more spacious */}
   586→      <aside
   587→        className={cn(
   588→          "hidden md:flex flex-col border-r border-border bg-sidebar transition-all duration-300 relative z-40",
   589→          collapsed ? "w-[72px]" : "w-[280px]"
   590→        )}
   591→      >
   592→        <SidebarContent
   593→          collapsed={collapsed}
   594→          userRole={userRole}
   595→          userName={userName}
   596→          pathname={pathname}
   597→          onNavigate={handleNavigate}
   598→          badgeCounts={navBadgeData}
   599→        />
   600→        <Button
   601→          variant="ghost"
   602→          size="icon"
   603→          className="absolute top-24 -right-3 z-10 h-7 w-7 rounded-full border bg-background shadow-sm hidden md:flex"
   604→          onClick={() => setCollapsed(!collapsed)}
   605→          aria-label="Toggle sidebar"
   606→        >
   607→          <ChevronLeft className={cn("h-3.5 w-3.5 transition-transform", collapsed && "rotate-180")} />
   608→        </Button>
   609→      </aside>
   610→
   611→      {/* Mobile Sidebar */}
   612→      <Sheet open={mobileOpen} onOpenChange={setMobileOpen}>
   613→        <SheetContent side="left" className="w-[280px] p-0 bg-sidebar">
   614→          <div className="flex items-center justify-end p-2 pb-0">
   615→            <Button
   616→              variant="ghost"
   617→              size="icon"
   618→              className="h-8 w-8"
   619→              onClick={() => setMobileOpen(false)}
   620→              aria-label="Close menu"
   621→            >
   622→              <X className="h-4 w-4" />
   623→            </Button>
   624→          </div>
   625→          <SidebarContent
   626→            collapsed={false}
   627→            userRole={userRole}
   628→            userName={userName}
   629→            pathname={pathname}
   630→            onNavigate={handleNavigate}
   631→            badgeCounts={navBadgeData}
   632→          />
   633→        </SheetContent>
   634→      </Sheet>
   635→
   636→      {/* Main Content */}
   637→      <div className="flex-1 flex flex-col min-w-0">
   638→        {/* Header - taller and more prominent */}
   639→        <header className="h-14 sm:h-16 border-b border-border bg-card flex items-center justify-between px-3 sm:px-5 sticky top-0 z-30">
   640→          <div className="flex items-center gap-3">
   641→            <Button variant="ghost" size="icon" className="md:hidden" aria-label="Open menu" onClick={() => setMobileOpen(true)}>
   642→              <Menu className="h-5 w-5" />
   643→            </Button>
   644→            <h2 className="text-sm sm:text-base font-semibold text-foreground truncate">
   645→              {allNavItems.find((i) => pathname === i.href || (i.href !== "/dashboard" && pathname.startsWith(i.href + "/")))?.title || "Dashboard"}
   646→            </h2>
   647→          </div>
   648→
   649→          <div className="flex items-center gap-1 sm:gap-2">
   650→            {/* Theme Selector Dropdown */}
   651→            <DropdownMenu>
   652→              <DropdownMenuTrigger asChild>
   653→                <Button variant="ghost" size="icon" className="h-8 w-8 sm:h-9 sm:w-9" aria-label="Change theme">
   654→                  {theme === "system" ? (
   655→                    <Monitor className="h-4 w-4" />
   656→                  ) : theme === "dark" ? (
   657→                    <Moon className="h-4 w-4" />
   658→                  ) : theme === "bluelight" ? (
   659→                    <Eye className="h-4 w-4 text-amber-600" />
   660→                  ) : (
   661→                    <Sun className="h-4 w-4" />
   662→                  )}
   663→                </Button>
   664→              </DropdownMenuTrigger>
   665→              <DropdownMenuContent align="end" className="w-48">
   666→                <DropdownMenuItem onClick={() => setTheme("light")} className={cn("flex items-center gap-3", theme === "light" && "bg-accent")}>
   667→                  <Sun className="h-4 w-4" />
   668→                  <span>Light Mode</span>
   669→                  {theme === "light" && <Check className="h-3 w-3 ml-auto text-primary" />}
   670→                </DropdownMenuItem>
   671→                <DropdownMenuItem onClick={() => setTheme("dark")} className={cn("flex items-center gap-3", theme === "dark" && "bg-accent")}>
   672→                  <Moon className="h-4 w-4" />
   673→                  <span>Dark Mode</span>
   674→                  {theme === "dark" && <Check className="h-3 w-3 ml-auto text-primary" />}
   675→                </DropdownMenuItem>
   676→                <DropdownMenuItem onClick={() => setTheme("bluelight")} className={cn("flex items-center gap-3", theme === "bluelight" && "bg-accent")}>
   677→                  <Eye className="h-4 w-4 text-amber-600" />
   678→                  <div className="flex flex-col">
   679→                    <span>Blue Light</span>
   680→                    <span className="text-[10px] text-muted-foreground">Eye protection for long use</span>
   681→                  </div>
   682→                  {theme === "bluelight" && <Check className="h-3 w-3 ml-auto text-primary" />}
   683→                </DropdownMenuItem>
   684→                <DropdownMenuSeparator />
   685→                <DropdownMenuItem onClick={() => setTheme("system")} className={cn("flex items-center gap-3", theme === "system" && "bg-accent")}>
   686→                  <Monitor className="h-4 w-4" />
   687→                  <span>System Default</span>
   688→                  {theme === "system" && <Check className="h-3 w-3 ml-auto text-primary" />}
   689→                </DropdownMenuItem>
   690→              </DropdownMenuContent>
   691→            </DropdownMenu>
   692→
   693→            {/* Notifications Sheet — better scroll on all devices */}
   694→            <Sheet open={notifOpen} onOpenChange={setNotifOpen}>
   695→              <SheetTrigger asChild>
   696→                <Button variant="ghost" size="icon" className="relative h-8 w-8 sm:h-9 sm:w-9" aria-label="Notifications">
   697→                  <Bell className="h-4 w-4" />
   698→                  {unreadCount > 0 && (
   699→                    <Badge className="absolute -top-1 -right-1 h-5 w-5 p-0 flex items-center justify-center text-[10px]">
   700→                      {unreadCount > 9 ? "9+" : unreadCount}
   701→                    </Badge>
   702→                  )}
   703→                </Button>
   704→              </SheetTrigger>
   705→              <SheetContent className="w-full sm:max-w-md p-0 overflow-hidden flex flex-col">
   706→                <SheetHeader className="p-4 pb-3 border-b pr-10 shrink-0">
   707→                  <div className="flex items-center justify-between">
   708→                    <SheetTitle className="text-sm font-semibold">Notifications</SheetTitle>
   709→                    {unreadCount > 0 && (
   710→                      <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={markAllAsRead}>
   711→                        <Check className="h-3 w-3 mr-1" /> Mark all read
   712→                      </Button>
   713→                    )}
   714→                  </div>
   715→                </SheetHeader>
   716→                <ScrollArea className="flex-1">
   717→                  {notifications.length === 0 ? (
   718→                    <div className="p-8 text-center text-sm text-muted-foreground">
   719→                      <Bell className="h-8 w-8 mx-auto mb-2 opacity-20" />
   720→                      <p>No notifications yet</p>
   721→                    </div>
   722→                  ) : (
   723→                    <div>
   724→                      {notifications.slice(0, 50).map((notif) => {
   725→                        const NotifIcon = notifIcons[notif.type] || Info;
   726→                        return (
   727→                          <div
   728→                            key={notif.id}
   729→                            className={cn(
   730→                              "flex items-start gap-3 p-3 hover:bg-accent/50 cursor-pointer transition-colors border-b last:border-0",
   731→                              !notif.isRead && "bg-primary/5"
   732→                            )}
   733→                            onClick={() => handleNotificationClick(notif)}
   734→                          >
   735→                            <div className={cn(
   736→                              "mt-0.5 h-7 w-7 rounded-full flex items-center justify-center shrink-0",
   737→                              notif.type === "ERROR" || notif.type === "WARNING" ? "bg-red-100 text-red-600 dark:bg-red-900/30 dark:text-red-400" :
   738→                              notif.type === "SUCCESS" ? "bg-green-100 text-green-600 dark:bg-green-900/30 dark:text-green-400" :
   739→                              notif.type === "TASK" || notif.type === "APPROVAL" ? "bg-blue-100 text-blue-600 dark:bg-blue-900/30 dark:text-blue-400" :
   740→                              notif.type === "AGENT" ? "bg-purple-100 text-purple-600 dark:bg-purple-900/30 dark:text-purple-400" :
   741→                              "bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400"
   742→                            )}>
   743→                              <NotifIcon className="h-3.5 w-3.5" />
   744→                            </div>
   745→                            <div className="flex-1 min-w-0">
   746→                              <div className="flex items-center justify-between gap-2">
   747→                                <span className={cn("text-xs font-medium", !notif.isRead && "font-semibold")}>
   748→                                  {safeText(notif.title, "")}
   749→                                </span>
   750→                                <span className="text-[10px] text-muted-foreground whitespace-nowrap">
   751→                                  {formatRelativeTime(notif.createdAt)}
   752→                                </span>
   753→                              </div>
   754→                              <p className="text-xs text-muted-foreground line-clamp-2 mt-0.5">
   755→                                {safeText(notif.message, "")}
   756→                              </p>
   757→                            </div>
   758→                            <Button
   759→                              variant="ghost"
   760→                              size="icon"
   761→                              className="h-5 w-5 shrink-0 opacity-40 hover:opacity-100"
   762→                              onClick={(e) => {
   763→                                e.stopPropagation();
   764→                                deleteNotification(notif.id);
   765→                              }}
   766→                              aria-label="Dismiss notification"
   767→                            >
   768→                              <Trash2 className="h-3 w-3" />
   769→                            </Button>
   770→                          </div>
   771→                        );
   772→                      })}
   773→                      {/* W8: Show indicator when notifications are capped at 50 */}
   774→                      {notifications.length > 50 && (
   775→                        <span className="text-xs text-center text-muted-foreground block py-2">
   776→                          View all {notifications.length} notifications
   777→                        </span>
   778→                      )}
   779→                    </div>
   780→                  )}
   781→                </ScrollArea>
   782→              </SheetContent>
   783→            </Sheet>
   784→
   785→            <DropdownMenu>
   786→              <DropdownMenuTrigger asChild>
   787→                <Button variant="ghost" className="flex items-center gap-2 h-8 sm:h-9">
   788→                  <Avatar className="h-8 w-8">
   789→                    <AvatarFallback className="bg-primary text-primary-foreground text-xs font-bold">
   790→                      {userName.split(" ").filter(Boolean).map((n) => n[0]).join("").toUpperCase().slice(0, 2)}
   791→                    </AvatarFallback>
   792→                  </Avatar>
   793→                  <span className="hidden sm:block text-sm font-medium">{userName}</span>
   794→                </Button>
   795→              </DropdownMenuTrigger>
   796→              <DropdownMenuContent align="end" className="w-56">
   797→                <div className="p-2">
   798→                  <p className="text-sm font-medium">{userName}</p>
   799→                  <p className="text-xs text-muted-foreground">{userEmail}</p>
   800→                  <Badge variant="secondary" className="mt-1 text-xs">{userRole.replace("_", " ")}</Badge>
   801→                </div>
   802→                <DropdownMenuSeparator />
   803→                <DropdownMenuItem onClick={() => router.push("/dashboard/settings")}>
   804→                  <Settings className="mr-2 h-4 w-4" /> Settings
   805→                </DropdownMenuItem>
   806→                <DropdownMenuSeparator />
   807→                <DropdownMenuItem onClick={async () => { await signOut({ redirect: false }); router.push("/login"); }}>
   808→                  <LogOut className="mr-2 h-4 w-4" /> Sign Out
   809→                </DropdownMenuItem>
   810→              </DropdownMenuContent>
   811→            </DropdownMenu>
   812→          </div>
   813→        </header>
   814→
   815→        {/* Page Content - more padding */}
   816→        <main className="flex-1 p-3 sm:p-4 md:p-6 lg:p-8 overflow-auto">{children}</main>
   817→      </div>
   818→
   819→      {/* Agentation — visual feedback tool (all users) */}
   820→      <Agentation />
   821→    </div>
   822→  );
