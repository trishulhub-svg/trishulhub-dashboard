    41→    41→async function getLarkTaskIds(taskIds: string[]): Promise<Record<string, string>> {
    42→    42→  if (taskIds.length === 0) return {}
    43→    43→  try {
    44→    44→    const rows = await db.$queryRawUnsafe<Array<{ taskId: string; larkTaskId: string }>>(
    45→    45→      'SELECT "taskId", "larkTaskId" FROM "LarkTaskMapping" WHERE "taskId" IN (?)',
    46→    46→      taskIds
    47→    47→    )
    48→    48→    const map: Record<string, string> = {}
    49→    49→    for (const r of rows) map[r.taskId] = r.larkTaskId
    50→    50→    return map
    51→    51→  } catch {
    52→    52→    return {}
    53→    53→  }
    54→    54→}

    57→    57→async function checkAssigneeLeave(db: any, userId: string, deadline: Date): Promise<{ name: string; leaveType: string; startDate: Date; endDate: Date } | null> {

    89→    89→  }
    90→    90→
    91→    91→  const userRole = session.user.role
    92→    92→  const userId = session.user.id
    93→    93→
    94→    94→  const { searchParams } = new URL(req.url)
    95→    95→  const projectId = searchParams.get("projectId")

   121→   121→      where.createdBy = userId

   131→   131→      db.task.findMany({
   132→   132→        where,
   133→   133→        orderBy: { createdAt: "desc" },
   134→   134→        take: limit,
   135→   135→        skip: (page - 1) * limit,
   136→   136→      }),
   137→   137→      db.task.count({ where }),
   138→   138→    ])
   139→   139→
   140→   140→    // Resolve userIds to names for assignee, approver, and creator
   141→   141→    const userIds = new Set<string>()
   142→   142→    for (const t of tasks) {
   143→   143→      if (t.assignedTo) userIds.add(t.assignedTo)

   151→   151→        select: { id: true, name: true }

   156→   156→    // Fetch Lark task IDs for all returned tasks

   162→   162→      approvedByName: t.approvedBy ? (userMap[t.approvedBy] || null) : null,
   163→   163→      createdByName: t.createdBy ? (userMap[t.createdBy] || null) : null,

   173→   173→  }
   174→   174→
   175→   175→  // ── Non-admin users: RBAC-filtered view ──
   176→   176→  // C20: Visibility filter moved into Prisma where clause so total count is accurate
   177→   177→  const assignedProjectIds = await getAssignedProjectIds(userId, userRole)
   178→   178→
   179→   179→  // Build filter conditions (separate from visibility)
   180→   180→  const filterConditions: Prisma.TaskWhereInput[] = []
   181→   181→
   182→   182→  // If a specific projectId is requested, verify access and restrict to it
   183→   183→  if (projectId) {
   184→   184→    // I9: Removed unnecessary `as string[]` cast — TypeScript narrows after truthiness check
   185→   185→    if (assignedProjectIds && !assignedProjectIds.includes(projectId)) {
   186→   186→      return NextResponse.json({ tasks: [], total: 0, page: 1, totalPages: 0 })
   187→   187→    }
   188→   188→    filterConditions.push({ projectId })
   189→   189→  } else if (standaloneFilter === "true") {
   190→   190→    // Standalone tasks — no project filter, but RBAC still applies
   191→   191→    filterConditions.push({ projectId: null })
   192→   192→  }
   193→   193→
   194→   194→  // assignedTo filter — non-admin: only allow "current"
   195→   195→  if (assignedToFilter === "current") {
   196→   196→    filterConditions.push({ assignedTo: userId })
   197→   197→  }
   198→   198→  // else: ignore any other value for non-admin
   199→   199→
   200→   200→  // createdBy filter — non-admin: only allow "current"
   201→   201→  if (createdByFilter === "current") {
   202→   202→    filterConditions.push({ createdBy: userId })
   203→   203→  }
   204→   204→  // else: ignore any other value for non-admin
   205→   205→
   206→   206→  // category filter
   207→   207→  if (categoryFilter) {
   208→   208→    filterConditions.push({ category: categoryFilter })
   209→   209→  }
   210→   210→
   211→   211→  // C20: Visibility filter in Prisma where clause
   212→   212→  // Non-admin can only see tasks assigned to them or created by them
   213→   213→  const visibilityOr: Prisma.TaskWhereInput[] = [
   214→   214→    { assignedTo: userId },
   215→   215→    { createdBy: userId },
   216→   216→  ]
   217→   217→
   218→   218→  const where: Prisma.TaskWhereInput = {
   219→   219→    AND: [
   220→   220→      ...filterConditions,
   221→   221→      { OR: visibilityOr },
   222→   222→    ]
   223→   223→  }
   224→   224→
   225→   225→  const [tasks, total] = await Promise.all([
   226→   226→    db.task.findMany({
   227→   227→      where,
   228→   228→      orderBy: { createdAt: "desc" },
   229→   229→      take: limit,
   230→   230→      skip: (page - 1) * limit,
   231→   231→    }),
   232→   232→    db.task.count({ where }),
   233→   233→  ])
   234→   234→
   235→   235→  // Resolve userIds to names for assignee, approver, and creator
   236→   236→  const userIds = new Set<string>()
   237→   237→  for (const t of tasks) {
   238→   238→    if (t.assignedTo) userIds.add(t.assignedTo)
   239→   239→    if (t.approvedBy) userIds.add(t.approvedBy)
   240→   240→    if (t.createdBy) userIds.add(t.createdBy)
   241→   241→  }
   242→   242→  let userMap: Record<string, string> = {}
   243→   243→  if (userIds.size > 0) {
   244→   244→    const users = await db.user.findMany({
   245→   245→      where: { id: { in: Array.from(userIds) } },
   246→   246→      select: { id: true, name: true }
   247→   247→    })
   248→   248→    for (const u of users) userMap[u.id] = u.name
   249→   249→  }
   250→   250→
   251→   251→  // Fetch Lark task IDs for all returned tasks
   252→   252→  const larkIds = await getLarkTaskIds(tasks.map(t => t.id))
   253→   253→
   254→   254→  const enriched = tasks.map(t => ({
   255→   255→    ...serializeTask(t),
   256→   256→    assignedToName: t.assignedTo ? (userMap[t.assignedTo] || null) : null,
   257→   257→    approvedByName: t.approvedBy ? (userMap[t.approvedBy] || null) : null,
   258→   258→    createdByName: t.createdBy ? (userMap[t.createdBy] || null) : null,
   259→   259→    larkTaskId: larkIds[t.id] || null,
   260→   260→  }))
   261→   261→  void logAudit({
   262→   262→    userId, userName: session.user.name || "unknown", userRole,
   263→   263→    department: "TEAM_WORK", page: "tasks", action: "READ",
   264→   264→    description: buildDescription("READ", "tasks"),
   265→   265→    ipAddress: getIpAddress(req), userAgent: getUserAgent(req),
   266→   266→  })
   267→   267→  return NextResponse.json({ tasks: enriched, total, page, totalPages: Math.ceil(total / limit) })
   268→   268→  } catch (error: unknown) {
   269→   269→    console.error("[tasks] GET error:", error instanceof Error ? error.message : String(error))
   270→   270→    return NextResponse.json({ error: "An error occurred" }, { status: 500 })
   271→   271→  }
   272→   272→}

   274→   274→export async function POST(req: NextRequest) {
   275→   275→  try {
   276→   276→  await ensureAllTables()
   277→   277→
   278→   278→  const session = await getServerSession(authOptions)
   279→   279→  if (!session?.user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
   280→   280→
   281→   281→  // Rate limit
   282→   282→  const rl = rateLimit(`tasks-write-${session.user.id}`, RATE_LIMITS.crmWrite.limit, RATE_LIMITS.crmWrite.windowMs)
   283→   283→  if (!rl.success) {
   284→   284→    return NextResponse.json({ error: "Rate limit exceeded" }, { status: 429 })
   285→   285→  }
   286→   286→
   287→   287→  const userRole = session.user.role
   288→   288→  const userId = session.user.id
   289→   289→  let body: Record<string, unknown>
   290→   290→  try { body = await req.json() } catch {
   291→   291→    return NextResponse.json({ error: "Invalid JSON body" }, { status: 400 })
   292→   292→  }
   293→   293→
   294→   294→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   295→   295→  // ROLE RESTRICTION: Only SUPER_ADMIN and ADMIN can create tasks
   296→   296→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   297→   297→  if (!isAdmin(userRole)) {
   298→   298→    return NextResponse.json({ error: "Forbidden: Only admin and superadmin can create tasks" }, { status: 403 })
   299→   299→  }
   300→   300→
   301→   301→  // Title is required
   302→   302→  if (!body.title) {
   303→   303→    return NextResponse.json({ error: "Task title is required" }, { status: 400 })
   304→   304→  }
   305→   305→
   306→   306→  // C21: Title length limit
   307→   307→  if (String(body.title).length > 500) {
   308→   308→    return NextResponse.json({ error: "Title must be 500 characters or less" }, { status: 400 })
   309→   309→  }
   310→   310→
   311→   311→  // C21: Description length cap
   312→   312→  if (body.description && String(body.description).length > 50000) {
   313→   313→    return NextResponse.json({ error: "Description must be 50000 characters or less" }, { status: 400 })
   314→   314→  }
   315→   315→
   316→   316→  // C21: Deadline validation
   317→   317→  if (body.deadline) {
   318→   318→    const d = new Date(String(body.deadline))
   319→   319→    if (isNaN(d.getTime())) {
   320→   320→      return NextResponse.json({ error: "Invalid deadline date" }, { status: 400 })
   321→   321→    }
   322→   322→  }
   323→   323→
   324→   324→  // I5: Removed dead code — the isAdmin check above already returns 403 for non-admins
   325→   325→  // If assignedTo is provided and is NOT the creator, verify the assignee exists
   326→   326→  if (body.assignedTo && String(body.assignedTo) !== userId) {
   327→   327→    // Verify the assignee exists and is an active user
   328→   328→    const assigneeExists = await db.user.findFirst({
   329→   329→      where: { id: String(body.assignedTo), isActive: true }
   330→   330→    })
   331→   331→    if (!assigneeExists) {
   332→   332→      return NextResponse.json({ error: "Assigned user not found or inactive" }, { status: 400 })
   333→   333→    }
   334→   334→  }
   335→   335→
   336→   336→  // projectId is optional now — standalone tasks are allowed
   337→   337→  // But if projectId IS provided, check project membership for non-admins
   338→   338→  if (body.projectId) {
   339→   339→    if (!isAdmin(userRole)) {
   340→   340→      const membership = await db.projectMember.findFirst({
   341→   341→        where: { userId, projectId: String(body.projectId) }
   342→   342→      })
   343→   343→      if (!membership) {
   344→   344→        return NextResponse.json({ error: "Forbidden: You can only create tasks in your assigned projects" }, { status: 403 })
   345→   345→      }
   346→   346→    }
   347→   347→  }
   348→   348→
   349→   349→  // Validate status
   350→   350→  const taskStatus = body.status ? String(body.status) : "TODO"
   351→   351→  if (!VALID_TASK_STATUSES.includes(taskStatus)) {
   352→   352→    return NextResponse.json({ error: `Invalid status. Must be one of: ${VALID_TASK_STATUSES.join(", ")}` }, { status: 400 })
   353→   353→  }
   354→   354→
   355→   355→  // Validate priority
   356→   356→  const taskPriority = body.priority ? String(body.priority) : "MEDIUM"
   357→   357→  if (!VALID_TASK_PRIORITIES.includes(taskPriority)) {
   358→   358→    return NextResponse.json({ error: `Invalid priority. Must be one of: ${VALID_TASK_PRIORITIES.join(", ")}` }, { status: 400 })
   359→   359→  }
   360→   360→
   361→   361→  // Validate category
   362→   362→  const taskCategory = body.category ? String(body.category) : "GENERAL"
   363→   363→  if (!VALID_TASK_CATEGORIES.includes(taskCategory)) {
   364→   364→    return NextResponse.json({ error: `Invalid category. Must be one of: ${VALID_TASK_CATEGORIES.join(", ")}` }, { status: 400 })
   365→   365→  }
   366→   366→
   367→   367→  // SECURITY: Whitelist allowed fields only (prevent mass assignment)
   368→   368→  const data: Prisma.TaskUncheckedCreateInput = {
   369→   369→    title: String(body.title),
   370→   370→    description: body.description ? String(body.description) : null,
   371→   371→    status: taskStatus,
   372→   372→    priority: taskPriority,
   373→   373→    category: taskCategory,
   374→   374→    createdBy: userId,
   375→   375→    projectId: body.projectId ? String(body.projectId) : null,
   376→   376→    assignedTo: body.assignedTo ? String(body.assignedTo) : null,
   377→   377→    assigneeType: body.assigneeType ? String(body.assigneeType) : "HUMAN",
   378→   378→    deadline: body.deadline ? new Date(String(body.deadline)) : null,
   379→   379→  }
   380→   380→
   381→   381→  // Check if assignee is on approved leave during the task period
   382→   382→  if (data.assignedTo && data.deadline) {
   383→   383→    const assigneeLeave = await checkAssigneeLeave(db, data.assignedTo as string, data.deadline as Date)
   384→   384→    if (assigneeLeave) {
   385→   385→      return NextResponse.json({
   386→   386→        error: `Cannot assign task: ${assigneeLeave.name} is on ${assigneeLeave.leaveType.replace("_", " ").toLowerCase()} leave from ${new Date(assigneeLeave.startDate).toLocaleDateString()} to ${new Date(assigneeLeave.endDate).toLocaleDateString()}`,
   387→   387→      }, { status: 400 })
   388→   388→    }
   389→   389→  }
   390→   390→
   391→   391→  const task = await db.task.create({ data })
   392→   392→
   393→   393→  // Send notification to assignee when a task is created with an assignee
   394→   394→  if (data.assignedTo && (data.assignedTo as string) !== userId) {
   395→   395→    const deadlineStr = data.deadline ? ` (Due: ${new Date(data.deadline as Date).toLocaleDateString()})` : ""
   396→   396→    await sendNotification(
   397→   397→      data.assignedTo as string,
   398→   398→      "New Task Assigned",
   399→   399→      `You have been assigned a new task: ${String(body.title)}${deadlineStr}`,
   400→   400→      "TASK",
   401→   401→      "/dashboard/tasks"
   402→   402→    )
   403→   403→  }
   404→   404→
   405→   405→  // Background: sync tasks to Git (fire-and-forget) — only when projectId exists
   406→   406→  if (data.projectId) {
   407→   407→    syncTasksToGit().catch(() => {})
   408→   408→  }
   409→   409→  // Background: sync new task to Lark (fire-and-forget)
   410→   410→  syncTaskToLark(task.id, {
   411→   411→    title: task.title,
   412→   412→    description: task.description || undefined,
   413→   413→    status: task.status,
   414→   414→    priority: task.priority,
   415→   415→    assignedTo: task.assignedTo || undefined,
