   107→function formatRelativeTime(dateStr: string | null | undefined): string {
   108→  if (!dateStr) return "N/A"
   109→  const date = new Date(dateStr)
   110→  if (isNaN(date.getTime())) return "N/A"
   111→  const now = new Date()
   112→  const diffMs = now.getTime() - date.getTime()
   113→  const diffSec = Math.floor(diffMs / 1000)
   114→  if (diffSec < 60) return "just now"
   115→  const diffMin = Math.floor(diffSec / 60)
   116→  if (diffMin < 60) return `${diffMin}m ago`
   117→  const diffHr = Math.floor(diffMin / 60)
   118→  if (diffHr < 24) return `${diffHr}h ago`
   119→  const diffDay = Math.floor(diffHr / 24)
   120→  if (diffDay < 7) return `${diffDay}d ago`
   121→  return formatDateTime(dateStr)
   122→}

   124→/** Safe helper: extract initials from a potentially null user name */
   125→function getInitials(userName: string | null | undefined): string {
   126→  if (!userName) return "?"
   127→  return userName.split(" ").filter(Boolean).map(n => n[0] || "").join("").toUpperCase().slice(0, 2) || "?"
   128→}
   129→
   130→/** Safe helper: format role for display */
   131→function formatRole(role: string | null | undefined): string {
   132→  if (!role) return "—"
   133→  return role.replace(/_/g, " ")
   134→}

   136→/**
   137→ * Local error boundary for the Audit Trail page.
   138→ * Catches render-time errors so they show a useful message
   139→ * instead of the generic DashboardError boundary.
   140→ */
   141→class AuditTrailErrorBoundary extends React.Component<
   142→  { children: React.ReactNode },
   143→  { hasError: boolean; error: Error | null }
   144→> {
   145→  constructor(props: { children: React.ReactNode }) {
   146→    super(props)
   147→    this.state = { hasError: false, error: null }
   148→  }
   149→  static getDerivedStateFromError(error: Error) {
   150→    return { hasError: true, error }
   151→  }
   152→  componentDidCatch(error: Error, info: React.ErrorInfo) {
   153→    console.error("[AuditTrailErrorBoundary]", error.message, info.componentStack)
   154→  }
   155→  render() {
   156→    if (this.state.hasError) {
   157→      return (
   158→        <div className="min-h-[40vh] flex items-center justify-center p-6">
   159→          <Card className="max-w-md w-full border-red-200 dark:border-red-900/50">
   160→            <CardContent className="p-6 space-y-4">
   161→              <div className="flex justify-center">
   162→                <div className="h-14 w-14 rounded-full bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   163→                  <AlertTriangle className="h-7 w-7 text-red-600 dark:text-red-400" />
   164→                </div>
   165→              </div>
   166→              <div className="text-center space-y-2">
   167→                <h2 className="text-lg font-bold">Audit Trail Error</h2>
   168→                <p className="text-sm text-muted-foreground">
   169→                  Something went wrong loading the audit trail. This is likely a data or rendering issue.
   170→                </p>
   171→                <details className="text-left">
   172→                  <summary className="text-xs text-muted-foreground cursor-pointer hover:text-foreground">
   173→                    Error details
   174→                  </summary>
   175→                  <pre className="mt-2 text-xs bg-muted p-3 rounded-md overflow-auto max-h-32 text-red-600 dark:text-red-400">
   176→                    {this.state.error?.message || "Unknown error"}
   177→                  </pre>
   178→                </details>
   179→              </div>
   180→              <div className="flex gap-2 justify-center">
   181→                <Button onClick={() => this.setState({ hasError: false, error: null })}>
   182→                  <RotateCcw className="h-4 w-4 mr-2" /> Try Again
   183→                </Button>
   184→              </div>
   185→            </CardContent>
   186→          </Card>
   187→        </div>
   188→      )
   189→    }
   190→    return this.props.children
   191→  }
   192→}
