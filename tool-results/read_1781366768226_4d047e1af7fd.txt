   273→              <DropdownMenuTrigger asChild>
   274→                <button
   275→                  type="button"
   276→                  className="inline-flex items-center gap-1 px-2 py-1 rounded-full text-[10px] font-semibold bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 hover:bg-emerald-500/20 border border-emerald-500/20 dark:border-emerald-500/10 shadow-sm transition-colors"

   278→                  <Globe className="h-2.5 w-2.5" />

   282→              <DropdownMenuContent align="start" className="w-48">
   283→                {pWebsites.map((w, i) => {

   303→    </div>

   307→// ━━ SortableProjectCard — wraps KanbanProjectCard with useSortable ━━
   308→function SortableProjectCard({
   309→  project,
   310→  onCardClick,
   311→  isAdminUser,
   312→  onEdit,
   313→  onDelete,
   314→  onHover,
   315→  pendingCount,
   316→  onPendingClick,
   317→}: {
   318→  project: Record<string, unknown>;
   319→  onCardClick: () => void;
   320→  isAdminUser: boolean;
   321→  onEdit: (project: Record<string, unknown>, e: React.MouseEvent) => void;
   322→  onDelete: (projectId: string, e: React.MouseEvent) => void;
   323→  onHover?: () => void;
   324→  pendingCount?: number;
   325→  onPendingClick?: () => void;
   326→}) {
   327→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({
   328→    id: safeText(project.id, ""),
   329→  });
   330→  const style = {
   331→    transform: CSS.Transform.toString(transform),
   332→    transition,
   333→    opacity: isDragging ? 0.5 : 1,
   334→  };
   335→
   336→  return (
   337→    <div ref={setNodeRef} style={style} {...attributes} {...listeners}>
   338→      <KanbanProjectCard
   339→        project={project}
   340→        onClick={onCardClick}
   341→        isAdminUser={isAdminUser}
   342→        onEdit={onEdit}
   343→        onDelete={onDelete}
   344→        isDragging={false}
   345→        onHover={onHover}
   346→        pendingCount={pendingCount}
   347→        onPendingClick={onPendingClick}
   348→      />
   349→    </div>
   350→  );
   351→}

   353→// ━━ DroppableKanbanColumn — wraps column card list with useDroppable + SortableContext ━━
   354→function DroppableKanbanColumn({
   355→  col,
   356→  projects,
   357→  isAdminUser,
   358→  onCardClick,
   359→  onEdit,
   360→  onDelete,
   361→  isDimmed,
   362→  activeId,
   363→  onHover,
   364→  pendingTaskCounts,
   365→}: {
   366→  col: typeof KANBAN_COLUMNS[number];
   367→  projects: Record<string, unknown>[];
   368→  isAdminUser: boolean;
   369→  onCardClick: (project: Record<string, unknown>) => void;
   370→  onEdit: (project: Record<string, unknown>, e: React.MouseEvent) => void;
   371→  onDelete: (projectId: string, e: React.MouseEvent) => void;
   372→  isDimmed: boolean;
   373→  activeId: string | null;
   374→  onHover?: (pid: string) => void;
   375→  pendingTaskCounts?: Record<string, number>;
   376→}) {
   377→  const { setNodeRef, isOver } = useDroppable({ id: col.key });
   378→
   379→  return (
   380→    <div
   381→      className={cn(
   382→        "flex-shrink-0 w-[260px] sm:w-[280px] rounded-xl border transition-all duration-300 snap-start relative overflow-hidden",
   383→        "bg-gradient-to-b from-white/80 to-white/50 dark:from-gray-900/60 dark:to-gray-900/30 backdrop-blur-xl",
   384→        "border-gray-200/80 dark:border-gray-700/50",
   385→        "hover:border-gray-300 dark:hover:border-gray-600",
   386→        col.glowColor,
   387→        isDimmed && "opacity-40 pointer-events-none",
   388→        isOver && !isDimmed && `ring-2 ${col.accentRing} border-primary/40 bg-primary/[0.04] dark:bg-primary/[0.08] shadow-lg`
   389→      )}
   390→      style={{ minHeight: "calc(100vh - 300px)" }}
   391→    >
   392→      {/* Left accent bar */}
   393→      <div className={cn("absolute left-0 top-0 bottom-0 w-[3px] rounded-l-xl", col.accentBar, isOver && !isDimmed ? "opacity-100" : "opacity-60")} />
   394→
   395→      {/* Column Header */}
   396→      <div className="px-3 py-2.5 border-b border-gray-200/50 dark:border-gray-700/40 pl-4">
   397→        <div className="flex items-center gap-2">
   398→          <span className={cn("h-2 w-2 rounded-full shrink-0 ring-2 ring-offset-1 ring-offset-white dark:ring-offset-gray-950", col.dot, col.dot.replace("bg-", "ring-"))} />
   399→          <h3 className="font-bold text-[12px] tracking-tight">{col.label}</h3>
   400→          <Badge
   401→            variant="secondary"
   402→            className="ml-auto h-5 min-w-[20px] px-1.5 text-[10px] font-bold justify-center bg-muted/80"
   403→          >
   404→            {projects.length}
   405→          </Badge>
   406→        </div>
   407→      </div>
   408→
   409→      {/* Card List — Droppable zone */}
   410→      <div
   411→        ref={setNodeRef}
   412→        className={cn(
   413→          "p-2 space-y-2 overflow-y-auto transition-all duration-300 pl-4",
   414→          isOver && !isDimmed && "bg-primary/[0.03] dark:bg-primary/[0.05]"
   415→        )}
   416→        style={{ maxHeight: "calc(100vh - 380px)" }}
   417→      >
   418→        {projects.length === 0 && (
   419→          <div className="flex flex-col items-center justify-center py-8 text-center">
   420→            <div className="h-8 w-8 rounded-full bg-muted/50 flex items-center justify-center mb-2">
   421→              <FolderKanban className="h-4 w-4 text-muted-foreground/30" />
   422→            </div>
   423→            <p className="text-[11px] text-muted-foreground/50 font-medium">No projects</p>
   424→          </div>
   425→        )}
   426→        <SortableContext items={projects.map((p) => safeText(p.id, ""))} strategy={verticalListSortingStrategy}>
   427→          {projects.map((project) => {
   428→            const pId = safeText(project.id, "");
   429→            // Don't render the actively dragged card in the list
   430→            if (activeId === pId) return null;
   431→            return (
   432→              <SortableProjectCard
   433→                key={pId}
   434→                project={project}
   435→                onCardClick={() => onCardClick(project)}
   436→                isAdminUser={isAdminUser}
   437→                onEdit={onEdit}
   438→                onDelete={onDelete}
   439→                onHover={() => onHover && onHover(pId)}
   440→                pendingCount={pendingTaskCounts?.[pId]}
   441→                onPendingClick={() => onCardClick(project)}
   442→              />
   443→            );
   444→          })}
   445→        </SortableContext>
   446→      </div>
   447→    </div>
   448→  );
   449→}

   723→// ━━ Helper: Calculate and format period between two dates ━━
   724→function calcProjectPeriod(startDate: string, deadline: string): string | null {
   725→  if (!startDate || !deadline) return null;
   726→  const start = new Date(startDate);
   727→  const end = new Date(deadline);
   728→  if (isNaN(start.getTime()) || isNaN(end.getTime())) return null;
   729→  const diffMs = end.getTime() - start.getTime();
   730→  if (diffMs < 0) return "Start date is after deadline";
   731→  const days = Math.floor(diffMs / (1000 * 60 * 60 * 24));
   732→  if (days === 0) return "Same day";
   733→  const months = Math.floor(days / 30);
   734→  const remainDays = days % 30;
   735→  if (months > 0 && remainDays > 0) return `${months}m ${remainDays}d (${days} days)`;
   736→  if (months > 0) return `${months}m (${days} days)`;
   737→  return `${days} days`;
   738→}
