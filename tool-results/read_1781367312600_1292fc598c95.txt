   202→function formatRelativeTime(dateStr: string): string {
   203→  if (!dateStr) return "unknown";
   204→  const date = new Date(dateStr);
   205→  if (isNaN(date.getTime())) return "unknown";
   206→  const now = new Date();
   207→  const diffMs = now.getTime() - date.getTime();
   208→  const diffMin = Math.floor(diffMs / 60000);
   209→  if (diffMin < 1) return "just now";
   210→  if (diffMin < 60) return `${diffMin}m ago`;
   211→  const diffHr = Math.floor(diffMin / 60);
   212→  if (diffHr < 24) return `${diffHr}h ago`;
   213→  const diffDay = Math.floor(diffHr / 24);
   214→  if (diffDay < 7) return `${diffDay}d ago`;
   215→  return safeDateStr(date);
   216→}

   524→  const rawRole = session?.user?.role;
   525→  const userRole = rawRole && VALID_ROLES.includes(rawRole as typeof VALID_ROLES[number])
   526→    ? rawRole
   527→    : "DEVELOPER";
   528→  const userName = session?.user?.name || "User";
   529→  const userEmail = session?.user?.email || "";
   530→  const userId = session?.user?.id || "";
   531→
   532→  const unreadCount = useMemo(() => notifications.filter((n) => !n.isRead).length, [notifications]);
   533→
   534→  // Badge count mapping: use API response directly (role-aware for all users)
   535→
   536→  const fetchNotifications = useCallback(async () => {
   537→    if (!userId) return;
   538→    try {
   539→      const res = await fetch("/api/notifications", { credentials: "include" });
   540→      if (res.ok) {
   541→        const data = await res.json();
   542→        setNotifications(safeArray<NotificationItem>(data));
   543→      }
   544→    } catch (err) {
   545→      console.error("Failed to fetch notifications:", err);
   546→    }
   547→  }, [userId]);
   548→
   549→  const fetchPendingCounts = useCallback(async () => {
   550→    try {
   551→      const res = await fetch("/api/approvals/pending-counts", { credentials: "include" });
   552→      if (res.ok) {
   553→        const data = await res.json();
   554→        // API returns a flat map of nav-href → count
   555→        if (data && typeof data === "object" && !data.error) {
   556→          setNavBadgeData(data as NavBadgeMap);
   557→          // Backward compat: also populate old format for any other consumers
   558→          setPendingCounts({
   559→            approvals: (data["/dashboard/approvals"] || 0) as number,
   560→            leaveRequests: (data["/dashboard/team"] || 0) as number,
   561→            tasksAwaitingApproval: (data["/dashboard/projects"] || 0) as number,
   562→            total: Object.values(data).reduce((sum: number, v) => sum + (v as number), 0),
   563→          });
   564→        }
   565→      }
   566→    } catch (err) {
   567→      console.error("Failed to fetch pending counts:", err);
   568→    }
   569→  }, []);
   570→
   571→  useEffect(() => {
   572→    if (status === "unauthenticated") {
   573→      router.push("/login");
   574→    }
   575→    // SECURITY: CLIENT role users should be redirected to /portal, not /dashboard
   576→    if (status === "authenticated" && userRole === "CLIENT") {
   577→      router.push("/portal");
   578→    }
   579→  }, [status, router, userRole]);
   580→
   581→  // Auto-collapse sidebar when navigating to workspace landing page
   582→  useEffect(() => {
   583→    if (pathname === "/dashboard/workspace") {
   584→      setCollapsed(true);
   585→    }
   586→  }, [pathname]);
   587→
   588→  // PERF: Defer notification + counts fetch by 200ms so page data loads first.
   589→  // Notifications are non-critical UI — they should not compete with the
   590→  // page's own API calls for network bandwidth on navigation.
   591→  useEffect(() => {
   592→    if (session) {
   593→      const timer = setTimeout(() => {
   594→        fetchNotifications();
   595→        fetchPendingCounts();
   596→      }, 200);
   597→      const interval = setInterval(() => {
   598→        fetchNotifications();
   599→        fetchPendingCounts();
   600→      }, 45000);
   601→      return () => {
   602→        clearTimeout(timer);
   603→        clearInterval(interval);
   604→      };
   605→    }
   606→  }, [session, fetchNotifications, fetchPendingCounts]);
   607→
   608→  const markAsRead = useCallback(async (notifId: string) => {
   609→    try {
   610→      await fetch("/api/notifications", {
   611→        method: "PATCH",
   612→        headers: { "Content-Type": "application/json" },
   613→        credentials: "include",
   614→        body: JSON.stringify({ id: notifId, isRead: true }),
   615→      });
   616→      setNotifications((prev) =>
   617→        prev.map((n) => (n.id === notifId ? { ...n, isRead: true } : n))
   618→      );
   619→    } catch (err) {
   620→      console.error("Failed to mark notification as read:", err);
   621→    }
   622→  }, []);
   623→
   624→  const markAllAsRead = useCallback(async () => {
   625→    try {
   626→      // PERF FIX: Single batch request instead of N parallel requests
   627→      const res = await fetch("/api/notifications", {
   628→        method: "PATCH",
   629→        headers: { "Content-Type": "application/json" },
   630→        credentials: "include",
   631→        body: JSON.stringify({ markAllRead: true }),
   632→      });
   633→      if (res.ok) {
   634→        setNotifications((prev) => prev.map((n) => ({ ...n, isRead: true })));
   635→      }
   636→    } catch (err) {
   637→      console.error("Failed to mark all notifications as read:", err);
   638→    }
   639→  }, []);
   640→
   641→  const deleteNotification = useCallback(async (notifId: string) => {
   642→    try {
   643→      await fetch(`/api/notifications?id=${notifId}`, {
   644→        method: "DELETE",
   645→        credentials: "include",
   646→      });
   647→      setNotifications((prev) => prev.filter((n) => n.id !== notifId));
   648→    } catch (err) {
   649→      console.error("Failed to delete notification:", err);
   650→    }
   651→  }, []);
   652→
   653→  // I7: Extracted notification click handler from inline async in JSX
   654→  const handleNotificationClick = async (notif: NotificationItem) => {
   655→    try {
   656→      if (!notif.isRead) await markAsRead(notif.id);
   657→    } catch {}
   658→    if (notif.link && notif.link.startsWith("/")) {
   659→      router.push(notif.link);
   660→      setNotifOpen(false);
   661→    }
   662→  };
   663→
   664→  const handleNavigate = (href: string) => {
   665→    router.push(href);
   666→    setMobileOpen(false);
   667→  };
   668→
   669→  // Safety timeout for session loading — show fallback after 15s
   670→  const [sessionTimedOut, setSessionTimedOut] = useState(false);
   671→  useEffect(() => {
   672→    if (status !== "loading") return;
   673→    const t = setTimeout(() => setSessionTimedOut(true), 15000);
   674→    return () => clearTimeout(t);
   675→  }, [status]);
   676→
   677→  if (status === "loading" && !sessionTimedOut) {
   678→    return <LoadingScreen />;
   679→  }
   680→
   681→  if (sessionTimedOut && status === "loading") {
   682→    return (
   683→      <div className="fixed inset-0 z-50 flex min-h-screen items-center justify-center bg-background">
   684→        <div className="flex flex-col items-center gap-4">
   685→          <p className="text-sm text-muted-foreground">Session is taking too long to load...</p>
   686→          <button
   687→            onClick={() => signOut({ callbackUrl: "/login" })}
   688→            className="rounded-lg bg-primary px-4 py-2 text-sm text-primary-foreground hover:bg-primary/90"
   689→          >
   690→            Sign In Again
   691→          </button>
   692→        </div>
   693→      </div>
   694→    );
   695→  }
   696→
   697→  if (!session) return null;
   698→  // W3: Prevent CLIENT users from seeing any dashboard content before redirect
   699→  if (userRole === "CLIENT") return null;
   700→
   701→  return (
   702→    <div className="min-h-screen flex bg-background">
   703→      {/* Desktop Sidebar - wider and more spacious */}
   704→      <aside
   705→        className={cn(
   706→          "hidden md:flex flex-col border-r border-border bg-sidebar transition-all duration-300 relative z-40 liquid-glass-sidebar",
   707→          collapsed ? "w-[72px]" : "w-[280px]"
   708→        )}
   709→      >
   710→        <SidebarContent
   711→          collapsed={collapsed}
   712→          userRole={userRole}
   713→          userName={userName}
   714→          pathname={pathname}
   715→          onNavigate={handleNavigate}
   716→          badgeCounts={navBadgeData}
   717→        />
   718→        <Button
   719→          variant="ghost"
   720→          size="icon"
   721→          className="absolute top-24 -right-3 z-10 h-7 w-7 rounded-full border bg-background shadow-sm hidden md:flex"
   722→          onClick={() => setCollapsed(!collapsed)}
   723→          aria-label="Toggle sidebar"
   724→        >
   725→          <ChevronLeft className={cn("h-3.5 w-3.5 transition-transform", collapsed && "rotate-180")} />
   726→        </Button>
   727→      </aside>
   728→
   729→      {/* Mobile Sidebar */}
   730→      <Sheet open={mobileOpen} onOpenChange={setMobileOpen}>
   731→        <SheetContent side="left" className="w-[280px] p-0 bg-sidebar">
   732→          <div className="flex items-center justify-end p-2 pb-0">
   733→            <Button
   734→              variant="ghost"
   735→              size="icon"
   736→              className="h-8 w-8"
   737→              onClick={() => setMobileOpen(false)}
   738→              aria-label="Close menu"
   739→            >
   740→              <X className="h-4 w-4" />
   741→            </Button>
   742→          </div>
   743→          <SidebarContent
   744→            collapsed={false}
   745→            userRole={userRole}
   746→            userName={userName}
   747→            pathname={pathname}
   748→            onNavigate={handleNavigate}
   749→            badgeCounts={navBadgeData}
   750→          />
   751→        </SheetContent>
   752→      </Sheet>
   753→
   754→      {/* Main Content */}
   755→      <div className="flex-1 flex flex-col min-w-0">
   756→        {/* Header - taller and more prominent */}
   757→        <header className="h-14 sm:h-16 glass-topbar flex items-center justify-between px-3 sm:px-5 sticky top-0 z-30">
   758→          <div className="flex items-center gap-3">
   759→            <Button variant="ghost" size="icon" className="md:hidden" aria-label="Open menu" onClick={() => setMobileOpen(true)}>
   760→              <Menu className="h-5 w-5" />
   761→            </Button>
   762→            <h2 className="text-sm sm:text-base font-semibold text-foreground truncate">
   763→              {allNavItems.find((i) => pathname === i.href || (i.href !== "/dashboard" && pathname.startsWith(i.href + "/")))?.title || "Dashboard"}
   764→            </h2>
   765→          </div>
   766→
   767→          <div className="flex items-center gap-1 sm:gap-2">
   768→            {/* Theme Selector Dropdown */}
   769→            <DropdownMenu>
   770→              <DropdownMenuTrigger asChild>
   771→                <Button variant="ghost" size="icon" className="h-8 w-8 sm:h-9 sm:w-9" aria-label="Change theme">
   772→                  {theme === "system" ? (
   773→                    <Monitor className="h-4 w-4" />
   774→                  ) : theme === "dark" ? (
   775→                    <Moon className="h-4 w-4" />
   776→                  ) : theme === "bluelight" ? (
   777→                    <Eye className="h-4 w-4 text-amber-600" />
   778→                  ) : (
   779→                    <Sun className="h-4 w-4" />
   780→                  )}
   781→                </Button>
   782→              </DropdownMenuTrigger>
   783→              <DropdownMenuContent align="end" className="w-48">
   784→                <DropdownMenuItem onClick={() => setTheme("light")} className={cn("flex items-center gap-3", theme === "light" && "bg-accent")}>
   785→                  <Sun className="h-4 w-4" />
   786→                  <span>Light Mode</span>
   787→                  {theme === "light" && <Check className="h-3 w-3 ml-auto text-primary" />}
   788→                </DropdownMenuItem>
   789→                <DropdownMenuItem onClick={() => setTheme("dark")} className={cn("flex items-center gap-3", theme === "dark" && "bg-accent")}>
   790→                  <Moon className="h-4 w-4" />
   791→                  <span>Dark Mode</span>
   792→                  {theme === "dark" && <Check className="h-3 w-3 ml-auto text-primary" />}
   793→                </DropdownMenuItem>
   794→                <DropdownMenuItem onClick={() => setTheme("bluelight")} className={cn("flex items-center gap-3", theme === "bluelight" && "bg-accent")}>
   795→                  <Eye className="h-4 w-4 text-amber-600" />
   796→                  <div className="flex flex-col">
   797→                    <span>Blue Light</span>
   798→                    <span className="text-[10px] text-muted-foreground">Eye protection for long use</span>
   799→                  </div>
   800→                  {theme === "bluelight" && <Check className="h-3 w-3 ml-auto text-primary" />}
   801→                </DropdownMenuItem>
   802→                <DropdownMenuSeparator />
   803→                <DropdownMenuItem onClick={() => setTheme("system")} className={cn("flex items-center gap-3", theme === "system" && "bg-accent")}>
   804→                  <Monitor className="h-4 w-4" />
   805→                  <span>System Default</span>
   806→                  {theme === "system" && <Check className="h-3 w-3 ml-auto text-primary" />}
   807→                </DropdownMenuItem>
   808→              </DropdownMenuContent>
   809→            </DropdownMenu>
   810→
   811→            {/* Notifications Sheet — better scroll on all devices */}
   812→            <Sheet open={notifOpen} onOpenChange={setNotifOpen}>
   813→              <SheetTrigger asChild>
   814→                <Button variant="ghost" size="icon" className="relative h-8 w-8 sm:h-9 sm:w-9" aria-label="Notifications">
   815→                  <Bell className="h-4 w-4" />
   816→                  {unreadCount > 0 && (
   817→                    <Badge className="absolute -top-1 -right-1 h-5 w-5 p-0 flex items-center justify-center text-[10px]">
   818→                      {unreadCount > 9 ? "9+" : unreadCount}
   819→                    </Badge>
   820→                  )}
   821→                </Button>
   822→              </SheetTrigger>
   823→              <SheetContent className="w-full sm:max-w-md p-0 overflow-hidden flex flex-col">
   824→                <SheetHeader className="p-4 pb-3 border-b pr-10 shrink-0">
   825→                  <div className="flex items-center justify-between">
   826→                    <SheetTitle className="text-sm font-semibold">Notifications</SheetTitle>
   827→                    {unreadCount > 0 && (
   828→                      <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={markAllAsRead}>
   829→                        <Check className="h-3 w-3 mr-1" /> Mark all read
   830→                      </Button>
   831→                    )}
   832→                  </div>
   833→                </SheetHeader>
   834→                <ScrollArea className="flex-1">
   835→                  {notifications.length === 0 ? (
   836→                    <div className="p-8 text-center text-sm text-muted-foreground">
   837→                      <Bell className="h-8 w-8 mx-auto mb-2 opacity-20" />
   838→                      <p>No notifications yet</p>
   839→                    </div>
   840→                  ) : (
   841→                    <div>
   842→                      {notifications.slice(0, 50).map((notif) => {
   843→                        const NotifIcon = notifIcons[notif.type] || Info;
   844→                        return (
   845→                          <div
   846→                            key={notif.id}
   847→                            className={cn(
   848→                              "flex items-start gap-3 p-3 hover:bg-accent/50 cursor-pointer transition-colors border-b last:border-0",
   849→                              !notif.isRead && "bg-primary/5"
   850→                            )}
   851→                            onClick={() => handleNotificationClick(notif)}
   852→                          >
   853→                            <div className={cn(
   854→                              "mt-0.5 h-7 w-7 rounded-full flex items-center justify-center shrink-0",
   855→                              notif.type === "ERROR" || notif.type === "WARNING" ? "bg-red-100 text-red-600 dark:bg-red-900/30 dark:text-red-400" :
   856→                              notif.type === "SUCCESS" ? "bg-green-100 text-green-600 dark:bg-green-900/30 dark:text-green-400" :
   857→                              notif.type === "TASK" || notif.type === "APPROVAL" ? "bg-blue-100 text-blue-600 dark:bg-blue-900/30 dark:text-blue-400" :
   858→                              notif.type === "AGENT" ? "bg-purple-100 text-purple-600 dark:bg-purple-900/30 dark:text-purple-400" :
   859→                              "bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400"
   860→                            )}>
   861→                              <NotifIcon className="h-3.5 w-3.5" />
   862→                            </div>
   863→                            <div className="flex-1 min-w-0">
   864→                              <div className="flex items-center justify-between gap-2">
   865→                                <span className={cn("text-xs font-medium", !notif.isRead && "font-semibold")}>
   866→                                  {safeText(notif.title, "")}
   867→                                </span>
   868→                                <span className="text-[10px] text-muted-foreground whitespace-nowrap">
   869→                                  {formatRelativeTime(notif.createdAt)}
   870→                                </span>
   871→                              </div>
   872→                              <p className="text-xs text-muted-foreground line-clamp-2 mt-0.5">
   873→                                {safeText(notif.message, "")}
   874→                              </p>
   875→                            </div>
   876→                            <Button
   877→                              variant="ghost"
   878→                              size="icon"
   879→                              className="h-5 w-5 shrink-0 opacity-40 hover:opacity-100"
   880→                              onClick={(e) => {
   881→                                e.stopPropagation();
   882→                                deleteNotification(notif.id);
   883→                              }}
   884→                              aria-label="Dismiss notification"
   885→                            >
   886→                              <Trash2 className="h-3 w-3" />
   887→                            </Button>
   888→                          </div>
   889→                        );
   890→                      })}
   891→                      {/* W8: Show indicator when notifications are capped at 50 */}
   892→                      {notifications.length > 50 && (
   893→                        <span className="text-xs text-center text-muted-foreground block py-2">
   894→                          View all {notifications.length} notifications
   895→                        </span>
   896→                      )}
   897→                    </div>
   898→                  )}
   899→                </ScrollArea>
   900→              </SheetContent>
   901→            </Sheet>
   902→
   903→            <DropdownMenu>
   904→              <DropdownMenuTrigger asChild>
   905→                <Button variant="ghost" className="flex items-center gap-2 h-8 sm:h-9">
   906→                  <Avatar className="h-8 w-8">
   907→                    <AvatarFallback className="bg-primary text-primary-foreground text-xs font-bold">
   908→                      {userName.split(" ").filter(Boolean).map((n) => n[0]).join("").toUpperCase().slice(0, 2)}
   909→                    </AvatarFallback>
   910→                  </Avatar>
   911→                  <span className="hidden sm:block text-sm font-medium">{userName}</span>
   912→                </Button>
   913→              </DropdownMenuTrigger>
   914→              <DropdownMenuContent align="end" className="w-56">
   915→                <div className="p-2">
   916→                  <p className="text-sm font-medium">{userName}</p>
   917→                  <p className="text-xs text-muted-foreground">{userEmail}</p>
   918→                  <Badge variant="secondary" className="mt-1 text-xs">{userRole.replace("_", " ")}</Badge>
   919→                </div>
   920→                <DropdownMenuSeparator />
   921→                <DropdownMenuItem onClick={() => router.push("/dashboard/settings")}>
   922→                  <Settings className="mr-2 h-4 w-4" /> Settings
   923→                </DropdownMenuItem>
   924→                <DropdownMenuSeparator />
   925→                <DropdownMenuItem onClick={async () => { await signOut({ redirect: false }); router.push("/login"); }}>
   926→                  <LogOut className="mr-2 h-4 w-4" /> Sign Out
   927→                </DropdownMenuItem>
   928→              </DropdownMenuContent>
   929→            </DropdownMenu>
   930→          </div>
   931→        </header>
   932→
   933→        {/* Page Content - more padding */}
   934→        <main className="flex-1 p-3 sm:p-4 md:p-6 lg:p-8 overflow-auto">{children}</main>
   935→      </div>
   936→
   937→      {/* Agentation — visual feedback tool (superadmin + Test developer only) */}
   938→      {(userRole === "SUPER_ADMIN" || userName === "Test") && <Agentation />}
   939→    </div>
   940→  );
   941→}
