   391→/**
   392→ * Validates data against a schema and returns either the validated data or an error response
   393→ * @param schema - The Zod schema to validate against
   394→ * @param data - The unknown data to validate
   395→ * @returns An object with either success=true and validated data, or success=false and an error message
   396→ */
   397→export function validateRequest<T>(schema: z.ZodSchema<T>, data: unknown): { success: true; data: T } | { success: false; error: string } {
   398→  const result = schema.safeParse(data)
   399→  if (!result.success) {
   400→    const firstError = result.error.issues?.[0]
   401→    return { success: false, error: firstError?.message || "Invalid input" }
   402→  }
   403→  return { success: true, data: result.data }
   404→}
