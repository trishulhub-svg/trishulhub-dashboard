     5→// 2. Automatically creates missing tables and adds missing columns
     6→// 3. Covers ALL 50 models — no manual SQL maintenance needed

     8→// WHEN TO RUN: Automatically via src/instrumentation.ts on every server cold start.
     9→// This is a safety net — the primary sync should be done via `prisma db push`.

    11→// IMPORTANT: We use "try ALTER TABLE, catch duplicate column" instead of
    12→// PRAGMA table_info because Turso/libSQL returns BigInt values in PRAGMA
    13→// results which Prisma cannot serialize, causing a TypeError that silently
    14→// skips all column migrations.

   448→/**
   449→ * Helper to safely check if a column exists in a table.
   450→ * Uses a SELECT query instead of PRAGMA table_info to avoid BigInt serialization
   451→ * errors that occur with Turso/libSQL adapter.
   452→ */
   453→async function columnExists(table: string, column: string): Promise<boolean> {
   454→  try {
   455→    // Use a safe approach: try to SELECT the column with LIMIT 0.
   456→    // This validates the column exists without fetching any data.
   457→    await db.$executeRawUnsafe(`SELECT "${column}" FROM "${table}" LIMIT 0`)
   458→    return true
   459→  } catch {
   460→    return false
   461→  }
   462→}
