    99→function formatRelativeTime(dateStr: string | null | undefined): string {
   100→  if (!dateStr) return "N/A"
   101→  const date = new Date(dateStr)
   102→  if (isNaN(date.getTime())) return "N/A"
   103→  const now = new Date()
   104→  const diffMs = now.getTime() - date.getTime()
   105→  const diffSec = Math.floor(diffMs / 1000)
   106→  if (diffSec < 60) return "just now"
   107→  const diffMin = Math.floor(diffSec / 60)
   108→  if (diffMin < 60) return `${diffMin}m ago`
   109→  const diffHr = Math.floor(diffMin / 60)
   110→  if (diffHr < 24) return `${diffHr}h ago`
   111→  const diffDay = Math.floor(diffHr / 24)
   112→  if (diffDay < 7) return `${diffDay}d ago`
   113→  return formatDateTime(dateStr)
   114→}

   116→/** Safe helper: extract initials from a potentially null user name */
   117→function getInitials(userName: string | null | undefined): string {
   118→  if (!userName) return "?"
   119→  return userName.split(" ").filter(Boolean).map(n => n[0] || "").join("").toUpperCase().slice(0, 2) || "?"
   120→}
   121→
   122→/** Safe helper: format role for display */
   123→function formatRole(role: string | null | undefined): string {
   124→  if (!role) return "—"
   125→  return role.replace(/_/g, " ")
   126→}

   128→/**
   129→ * Local error boundary for the Audit Trail page.
   130→ * Catches render-time errors so they show a useful message
   131→ * instead of the generic DashboardError boundary.
   132→ */
   133→class AuditTrailErrorBoundary extends React.Component<
   134→  { children: React.ReactNode },
   135→  { hasError: boolean; error: Error | null }
   136→> {
   137→  constructor(props: { children: React.ReactNode }) {
   138→    super(props)
   139→    this.state = { hasError: false, error: null }
   140→  }
   141→  static getDerivedStateFromError(error: Error) {
   142→    return { hasError: true, error }
   143→  }
   144→  componentDidCatch(error: Error, info: React.ErrorInfo) {
   145→    console.error("[AuditTrailErrorBoundary]", error.message, info.componentStack)
   146→  }
   147→  render() {
   148→    if (this.state.hasError) {
   149→      return (
   150→        <div className="min-h-[40vh] flex items-center justify-center p-6">
   151→          <Card className="max-w-md w-full border-red-200 dark:border-red-900/50">
   152→            <CardContent className="p-6 space-y-4">
   153→              <div className="flex justify-center">
   154→                <div className="h-14 w-14 rounded-full bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   155→                  <AlertTriangle className="h-7 w-7 text-red-600 dark:text-red-400" />
   156→                </div>
   157→              </div>
   158→              <div className="text-center space-y-2">
   159→                <h2 className="text-lg font-bold">Audit Trail Error</h2>
   160→                <p className="text-sm text-muted-foreground">
   161→                  Something went wrong loading the audit trail. This is likely a data or rendering issue.
   162→                </p>
   163→                <details className="text-left">
   164→                  <summary className="text-xs text-muted-foreground cursor-pointer hover:text-foreground">
   165→                    Error details
   166→                  </summary>
   167→                  <pre className="mt-2 text-xs bg-muted p-3 rounded-md overflow-auto max-h-32 text-red-600 dark:text-red-400">
   168→                    {this.state.error?.message || "Unknown error"}
   169→                  </pre>
   170→                </details>
   171→              </div>
   172→              <div className="flex gap-2 justify-center">
   173→                <Button onClick={() => this.setState({ hasError: false, error: null })}>
   174→                  <RotateCcw className="h-4 w-4 mr-2" /> Try Again
   175→                </Button>
   176→              </div>
   177→            </CardContent>
   178→          </Card>
   179→        </div>
   180→      )
   181→    }
   182→    return this.props.children
   183→  }
   184→}
