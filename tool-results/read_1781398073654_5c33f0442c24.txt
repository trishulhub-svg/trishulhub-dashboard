   101→function formatRelativeTime(dateStr: string | null | undefined): string {
   102→  if (!dateStr) return "N/A"
   103→  const date = new Date(dateStr)
   104→  if (isNaN(date.getTime())) return "N/A"
   105→  const now = new Date()
   106→  const diffMs = now.getTime() - date.getTime()
   107→  const diffSec = Math.floor(diffMs / 1000)
   108→  if (diffSec < 60) return "just now"
   109→  const diffMin = Math.floor(diffSec / 60)
   110→  if (diffMin < 60) return `${diffMin}m ago`
   111→  const diffHr = Math.floor(diffMin / 60)
   112→  if (diffHr < 24) return `${diffHr}h ago`
   113→  const diffDay = Math.floor(diffHr / 24)
   114→  if (diffDay < 7) return `${diffDay}d ago`
   115→  return formatDateTime(dateStr)
   116→}

   118→/** Safe helper: extract initials from a potentially null user name */
   119→function getInitials(userName: string | null | undefined): string {
   120→  if (!userName) return "?"
   121→  return userName.split(" ").filter(Boolean).map(n => n[0] || "").join("").toUpperCase().slice(0, 2) || "?"
   122→}
   123→
   124→/** Safe helper: format role for display */
   125→function formatRole(role: string | null | undefined): string {
   126→  if (!role) return "—"
   127→  return role.replace(/_/g, " ")
   128→}

   130→/**
   131→ * Local error boundary for the Audit Trail page.
   132→ * Catches render-time errors so they show a useful message
   133→ * instead of the generic DashboardError boundary.
   134→ */
   135→class AuditTrailErrorBoundary extends React.Component<
   136→  { children: React.ReactNode },
   137→  { hasError: boolean; error: Error | null }
   138→> {
   139→  constructor(props: { children: React.ReactNode }) {
   140→    super(props)
   141→    this.state = { hasError: false, error: null }
   142→  }
   143→  static getDerivedStateFromError(error: Error) {
   144→    return { hasError: true, error }
   145→  }
   146→  componentDidCatch(error: Error, info: React.ErrorInfo) {
   147→    console.error("[AuditTrailErrorBoundary]", error.message, info.componentStack)
   148→  }
   149→  render() {
   150→    if (this.state.hasError) {
   151→      return (
   152→        <div className="min-h-[40vh] flex items-center justify-center p-6">
   153→          <Card className="max-w-md w-full border-red-200 dark:border-red-900/50">
   154→            <CardContent className="p-6 space-y-4">
   155→              <div className="flex justify-center">
   156→                <div className="h-14 w-14 rounded-full bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   157→                  <AlertTriangle className="h-7 w-7 text-red-600 dark:text-red-400" />
   158→                </div>
   159→              </div>
   160→              <div className="text-center space-y-2">
   161→                <h2 className="text-lg font-bold">Audit Trail Error</h2>
   162→                <p className="text-sm text-muted-foreground">
   163→                  Something went wrong loading the audit trail. This is likely a data or rendering issue.
   164→                </p>
   165→                <details className="text-left">
   166→                  <summary className="text-xs text-muted-foreground cursor-pointer hover:text-foreground">
   167→                    Error details
   168→                  </summary>
   169→                  <pre className="mt-2 text-xs bg-muted p-3 rounded-md overflow-auto max-h-32 text-red-600 dark:text-red-400">
   170→                    {this.state.error?.message || "Unknown error"}
   171→                  </pre>
   172→                </details>
   173→              </div>
   174→              <div className="flex gap-2 justify-center">
   175→                <Button onClick={() => this.setState({ hasError: false, error: null })}>
   176→                  <RotateCcw className="h-4 w-4 mr-2" /> Try Again
   177→                </Button>
   178→              </div>
   179→            </CardContent>
   180→          </Card>
   181→        </div>
   182→      )
   183→    }
   184→    return this.props.children
   185→  }
   186→}
