   101→   101→function formatRelativeTime(dateStr: string | null | undefined): string {
   102→   102→  if (!dateStr) return "N/A"
   103→   103→  const date = new Date(dateStr)
   104→   104→  if (isNaN(date.getTime())) return "N/A"
   105→   105→  const now = new Date()
   106→   106→  const diffMs = now.getTime() - date.getTime()
   107→   107→  const diffSec = Math.floor(diffMs / 1000)
   108→   108→  if (diffSec < 60) return "just now"
   109→   109→  const diffMin = Math.floor(diffSec / 60)
   110→   110→  if (diffMin < 60) return `${diffMin}m ago`
   111→   111→  const diffHr = Math.floor(diffMin / 60)
   112→   112→  if (diffHr < 24) return `${diffHr}h ago`
   113→   113→  const diffDay = Math.floor(diffHr / 24)
   114→   114→  if (diffDay < 7) return `${diffDay}d ago`
   115→   115→  return formatDateTime(dateStr)
   116→   116→}

   118→   118→/** Safe helper: extract initials from a potentially null user name */
   119→   119→function getInitials(userName: string | null | undefined): string {
   120→   120→  if (!userName) return "?"
   121→   121→  return userName.split(" ").filter(Boolean).map(n => n[0] || "").join("").toUpperCase().slice(0, 2) || "?"
   122→   122→}
