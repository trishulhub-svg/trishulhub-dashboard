    50→  { key: "ALL", label: "All" },
    51→  { key: "TODO", label: "To Do" },
    52→  { key: "IN_PROGRESS", label: "In Progress" },
    53→  { key: "REVIEW", label: "Review" },

    57→
    58→export default function TodosPage() {
    59→  const params = useParams();
    60→  const router = useRouter();
    61→  const pathname = usePathname();
    62→  const { data: session, status: sessionStatus } = useSession();
    63→  const queryClient = useQueryClient();
    64→
    65→  const rawProjectId = params?.projectId;
    66→  const projectId = typeof rawProjectId === "string"
    67→    ? rawProjectId
    68→    : Array.isArray(rawProjectId)
    69→      ? String(rawProjectId[0] ?? "")
    70→      : "";
    71→
    72→  const userRole = session?.user?.role || "DEVELOPER";
    73→  const userId = session?.user?.id || "";
    74→  const isAdminUser = userRole === "SUPER_ADMIN" || userRole === "ADMIN";
    75→
    76→  const [filter, setFilter] = useState<FilterTab>("ALL");
    77→  const [search, setSearch] = useState("");
    78→  const [selectedTask, setSelectedTask] = useState<Record<string, unknown> | null>(null);
    79→  const [taskDetailOpen, setTaskDetailOpen] = useState(false);
    80→  const [togglingId, setTogglingId] = useState<string | null>(null);
    81→
    82→  const handle401 = (res: Response) => {
    83→    if (res.status === 401) { window.location.href = "/login"; return true; }
    84→    return false;
    85→  };
    86→
    87→  // ── React Query ──
    88→  const { data: projectData, isLoading: projectLoading } = useQuery({
    89→    queryKey: ["project", projectId],
    90→    queryFn: async () => {
    91→      if (!projectId) return null;
    92→      const res = await fetch(`/api/projects?projectId=${projectId}`, { credentials: "include" });
    93→      if (res.status === 401) { window.location.href = "/login"; throw new Error("Unauthorized"); }
    94→      if (!res.ok) throw new Error("Failed to load project");
    95→      const raw = deepSanitize(await res.json());
    96→      if (Array.isArray(raw) && raw.length > 0) return raw[0] as Record<string, unknown>;
    97→      if (raw && typeof raw === "object" && (raw as Record<string, unknown>).id) return raw as Record<string, unknown>;
    98→      if (Array.isArray((raw as Record<string, unknown>)?.data) && ((raw as Record<string, unknown>).data as unknown[]).length > 0) {
    99→        return ((raw as Record<string, unknown>).data as unknown[])[0] as Record<string, unknown>;
   100→      }
   101→      return null;
   102→    },
   103→    enabled: !!projectId,
   104→    staleTime: 30 * 1000,
   105→    retry: 1,
   106→  });
   107→
   108→  const { data: tasksData = [], isLoading: tasksLoading } = useQuery({
   109→    queryKey: ["project-tasks", projectId],
   110→    queryFn: async () => {
   111→      if (!projectId) return [];
   112→      const res = await fetch(`/api/tasks?projectId=${projectId}`, { credentials: "include" });
   113→      if (res.status === 401) { window.location.href = "/login"; throw new Error("Unauthorized"); }
   114→      if (!res.ok) throw new Error("Failed to load tasks");
   115→      const td = deepSanitize(await res.json());
   116→      return Array.isArray((td as Record<string, unknown>)?.tasks) ? (td as Record<string, unknown>).tasks as unknown[] : Array.isArray(td) ? td : (Array.isArray((td as Record<string, unknown>)?.data) ? (td as Record<string, unknown>).data as unknown[] : []);
   117→    },
   118→    enabled: !!projectId,
   119→    staleTime: 30 * 1000,
   120→    retry: 1,
   121→  });
   122→
   123→  const project = projectData;
   124→  const projectName = project ? extractStr(project, "name", "Untitled") : "";
   125→
   126→  // Audit fix: show ALL task statuses so developers can see REVIEW and AWAITING_APPROVAL
   127→  const visibleTasks = tasksData;
   128→
   129→  // Apply search and filter
   130→  const filteredTasks = useMemo(() => {
   131→    return visibleTasks.filter((t) => {
   132→      const title = extractStr(t, "title", "").toLowerCase();
   133→      const desc = extractStr(t, "description", "").toLowerCase();
   134→      const status = extractStr(t, "status", "");
   135→      const matchesSearch = !search || title.includes(search.toLowerCase()) || desc.includes(search.toLowerCase());
   136→      const matchesFilter = filter === "ALL" || status === filter;
   137→      return matchesSearch && matchesFilter;
   138→    });
   139→  }, [visibleTasks, search, filter]);
   140→
   141→  // Group filtered tasks by section
   142→  const todoTasks = filteredTasks.filter((t) => extractStr(t, "status", "") === "TODO");
   143→  const inProgressTasks = filteredTasks.filter((t) => extractStr(t, "status", "") === "IN_PROGRESS");
   144→  const reviewTasks = filteredTasks.filter((t) => extractStr(t, "status", "") === "REVIEW");
   145→  const awaitingTasks = filteredTasks.filter((t) => extractStr(t, "status", "") === "AWAITING_APPROVAL");
   146→  const doneTasks = filteredTasks.filter((t) => extractStr(t, "status", "") === "DONE");
   147→  const remainingCount = visibleTasks.filter((t) => extractStr(t, "status", "") !== "DONE").length;
   148→
   149→  // Toggle task done/todo
   150→  const handleToggleDone = async (taskId: string, currentStatus: string) => {
   151→    setTogglingId(taskId);
   152→    try {
   153→      const newStatus = currentStatus === "DONE" ? "TODO" : "DONE";
   154→      const res = await fetch("/api/tasks", {
   155→        method: "PATCH",
   156→        headers: { "Content-Type": "application/json" },
   157→        credentials: "include",
   158→        body: JSON.stringify({ id: taskId, status: newStatus }),
   159→      });
   160→      if (res.ok) {
   161→        const updated = await res.json().catch(() => null);
   162→        const finalStatus = updated?.status || newStatus;
   163→        if (finalStatus === "AWAITING_APPROVAL" && newStatus === "DONE") {
   164→          toast.success("Task submitted for approval");
   165→        } else if (finalStatus === "DONE") {
   166→          toast.success("Task completed");
   167→        } else {
   168→          toast.success("Task moved back to To Do");
   169→        }
   170→        queryClient.invalidateQueries({ queryKey: ["project-tasks", projectId] });
   171→        // Audit fix: also invalidate global todos cache for consistency
   172→        queryClient.invalidateQueries({ queryKey: ["my-tasks-all"] });
   173→      } else {
   174→        if (handle401(res)) return;
   175→        const err = await res.json().catch(() => null);
   176→        toast.error(err?.error || "Failed to update task");
   177→      }
   178→    } catch {
   179→      toast.error("Failed to update task");
   180→    } finally {
   181→      setTogglingId(null);
   182→    }
   183→  };
   184→
   185→  // Check if deadline is overdue
   186→  const isOverdue = (deadline: string) => {
   187→    if (!deadline) return false;
   188→    return new Date(deadline) < new Date();
   189→  };
   190→
   191→  const isLoading = sessionStatus === "loading" || projectLoading || tasksLoading;
   192→
   193→  if (isLoading) {
   194→    return (
   195→      <div className="space-y-5">
   196→        <div className="flex items-center gap-3">
   197→          <Skeleton className="h-9 w-9 rounded-lg" />
   198→          <div className="space-y-1.5">
   199→            <Skeleton className="h-7 w-48" />
   200→            <Skeleton className="h-4 w-32" />
   201→          </div>
   202→        </div>
   203→        <div className="flex items-center gap-3">
   204→          <Skeleton className="h-9 w-64" />
   205→          <div className="flex gap-1.5">
   206→            {[1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-8 w-20 rounded-lg" />)}
   207→          </div>
   208→        </div>
   209→        <div className="space-y-2">
   210→          {[1, 2, 3, 4, 5].map((i) => (
   211→            <Skeleton key={i} className="h-20 w-full rounded-xl" />
   212→          ))}
   213→        </div>
   214→      </div>
   215→    );
   216→  }
   217→
   218→  if (!projectId || !project) {
   219→    return (
   220→      <div className="text-center py-16">
   221→        <div className="h-16 w-16 mx-auto rounded-2xl bg-muted/50 flex items-center justify-center mb-4">
   222→          <ListTodo className="h-8 w-8 text-muted-foreground/40" />
   223→        </div>
   224→        <p className="text-muted-foreground mb-4 font-medium">Project not found</p>
   225→        <Button variant="outline" onClick={() => router.push("/dashboard/projects")} className="gap-2">
   226→          <ArrowLeft className="h-4 w-4" /> Back to Projects
   227→        </Button>
   228→      </div>
   229→    );
   230→  }
   231→
   232→  return (
   233→    <div className="space-y-5">
   234→      {/* Header */}
   235→      <div className="flex items-start gap-3">
   236→        <Button variant="ghost" size="icon" onClick={() => router.push(`/dashboard/projects/${projectId}`)} aria-label="Back to project" className="mt-0.5 hover:bg-muted/80 rounded-lg hover:scale-105 transition-all duration-200">
   237→          <ArrowLeft className="h-4 w-4" />
   238→        </Button>
   239→        <div className="flex-1 min-w-0">
   240→          <div className="flex items-center gap-2.5 flex-wrap">
   241→            <h1 className="text-2xl font-extrabold tracking-tight">{safeText(projectName, "Untitled")}</h1>
   242→            <Badge variant="secondary" className="font-medium bg-primary/10 text-primary border-primary/20">
   243→              <ClipboardCheck className="h-3 w-3 mr-1" /> My Tasks
   244→            </Badge>
   245→          </div>
   246→          <p className="text-sm text-muted-foreground mt-1">
   247→            {String(remainingCount)} task{remainingCount !== 1 ? "s" : ""} remaining
   248→          </p>
   249→        </div>
   250→      </div>
   251→
   252→      {/* ── View Tabs ── */}
   253→      <div className="flex items-center gap-1 bg-muted/60 rounded-lg p-1 w-fit">
   254→        <Button
   255→          variant={pathname.endsWith("/todos") ? "ghost" : "default"}
   256→          size="sm"
   257→          className={cn(
   258→            "gap-1.5 text-xs transition-all",
   259→            pathname.endsWith("/todos") && "text-muted-foreground hover:text-foreground",
   260→            !pathname.endsWith("/todos") && "shadow-sm"
   261→          )}
   262→          onClick={() => router.push(`/dashboard/projects/${projectId}`)}
   263→        >
   264→          <LayoutDashboard className="h-3.5 w-3.5" />
   265→          Overview
   266→          {isAdminUser && <Badge variant="secondary" className="ml-1 h-4 px-1 text-[10px]">Admin</Badge>}
   267→        </Button>
   268→        <Button
   269→          variant={pathname.endsWith("/todos") ? "default" : "ghost"}
   270→          size="sm"
   271→          className={cn(
   272→            "gap-1.5 text-xs transition-all",
   273→            !pathname.endsWith("/todos") && "text-muted-foreground hover:text-foreground",
   274→            pathname.endsWith("/todos") && "shadow-sm"
   275→          )}
   276→          onClick={() => router.push(`/dashboard/projects/${projectId}/todos`)}
   277→        >
   278→          <ClipboardCheck className="h-3.5 w-3.5" />
   279→          My Tasks
   280→          {!isAdminUser && <Badge variant="secondary" className="ml-1 h-4 px-1 text-[10px]">You</Badge>}
   281→        </Button>
   282→      </div>
   283→
   284→      {/* Search & Filters */}
   285→      <div className="flex items-center gap-3 flex-wrap">
   286→        <div className="relative flex-1 max-w-sm">
   287→          <Search className="absolute left-3 top-2.5 h-4 w-4 text-muted-foreground/60" />
   288→          <Input
   289→            placeholder="Search tasks..."
   290→            value={search}
   291→            onChange={(e) => setSearch(e.target.value)}
   292→            className="pl-9 bg-white/60 dark:bg-white/[0.04] backdrop-blur-md border-gray-200/80 dark:border-gray-700/50 focus:bg-white dark:focus:bg-white/[0.06] transition-all"
   293→            aria-label="Search tasks"
   294→          />
   295→        </div>
   296→        <div className="flex gap-1.5 flex-wrap">
   297→          {FILTER_TABS.map((tab) => {
   298→            const isActive = filter === tab.key;
   299→            const count =
   300→              tab.key === "ALL" ? visibleTasks.length :
   301→              tab.key === "TODO" ? visibleTasks.filter((t) => extractStr(t, "status", "") === "TODO").length :
   302→              tab.key === "IN_PROGRESS" ? visibleTasks.filter((t) => extractStr(t, "status", "") === "IN_PROGRESS").length :
   303→              tab.key === "REVIEW" ? visibleTasks.filter((t) => extractStr(t, "status", "") === "REVIEW").length :
   304→              tab.key === "AWAITING_APPROVAL" ? visibleTasks.filter((t) => extractStr(t, "status", "") === "AWAITING_APPROVAL").length :
   305→              visibleTasks.filter((t) => extractStr(t, "status", "") === "DONE").length;
   306→            return (
   307→              <Button
   308→                key={tab.key}
   309→                variant={isActive ? "default" : "outline"}
   310→                size="sm"
   311→                className={cn(
   312→                  "h-8 text-xs transition-all duration-200 gap-1.5",
   313→                  isActive ? "shadow-md shadow-primary/20" : "hover:bg-muted/80 border-gray-200/80 dark:border-gray-700/50",
   314→                )}
   315→                onClick={() => setFilter(tab.key)}
   316→              >
   317→                {tab.label}
   318→                <Badge variant="secondary" className={cn("h-4 px-1 text-[10px] font-bold", isActive ? "bg-white/20 text-white" : "bg-muted/80")}>
   319→                  {String(count)}
   320→                </Badge>
   321→              </Button>
   322→            );
   323→          })}
   324→        </div>
   325→      </div>
   326→
   327→      {/* Empty state */}
   328→      {filteredTasks.length === 0 && (
   329→        <div className="text-center py-16">
   330→          <div className="h-16 w-16 mx-auto rounded-2xl bg-muted/50 flex items-center justify-center mb-4">
   331→            <CheckCircle2 className="h-8 w-8 text-muted-foreground/30" />
   332→          </div>
   333→          <p className="text-muted-foreground mb-1 font-medium">
   334→            {search ? "No tasks match your search" : filter !== "ALL" ? `No ${filter.replace("_", " ").toLowerCase()} tasks` : "No tasks yet"}
   335→          </p>
   336→          <p className="text-sm text-muted-foreground/60">
   337→            {search ? "Try a different search term" : "Tasks assigned to this project will appear here"}
   338→          </p>
   339→        </div>
   340→      )}
   341→
   342→      {/* ── To Do Section ── */}
   343→      {filter === "ALL" && todoTasks.length > 0 && (
   344→        <section>
   345→          <div className="flex items-center gap-2 mb-3 px-1">
   346→            <Circle className="h-4 w-4 text-gray-400" />
   347→            <h2 className="text-sm font-bold text-muted-foreground">To Do</h2>
   348→            <Badge variant="secondary" className="text-[10px] font-bold">{String(todoTasks.length)}</Badge>
   349→          </div>
   350→          <div className="space-y-2">
   351→            {todoTasks.map((task) => (
   352→              <TaskCard
   353→                key={extractStr(task, "id", "")}
   354→                task={task}
   355→                isDone={false}
   356→                isToggling={togglingId === extractStr(task, "id", "")}
   357→                onToggle={() => handleToggleDone(extractStr(task, "id", ""), extractStr(task, "status", ""))}
   358→                onClick={() => { setSelectedTask(task as Record<string, unknown>); setTaskDetailOpen(true); }}
   359→              />
   360→            ))}
   361→          </div>
   362→        </section>
   363→      )}
   364→
   365→      {/* ── In Progress Section ── */}
   366→      {filter === "ALL" && inProgressTasks.length > 0 && (
   367→        <section>
   368→          <div className="flex items-center gap-2 mb-3 px-1">
   369→            <div className="h-3.5 w-3.5 rounded-sm bg-blue-400" />
   370→            <h2 className="text-sm font-bold text-muted-foreground">In Progress</h2>
   371→            <Badge variant="secondary" className="text-[10px] font-bold">{String(inProgressTasks.length)}</Badge>
   372→          </div>
   373→          <div className="space-y-2">
   374→            {inProgressTasks.map((task) => (
   375→              <TaskCard
   376→                key={extractStr(task, "id", "")}
   377→                task={task}
   378→                isDone={false}
   379→                isToggling={togglingId === extractStr(task, "id", "")}
   380→                onToggle={() => handleToggleDone(extractStr(task, "id", ""), extractStr(task, "status", ""))}
   381→                onClick={() => { setSelectedTask(task as Record<string, unknown>); setTaskDetailOpen(true); }}
   382→              />
   383→            ))}
   384→          </div>
   385→        </section>
   386→      )}
   387→
   388→      {/* ── Review Section (audit fix: was hidden) ── */}
   389→      {filter === "ALL" && reviewTasks.length > 0 && (
   390→        <section>
   391→          <div className="flex items-center gap-2 mb-3 px-1">
   392→            <div className="h-3.5 w-3.5 rounded-sm bg-yellow-400" />
   393→            <h2 className="text-sm font-bold text-muted-foreground">In Review</h2>
   394→            <Badge variant="secondary" className="text-[10px] font-bold">{String(reviewTasks.length)}</Badge>
   395→          </div>
   396→          <div className="space-y-2">
   397→            {reviewTasks.map((task) => (
   398→              <TaskCard
   399→                key={extractStr(task, "id", "")}
   400→                task={task}
   401→                isDone={false}
   402→                isToggling={togglingId === extractStr(task, "id", "")}
   403→                onToggle={() => handleToggleDone(extractStr(task, "id", ""), extractStr(task, "status", ""))}
   404→                onClick={() => { setSelectedTask(task as Record<string, unknown>); setTaskDetailOpen(true); }}
   405→              />
   406→            ))}
   407→          </div>
   408→        </section>
   409→      )}
   410→
   411→      {/* ── Awaiting Approval Section (audit fix: was hidden) ── */}
   412→      {filter === "ALL" && awaitingTasks.length > 0 && (
   413→        <section>
   414→          <div className="flex items-center gap-2 mb-3 px-1">
   415→            <Clock className="h-4 w-4 text-orange-400" />
   416→            <h2 className="text-sm font-bold text-muted-foreground">Pending Approval</h2>
   417→            <Badge variant="secondary" className="text-[10px] font-bold">{String(awaitingTasks.length)}</Badge>
   418→          </div>
   419→          <div className="space-y-2">
   420→            {awaitingTasks.map((task) => (
   421→              <TaskCard
   422→                key={extractStr(task, "id", "")}
   423→                task={task}
   424→                isDone={false}
   425→                isToggling={togglingId === extractStr(task, "id", "")}
   426→                onToggle={() => handleToggleDone(extractStr(task, "id", ""), extractStr(task, "status", ""))}
   427→                onClick={() => { setSelectedTask(task as Record<string, unknown>); setTaskDetailOpen(true); }}
   428→              />
   429→            ))}
   430→          </div>
   431→        </section>
   432→      )}
   433→
   434→      {/* ── Completed Section ── */}
   435→      {filter === "ALL" && doneTasks.length > 0 && (
   436→        <section>
   437→          <div className="flex items-center gap-2 mb-3 px-1">
   438→            <CheckCircle2 className="h-4 w-4 text-green-500" />
   439→            <h2 className="text-sm font-bold text-muted-foreground">Completed</h2>
   440→            <Badge variant="secondary" className="text-[10px] font-bold">{String(doneTasks.length)}</Badge>
   441→          </div>
   442→          <div className="space-y-2">
   443→            {doneTasks.map((task) => (
   444→              <TaskCard
   445→                key={extractStr(task, "id", "")}
   446→                task={task}
   447→                isDone={true}
   448→                isToggling={togglingId === extractStr(task, "id", "")}
   449→                onToggle={() => handleToggleDone(extractStr(task, "id", ""), extractStr(task, "status", ""))}
   450→                onClick={() => { setSelectedTask(task as Record<string, unknown>); setTaskDetailOpen(true); }}
   451→              />
   452→            ))}
   453→          </div>
   454→        </section>
   455→      )}
   456→
   457→      {/* ── Single filter view (flat list) — only when NOT ALL, since ALL uses grouped sections above ── */}
   458→      {filter !== "ALL" && filteredTasks.length > 0 && (
   459→        <div className="space-y-2">
   460→          {(filter === "TODO" ? todoTasks : filter === "IN_PROGRESS" ? inProgressTasks : filter === "REVIEW" ? reviewTasks : filter === "AWAITING_APPROVAL" ? awaitingTasks : doneTasks).map((task) => (
   461→            <TaskCard
   462→              key={extractStr(task, "id", "")}
   463→              task={task}
   464→              isDone={extractStr(task, "status", "") === "DONE"}
   465→              isToggling={togglingId === extractStr(task, "id", "")}
   466→              onToggle={() => handleToggleDone(extractStr(task, "id", ""), extractStr(task, "status", ""))}
   467→              onClick={() => { setSelectedTask(task as Record<string, unknown>); setTaskDetailOpen(true); }}
   468→            />
   469→          ))}
   470→        </div>
   471→      )}
   472→
   473→      {/* ── Task Detail Dialog ── */}
   474→      <Dialog open={taskDetailOpen} onOpenChange={setTaskDetailOpen}>
   475→        <DialogContent className="max-w-lg">
   476→          {selectedTask && (() => {
   477→            const dtId = extractStr(selectedTask, "id", "");
   478→            const dtTitle = extractStr(selectedTask, "title", "Untitled");
   479→            const dtDesc = extractStr(selectedTask, "description", "");
   480→            const dtPriority = extractStr(selectedTask, "priority", "MEDIUM");
   481→            const dtStatus = extractStr(selectedTask, "status", "TODO");
   482→            const dtAssigneeType = extractStr(selectedTask, "assigneeType", "HUMAN");
   483→            const dtAssignedTo = extractStr(selectedTask, "assignedTo", "");
   484→            const dtDeadline = extractStr(selectedTask, "deadline", "");
   485→            const dtCreatedAt = extractStr(selectedTask, "createdAt", "");
   486→            const dtUpdatedAt = extractStr(selectedTask, "updatedAt", "");
   487→            const dtCompletedAt = extractStr(selectedTask, "completedAt", "");
   488→            const dtApprovedBy = extractStr(selectedTask, "approvedBy", "");
   489→            const dtApprovedAt = extractStr(selectedTask, "approvedAt", "");
   490→            const isDone = dtStatus === "DONE";
   491→            const overdue = dtDeadline && isOverdue(dtDeadline) && !isDone;
   492→            return (
   493→              <>
   494→                <DialogHeader>
   495→                  <div className="flex items-center justify-between pr-6">
   496→                    <DialogTitle className={cn("text-lg font-bold", isDone && "line-through text-muted-foreground")}>{safeText(dtTitle, "Untitled")}</DialogTitle>
   497→                    <Badge className={`shrink-0 font-semibold ${priorityColors[dtPriority] || ""}`}>
   498→                      {safeText(dtPriority, "MEDIUM")}
   499→                    </Badge>
   500→                  </div>
   501→                  <div className="flex items-center gap-2 mt-1">
   502→                    <Badge className={`${taskStatusColors[dtStatus] || "bg-gray-100 text-gray-800"} text-xs font-medium`}>{safeText(dtStatus, "TODO").replace("_", " ")}</Badge>
   503→                    <DialogDescription className="text-xs text-muted-foreground">
   504→                      Created {dtCreatedAt ? safeDate(dtCreatedAt, "N/A") : "N/A"}
   505→                    </DialogDescription>
   506→                  </div>
   507→                </DialogHeader>
   508→                <div className="space-y-4 mt-2">
   509→                  {/* Assignee & Deadline Meta */}
   510→                  <div className="flex flex-wrap items-center gap-3 p-2.5 rounded-lg bg-muted/30 border">
   511→                    {dtAssigneeType === "AI" ? (
   512→                      <span className="flex items-center gap-1.5 text-xs text-muted-foreground"><Bot className="h-3.5 w-3.5" /> System</span>
   513→                    ) : (
   514→                      <span className="flex items-center gap-1.5 text-xs text-muted-foreground"><User className="h-3.5 w-3.5" /> {extractStr(selectedTask, "assignedToName", "") || safeText(dtAssignedTo) || "Unassigned"}</span>
   515→                    )}
   516→                    {dtDeadline && (
   517→                      <span className={cn("flex items-center gap-1.5 text-xs", overdue ? "text-red-500 font-medium" : "text-muted-foreground")}>
   518→                        <CalendarDays className="h-3.5 w-3.5" /> {safeDate(dtDeadline, "")}
   519→                        {overdue && " (Overdue)"}
   520→                      </span>
   521→                    )}
   522→                  </div>
   523→
   524→                  {/* Description */}
   525→                  <div>
   526→                    <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wider mb-2">Description</p>
   527→                    {dtDesc ? (
   528→                      <div className="text-sm leading-relaxed whitespace-pre-wrap bg-muted/40 rounded-xl p-3.5 border">
   529→                        {safeText(dtDesc)}
   530→                      </div>
   531→                    ) : (
   532→                      <p className="text-sm text-muted-foreground italic">No description provided.</p>
   533→                    )}
   534→                  </div>
   535→
   536→                  {/* Timestamps */}
   537→                  <div>
   538→                    <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wider mb-2">Timeline</p>
   539→                    <div className="grid grid-cols-2 gap-2">
   540→                      <div className="rounded-lg bg-muted/30 p-2.5 border">
   541→                        <p className="text-[10px] text-muted-foreground uppercase tracking-wider font-medium">Created</p>
   542→                        <p className="text-xs font-medium mt-0.5">{dtCreatedAt ? safeDate(dtCreatedAt, "N/A") : "N/A"}</p>
   543→                      </div>
   544→                      <div className="rounded-lg bg-muted/30 p-2.5 border">
   545→                        <p className="text-[10px] text-muted-foreground uppercase tracking-wider font-medium">Last Updated</p>
   546→                        <p className="text-xs font-medium mt-0.5">{dtUpdatedAt ? safeDate(dtUpdatedAt, "N/A") : "N/A"}</p>
   547→                      </div>
   548→                      {dtCompletedAt && (
   549→                        <div className="rounded-lg bg-green-50/50 dark:bg-green-900/10 p-2.5 border border-green-200/50 dark:border-green-900/20">
   550→                          <p className="text-[10px] text-green-600 dark:text-green-400 uppercase tracking-wider font-medium">Completed</p>
   551→                          <p className="text-xs font-medium mt-0.5">{safeDate(dtCompletedAt, "N/A")}</p>
   552→                        </div>
   553→                      )}
   554→                      {dtApprovedAt && (
   555→                        <div className="rounded-lg bg-green-50/50 dark:bg-green-900/10 p-2.5 border border-green-200/50 dark:border-green-900/20">
   556→                          <p className="text-[10px] text-green-600 dark:text-green-400 uppercase tracking-wider font-medium">Approved</p>
   557→                          <p className="text-xs font-medium mt-0.5">{safeDate(dtApprovedAt, "N/A")}</p>
   558→                        </div>
   559→                      )}
   560→                    </div>
   561→                  </div>
   562→                </div>
   563→              </>
   564→            );
   565→          })()}
   566→        </DialogContent>
   567→      </Dialog>
   568→    </div>
   569→  );

   573→function TaskCard({
   574→  task,
   575→  isDone,
   576→  isToggling,
   577→  onToggle,
   578→  onClick,
   579→}: {
   580→  task: unknown;

   582→  isToggling: boolean;
