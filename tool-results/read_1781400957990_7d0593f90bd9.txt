   236→/** Single personal task item (checkbox + title + meta) — upgraded with category & hover scale */
   237→function PersonalTaskItem({ task, togglingId, onToggleDone, index = 0 }: {
   238→  task: unknown; togglingId: string | null; onToggleDone: (id: string) => void; index?: number;
   239→}) {
   240→  const taskId = extractStr(task, "id", "");
   241→  const title = extractStr(task, "title", "Untitled");
   242→  const status = extractStr(task, "status", "TODO");
   243→  const priority = extractStr(task, "priority", "MEDIUM");
   244→  const deadline = extractStr(task, "deadline", "");
   245→  const category = extractStr(task, "category", "GENERAL");
   246→  const isAwaiting = status === "AWAITING_APPROVAL";
   247→  const isDone = status === "DONE";
   248→  const isOverdue = deadline && new Date(deadline) < new Date() && !isAwaiting && !isDone;
   249→  const isToggling = togglingId === taskId;
   250→  const priorityDot = priorityDotColors[priority] || "bg-gray-400";
   251→  const priorityBorder = priorityBorderColors[priority] || "border-l-gray-300";
   252→
   253→  return (
   254→    <div
   255→      className={cn(
   256→        "flex items-center gap-3 p-3 rounded-lg border border-l-[3px] transition-all duration-200",
   257→        "bg-white/70 dark:bg-white/[0.03] border-gray-200/60 dark:border-gray-700/40",
   258→        "hover:bg-white dark:hover:bg-white/[0.06] hover:shadow-sm hover:scale-[1.005]",
   259→        priorityBorder,
   260→        isDone && "opacity-50", isAwaiting && "opacity-60",
   261→        "animate-slide-up",
   262→      )}
   263→      style={{ animationDelay: `${index * 50}ms` }}
   264→    >
   265→      <button
   266→        type="button"
   267→        onClick={(e) => { e.stopPropagation(); onToggleDone(taskId); }}
   268→        disabled={isToggling || isAwaiting}
   269→        className={cn(
   270→          "shrink-0 h-[18px] w-[18px] rounded border-2 flex items-center justify-center",
   271→          "transition-all duration-200 hover:scale-110",
   272→          isToggling && "animate-pulse",
   273→          isDone && "border-green-500 bg-green-500",
   274→          isAwaiting && "cursor-default",
   275→          !isDone && !isAwaiting && "hover:border-primary/60 hover:bg-primary/5",
   276→          !isDone && "border-gray-300 dark:border-gray-600",
   277→        )}
   278→      >
   279→        {isDone && <Check className="h-3 w-3 text-white" />}
   280→        {isAwaiting && <CheckCircle2 className="h-3 w-3 text-orange-400" />}
   281→      </button>
   282→      <div className="flex-1 min-w-0">
   283→        <p className={cn("text-sm leading-snug", (isDone || isAwaiting) && "line-through text-muted-foreground/70", !isDone && !isAwaiting && "font-medium")}>
   284→          {safeText(title)}
   285→        </p>
   286→      </div>
   287→      <div className="flex items-center gap-2.5 shrink-0">
   288→        <CategoryBadge category={category} />
   289→        <span className={cn("h-2 w-2 rounded-full shrink-0", priorityDot)} title={priority} />
   290→        {deadline && (
   291→          <span className={cn("text-[11px] flex items-center gap-1", isOverdue ? "text-red-500 font-medium" : "text-muted-foreground")}>
   292→            {isOverdue ? <AlertTriangle className="h-3 w-3" /> : <Clock className="h-3 w-3" />}
   293→            {safeDate(deadline)}
   294→          </span>
   295→        )}
   296→        {isAwaiting && (
   297→          <Badge className="text-[10px] bg-orange-100 text-orange-700 dark:bg-orange-900/30 dark:text-orange-300 font-medium">
   298→            Reviewing
   299→          </Badge>
   300→        )}
   301→      </div>
   302→    </div>
   303→  );
   304→}
