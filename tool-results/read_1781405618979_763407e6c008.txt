   275→                <button
   276→                  type="button"
   277→                  className="inline-flex items-center gap-1 px-2 py-1 rounded-full text-[10px] font-semibold bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 hover:bg-emerald-500/20 border border-emerald-500/20 dark:border-emerald-500/10 shadow-sm transition-colors"
   278→                >
   279→                  <Globe className="h-2.5 w-2.5" />

   283→              <DropdownMenuContent align="start" className="w-48">
   284→                {pWebsites.map((w, i) => {
   285→                  const wUrl = safeText(w.url, "");

   308→// ━━ SortableProjectCard — wraps KanbanProjectCard with useSortable ━━
   309→function SortableProjectCard({
   310→  project,
   311→  onCardClick,
   312→  isAdminUser,
   313→  onEdit,
   314→  onDelete,
   315→  onHover,
   316→  pendingCount,
   317→  onPendingClick,
   318→}: {
   319→  project: Record<string, unknown>;
   320→  onCardClick: () => void;
   321→  isAdminUser: boolean;
   322→  onEdit: (project: Record<string, unknown>, e: React.MouseEvent) => void;
   323→  onDelete: (projectId: string, e: React.MouseEvent) => void;
   324→  onHover?: () => void;
   325→  pendingCount?: number;
   326→  onPendingClick?: () => void;
   327→}) {
   328→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({
   329→    id: safeText(project.id, ""),
   330→  });
   331→  const style = {
   332→    transform: CSS.Transform.toString(transform),
   333→    transition,
   334→    opacity: isDragging ? 0.5 : 1,
   335→  };
   336→
   337→  return (
   338→    <div ref={setNodeRef} style={style} {...attributes} {...listeners}>
   339→      <KanbanProjectCard
   340→        project={project}
   341→        onClick={onCardClick}
   342→        isAdminUser={isAdminUser}
   343→        onEdit={onEdit}
   344→        onDelete={onDelete}
   345→        isDragging={false}
   346→        onHover={onHover}
   347→        pendingCount={pendingCount}
   348→        onPendingClick={onPendingClick}
   349→      />
   350→    </div>
   351→  );
   352→}

   354→// ━━ DroppableKanbanColumn — wraps column card list with useDroppable + SortableContext ━━
   355→function DroppableKanbanColumn({
   356→  col,
   357→  projects,
   358→  isAdminUser,
   359→  onCardClick,
   360→  onEdit,
   361→  onDelete,
   362→  isDimmed,
   363→  activeId,
   364→  onHover,
   365→  pendingTaskCounts,
   366→}: {
   367→  col: typeof KANBAN_COLUMNS[number];
   368→  projects: Record<string, unknown>[];
   369→  isAdminUser: boolean;
   370→  onCardClick: (project: Record<string, unknown>) => void;
   371→  onEdit: (project: Record<string, unknown>, e: React.MouseEvent) => void;
   372→  onDelete: (projectId: string, e: React.MouseEvent) => void;
   373→  isDimmed: boolean;
   374→  activeId: string | null;
   375→  onHover?: (pid: string) => void;
   376→  pendingTaskCounts?: Record<string, number>;
   377→}) {
   378→  const { setNodeRef, isOver } = useDroppable({ id: col.key });
   379→
   380→  return (
   381→    <div
   382→      className={cn(
   383→        "flex-shrink-0 w-[260px] sm:w-[280px] rounded-xl border transition-all duration-300 snap-start relative overflow-hidden",
   384→        "bg-gradient-to-b from-white/80 to-white/50 dark:from-gray-900/60 dark:to-gray-900/30 backdrop-blur-xl",
   385→        "border-gray-200/80 dark:border-gray-700/50",
   386→        "hover:border-gray-300 dark:hover:border-gray-600",
   387→        col.glowColor,
   388→        isDimmed && "opacity-40 pointer-events-none",
   389→        isOver && !isDimmed && `ring-2 ${col.accentRing} border-primary/40 bg-primary/[0.04] dark:bg-primary/[0.08] shadow-lg`
   390→      )}
   391→      style={{ minHeight: "calc(100vh - 300px)" }}
   392→    >
   393→      {/* Left accent bar */}
   394→      <div className={cn("absolute left-0 top-0 bottom-0 w-[3px] rounded-l-xl", col.accentBar, isOver && !isDimmed ? "opacity-100" : "opacity-60")} />
   395→
   396→      {/* Column Header */}
   397→      <div className="px-3 py-2.5 border-b border-gray-200/50 dark:border-gray-700/40 pl-4">
   398→        <div className="flex items-center gap-2">
   399→          <span className={cn("h-2 w-2 rounded-full shrink-0 ring-2 ring-offset-1 ring-offset-white dark:ring-offset-gray-950", col.dot, col.dot.replace("bg-", "ring-"))} />
   400→          <h3 className="font-bold text-[12px] tracking-tight">{col.label}</h3>
   401→          <Badge
   402→            variant="secondary"
   403→            className="ml-auto h-5 min-w-[20px] px-1.5 text-[10px] font-bold justify-center bg-muted/80"
   404→          >
   405→            {projects.length}
   406→          </Badge>
   407→        </div>
   408→      </div>
   409→
   410→      {/* Card List — Droppable zone */}
   411→      <div
   412→        ref={setNodeRef}
   413→        className={cn(
   414→          "p-2 space-y-2 overflow-y-auto transition-all duration-300 pl-4",
   415→          isOver && !isDimmed && "bg-primary/[0.03] dark:bg-primary/[0.05]"
   416→        )}
   417→        style={{ maxHeight: "calc(100vh - 380px)" }}
   418→      >
   419→        {projects.length === 0 && (
   420→          <div className="flex flex-col items-center justify-center py-8 text-center">
   421→            <div className="h-8 w-8 rounded-full bg-muted/50 flex items-center justify-center mb-2">
   422→              <FolderKanban className="h-4 w-4 text-muted-foreground/30" />
   423→            </div>
   424→            <p className="text-[11px] text-muted-foreground/50 font-medium">No projects</p>
   425→          </div>
   426→        )}
   427→        <SortableContext items={projects.map((p) => safeText(p.id, ""))} strategy={verticalListSortingStrategy}>
   428→          {projects.map((project) => {
   429→            const pId = safeText(project.id, "");
   430→            // Don't render the actively dragged card in the list
   431→            if (activeId === pId) return null;
   432→            return (
   433→              <SortableProjectCard
   434→                key={pId}
   435→                project={project}
   436→                onCardClick={() => onCardClick(project)}
   437→                isAdminUser={isAdminUser}
   438→                onEdit={onEdit}
   439→                onDelete={onDelete}
   440→                onHover={() => onHover && onHover(pId)}
   441→                pendingCount={pendingTaskCounts?.[pId]}
   442→                onPendingClick={() => onCardClick(project)}
   443→              />
   444→            );
   445→          })}
   446→        </SortableContext>
   447→      </div>
   448→    </div>
   449→  );
   450→}

   724→// ━━ Helper: Calculate and format period between two dates ━━
   725→function calcProjectPeriod(startDate: string, deadline: string): string | null {
   726→  if (!startDate || !deadline) return null;
   727→  const start = new Date(startDate);
   728→  const end = new Date(deadline);
   729→  if (isNaN(start.getTime()) || isNaN(end.getTime())) return null;
   730→  const diffMs = end.getTime() - start.getTime();
   731→  if (diffMs < 0) return "Start date is after deadline";
   732→  const days = Math.floor(diffMs / (1000 * 60 * 60 * 24));
   733→  if (days === 0) return "Same day";
   734→  const months = Math.floor(days / 30);
   735→  const remainDays = days % 30;
   736→  if (months > 0 && remainDays > 0) return `${months}m ${remainDays}d (${days} days)`;
   737→  if (months > 0) return `${months}m (${days} days)`;
   738→  return `${days} days`;
   739→}
