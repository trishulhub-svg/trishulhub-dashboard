    21→// ━━ CIRCULAR SYNC GUARD ━━
    22→
    23→/** In-memory map: "taskId:direction" → timestamp. Prevents TH→Lark→TH→Lark loops.
    24→ *  Cooldown: 10 seconds.
    25→ */
    26→const _syncGuard = new Map<string, number>()
    27→const SYNC_GUARD_COOLDOWN_MS = 10_000
    28→
    29→function markSyncGuard(taskId: string, direction: string): void {
    30→  _syncGuard.set(`${taskId}:${direction}`, Date.now())
    31→}
    32→
    33→function isSyncGuarded(taskId: string, direction: string): boolean {
    34→  const ts = _syncGuard.get(`${taskId}:${direction}`)
    35→  if (!ts) return false
    36→  if (Date.now() - ts > SYNC_GUARD_COOLDOWN_MS) {
    37→    _syncGuard.delete(`${taskId}:${direction}`)
    38→    return false
    39→  }
    40→  return true
    41→}
    42→
    43→// ━━ WEBHOOK EVENT DEDUPLICATION ━━
    44→
    45→/** In-memory set of processed event_ids. Prevents duplicate processing from Lark retries.
    46→ *  Window: 30 seconds.
    47→ */
    48→const _processedEvents = new Map<string, number>()
    49→const DEDUP_WINDOW_MS = 30_000
    50→
    51→function isEventDuplicate(eventId: string): boolean {
    52→  const ts = _processedEvents.get(eventId)
    53→  if (!ts) return false
    54→  if (Date.now() - ts > DEDUP_WINDOW_MS) {
    55→    _processedEvents.delete(eventId)
    56→    return false
    57→  }
    58→  return true
    59→}
    60→
    61→function markEventProcessed(eventId: string): void {
    62→  _processedEvents.set(eventId, Date.now())
    63→}
    64→
    65→// Periodic cleanup of stale entries (every 60s)
    66→if (typeof setInterval !== "undefined") {
    67→  setInterval(() => {
    68→    const now = Date.now()
    69→    for (const [key, ts] of _syncGuard) {
    70→      if (now - ts > SYNC_GUARD_COOLDOWN_MS) _syncGuard.delete(key)
    71→    }
    72→    for (const [key, ts] of _processedEvents) {
    73→      if (now - ts > DEDUP_WINDOW_MS) _processedEvents.delete(key)
    74→    }
    75→  }, 60_000)
    76→}

    79→ * Log a sync operation to the LarkSyncLog table.
    80→ */
    81→async function logSync(params: {
    82→  direction: SyncDirection
    83→  action: SyncAction
    84→  status: SyncStatus
    85→  taskId?: string
    86→  larkTaskId?: string
    87→  larkTaskListId?: string
    88→  projectId?: string
    89→  userId?: string
    90→  error?: string
    91→  metadata?: Record<string, unknown>
    92→}): Promise<void> {
    93→  try {
    94→    await db.$executeRawUnsafe(
    95→      `INSERT INTO "LarkSyncLog" ("id", "direction", "action", "status", "taskId", "larkTaskId", "larkTaskListId", "projectId", "userId", "error", "metadata", "createdAt") VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)`,
    96→      `sync_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
    97→      params.direction,
    98→      params.action,
    99→      params.status,
   100→      params.taskId || null,
   101→      params.larkTaskId || null,
   102→      params.larkTaskListId || null,
   103→      params.projectId || null,
   104→      params.userId || null,
   105→      params.error || null,
   106→      params.metadata ? JSON.stringify(params.metadata) : null
   107→    )
   108→  } catch (err) {
   109→    console.error("[lark/sync] Failed to log sync:", err)
   110→  }
   111→}

   113→/**
   114→ * Get the Lark open_id for a TrishulHub user.
   115→ * Checks LarkUserMapping table first, then tries email lookup.
   116→ */
   117→export async function getUserLarkOpenId(userId: string): Promise<string | null> {
   118→  try {
   119→    // Check mapping table
   120→    const rows = await db.$queryRawUnsafe<Array<{ larkOpenId: string }>>(
   121→      'SELECT "larkOpenId" FROM "LarkUserMapping" WHERE "userId" = ?',
   122→      userId
   123→    )
   124→    if (rows.length > 0 && rows[0].larkOpenId) {
   125→      return rows[0].larkOpenId
   126→    }
   127→
   128→    // Try email lookup
   129→    const user = await db.user.findUnique({ where: { id: userId }, select: { email: true } })
   130→    if (user?.email) {
   131→      const larkUser = await lookupUserByEmail(user.email)
   132→      if (larkUser?.open_id) {
   133→        // Auto-save the mapping
   134→        await db.$executeRawUnsafe(
   135→          'INSERT OR IGNORE INTO "LarkUserMapping" ("id", "userId", "larkOpenId", "larkName", "larkEmail", "matchedBy", "createdAt") VALUES (?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)',
   136→          `map_${userId}_${Date.now()}`,
   137→          userId,
   138→          larkUser.open_id,
   139→          larkUser.name || "",
   140→          larkUser.email || "",
   141→          "email_auto"
   142→        )
   143→        return larkUser.open_id
   144→      }
   145→    }
   146→
   147→    return null
   148→  } catch (err) {
   149→    console.error("[lark/sync] Failed to get user Lark ID:", err)
   150→    return null
   151→  }
   152→}

   229→/**
   230→ * Sync a task creation from TrishulHub to Lark.
   231→ * Call this AFTER the task is saved in the DB.
   232→ */
   233→export async function syncTaskToLark(
   234→  taskId: string,
   235→  data: {
   236→    title: string
   237→    description?: string
   238→    status: string
   239→    priority: string
   240→    assignedTo?: string
   241→    projectId?: string
   242→    deadline?: string | Date
   243→  },
   244→  userId: string
   245→): Promise<void> {
   246→  const config = await getLarkConfig()
   247→  if (!config?.enabled) return
   248→
   249→  try {
   250→    // Get or create the task list for the project
   251→    let tasklistId: string | undefined
   252→
   253→    if (data.projectId) {
   254→      const project = await db.project.findUnique({ where: { id: data.projectId }, select: { name: true } })
   255→      if (project) {
   256→        // Try to get or create the project task list (retry once on failure)
   257→        let result = await getOrCreateProjectTaskList(data.projectId, project.name)
   258→        if (!result) {
   259→          // Retry once — transient API errors can cause first attempt to fail
   260→          console.warn(`[lark/sync] Retrying task list creation for project ${data.projectId} (${project.name})`)
   261→          result = await getOrCreateProjectTaskList(data.projectId, project.name)
   262→        }
   263→        if (result) {
   264→          tasklistId = result.tasklistId
   265→          if (result.created) {
   266→            console.log(`[lark/sync] Created new Lark task list for project: ${project.name}`)
   267→          }
   268→        } else {
   269→          console.error(`[lark/sync] FAILED to get/create task list for project ${data.projectId} (${project.name}) — aborting sync, task will NOT fall to default list`)
   270→          await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Failed to create project task list for "${project.name}"` })
   271→          // Don't fall back to default — if the task has a project, it MUST go to that project's list
   272→          return
   273→        }
   274→      }
   275→    }
   276→
   277→    if (!tasklistId) {
   278→      // No project — use a default "TrishulHub Tasks" list
   279→      const defaultResult = await getOrCreateProjectTaskList("__default__", "TrishulHub Tasks")
   280→      if (!defaultResult) {
   281→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Failed to get/create default task list" })
   282→        return
   283→      }
   284→      tasklistId = defaultResult.tasklistId
   285→    }
   286→
   287→    // Resolve assignee's Lark open_id
   288→    let assigneeOpenId: string | undefined
   289→    if (data.assignedTo) {
   290→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   291→    }
   292→
   293→    // Convert deadline to unix ms
   294→    let dueTimestamp: number | undefined
   295→    if (data.deadline) {
   296→      dueTimestamp = new Date(data.deadline).getTime()
   297→    }
   298→
   299→    // Create task — NO extra, NO status, NO priority (not supported by Lark create API)
   300→    const larkTask = await createTask(tasklistId, {
   301→      title: data.title,
   302→      description: data.description,
   303→      assigneeOpenId,
   304→      dueTimestamp,
   305→    })
   306→
   307→    if (larkTask) {
   308→      await saveLarkTaskMapping(taskId, larkTask.task_id, tasklistId)
   309→
   310→      // Now set status & priority via separate update
   311→      await updateTask(larkTask.task_id, {
   312→        status: data.status,
   313→        priority: data.priority,
   314→      })
   315→
   316→      // Add assignee as tasklist member (editor role) so the tasklist
   317→      // appears in the user's Lark Task Center
   318→      if (assigneeOpenId) {
   319→        await addTaskListMember(tasklistId, [
   320→          { id: assigneeOpenId, role: "editor" },
   321→        ])
   322→      }
   323→
   324→      // Mark circular guard to prevent webhook from echoing back
   325→      markSyncGuard(taskId, "TO_LARK")
   326→
   327→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "SUCCESS", taskId, larkTaskId: larkTask.task_id, larkTaskListId: tasklistId, projectId: data.projectId, userId })
   328→    } else {
   329→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Lark API returned null task" })
   330→    }
   331→  } catch (err) {
   332→    const msg = err instanceof Error ? err.message : String(err)
   333→    await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: msg })
   334→  }
   335→}

   337→/**
   338→ * Sync a task update from TrishulHub to Lark.
   339→ */
   340→export async function syncTaskUpdateToLark(
   341→  taskId: string,
   342→  data: {
   343→    title?: string
   344→    description?: string
   345→    status?: string
   346→    priority?: string
   347→    assignedTo?: string
   348→    deadline?: string | Date | null
   349→  },
   350→  userId: string
   351→): Promise<void> {
   352→  const config = await getLarkConfig()
   353→  if (!config?.enabled) return
   354→
   355→  // Circular sync guard: skip if we just synced FROM Lark for this task
   356→  if (isSyncGuarded(taskId, "FROM_LARK")) {
   357→    console.log(`[lark/sync] Skipping TO_LARK update for ${taskId} — circular guard active`)
   358→    return
   359→  }
   360→
   361→  try {
   362→    const mapping = await getLarkTaskMapping(taskId)
   363→    if (!mapping) {
   364→      // No mapping exists — try full sync (task may have been created before Lark was enabled)
   365→      const task = await db.task.findUnique({ where: { id: taskId } })
   366→      if (task) {
   367→        await syncTaskToLark(taskId, {
   368→          title: task.title,
   369→          description: task.description || undefined,
   370→          status: task.status,
   371→          priority: task.priority,
   372→          assignedTo: task.assignedTo || undefined,
   373→          projectId: task.projectId || undefined,
   374→          deadline: task.deadline || undefined,
   375→        }, userId)
   376→      }
   377→      return
   378→    }
   379→
   380→    // Resolve assignee's Lark open_id
   381→    let assigneeOpenId: string | undefined
   382→    if (data.assignedTo) {
   383→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   384→    }
   385→
   386→    // Convert deadline to unix ms
   387→    let dueTimestamp: number | undefined
   388→    if (data.deadline) {
   389→      dueTimestamp = new Date(data.deadline).getTime()
   390→    }
   391→
   392→    // Update task — NO extra, NO assignee in body (use member APIs instead)
   393→    const larkTask = await updateTask(mapping.larkTaskId, {
   394→      title: data.title,
   395→      description: data.description,
   396→      status: data.status,
   397→      priority: data.priority,
   398→      dueTimestamp,
   399→    })
   400→
   401→    // Handle assignee change via member API
   402→    if (data.assignedTo !== undefined && assigneeOpenId) {
   403→      await addTaskMember(mapping.larkTaskId, [
   404→        { id: assigneeOpenId, role: "assignee" },
   405→      ])
   406→    }
   407→
   408→    if (larkTask) {
   409→      // Mark circular guard
   410→      markSyncGuard(taskId, "TO_LARK")
   411→
   412→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   413→    } else {
   414→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark API returned null" })
   415→    }
   416→  } catch (err) {
   417→    const msg = err instanceof Error ? err.message : String(err)
   418→    await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, userId, error: msg })
   419→  }
   420→}

   422→/**
   423→ * Sync a task deletion from TrishulHub to Lark.
   424→ */
   425→export async function syncTaskDeleteToLark(taskId: string, userId: string): Promise<void> {
   426→  const config = await getLarkConfig()
   427→  if (!config?.enabled) return
   428→
   429→  try {
   430→    const mapping = await getLarkTaskMapping(taskId)
   431→    if (!mapping) {
   432→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "SKIPPED", taskId, userId, error: "No Lark mapping found" })
   433→      return
   434→    }
   435→
   436→    const success = await larkDeleteTask(mapping.larkTaskListId, mapping.larkTaskId)
   437→    await removeLarkTaskMapping(taskId)
   438→
   439→    if (success) {
   440→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   441→    } else {
   442→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark delete API failed" })
   443→    }
   444→  } catch (err) {
   445→    const msg = err instanceof Error ? err.message : String(err)
   446→    await logSync({ direction: "TO_LARK", action: "DELETE", status: "FAILED", taskId, userId, error: msg })
   447→  }
   448→}

   450→// ━━ LARK → TRISHULHUB SYNC (Webhook-driven) ━━
   451→
   452→/**
   453→ * Handle incoming Lark webhook events.
   454→ * @param eventId — Lark event_id for deduplication
   455→ * @param eventType — Lark event type
   456→ * @param eventData — Parsed event payload
   457→ */
   458→export async function handleLarkWebhookEvent(
   459→  eventId: string,
   460→  eventType: string,
   461→  eventData: {
   462→    task_id: string
   463→    tasklist_id: string
   464→    operator?: { open_id: string }
   465→    [key: string]: unknown
   466→  }
   467→): Promise<{ success: boolean; message: string }> {
   468→  const config = await getLarkConfig()
   469→  if (!config?.enabled) {
   470→    return { success: false, message: "Lark sync is disabled" }
   471→  }
   472→
   473→  // ── Dedup check ──
   474→  if (isEventDuplicate(eventId)) {
   475→    console.log(`[lark/sync] Dedup: skipping duplicate event ${eventId}`)
   476→    return { success: true, message: "Duplicate event, skipped" }
   477→  }
   478→  markEventProcessed(eventId)
   479→
   480→  const { task_id, tasklist_id } = eventData
   481→  console.log(`[lark/sync] Processing event: ${eventType} (id: ${eventId}) for task: ${task_id}`)
   482→  console.log(`[lark/sync] Event details: tasklist_id=${tasklist_id}, operator=${eventData.operator?.open_id || 'none'}`)
   483→
   484→  try {
   485→    // Find existing mapping
   486→    const thTaskId = await getTaskIdByLarkId(task_id)
   487→    console.log(`[lark/sync] Mapping lookup: larkTaskId=${task_id} → thTaskId=${thTaskId || 'NOT FOUND'}`)
   488→
   489→    if (eventType === "task.task.deleted") {
   490→      if (!thTaskId) {
   491→        return { success: true, message: "Task not mapped, ignoring delete" }
   492→      }
   493→      // Delete the TrishulHub task
   494→      await db.task.delete({ where: { id: thTaskId } })
   495→      await removeLarkTaskMapping(thTaskId)
   496→      return { success: true, message: "Task deleted from TrishulHub" }
   497→    }
   498→
   499→    // For create/update/complete, we need the full task data from Lark
   500→    // Add 1s delay to give Lark eventual-consistency time to propagate
   501→    await new Promise((resolve) => setTimeout(resolve, 1000))
   502→
   503→    const larkTask = await getTask(tasklist_id, task_id)
   504→    if (larkTask) {
   505→      console.log(`[lark/sync] Fetched from Lark: title="${larkTask.title}", status="${larkTask.status}", priority="${larkTask.priority}", assignee="${larkTask.assignee?.open_id || 'none'}"`)
   506→    } else {
   507→      console.warn(`[lark/sync] getTask returned null for task ${task_id} in list ${tasklist_id}`)
   508→    }
   509→
   510→    // Fallback: if getTask returns null (e.g. completed tasks return 99404),
   511→    // but we have a mapping, infer status from event type instead of blindly marking DONE
   512→    if (!larkTask) {
   513→      if (thTaskId) {
   514→        // Lark API returns null/error for completed tasks on free tier (error 99404).
   515→        // Only mark as DONE if the event type suggests completion.
   516→        const likelyCompleted = eventType.includes("done") || eventType.includes("complete") || eventType.includes("finish");
   517→        const newStatus = likelyCompleted ? "DONE" : "IN_PROGRESS";
   518→        const fallbackData: Record<string, unknown> = { status: newStatus };
   519→        if (newStatus === "DONE") fallbackData.completedAt = new Date();
   520→
   521→        const existingTask = await db.task.findUnique({ where: { id: thTaskId }, select: { status: true } });
   522→        console.warn(`[lark/sync] getTask returned null for mapped task ${thTaskId}. Event: ${eventType}. Assuming status="${newStatus}" (was "${existingTask?.status || 'unknown'}")`);
   523→
   524→        await db.task.update({ where: { id: thTaskId }, data: fallbackData })
   525→        markSyncGuard(thTaskId, "FROM_LARK")
   526→        await logSync({
   527→          direction: "FROM_LARK",
   528→          action: "STATUS_CHANGE",
   529→          status: "SUCCESS",
   530→          taskId: thTaskId,
   531→          larkTaskId: task_id,
   532→          larkTaskListId: tasklist_id,
   533→          metadata: { inferredStatus: newStatus, eventType },
   534→        })
   535→        return { success: true, message: `Task ${thTaskId} updated to ${newStatus} (getTask returned null, inferred from event type)` }
   536→      }
   537→      return { success: false, message: "Failed to fetch task from Lark and no existing mapping" }
   538→    }
   539→
   540→    // Circular sync guard: skip if we just synced TO_LARK for this task
   541→    if (thTaskId && isSyncGuarded(thTaskId, "TO_LARK")) {
   542→      console.log(`[lark/sync] Skipping FROM_LARK for ${thTaskId} — circular guard active`)
   543→      return { success: true, message: "Circular guard: skipped" }
   544→    }
   545→
   546→    // Map Lark task data to TrishulHub format
   547→    const title = larkTask.title
   548→    const description = larkTask.description
   549→    const status = STATUS_FROM_LARK[larkTask.status] || "TODO"
   550→    console.log(`[lark/sync] Status mapping: lark="${larkTask.status}" → th="${status}"`)
   551→    const priority = PRIORITY_FROM_LARK[larkTask.priority] || "MEDIUM"
   552→
   553→    // Resolve assignee
   554→    let assignedTo: string | undefined
   555→    if (larkTask.assignee?.open_id) {
   556→      assignedTo = await getUserIdByLarkOpenId(larkTask.assignee.open_id) || undefined
   557→    }
   558→
   559→    // Resolve due date
   560→    const deadline = larkTask.due?.timestamp ? new Date(parseInt(larkTask.due.timestamp)) : null
   561→
   562→    if (thTaskId) {
   563→      // UPDATE existing task
   564→      const updateData: Record<string, unknown> = {}
   565→      if (title) updateData.title = title
   566→      if (description !== undefined) updateData.description = description
   567→      updateData.status = status
   568→      updateData.priority = priority
   569→      if (assignedTo !== undefined) updateData.assignedTo = assignedTo
   570→      if (deadline !== null) updateData.deadline = deadline
   571→
   572→      // Set completedAt if status changed to DONE
   573→      if (status === "DONE") {
   574→        updateData.completedAt = new Date()
   575→      }
   576→
   577→      await db.task.update({ where: { id: thTaskId }, data: updateData })
   578→      console.log(`[lark/sync] Updated TH task ${thTaskId}: status=${updateData.status}, priority=${updateData.priority}, assignedTo=${updateData.assignedTo || 'unchanged'}`)
   579→
   580→      // Mark circular guard so our update handler doesn't echo back
   581→      markSyncGuard(thTaskId, "FROM_LARK")
   582→
   583→      await logSync({
   584→        direction: "FROM_LARK",
   585→        action: "UPDATE",
   586→        status: "SUCCESS",
   587→        taskId: thTaskId,
   588→        larkTaskId: task_id,
   589→        larkTaskListId: tasklist_id,
   590→      })
   591→
   592→      return { success: true, message: `Task ${thTaskId} updated from Lark` }
   593→    } else {
   594→      // CREATE new task (originated in Lark)
   595→      const newTask = await db.task.create({
   596→        data: {
   597→          title,
   598→          description: description || null,
   599→          status,
   600→          priority,
   601→          assignedTo: assignedTo || null,
   602→          deadline,
   603→          assigneeType: "HUMAN",
   604→          category: "GENERAL",
   605→          projectId: await findProjectByTaskList(tasklist_id),
   606→        },
   607→      })
   608→
   609→      // Save mapping
   610→      await saveLarkTaskMapping(newTask.id, task_id, tasklist_id)
   611→
   612→      // Mark guard for the newly created task
   613→      markSyncGuard(newTask.id, "FROM_LARK")
   614→
   615→      return { success: true, message: `New task ${newTask.id} created from Lark` }
   616→    }
   617→  } catch (err) {
   618→    const msg = err instanceof Error ? err.message : String(err)
   619→    console.error("[lark/sync] Webhook handling error:", msg)
   620→    return { success: false, message: msg }
   621→  }
   622→}

   660→ * Remove a TrishulHub project member from the corresponding Lark task list.
   661→ * Called fire-and-forget from the project members DELETE route.
   662→ */
   663→export async function removeProjectMemberFromLarkTaskList(
   664→  projectId: string,
   665→  userId: string
   666→): Promise<void> {
   667→  const config = await getLarkConfig()
   668→  if (!config?.enabled) return
   669→
   670→  try {
   671→    const { getAppSetting, delAppSetting } = await import("@/lib/db")
   672→    const raw = await getAppSetting(`lark_tasklist_${projectId}`)
   673→    if (!raw) return
   674→
   675→    const parsed = JSON.parse(raw)
   676→    const tasklistId = parsed.tasklistId
   677→    if (!tasklistId) return
   678→
   679→    // Get user's Lark open_id
   680→    const openId = await getUserLarkOpenId(userId)
   681→    if (!openId) return
   682→
   683→    // Remove from task list
   684→    await removeTaskListMember(tasklistId, [{ id: openId }])
   685→    console.log(`[lark/sync] Removed user ${userId} (${openId}) from tasklist ${tasklistId}`)
   686→
   687→    // Cleanup: check if project has no members left — if so, delete the tasklist
   688→    const memberCount = await db.projectMember.count({ where: { projectId } })

   692→      await delAppSetting(`lark_tasklist_${projectId}`)
   693→      console.log(`[lark/sync] Deleted empty tasklist ${tasklistId} for project ${projectId}`)
   694→    }
   695→  } catch (err) {
   696→    console.error("[lark/sync] removeProjectMemberFromLarkTaskList failed:", err instanceof Error ? err.message : err)
   697→  }
   698→}

   700→/**
   701→ * Find a project ID by looking up the tasklist mapping.
   702→ */
   703→async function findProjectByTaskList(tasklistId: string): Promise<string | null> {
   704→  try {
   705→    const rows = await db.$queryRawUnsafe<Array<{ key: string; value: string }>>(
   706→      'SELECT "key", "value" FROM "AppSetting" WHERE "key" LIKE ?',
   707→      'lark_tasklist_%'
   708→    )
   709→
   710→    for (const row of rows) {
   711→      try {
   712→        const parsed = JSON.parse(row.value)
   713→        if (parsed.tasklistId === tasklistId) {
   714→          const projectId = row.key.replace("lark_tasklist_", "")
   715→          if (projectId && projectId !== "__default__") {
   716→            const project = await db.project.findUnique({ where: { id: projectId }, select: { id: true } })
   717→            return project?.id || null
   718→          }
   719→        }
   720→      } catch {
   721→        continue
   722→      }
   723→    }
   724→  } catch {
   725→    // Ignore
   726→  }
   727→
   728→  return null
   729→}
