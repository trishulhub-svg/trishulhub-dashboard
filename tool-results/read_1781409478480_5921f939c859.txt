    21→    21→// ━━ CIRCULAR SYNC GUARD ━━
    22→    22→
    23→    23→/** In-memory map: "taskId:direction" → timestamp. Prevents TH→Lark→TH→Lark loops.
    24→    24→ *  Cooldown: 10 seconds.
    25→    25→ */
    26→    26→const _syncGuard = new Map<string, number>()
    27→    27→const SYNC_GUARD_COOLDOWN_MS = 10_000
    28→    28→
    29→    29→function markSyncGuard(taskId: string, direction: string): void {
    30→    30→  _syncGuard.set(`${taskId}:${direction}`, Date.now())
    31→    31→}
    32→    32→
    33→    33→function isSyncGuarded(taskId: string, direction: string): boolean {
    34→    34→  const ts = _syncGuard.get(`${taskId}:${direction}`)
    35→    35→  if (!ts) return false
    36→    36→  if (Date.now() - ts > SYNC_GUARD_COOLDOWN_MS) {
    37→    37→    _syncGuard.delete(`${taskId}:${direction}`)
    38→    38→    return false
    39→    39→  }
    40→    40→  return true
    41→    41→}
    42→    42→
    43→    43→// ━━ WEBHOOK EVENT DEDUPLICATION ━━
    44→    44→
    45→    45→/** In-memory set of processed event_ids. Prevents duplicate processing from Lark retries.
    46→    46→ *  Window: 30 seconds.
    47→    47→ */
    48→    48→const _processedEvents = new Map<string, number>()
    49→    49→const DEDUP_WINDOW_MS = 30_000
    50→    50→
    51→    51→function isEventDuplicate(eventId: string): boolean {
    52→    52→  const ts = _processedEvents.get(eventId)
    53→    53→  if (!ts) return false
    54→    54→  if (Date.now() - ts > DEDUP_WINDOW_MS) {
    55→    55→    _processedEvents.delete(eventId)
    56→    56→    return false
    57→    57→  }
    58→    58→  return true
    59→    59→}
    60→    60→
    61→    61→function markEventProcessed(eventId: string): void {
    62→    62→  _processedEvents.set(eventId, Date.now())
    63→    63→}
    64→    64→
    65→    65→// Periodic cleanup of stale entries (every 60s)
    66→    66→if (typeof setInterval !== "undefined") {
    67→    67→  setInterval(() => {
    68→    68→    const now = Date.now()
    69→    69→    for (const [key, ts] of _syncGuard) {
    70→    70→      if (now - ts > SYNC_GUARD_COOLDOWN_MS) _syncGuard.delete(key)
    71→    71→    }
    72→    72→    for (const [key, ts] of _processedEvents) {
    73→    73→      if (now - ts > DEDUP_WINDOW_MS) _processedEvents.delete(key)
    74→    74→    }
    75→    75→  }, 60_000)
    76→    76→}

   117→   117→export async function getUserLarkOpenId(userId: string): Promise<string | null> {
   118→   118→  try {
   119→   119→    // Check mapping table
   120→   120→    const rows = await db.$queryRawUnsafe<Array<{ larkOpenId: string }>>(
   121→   121→      'SELECT "larkOpenId" FROM "LarkUserMapping" WHERE "userId" = ?',
   122→   122→      userId

   124→   124→    if (rows.length > 0 && rows[0].larkOpenId) {
   125→   125→      return rows[0].larkOpenId

   128→   128→    // Try email lookup

   132→   132→      if (larkUser?.open_id) {
   133→   133→        // Auto-save the mapping

   136→   136→          `map_${userId}_${Date.now()}`,
   137→   137→          userId,
   138→   138→          larkUser.open_id,
   139→   139→          larkUser.name || "",
   140→   140→          larkUser.email || "",
   141→   141→          "email_auto"
   142→   142→        )
   143→   143→        return larkUser.open_id
   144→   144→      }
   145→   145→    }
   146→   146→
   147→   147→    return null
   148→   148→  } catch (err) {

   229→   229→/**
   230→   230→ * Sync a task creation from TrishulHub to Lark.
   231→   231→ * Call this AFTER the task is saved in the DB.
   232→   232→ */
   233→   233→export async function syncTaskToLark(
   234→   234→  taskId: string,
   235→   235→  data: {
   236→   236→    title: string
   237→   237→    description?: string
   238→   238→    status: string
   239→   239→    priority: string
   240→   240→    assignedTo?: string
   241→   241→    projectId?: string
   242→   242→    deadline?: string | Date
   243→   243→  },
   244→   244→  userId: string
   245→   245→): Promise<void> {
   246→   246→  const config = await getLarkConfig()
   247→   247→  if (!config?.enabled) return
   248→   248→
   249→   249→  try {
   250→   250→    // Get or create the task list for the project
   251→   251→    let tasklistId: string | undefined
   252→   252→
   253→   253→    if (data.projectId) {
   254→   254→      const project = await db.project.findUnique({ where: { id: data.projectId }, select: { name: true } })
   255→   255→      if (project) {
   256→   256→        // Try to get or create the project task list (retry once on failure)
   257→   257→        let result = await getOrCreateProjectTaskList(data.projectId, project.name)
   258→   258→        if (!result) {
   259→   259→          // Retry once — transient API errors can cause first attempt to fail
   260→   260→          console.warn(`[lark/sync] Retrying task list creation for project ${data.projectId} (${project.name})`)
   261→   261→          result = await getOrCreateProjectTaskList(data.projectId, project.name)
   262→   262→        }
   263→   263→        if (result) {
   264→   264→          tasklistId = result.tasklistId
   265→   265→          if (result.created) {
   266→   266→            console.log(`[lark/sync] Created new Lark task list for project: ${project.name}`)
   267→   267→          }
   268→   268→        } else {
   269→   269→          console.error(`[lark/sync] FAILED to get/create task list for project ${data.projectId} (${project.name}) — aborting sync, task will NOT fall to default list`)
   270→   270→          await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Failed to create project task list for "${project.name}"` })
   271→   271→          // Don't fall back to default — if the task has a project, it MUST go to that project's list
   272→   272→          return
   273→   273→        }
   274→   274→      }
   275→   275→    }
   276→   276→
   277→   277→    if (!tasklistId) {
   278→   278→      // No project — use a default "TrishulHub Tasks" list
   279→   279→      const defaultResult = await getOrCreateProjectTaskList("__default__", "TrishulHub Tasks")
   280→   280→      if (!defaultResult) {
   281→   281→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Failed to get/create default task list" })
   282→   282→        return
   283→   283→      }
   284→   284→      tasklistId = defaultResult.tasklistId
   285→   285→    }
   286→   286→
   287→   287→    // Resolve assignee's Lark open_id
   288→   288→    let assigneeOpenId: string | undefined
   289→   289→    if (data.assignedTo) {
   290→   290→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   291→   291→    }
   292→   292→
   293→   293→    // Convert deadline to unix ms
   294→   294→    let dueTimestamp: number | undefined
   295→   295→    if (data.deadline) {
   296→   296→      dueTimestamp = new Date(data.deadline).getTime()
   297→   297→    }
   298→   298→
   299→   299→    // Create task — NO extra, NO status, NO priority (not supported by Lark create API)
   300→   300→    const larkTask = await createTask(tasklistId, {
   301→   301→      title: data.title,
   302→   302→      description: data.description,
   303→   303→      assigneeOpenId,
   304→   304→      dueTimestamp,
   305→   305→    })
   306→   306→
   307→   307→    if (larkTask) {
   308→   308→      await saveLarkTaskMapping(taskId, larkTask.task_id, tasklistId)
   309→   309→
   310→   310→      // Now set status & priority via separate update
   311→   311→      await updateTask(larkTask.task_id, {
   312→   312→        status: data.status,
   313→   313→        priority: data.priority,
   314→   314→      })
   315→   315→
   316→   316→      // Add assignee as tasklist member (editor role) so the tasklist
   317→   317→      // appears in the user's Lark Task Center
   318→   318→      if (assigneeOpenId) {
   319→   319→        await addTaskListMember(tasklistId, [
   320→   320→          { id: assigneeOpenId, role: "editor" },
   321→   321→        ])
   322→   322→      }
   323→   323→
   324→   324→      // Mark circular guard to prevent webhook from echoing back
   325→   325→      markSyncGuard(taskId, "TO_LARK")
   326→   326→
   327→   327→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "SUCCESS", taskId, larkTaskId: larkTask.task_id, larkTaskListId: tasklistId, projectId: data.projectId, userId })
   328→   328→    } else {
   329→   329→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Lark API returned null task" })
   330→   330→    }
   331→   331→  } catch (err) {
   332→   332→    const msg = err instanceof Error ? err.message : String(err)
   333→   333→    await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: msg })
   334→   334→  }
   335→   335→}

   337→   337→/**
   338→   338→ * Sync a task update from TrishulHub to Lark.
   339→   339→ */
   340→   340→export async function syncTaskUpdateToLark(
   341→   341→  taskId: string,
   342→   342→  data: {
   343→   343→    title?: string
   344→   344→    description?: string
   345→   345→    status?: string
   346→   346→    priority?: string
   347→   347→    assignedTo?: string
   348→   348→    deadline?: string | Date | null
   349→   349→  },
   350→   350→  userId: string
   351→   351→): Promise<void> {
   352→   352→  const config = await getLarkConfig()
   353→   353→  if (!config?.enabled) return
   354→   354→
   355→   355→  // Circular sync guard: skip if we just synced FROM Lark for this task
   356→   356→  if (isSyncGuarded(taskId, "FROM_LARK")) {
   357→   357→    console.log(`[lark/sync] Skipping TO_LARK update for ${taskId} — circular guard active`)
   358→   358→    return
   359→   359→  }
   360→   360→
   361→   361→  try {
   362→   362→    const mapping = await getLarkTaskMapping(taskId)
   363→   363→    if (!mapping) {
   364→   364→      // No mapping exists — try full sync (task may have been created before Lark was enabled)
   365→   365→      const task = await db.task.findUnique({ where: { id: taskId } })
   366→   366→      if (task) {
   367→   367→        await syncTaskToLark(taskId, {
   368→   368→          title: task.title,
   369→   369→          description: task.description || undefined,
   370→   370→          status: task.status,
   371→   371→          priority: task.priority,
   372→   372→          assignedTo: task.assignedTo || undefined,
   373→   373→          projectId: task.projectId || undefined,
   374→   374→          deadline: task.deadline || undefined,
   375→   375→        }, userId)
   376→   376→      }
   377→   377→      return
   378→   378→    }
   379→   379→
   380→   380→    // Resolve assignee's Lark open_id
   381→   381→    let assigneeOpenId: string | undefined
   382→   382→    if (data.assignedTo) {
   383→   383→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   384→   384→    }
   385→   385→
   386→   386→    // Convert deadline to unix ms
   387→   387→    let dueTimestamp: number | undefined
   388→   388→    if (data.deadline) {
   389→   389→      dueTimestamp = new Date(data.deadline).getTime()
   390→   390→    }
   391→   391→
   392→   392→    // Update task — NO extra, NO assignee in body (use member APIs instead)
   393→   393→    const larkTask = await updateTask(mapping.larkTaskId, {
   394→   394→      title: data.title,
   395→   395→      description: data.description,
   396→   396→      status: data.status,
   397→   397→      priority: data.priority,
   398→   398→      dueTimestamp,
   399→   399→    })
   400→   400→
   401→   401→    // Handle assignee change via member API
   402→   402→    if (data.assignedTo !== undefined && assigneeOpenId) {
   403→   403→      await addTaskMember(mapping.larkTaskId, [
   404→   404→        { id: assigneeOpenId, role: "assignee" },
   405→   405→      ])
   406→   406→    }
   407→   407→
   408→   408→    if (larkTask) {
   409→   409→      // Mark circular guard
   410→   410→      markSyncGuard(taskId, "TO_LARK")
   411→   411→
   412→   412→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   413→   413→    } else {
   414→   414→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark API returned null" })
   415→   415→    }
   416→   416→  } catch (err) {
   417→   417→    const msg = err instanceof Error ? err.message : String(err)
   418→   418→    await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, userId, error: msg })
   419→   419→  }
   420→   420→}

   467→   467→): Promise<{ success: boolean; message: string }> {
   468→   468→  const config = await getLarkConfig()
   469→   469→  if (!config?.enabled) {
   470→   470→    return { success: false, message: "Lark sync is disabled" }
   471→   471→  }
   472→   472→
   473→   473→  // ── Dedup check ──
   474→   474→  if (isEventDuplicate(eventId)) {
   475→   475→    console.log(`[lark/sync] Dedup: skipping duplicate event ${eventId}`)
   476→   476→    return { success: true, message: "Duplicate event, skipped" }
   477→   477→  }
   478→   478→  markEventProcessed(eventId)
   479→   479→
   480→   480→  const { task_id, tasklist_id } = eventData
   481→   481→  console.log(`[lark/sync] Processing event: ${eventType} (id: ${eventId}) for task: ${task_id}`)
   482→   482→  console.log(`[lark/sync] Event details: tasklist_id=${tasklist_id}, operator=${eventData.operator?.open_id || 'none'}`)
   483→   483→
   484→   484→  try {
   485→   485→    // Find existing mapping
   486→   486→    const thTaskId = await getTaskIdByLarkId(task_id)
   487→   487→    console.log(`[lark/sync] Mapping lookup: larkTaskId=${task_id} → thTaskId=${thTaskId || 'NOT FOUND'}`)
   488→   488→
   489→   489→    if (eventType === "task.task.deleted") {
   490→   490→      if (!thTaskId) {
   491→   491→        return { success: true, message: "Task not mapped, ignoring delete" }
   492→   492→      }
   493→   493→      // Delete the TrishulHub task
   494→   494→      await db.task.delete({ where: { id: thTaskId } })
   495→   495→      await removeLarkTaskMapping(thTaskId)
   496→   496→      return { success: true, message: "Task deleted from TrishulHub" }
   497→   497→    }
   498→   498→
   499→   499→    // For create/update/complete, we need the full task data from Lark
   500→   500→    // Add 1s delay to give Lark eventual-consistency time to propagate
   501→   501→    await new Promise((resolve) => setTimeout(resolve, 1000))
   502→   502→
   503→   503→    const larkTask = await getTask(tasklist_id, task_id)
   504→   504→    if (larkTask) {
   505→   505→      console.log(`[lark/sync] Fetched from Lark: title="${larkTask.title}", status="${larkTask.status}", priority="${larkTask.priority}", assignee="${larkTask.assignee?.open_id || 'none'}"`)
   506→   506→    } else {
   507→   507→      console.warn(`[lark/sync] getTask returned null for task ${task_id} in list ${tasklist_id}`)
   508→   508→    }
   509→   509→
   510→   510→    // Fallback: if getTask returns null (e.g. completed tasks return 99404),
   511→   511→    // but we have a mapping, infer status from event type instead of blindly marking DONE
   512→   512→    if (!larkTask) {
   513→   513→      if (thTaskId) {
   514→   514→        // Lark API returns null/error for completed tasks on free tier (error 99404).
   515→   515→        // Only mark as DONE if the event type suggests completion.
   516→   516→        const likelyCompleted = eventType.includes("done") || eventType.includes("complete") || eventType.includes("finish");
   517→   517→        const newStatus = likelyCompleted ? "DONE" : "IN_PROGRESS";
   518→   518→        const fallbackData: Record<string, unknown> = { status: newStatus };
   519→   519→        if (newStatus === "DONE") fallbackData.completedAt = new Date();
   520→   520→
   521→   521→        const existingTask = await db.task.findUnique({ where: { id: thTaskId }, select: { status: true } });
   522→   522→        console.warn(`[lark/sync] getTask returned null for mapped task ${thTaskId}. Event: ${eventType}. Assuming status="${newStatus}" (was "${existingTask?.status || 'unknown'}")`);
   523→   523→
   524→   524→        await db.task.update({ where: { id: thTaskId }, data: fallbackData })
   525→   525→        markSyncGuard(thTaskId, "FROM_LARK")
   526→   526→        await logSync({
   527→   527→          direction: "FROM_LARK",
   528→   528→          action: "STATUS_CHANGE",
   529→   529→          status: "SUCCESS",
   530→   530→          taskId: thTaskId,
   531→   531→          larkTaskId: task_id,
   532→   532→          larkTaskListId: tasklist_id,
   533→   533→          metadata: { inferredStatus: newStatus, eventType },
   534→   534→        })
   535→   535→        return { success: true, message: `Task ${thTaskId} updated to ${newStatus} (getTask returned null, inferred from event type)` }
   536→   536→      }
   537→   537→      return { success: false, message: "Failed to fetch task from Lark and no existing mapping" }
   538→   538→    }
   539→   539→
   540→   540→    // Circular sync guard: skip if we just synced TO_LARK for this task
   541→   541→    if (thTaskId && isSyncGuarded(thTaskId, "TO_LARK")) {
   542→   542→      console.log(`[lark/sync] Skipping FROM_LARK for ${thTaskId} — circular guard active`)
   543→   543→      return { success: true, message: "Circular guard: skipped" }
   544→   544→    }
   545→   545→
   546→   546→    // Map Lark task data to TrishulHub format
   547→   547→    const title = larkTask.title
   548→   548→    const description = larkTask.description
   549→   549→    const status = STATUS_FROM_LARK[larkTask.status] || "TODO"
   550→   550→    console.log(`[lark/sync] Status mapping: lark="${larkTask.status}" → th="${status}"`)
   551→   551→    const priority = PRIORITY_FROM_LARK[larkTask.priority] || "MEDIUM"
   552→   552→
   553→   553→    // Resolve assignee
   554→   554→    let assignedTo: string | undefined
   555→   555→    if (larkTask.assignee?.open_id) {
   556→   556→      assignedTo = await getUserIdByLarkOpenId(larkTask.assignee.open_id) || undefined
   557→   557→    }
   558→   558→
   559→   559→    // Resolve due date
   560→   560→    const deadline = larkTask.due?.timestamp ? new Date(parseInt(larkTask.due.timestamp)) : null
   561→   561→
   562→   562→    if (thTaskId) {
   563→   563→      // UPDATE existing task
   564→   564→      const updateData: Record<string, unknown> = {}
   565→   565→      if (title) updateData.title = title
   566→   566→      if (description !== undefined) updateData.description = description
   567→   567→      updateData.status = status
   568→   568→      updateData.priority = priority
   569→   569→      if (assignedTo !== undefined) updateData.assignedTo = assignedTo
   570→   570→      if (deadline !== null) updateData.deadline = deadline
   571→   571→
   572→   572→      // Set completedAt if status changed to DONE
   573→   573→      if (status === "DONE") {
   574→   574→        updateData.completedAt = new Date()
   575→   575→      }
   576→   576→
   577→   577→      await db.task.update({ where: { id: thTaskId }, data: updateData })
   578→   578→      console.log(`[lark/sync] Updated TH task ${thTaskId}: status=${updateData.status}, priority=${updateData.priority}, assignedTo=${updateData.assignedTo || 'unchanged'}`)
   579→   579→
   580→   580→      // Mark circular guard so our update handler doesn't echo back
   581→   581→      markSyncGuard(thTaskId, "FROM_LARK")
   582→   582→
   583→   583→      await logSync({
   584→   584→        direction: "FROM_LARK",
   585→   585→        action: "UPDATE",
   586→   586→        status: "SUCCESS",
   587→   587→        taskId: thTaskId,
   588→   588→        larkTaskId: task_id,
   589→   589→        larkTaskListId: tasklist_id,
   590→   590→      })
   591→   591→
   592→   592→      return { success: true, message: `Task ${thTaskId} updated from Lark` }
   593→   593→    } else {
   594→   594→      // CREATE new task (originated in Lark)
   595→   595→      const newTask = await db.task.create({
   596→   596→        data: {
   597→   597→          title,
   598→   598→          description: description || null,
   599→   599→          status,
   600→   600→          priority,
   601→   601→          assignedTo: assignedTo || null,
   602→   602→          deadline,
   603→   603→          assigneeType: "HUMAN",
   604→   604→          category: "GENERAL",
   605→   605→          projectId: await findProjectByTaskList(tasklist_id),
   606→   606→        },
   607→   607→      })
   608→   608→
   609→   609→      // Save mapping
   610→   610→      await saveLarkTaskMapping(newTask.id, task_id, tasklist_id)
   611→   611→
   612→   612→      // Mark guard for the newly created task
   613→   613→      markSyncGuard(newTask.id, "FROM_LARK")
   614→   614→
   615→   615→      return { success: true, message: `New task ${newTask.id} created from Lark` }
   616→   616→    }
   617→   617→  } catch (err) {
   618→   618→    const msg = err instanceof Error ? err.message : String(err)
   619→   619→    console.error("[lark/sync] Webhook handling error:", msg)
   620→   620→    return { success: false, message: msg }
   621→   621→  }
   622→   622→}

   703→   703→async function findProjectByTaskList(tasklistId: string): Promise<string | null> {
   704→   704→  try {
   705→   705→    const rows = await db.$queryRawUnsafe<Array<{ key: string; value: string }>>(
   706→   706→      'SELECT "key", "value" FROM "AppSetting" WHERE "key" LIKE ?',
   707→   707→      'lark_tasklist_%'
   708→   708→    )
   709→   709→
   710→   710→    for (const row of rows) {
   711→   711→      try {
   712→   712→        const parsed = JSON.parse(row.value)
   713→   713→        if (parsed.tasklistId === tasklistId) {
   714→   714→          const projectId = row.key.replace("lark_tasklist_", "")
   715→   715→          if (projectId && projectId !== "__default__") {
   716→   716→            const project = await db.project.findUnique({ where: { id: projectId }, select: { id: true } })
   717→   717→            return project?.id || null
   718→   718→          }
   719→   719→        }
   720→   720→      } catch {
   721→   721→        continue
   722→   722→      }
   723→   723→    }
   724→   724→  } catch {
   725→   725→    // Ignore
   726→   726→  }
   727→   727→
   728→   728→  return null
