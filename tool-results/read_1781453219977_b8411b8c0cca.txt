   122→function formatRelativeTime(dateStr: string | null | undefined): string {
   123→  if (!dateStr) return "N/A"
   124→  const date = new Date(dateStr)
   125→  if (isNaN(date.getTime())) return "N/A"
   126→  const now = new Date()
   127→  const diffMs = now.getTime() - date.getTime()
   128→  const diffSec = Math.floor(diffMs / 1000)
   129→  if (diffSec < 60) return "just now"
   130→  const diffMin = Math.floor(diffSec / 60)
   131→  if (diffMin < 60) return `${diffMin}m ago`
   132→  const diffHr = Math.floor(diffMin / 60)
   133→  if (diffHr < 24) return `${diffHr}h ago`
   134→  const diffDay = Math.floor(diffHr / 24)
   135→  if (diffDay < 7) return `${diffDay}d ago`
   136→  return formatDateTime(dateStr)
   137→}

   139→/** Safe helper: extract initials from a potentially null user name */
   140→function getInitials(userName: string | null | undefined): string {
   141→  if (!userName) return "?"
   142→  return userName.split(" ").filter(Boolean).map(n => n[0] || "").join("").toUpperCase().slice(0, 2) || "?"
   143→}

   180→/**
   181→ * Local error boundary for the Audit Trail page.
   182→ * Catches render-time errors so they show a useful message
   183→ * instead of the generic DashboardError boundary.
   184→ */
   185→class AuditTrailErrorBoundary extends React.Component<
   186→  { children: React.ReactNode },
   187→  { hasError: boolean; error: Error | null }
   188→> {
   189→  constructor(props: { children: React.ReactNode }) {
   190→    super(props)
   191→    this.state = { hasError: false, error: null }
   192→  }
   193→  static getDerivedStateFromError(error: Error) {
   194→    return { hasError: true, error }
   195→  }
   196→  componentDidCatch(error: Error, info: React.ErrorInfo) {
   197→    console.error("[AuditTrailErrorBoundary]", error.message, info.componentStack)
   198→  }
   199→  render() {
   200→    if (this.state.hasError) {
   201→      return (
   202→        <div className="min-h-[40vh] flex items-center justify-center p-6">
   203→          <Card className="max-w-md w-full border-red-200 dark:border-red-900/50">
   204→            <CardContent className="p-6 space-y-4">
   205→              <div className="flex justify-center">
   206→                <div className="h-14 w-14 rounded-full bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   207→                  <AlertTriangle className="h-7 w-7 text-red-600 dark:text-red-400" />
   208→                </div>
   209→              </div>
   210→              <div className="text-center space-y-2">
   211→                <h2 className="text-lg font-bold">Audit Trail Error</h2>
   212→                <p className="text-sm text-muted-foreground">
   213→                  Something went wrong loading the audit trail. This is likely a data or rendering issue.
   214→                </p>
   215→                <details className="text-left">
   216→                  <summary className="text-xs text-muted-foreground cursor-pointer hover:text-foreground">
   217→                    Error details
   218→                  </summary>
   219→                  <pre className="mt-2 text-xs bg-muted p-3 rounded-md overflow-auto max-h-32 text-red-600 dark:text-red-400">
   220→                    {this.state.error?.message || "Unknown error"}
   221→                  </pre>
   222→                </details>
   223→              </div>
   224→              <div className="flex gap-2 justify-center">
   225→                <Button onClick={() => this.setState({ hasError: false, error: null })}>
   226→                  <RotateCcw className="h-4 w-4 mr-2" /> Try Again
   227→                </Button>
   228→              </div>
   229→            </CardContent>
   230→          </Card>
   231→        </div>
   232→      )
   233→    }
   234→    return this.props.children
   235→  }
   236→}
