    21→// ━━ CIRCULAR SYNC GUARD ━━
    22→
    23→/** In-memory map: "taskId:direction" → timestamp. Prevents TH→Lark→TH→Lark loops.
    24→ *  Cooldown: 10 seconds.
    25→ */
    26→const _syncGuard = new Map<string, number>()
    27→const SYNC_GUARD_COOLDOWN_MS = 10_000
    28→
    29→function markSyncGuard(taskId: string, direction: string): void {
    30→  _syncGuard.set(`${taskId}:${direction}`, Date.now())
    31→}
    32→
    33→function isSyncGuarded(taskId: string, direction: string): boolean {
    34→  const ts = _syncGuard.get(`${taskId}:${direction}`)
    35→  if (!ts) return false
    36→  if (Date.now() - ts > SYNC_GUARD_COOLDOWN_MS) {
    37→    _syncGuard.delete(`${taskId}:${direction}`)
    38→    return false
    39→  }
    40→  return true
    41→}
    42→
    43→// ━━ WEBHOOK EVENT DEDUPLICATION ━━
    44→
    45→/** In-memory set of processed event_ids. Prevents duplicate processing from Lark retries.
    46→ *  Window: 30 seconds.
    47→ */
    48→const _processedEvents = new Map<string, number>()
    49→const DEDUP_WINDOW_MS = 30_000
    50→
    51→function isEventDuplicate(eventId: string): boolean {
    52→  const ts = _processedEvents.get(eventId)
    53→  if (!ts) return false
    54→  if (Date.now() - ts > DEDUP_WINDOW_MS) {
    55→    _processedEvents.delete(eventId)
    56→    return false
    57→  }
    58→  return true
    59→}
    60→
    61→function markEventProcessed(eventId: string): void {
    62→  _processedEvents.set(eventId, Date.now())
    63→}
    64→
    65→// Periodic cleanup of stale entries (every 60s)
    66→if (typeof setInterval !== "undefined") {
    67→  setInterval(() => {
    68→    const now = Date.now()
    69→    for (const [key, ts] of _syncGuard) {
    70→      if (now - ts > SYNC_GUARD_COOLDOWN_MS) _syncGuard.delete(key)
    71→    }
    72→    for (const [key, ts] of _processedEvents) {
    73→      if (now - ts > DEDUP_WINDOW_MS) _processedEvents.delete(key)
    74→    }
    75→  }, 60_000)
    76→}

    82→  direction: SyncDirection
    83→  action: SyncAction
    84→  status: SyncStatus
    85→  taskId?: string

    87→  larkTaskListId?: string
    88→  projectId?: string

    95→      `INSERT INTO "LarkSyncLog" ("id", "direction", "action", "status", "taskId", "larkTaskId", "larkTaskListId", "projectId", "userId", "error", "metadata", "createdAt") VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)`,
    96→      `sync_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
    97→      params.direction,
    98→      params.action,
    99→      params.status,
   100→      params.taskId || null,
   101→      params.larkTaskId || null,
   102→      params.larkTaskListId || null,
   103→      params.projectId || null,
   104→      params.userId || null,

   113→/**
   114→ * Get the Lark open_id for a TrishulHub user.
   115→ * Checks LarkUserMapping table first, then tries email lookup.
   116→ */
   117→export async function getUserLarkOpenId(userId: string): Promise<string | null> {
   118→  try {
   119→    // Check mapping table
   120→    const rows = await db.$queryRawUnsafe<Array<{ larkOpenId: string }>>(
   121→      'SELECT "larkOpenId" FROM "LarkUserMapping" WHERE "userId" = ?',
   122→      userId
   123→    )
   124→    if (rows.length > 0 && rows[0].larkOpenId) {
   125→      return rows[0].larkOpenId
   126→    }
   127→
   128→    // Try email lookup
   129→    const user = await db.user.findUnique({ where: { id: userId }, select: { email: true } })
   130→    if (user?.email) {
   131→      const larkUser = await lookupUserByEmail(user.email)
   132→      if (larkUser?.open_id) {
   133→        // Auto-save the mapping
   134→        await db.$executeRawUnsafe(
   135→          'INSERT OR IGNORE INTO "LarkUserMapping" ("id", "userId", "larkOpenId", "larkName", "larkEmail", "matchedBy", "createdAt") VALUES (?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)',
   136→          `map_${userId}_${Date.now()}`,
   137→          userId,
   138→          larkUser.open_id,
   139→          larkUser.name || "",
   140→          larkUser.email || "",
   141→          "email_auto"
   142→        )
   143→        return larkUser.open_id
   144→      }
   145→    }
   146→
   147→    return null
   148→  } catch (err) {
   149→    console.error("[lark/sync] Failed to get user Lark ID:", err)
   150→    return null
   151→  }
   152→}

   229→/**
   230→ * Sync a task creation from TrishulHub to Lark.
   231→ * Call this AFTER the task is saved in the DB.
   232→ */
   233→export async function syncTaskToLark(
   234→  taskId: string,
   235→  data: {
   236→    title: string
   237→    description?: string
   238→    status: string
   239→    priority: string
   240→    assignedTo?: string
   241→    projectId?: string
   242→    deadline?: string | Date
   243→  },
   244→  userId: string
   245→): Promise<void> {
   246→  const config = await getLarkConfig()
   247→  if (!config?.enabled) return
   248→
   249→  try {
   250→    // Get or create the task list for the project
   251→    let tasklistId: string | undefined
   252→
   253→    if (data.projectId) {
   254→      const project = await db.project.findUnique({ where: { id: data.projectId }, select: { name: true } })
   255→      if (project) {
   256→        // Try to get or create the project task list (retry once on failure)
   257→        let result = await getOrCreateProjectTaskList(data.projectId, project.name)
   258→        if (!result) {
   259→          // Retry once — transient API errors can cause first attempt to fail
   260→          console.warn(`[lark/sync] Retrying task list creation for project ${data.projectId} (${project.name})`)
   261→          result = await getOrCreateProjectTaskList(data.projectId, project.name)
   262→        }
   263→        if (result) {
   264→          tasklistId = result.tasklistId
   265→          if (result.created) {
   266→            console.log(`[lark/sync] Created new Lark task list for project: ${project.name}`)
   267→          }
   268→        } else {
   269→          console.error(`[lark/sync] FAILED to get/create task list for project ${data.projectId} (${project.name}) — aborting sync, task will NOT fall to default list`)
   270→          await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Failed to create project task list for "${project.name}"` })
   271→          // Don't fall back to default — if the task has a project, it MUST go to that project's list
   272→          return
   273→        }
   274→      } else {
   275→        // Project exists in task data but not found in DB — don't silently fall to default
   276→        console.error(`[lark/sync] Project ${data.projectId} not found in DB — aborting sync`)
   277→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Project ${data.projectId} not found in database` })
   278→        return
   279→      }
   280→    }
   281→
   282→    if (!tasklistId) {
   283→      // No project — use a default "TrishulHub Tasks" list
   284→      const defaultResult = await getOrCreateProjectTaskList("__default__", "TrishulHub Tasks")
   285→      if (!defaultResult) {
   286→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Failed to get/create default task list" })
   287→        return
   288→      }
   289→      tasklistId = defaultResult.tasklistId
   290→    }
   291→
   292→    // Resolve assignee's Lark open_id
   293→    let assigneeOpenId: string | undefined
   294→    if (data.assignedTo) {
   295→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   296→    }
   297→
   298→    // Convert deadline to unix ms
   299→    let dueTimestamp: number | undefined
   300→    if (data.deadline) {
   301→      dueTimestamp = new Date(data.deadline).getTime()
   302→    }
   303→
   304→    // Create task — bare creation only (no members, no status, no priority on create API)
   305→    const larkTask = await createTask(tasklistId, {
   306→      title: data.title,
   307→      description: data.description,
   308→      dueTimestamp,
   309→    })
   310→
   311→    if (larkTask) {
   312→      await saveLarkTaskMapping(taskId, larkTask.task_id, tasklistId)
   313→
   314→      // Assign member via add_members endpoint (create API doesn't support members field)
   315→      if (assigneeOpenId) {
   316→        await addTaskMember(larkTask.task_id, [
   317→          { id: assigneeOpenId, role: "assignee" },
   318→        ])
   319→      }
   320→
   321→      // Set status & priority via separate update (create API doesn't support these)
   322→      await updateTask(larkTask.task_id, {
   323→        status: data.status,
   324→        priority: data.priority,
   325→      })
   326→
   327→      // Add assignee as tasklist member (editor role) so the tasklist
   328→      // appears in the user's Lark Task Center
   329→      if (assigneeOpenId) {
   330→        await addTaskListMember(tasklistId, [
   331→          { id: assigneeOpenId, role: "editor" },
   332→        ])
   333→      }
   334→
   335→      // Mark circular guard to prevent webhook from echoing back
   336→      markSyncGuard(taskId, "TO_LARK")
   337→
   338→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "SUCCESS", taskId, larkTaskId: larkTask.task_id, larkTaskListId: tasklistId, projectId: data.projectId, userId })
   339→    } else {
   340→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Lark API returned null task" })
   341→    }
   342→  } catch (err) {
   343→    const msg = err instanceof Error ? err.message : String(err)
   344→    await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: msg })
   345→  }
   346→}

   348→/**
   349→ * Sync a task update from TrishulHub to Lark.
   350→ */
   351→export async function syncTaskUpdateToLark(
   352→  taskId: string,
   353→  data: {
   354→    title?: string
   355→    description?: string
   356→    status?: string
   357→    priority?: string
   358→    assignedTo?: string
   359→    deadline?: string | Date | null
   360→  },
   361→  userId: string
   362→): Promise<void> {
   363→  const config = await getLarkConfig()
   364→  if (!config?.enabled) return
   365→
   366→  // Circular sync guard: skip if we just synced FROM Lark for this task
   367→  if (isSyncGuarded(taskId, "FROM_LARK")) {
   368→    console.log(`[lark/sync] Skipping TO_LARK update for ${taskId} — circular guard active`)
   369→    return
   370→  }
   371→
   372→  try {
   373→    const mapping = await getLarkTaskMapping(taskId)
   374→    if (!mapping) {
   375→      // No mapping exists — try full sync (task may have been created before Lark was enabled)
   376→      const task = await db.task.findUnique({ where: { id: taskId } })
   377→      if (task) {
   378→        await syncTaskToLark(taskId, {
   379→          title: task.title,
   380→          description: task.description || undefined,
   381→          status: task.status,
   382→          priority: task.priority,
   383→          assignedTo: task.assignedTo || undefined,
   384→          projectId: task.projectId || undefined,
   385→          deadline: task.deadline || undefined,
   386→        }, userId)
   387→      }
   388→      return
   389→    }
   390→
   391→    // Resolve assignee's Lark open_id
   392→    let assigneeOpenId: string | undefined
   393→    if (data.assignedTo) {
   394→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   395→    }
   396→
   397→    // Convert deadline to unix ms
   398→    let dueTimestamp: number | undefined
   399→    if (data.deadline) {
   400→      dueTimestamp = new Date(data.deadline).getTime()
   401→    }
   402→
   403→    // Update task — NO extra, NO assignee in body (use member APIs instead)
   404→    const larkTask = await updateTask(mapping.larkTaskId, {
   405→      title: data.title,
   406→      description: data.description,
   407→      status: data.status,
   408→      priority: data.priority,
   409→      dueTimestamp,
   410→    })
   411→
   412→    // Handle assignee change via member API
   413→    if (data.assignedTo !== undefined && assigneeOpenId) {
   414→      await addTaskMember(mapping.larkTaskId, [
   415→        { id: assigneeOpenId, role: "assignee" },
   416→      ])
   417→    }
   418→
   419→    if (larkTask) {
   420→      // Mark circular guard
   421→      markSyncGuard(taskId, "TO_LARK")
   422→
   423→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   424→    } else {
   425→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark API returned null" })
   426→    }
   427→  } catch (err) {
   428→    const msg = err instanceof Error ? err.message : String(err)
   429→    await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, userId, error: msg })
   430→  }
   431→}

   433→/**
   434→ * Sync a task deletion from TrishulHub to Lark.
   435→ */
   436→export async function syncTaskDeleteToLark(taskId: string, userId: string): Promise<void> {
   437→  const config = await getLarkConfig()
   438→  if (!config?.enabled) return
   439→
   440→  try {
   441→    const mapping = await getLarkTaskMapping(taskId)
   442→    if (!mapping) {
   443→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "SKIPPED", taskId, userId, error: "No Lark mapping found" })

   445→    }
   446→
   447→    const success = await larkDeleteTask(mapping.larkTaskListId, mapping.larkTaskId)
   448→    await removeLarkTaskMapping(taskId)
   449→
   450→    if (success) {
   451→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   452→    } else {
   453→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark delete API failed" })
   454→    }
   455→  } catch (err) {
   456→    const msg = err instanceof Error ? err.message : String(err)
   457→    await logSync({ direction: "TO_LARK", action: "DELETE", status: "FAILED", taskId, userId, error: msg })
   458→  }
   459→}

   461→// ━━ LARK → TRISHULHUB SYNC (Webhook-driven) ━━
   462→
   463→/**
   464→ * Handle incoming Lark webhook events.
   465→ * @param eventId — Lark event_id for deduplication
   466→ * @param eventType — Lark event type
   467→ * @param eventData — Parsed event payload
   468→ */
   469→export async function handleLarkWebhookEvent(
   470→  eventId: string,
   471→  eventType: string,
   472→  eventData: {
   473→    task_id: string
   474→    tasklist_id: string
   475→    operator?: { open_id: string }
   476→    [key: string]: unknown
   477→  }
   478→): Promise<{ success: boolean; message: string }> {
   479→  const config = await getLarkConfig()
   480→  if (!config?.enabled) {
   481→    return { success: false, message: "Lark sync is disabled" }
   482→  }
   483→
   484→  // ── Dedup check ──
   485→  if (isEventDuplicate(eventId)) {
   486→    console.log(`[lark/sync] Dedup: skipping duplicate event ${eventId}`)
   487→    return { success: true, message: "Duplicate event, skipped" }
   488→  }
   489→  markEventProcessed(eventId)
   490→
   491→  const { task_id, tasklist_id } = eventData
   492→  console.log(`[lark/sync] Processing event: ${eventType} (id: ${eventId}) for task: ${task_id}`)
   493→  console.log(`[lark/sync] Event details: tasklist_id=${tasklist_id}, operator=${eventData.operator?.open_id || 'none'}`)
   494→
   495→  try {
   496→    // Find existing mapping
   497→    const thTaskId = await getTaskIdByLarkId(task_id)
   498→    console.log(`[lark/sync] Mapping lookup: larkTaskId=${task_id} → thTaskId=${thTaskId || 'NOT FOUND'}`)
   499→
   500→    if (eventType === "task.task.deleted") {
   501→      if (!thTaskId) {
   502→        return { success: true, message: "Task not mapped, ignoring delete" }
   503→      }
   504→      // Delete the TrishulHub task
   505→      await db.task.delete({ where: { id: thTaskId } })
   506→      await removeLarkTaskMapping(thTaskId)
   507→
   508→      try {
   509→        const { logAudit } = await import("@/lib/audit-log")
   510→        await logAudit({
   511→          userId: eventData.operator?.open_id || "lark-webhook",
   512→          userName: "Lark Webhook",
   513→          userRole: "SYSTEM",
   514→          department: "TEAM_WORK",
   515→          page: "tasks",
   516→          action: "DELETE",
   517→          entityType: "Task",
   518→          entityId: thTaskId,
   519→          description: `Task deleted via Lark sync`,
   520→          status: "SUCCESS",
   521→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id }),
   522→        })
   523→      } catch { /* audit log is best-effort */ }
   524→
   525→      return { success: true, message: "Task deleted from TrishulHub" }
   526→    }
   527→
   528→    // For create/update/complete, we need the full task data from Lark
   529→    // Add 1s delay to give Lark eventual-consistency time to propagate
   530→    await new Promise((resolve) => setTimeout(resolve, 1000))
   531→
   532→    const larkTask = await getTask(tasklist_id, task_id)
   533→    if (larkTask) {
   534→      console.log(`[lark/sync] Fetched from Lark: title="${larkTask.title}", status="${larkTask.status}", priority="${larkTask.priority}", assignee="${larkTask.assignee?.open_id || 'none'}"`)
   535→    } else {
   536→      console.warn(`[lark/sync] getTask returned null for task ${task_id} in list ${tasklist_id}`)
   537→    }
   538→
   539→    // Fallback: if getTask returns null (e.g. completed tasks return 99404 on free tier),
   540→    // retry once after a short delay, then inspect event body for status clues.
   541→    if (!larkTask) {
   542→      if (thTaskId) {
   543→        // Retry getTask once after 2s — Lark eventual consistency may need more time
   544→        console.warn(`[lark/sync] getTask returned null for mapped task ${thTaskId}, retrying...`)
   545→        await new Promise((resolve) => setTimeout(resolve, 2000))
   546→        const retryTask = await getTask(tasklist_id, task_id)
   547→
   548→        if (retryTask) {
   549→          // Retry succeeded — process normally via the update path below
   550→          // Rewrite larkTask so the code below handles it
   551→          Object.assign(eventData, { _retryTask: retryTask })
   552→          // Fall through to the main update logic — we'll read from retryTask
   553→          const title = retryTask.title
   554→          const description = retryTask.description
   555→          const status = STATUS_FROM_LARK[retryTask.status] || "TODO"
   556→          const priority = PRIORITY_FROM_LARK[retryTask.priority] || "MEDIUM"
   557→
   558→          let assignedTo: string | undefined
   559→          if (retryTask.assignee?.open_id) {
   560→            assignedTo = await getUserIdByLarkOpenId(retryTask.assignee.open_id) || undefined
   561→          }
   562→
   563→          const deadline = retryTask.due?.timestamp ? new Date(parseInt(retryTask.due.timestamp)) : null
   564→
   565→          const updateData: Record<string, unknown> = {}
   566→          if (title) updateData.title = title
   567→          if (description !== undefined) updateData.description = description
   568→          updateData.status = status
   569→          updateData.priority = priority
   570→          if (assignedTo !== undefined) updateData.assignedTo = assignedTo
   571→          if (deadline !== null) updateData.deadline = deadline
   572→          if (status === "DONE") updateData.completedAt = new Date()
   573→
   574→          await db.task.update({ where: { id: thTaskId }, data: updateData })
   575→          markSyncGuard(thTaskId, "FROM_LARK")
   576→
   577→          const { logAudit } = await import("@/lib/audit-log")
   578→          await logAudit({
   579→            userId: eventData.operator?.open_id || "lark-webhook",
   580→            userName: "Lark Webhook",
   581→            userRole: "SYSTEM",
   582→            department: "TEAM_WORK",
   583→            page: "tasks",
   584→            action: "STATUS_CHANGE",
   585→            entityType: "Task",
   586→            entityId: thTaskId,
   587→            description: `Task status changed to "${status}" via Lark sync (retried)`,
   588→            newValue: status,
   589→            status: "SUCCESS",
   590→            metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, larkStatus: retryTask.status }),
   591→          })
   592→
   593→          await logSync({
   594→            direction: "FROM_LARK",
   595→            action: "STATUS_CHANGE",
   596→            status: "SUCCESS",
   597→            taskId: thTaskId,
   598→            larkTaskId: task_id,
   599→            larkTaskListId: tasklist_id,
   600→            metadata: { inferredStatus: status, eventType, retried: true },
   601→          })
   602→          return { success: true, message: `Task ${thTaskId} updated to ${status} (retried getTask)` }
   603→        }
   604→
   605→        // Both getTask calls failed — inspect event body for status change clues.
   606→        const eventBody = (eventData as Record<string, unknown>)
   607→        const changedFields = eventBody.changed_fields as string[] | undefined
   608→
   609→        // Determine if this is likely a completion:
   610→        // 1. Event type contains done/complete/finish keywords
   611→        // 2. OR changed_fields includes "status" (something changed, likely to done since getTask fails)
   612→        const likelyCompleted =
   613→          eventType.includes("done") || eventType.includes("complete") || eventType.includes("finish") ||
   614→          (Array.isArray(changedFields) && changedFields.includes("status"))
   615→
   616→        const newStatus = likelyCompleted ? "DONE" : "IN_PROGRESS";
   617→        const fallbackData: Record<string, unknown> = { status: newStatus };
   618→        if (newStatus === "DONE") fallbackData.completedAt = new Date();
   619→
   620→        const existingTask = await db.task.findUnique({ where: { id: thTaskId }, select: { status: true } });
   621→        console.warn(`[lark/sync] getTask returned null twice for mapped task ${thTaskId}. Event: ${eventType}. Assuming status="${newStatus}" (was "${existingTask?.status || 'unknown'}")`);
   622→
   623→        await db.task.update({ where: { id: thTaskId }, data: fallbackData })
   624→        markSyncGuard(thTaskId, "FROM_LARK")
   625→
   626→        const { logAudit } = await import("@/lib/audit-log")
   627→        await logAudit({
   628→          userId: eventData.operator?.open_id || "lark-webhook",
   629→          userName: "Lark Webhook",
   630→          userRole: "SYSTEM",
   631→          department: "TEAM_WORK",
   632→          page: "tasks",
   633→          action: "STATUS_CHANGE",
   634→          entityType: "Task",
   635→          entityId: thTaskId,
   636→          description: `Task status changed to "${newStatus}" via Lark sync (inferred — getTask failed)`,
   637→          newValue: newStatus,
   638→          status: "SUCCESS",
   639→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, inferred: true, previousStatus: existingTask?.status }),
   640→        })
   641→
   642→        await logSync({
   643→          direction: "FROM_LARK",
   644→          action: "STATUS_CHANGE",
   645→          status: "SUCCESS",
   646→          taskId: thTaskId,
   647→          larkTaskId: task_id,
   648→          larkTaskListId: tasklist_id,
   649→          metadata: { inferredStatus: newStatus, eventType, retried: true },
   650→        })
   651→        return { success: true, message: `Task ${thTaskId} updated to ${newStatus} (getTask null, inferred from event + changed_fields)` }
   652→      }
   653→      return { success: false, message: "Failed to fetch task from Lark and no existing mapping" }
   654→    }
   655→
   656→    // Circular sync guard: skip if we just synced TO_LARK for this task
   657→    if (thTaskId && isSyncGuarded(thTaskId, "TO_LARK")) {
   658→      console.log(`[lark/sync] Skipping FROM_LARK for ${thTaskId} — circular guard active`)
   659→      return { success: true, message: "Circular guard: skipped" }
   660→    }
   661→
   662→    // Map Lark task data to TrishulHub format
   663→    const title = larkTask.title
   664→    const description = larkTask.description
   665→    const status = STATUS_FROM_LARK[larkTask.status] || "TODO"
   666→    console.log(`[lark/sync] Status mapping: lark="${larkTask.status}" → th="${status}"`)
   667→    const priority = PRIORITY_FROM_LARK[larkTask.priority] || "MEDIUM"
   668→
   669→    // Resolve assignee
   670→    let assignedTo: string | undefined
   671→    if (larkTask.assignee?.open_id) {
   672→      assignedTo = await getUserIdByLarkOpenId(larkTask.assignee.open_id) || undefined
   673→    }
   674→
   675→    // Resolve due date
   676→    const deadline = larkTask.due?.timestamp ? new Date(parseInt(larkTask.due.timestamp)) : null
   677→
   678→    if (thTaskId) {
   679→      // UPDATE existing task
   680→      const updateData: Record<string, unknown> = {}
   681→      if (title) updateData.title = title
   682→      if (description !== undefined) updateData.description = description
   683→      updateData.status = status
   684→      updateData.priority = priority
   685→      if (assignedTo !== undefined) updateData.assignedTo = assignedTo
   686→      if (deadline !== null) updateData.deadline = deadline
   687→
   688→      // Set completedAt if status changed to DONE
   689→      if (status === "DONE") {
   690→        updateData.completedAt = new Date()
   691→      }
   692→
   693→      await db.task.update({ where: { id: thTaskId }, data: updateData })
   694→      console.log(`[lark/sync] Updated TH task ${thTaskId}: status=${updateData.status}, priority=${updateData.priority}, assignedTo=${updateData.assignedTo || 'unchanged'}`)
   695→
   696→      // Mark circular guard so our update handler doesn't echo back
   697→      markSyncGuard(thTaskId, "FROM_LARK")
   698→
   699→      // Log to both LarkSyncLog and AuditLog
   700→      await logSync({
   701→        direction: "FROM_LARK",
   702→        action: status === "DONE" ? "STATUS_CHANGE" : "UPDATE",
   703→        status: "SUCCESS",
   704→        taskId: thTaskId,
   705→        larkTaskId: task_id,
   706→        larkTaskListId: tasklist_id,
   707→      })
   708→
   709→      try {
   710→        const { logAudit } = await import("@/lib/audit-log")
   711→        await logAudit({
   712→          userId: eventData.operator?.open_id || "lark-webhook",
   713→          userName: "Lark Webhook",
   714→          userRole: "SYSTEM",
   715→          department: "TEAM_WORK",
   716→          page: "tasks",
   717→          action: status === "DONE" ? "STATUS_CHANGE" : "UPDATE",
   718→          entityType: "Task",
   719→          entityId: thTaskId,
   720→          description: `Task ${status === "DONE" ? "marked as DONE" : "updated"} via Lark sync`,
   721→          newValue: status,
   722→          status: "SUCCESS",
   723→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, larkStatus: larkTask.status }),
   724→        })
   725→      } catch { /* audit log is best-effort */ }
   726→
   727→      return { success: true, message: `Task ${thTaskId} updated from Lark` }
   728→    } else {
   729→      // CREATE new task (originated in Lark)
   730→      const newTask = await db.task.create({
   731→        data: {
   732→          title,
   733→          description: description || null,
   734→          status,
   735→          priority,
   736→          assignedTo: assignedTo || null,
   737→          deadline,
   738→          assigneeType: "HUMAN",
   739→          category: "GENERAL",
   740→          projectId: await findProjectByTaskList(tasklist_id),
   741→        },
   742→      })
   743→
   744→      // Save mapping
   745→      await saveLarkTaskMapping(newTask.id, task_id, tasklist_id)
   746→
   747→      // Mark guard for the newly created task
   748→      markSyncGuard(newTask.id, "FROM_LARK")
   749→
   750→      try {
   751→        const { logAudit } = await import("@/lib/audit-log")
   752→        await logAudit({
   753→          userId: eventData.operator?.open_id || "lark-webhook",
   754→          userName: "Lark Webhook",
   755→          userRole: "SYSTEM",
   756→          department: "TEAM_WORK",
   757→          page: "tasks",
   758→          action: "CREATE",
   759→          entityType: "Task",
   760→          entityId: newTask.id,
   761→          description: `Task created from Lark: "${title}"`,
   762→          newValue: status,
   763→          status: "SUCCESS",
   764→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id }),
   765→        })
   766→      } catch { /* audit log is best-effort */ }
   767→
   768→      return { success: true, message: `New task ${newTask.id} created from Lark` }
   769→    }
   770→  } catch (err) {
   771→    const msg = err instanceof Error ? err.message : String(err)
   772→    console.error("[lark/sync] Webhook handling error:", msg)
   773→    return { success: false, message: msg }
   774→  }
   775→}

   779→/**
   780→ * Add a TrishulHub project member to the corresponding Lark task list.
   781→ * Called fire-and-forget from the project members POST route.
   782→ */
   783→export async function addProjectMemberToLarkTaskList(
   784→  projectId: string,
   785→  userId: string
   786→): Promise<void> {
   787→  const config = await getLarkConfig()
   788→  if (!config?.enabled) return
   789→
   790→  try {
   791→    // Find the tasklist mapping for this project
   792→    const { getAppSetting } = await import("@/lib/db")
   793→    const raw = await getAppSetting(`lark_tasklist_${projectId}`)
   794→    if (!raw) return
   795→
   796→    const parsed = JSON.parse(raw)
   797→    const tasklistId = parsed.tasklistId
   798→    if (!tasklistId) return
   799→
   800→    // Get user's Lark open_id
   801→    const openId = await getUserLarkOpenId(userId)
   802→    if (!openId) return
   803→
   804→    // Add as editor to the task list
   805→    await addTaskListMember(tasklistId, [{ id: openId, role: "editor" }])
   806→    console.log(`[lark/sync] Added user ${userId} (${openId}) to tasklist ${tasklistId}`)
   807→  } catch (err) {
   808→    console.error("[lark/sync] addProjectMemberToLarkTaskList failed:", err instanceof Error ? err.message : err)
   809→  }
   810→}

   812→/**
   813→ * Remove a TrishulHub project member from the corresponding Lark task list.
   814→ * Called fire-and-forget from the project members DELETE route.
   815→ */
   816→export async function removeProjectMemberFromLarkTaskList(
   817→  projectId: string,
   818→  userId: string
   819→): Promise<void> {
   820→  const config = await getLarkConfig()
   821→  if (!config?.enabled) return
   822→
   823→  try {
   824→    const { getAppSetting, delAppSetting } = await import("@/lib/db")
   825→    const raw = await getAppSetting(`lark_tasklist_${projectId}`)
   826→    if (!raw) return
   827→
   828→    const parsed = JSON.parse(raw)
   829→    const tasklistId = parsed.tasklistId
   830→    if (!tasklistId) return
   831→
   832→    // Get user's Lark open_id
   833→    const openId = await getUserLarkOpenId(userId)
   834→    if (!openId) return
   835→
   836→    // Remove from task list
   837→    await removeTaskListMember(tasklistId, [{ id: openId }])
   838→    console.log(`[lark/sync] Removed user ${userId} (${openId}) from tasklist ${tasklistId}`)
   839→
   840→    // Cleanup: check if project has no members left — if so, delete the tasklist
   841→    const memberCount = await db.projectMember.count({ where: { projectId } })
   842→    if (memberCount === 0) {
   843→      const { deleteTaskList } = await import("./client")
   844→      await deleteTaskList(tasklistId)
   845→      await delAppSetting(`lark_tasklist_${projectId}`)
   846→      console.log(`[lark/sync] Deleted empty tasklist ${tasklistId} for project ${projectId}`)
   847→    }
   848→  } catch (err) {
   849→    console.error("[lark/sync] removeProjectMemberFromLarkTaskList failed:", err instanceof Error ? err.message : err)
   850→  }
   851→}

   853→/**
   854→ * Find a project ID by looking up the tasklist mapping.
   855→ */
   856→async function findProjectByTaskList(tasklistId: string): Promise<string | null> {
   857→  try {
   858→    const rows = await db.$queryRawUnsafe<Array<{ key: string; value: string }>>(
   859→      'SELECT "key", "value" FROM "AppSetting" WHERE "key" LIKE ?',
   860→      'lark_tasklist_%'
   861→    )
   862→
   863→    for (const row of rows) {
   864→      try {
   865→        const parsed = JSON.parse(row.value)
   866→        if (parsed.tasklistId === tasklistId) {
   867→          const projectId = row.key.replace("lark_tasklist_", "")
   868→          if (projectId && projectId !== "__default__") {
   869→            const project = await db.project.findUnique({ where: { id: projectId }, select: { id: true } })
   870→            return project?.id || null
   871→          }
   872→        }
   873→      } catch {
   874→        continue
   875→      }
   876→    }
   877→  } catch {
   878→    // Ignore
   879→  }
   880→
   881→  return null
