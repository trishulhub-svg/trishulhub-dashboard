   205→function formatRelativeTime(dateStr: string): string {
   206→  if (!dateStr) return "unknown";
   207→  const date = new Date(dateStr);
   208→  if (isNaN(date.getTime())) return "unknown";
   209→  const now = new Date();
   210→  const diffMs = now.getTime() - date.getTime();
   211→  const diffMin = Math.floor(diffMs / 60000);
   212→  if (diffMin < 1) return "just now";
   213→  if (diffMin < 60) return `${diffMin}m ago`;
   214→  const diffHr = Math.floor(diffMin / 60);
   215→  if (diffHr < 24) return `${diffHr}h ago`;
   216→  const diffDay = Math.floor(diffHr / 24);
   217→  if (diffDay < 7) return `${diffDay}d ago`;
   218→  return safeDateStr(date);
   219→}

   896→                      {/* W8: Show indicator when notifications are capped at 50 */}

   907→
   908→            <DropdownMenu>
   909→              <DropdownMenuTrigger asChild>
   910→                <Button variant="ghost" className="flex items-center gap-2 h-8 sm:h-9">
   911→                  <Avatar className="h-8 w-8">
   912→                    <AvatarFallback className="bg-primary text-primary-foreground text-xs font-bold">
   913→                      {userName.split(" ").filter(Boolean).map((n) => n[0]).join("").toUpperCase().slice(0, 2)}
   914→                    </AvatarFallback>
   915→                  </Avatar>
   916→                  <span className="hidden sm:block text-sm font-medium">{userName}</span>
   917→                </Button>
   918→              </DropdownMenuTrigger>
   919→              <DropdownMenuContent align="end" className="w-56">

   921→                  <p className="text-sm font-medium">{userName}</p>

   941→        </header>
   942→
   943→        {/* Page Content - more padding */}

   946→
   947→      {/* Agentation — visual feedback tool (all users) */}
   948→      <Agentation />
   949→
   950→      {/* Floating Task Boards — accessible from all dashboard pages */}
   951→      <FloatingBoardRenderer />

   953→    </FloatingBoardProvider>

   955→}

   957→// ─── Bridge: Listens for logout event to auto-minimize floating boards ──
   958→function LogoutBridge() {
   959→  const { signalLogout } = useFloatingBoards();
   960→  useEffect(() => {
   961→    const handler = () => signalLogout();
   962→    window.addEventListener("trishulhub:logout", handler);
   963→    return () => window.removeEventListener("trishulhub:logout", handler);
   964→  }, [signalLogout]);
   965→  return null;
   966→}
