    21→// ━━ CIRCULAR SYNC GUARD ━━
    22→
    23→/** In-memory map: "taskId:direction" → timestamp. Prevents TH→Lark→TH→Lark loops.
    24→ *  Cooldown: 10 seconds.
    25→ */
    26→const _syncGuard = new Map<string, number>()
    27→const SYNC_GUARD_COOLDOWN_MS = 10_000
    28→
    29→function markSyncGuard(taskId: string, direction: string): void {
    30→  _syncGuard.set(`${taskId}:${direction}`, Date.now())
    31→}
    32→
    33→function isSyncGuarded(taskId: string, direction: string): boolean {
    34→  const ts = _syncGuard.get(`${taskId}:${direction}`)
    35→  if (!ts) return false
    36→  if (Date.now() - ts > SYNC_GUARD_COOLDOWN_MS) {
    37→    _syncGuard.delete(`${taskId}:${direction}`)
    38→    return false
    39→  }
    40→  return true
    41→}
    42→
    43→// ━━ WEBHOOK EVENT DEDUPLICATION ━━
    44→
    45→/** In-memory set of processed event_ids. Prevents duplicate processing from Lark retries.
    46→ *  Window: 30 seconds.
    47→ */
    48→const _processedEvents = new Map<string, number>()
    49→const DEDUP_WINDOW_MS = 30_000
    50→
    51→function isEventDuplicate(eventId: string): boolean {
    52→  const ts = _processedEvents.get(eventId)
    53→  if (!ts) return false
    54→  if (Date.now() - ts > DEDUP_WINDOW_MS) {
    55→    _processedEvents.delete(eventId)
    56→    return false
    57→  }
    58→  return true
    59→}
    60→
    61→function markEventProcessed(eventId: string): void {
    62→  _processedEvents.set(eventId, Date.now())
    63→}
    64→
    65→// Periodic cleanup of stale entries (every 60s)
    66→if (typeof setInterval !== "undefined") {
    67→  setInterval(() => {
    68→    const now = Date.now()
    69→    for (const [key, ts] of _syncGuard) {
    70→      if (now - ts > SYNC_GUARD_COOLDOWN_MS) _syncGuard.delete(key)
    71→    }
    72→    for (const [key, ts] of _processedEvents) {
    73→      if (now - ts > DEDUP_WINDOW_MS) _processedEvents.delete(key)
    74→    }
    75→  }, 60_000)
    76→}

   113→/**
   114→ * Get the Lark open_id for a TrishulHub user.
   115→ * Checks LarkUserMapping table first, then tries email lookup.
   116→ */
   117→export async function getUserLarkOpenId(userId: string): Promise<string | null> {
   118→  try {
   119→    // Check mapping table
   120→    const rows = await db.$queryRawUnsafe<Array<{ larkOpenId: string }>>(
   121→      'SELECT "larkOpenId" FROM "LarkUserMapping" WHERE "userId" = ?',
   122→      userId
   123→    )
   124→    if (rows.length > 0 && rows[0].larkOpenId) {
   125→      return rows[0].larkOpenId
   126→    }
   127→
   128→    // Try email lookup
   129→    const user = await db.user.findUnique({ where: { id: userId }, select: { email: true } })
   130→    if (user?.email) {
   131→      const larkUser = await lookupUserByEmail(user.email)
   132→      if (larkUser?.open_id) {
   133→        // Auto-save the mapping
   134→        await db.$executeRawUnsafe(
   135→          'INSERT OR IGNORE INTO "LarkUserMapping" ("id", "userId", "larkOpenId", "larkName", "larkEmail", "matchedBy", "createdAt") VALUES (?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)',
   136→          `map_${userId}_${Date.now()}`,
   137→          userId,
   138→          larkUser.open_id,
   139→          larkUser.name || "",
   140→          larkUser.email || "",
   141→          "email_auto"
   142→        )
   143→        return larkUser.open_id
   144→      }
   145→    }
   146→
   147→    return null
   148→  } catch (err) {
   149→    console.error("[lark/sync] Failed to get user Lark ID:", err)
   150→    return null
   151→  }
   152→}

   229→/**
   230→ * Sync a task creation from TrishulHub to Lark.
   231→ * Call this AFTER the task is saved in the DB.
   232→ */
   233→export async function syncTaskToLark(
   234→  taskId: string,
   235→  data: {
   236→    title: string
   237→    description?: string
   238→    status: string
   239→    priority: string
   240→    assignedTo?: string
   241→    projectId?: string
   242→    deadline?: string | Date
   243→  },
   244→  userId: string
   245→): Promise<void> {
   246→  const config = await getLarkConfig()
   247→  if (!config?.enabled) return
   248→
   249→  try {
   250→    // Get or create the task list for the project
   251→    let tasklistId: string | undefined
   252→
   253→    if (data.projectId) {
   254→      const project = await db.project.findUnique({ where: { id: data.projectId }, select: { name: true } })
   255→      if (project) {
   256→        // Try to get or create the project task list (retry once on failure)
   257→        let result = await getOrCreateProjectTaskList(data.projectId, project.name)
   258→        if (!result) {
   259→          // Retry once — transient API errors can cause first attempt to fail
   260→          console.warn(`[lark/sync] Retrying task list creation for project ${data.projectId} (${project.name})`)
   261→          result = await getOrCreateProjectTaskList(data.projectId, project.name)
   262→        }
   263→        if (result) {
   264→          tasklistId = result.tasklistId
   265→          if (result.created) {
   266→            console.log(`[lark/sync] Created new Lark task list for project: ${project.name}`)
   267→          }
   268→        } else {
   269→          console.error(`[lark/sync] FAILED to get/create task list for project ${data.projectId} (${project.name}) — aborting sync, task will NOT fall to default list`)
   270→          await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Failed to create project task list for "${project.name}"` })
   271→          // Don't fall back to default — if the task has a project, it MUST go to that project's list
   272→          return
   273→        }
   274→      } else {
   275→        // Project exists in task data but not found in DB — don't silently fall to default
   276→        console.error(`[lark/sync] Project ${data.projectId} not found in DB — aborting sync`)
   277→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Project ${data.projectId} not found in database` })
   278→        return
   279→      }
   280→    }
   281→
   282→    if (!tasklistId) {
   283→      // No project — use a default "TrishulHub Tasks" list
   284→      const defaultResult = await getOrCreateProjectTaskList("__default__", "TrishulHub Tasks")
   285→      if (!defaultResult) {
   286→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Failed to get/create default task list" })
   287→        return
   288→      }
   289→      tasklistId = defaultResult.tasklistId
   290→    }
   291→
   292→    // Resolve assignee's Lark open_id
   293→    let assigneeOpenId: string | undefined
   294→    if (data.assignedTo) {
   295→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   296→    }
   297→
   298→    // Convert deadline to unix ms
   299→    let dueTimestamp: number | undefined
   300→    if (data.deadline) {
   301→      dueTimestamp = new Date(data.deadline).getTime()
   302→    }
   303→
   304→    // Create task — bare creation only (no members, no status, no priority on create API)
   305→    const larkTask = await createTask(tasklistId, {
   306→      title: data.title,
   307→      description: data.description,
   308→      dueTimestamp,
   309→    })
   310→
   311→    if (larkTask) {
   312→      await saveLarkTaskMapping(taskId, larkTask.task_id, tasklistId)
   313→
   314→      // Set description, due, status & priority via separate update
   315→      // (create API only accepts tasklist_id + summary)
   316→      await updateTask(larkTask.task_id, {
   317→        description: data.description,
   318→        status: data.status,
   319→        priority: data.priority,
   320→        dueTimestamp,
   321→      })
   322→
   323→      // Assign member via add_members endpoint (create API doesn't support members)
   324→      if (assigneeOpenId) {
   325→        await addTaskMember(larkTask.task_id, [
   326→          { id: assigneeOpenId, role: "assignee" },
   327→        ])
   328→      }
   329→
   330→      // Add assignee as tasklist member (editor role) so the tasklist
   331→      // appears in the user's Lark Task Center
   332→      if (assigneeOpenId) {
   333→        await addTaskListMember(tasklistId, [
   334→          { id: assigneeOpenId, role: "editor" },
   335→        ])
   336→      }
   337→
   338→      // Mark circular guard to prevent webhook from echoing back
   339→      markSyncGuard(taskId, "TO_LARK")
   340→
   341→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "SUCCESS", taskId, larkTaskId: larkTask.task_id, larkTaskListId: tasklistId, projectId: data.projectId, userId })
   342→    } else {
   343→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Lark API returned null task" })
   344→    }
   345→  } catch (err) {
   346→    const msg = err instanceof Error ? err.message : String(err)
   347→    await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: msg })
   348→  }
   349→}

   351→/**
   352→ * Sync a task update from TrishulHub to Lark.
   353→ */
   354→export async function syncTaskUpdateToLark(
   355→  taskId: string,
   356→  data: {
   357→    title?: string
   358→    description?: string
   359→    status?: string
   360→    priority?: string
   361→    assignedTo?: string
   362→    deadline?: string | Date | null
   363→  },
   364→  userId: string
   365→): Promise<void> {
   366→  const config = await getLarkConfig()
   367→  if (!config?.enabled) return
   368→
   369→  // Circular sync guard: skip if we just synced FROM Lark for this task
   370→  if (isSyncGuarded(taskId, "FROM_LARK")) {
   371→    console.log(`[lark/sync] Skipping TO_LARK update for ${taskId} — circular guard active`)
   372→    return
   373→  }
   374→
   375→  try {
   376→    const mapping = await getLarkTaskMapping(taskId)
   377→    if (!mapping) {
   378→      // No mapping exists — try full sync (task may have been created before Lark was enabled)
   379→      const task = await db.task.findUnique({ where: { id: taskId } })
   380→      if (task) {
   381→        await syncTaskToLark(taskId, {
   382→          title: task.title,
   383→          description: task.description || undefined,
   384→          status: task.status,
   385→          priority: task.priority,
   386→          assignedTo: task.assignedTo || undefined,
   387→          projectId: task.projectId || undefined,
   388→          deadline: task.deadline || undefined,
   389→        }, userId)
   390→      }
   391→      return
   392→    }
   393→
   394→    // Resolve assignee's Lark open_id
   395→    let assigneeOpenId: string | undefined
   396→    if (data.assignedTo) {
   397→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   398→    }
   399→
   400→    // Convert deadline to unix ms
   401→    let dueTimestamp: number | undefined
   402→    if (data.deadline) {
   403→      dueTimestamp = new Date(data.deadline).getTime()
   404→    }
   405→
   406→    // Update task — NO extra, NO assignee in body (use member APIs instead)
   407→    const larkTask = await updateTask(mapping.larkTaskId, {
   408→      title: data.title,
   409→      description: data.description,
   410→      status: data.status,
   411→      priority: data.priority,
   412→      dueTimestamp,
   413→    })
   414→
   415→    // Handle assignee change via member API
   416→    if (data.assignedTo !== undefined && assigneeOpenId) {
   417→      await addTaskMember(mapping.larkTaskId, [
   418→        { id: assigneeOpenId, role: "assignee" },
   419→      ])
   420→    }
   421→
   422→    if (larkTask) {
   423→      // Mark circular guard
   424→      markSyncGuard(taskId, "TO_LARK")
   425→
   426→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   427→    } else {
   428→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark API returned null" })
   429→    }
   430→  } catch (err) {
   431→    const msg = err instanceof Error ? err.message : String(err)
   432→    await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, userId, error: msg })
   433→  }
   434→}

   439→export async function syncTaskDeleteToLark(taskId: string, userId: string): Promise<void> {

   456→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark delete API failed" })
   457→    }
   458→  } catch (err) {
   459→    const msg = err instanceof Error ? err.message : String(err)
   460→    await logSync({ direction: "TO_LARK", action: "DELETE", status: "FAILED", taskId, userId, error: msg })
   461→  }

   464→// ━━ LARK → TRISHULHUB SYNC (Webhook-driven) ━━
   465→
   466→/**
   467→ * Handle incoming Lark webhook events.
   468→ * @param eventId — Lark event_id for deduplication
   469→ * @param eventType — Lark event type
   470→ * @param eventData — Parsed event payload
   471→ */
   472→export async function handleLarkWebhookEvent(
   473→  eventId: string,
   474→  eventType: string,
   475→  eventData: {
   476→    task_id: string
   477→    tasklist_id: string
   478→    operator?: { open_id: string }
   479→    [key: string]: unknown
   480→  }
   481→): Promise<{ success: boolean; message: string }> {
   482→  const config = await getLarkConfig()
   483→  if (!config?.enabled) {
   484→    return { success: false, message: "Lark sync is disabled" }
   485→  }
   486→
   487→  // ── Dedup check ──
   488→  if (isEventDuplicate(eventId)) {
   489→    console.log(`[lark/sync] Dedup: skipping duplicate event ${eventId}`)
   490→    return { success: true, message: "Duplicate event, skipped" }
   491→  }
   492→  markEventProcessed(eventId)
   493→
   494→  const { task_id, tasklist_id } = eventData
   495→  console.log(`[lark/sync] Processing event: ${eventType} (id: ${eventId}) for task: ${task_id}`)
   496→  console.log(`[lark/sync] Event details: tasklist_id=${tasklist_id}, operator=${eventData.operator?.open_id || 'none'}`)
   497→
   498→  try {
   499→    // Find existing mapping
   500→    const thTaskId = await getTaskIdByLarkId(task_id)
   501→    console.log(`[lark/sync] Mapping lookup: larkTaskId=${task_id} → thTaskId=${thTaskId || 'NOT FOUND'}`)
   502→
   503→    if (eventType === "task.task.deleted") {
   504→      if (!thTaskId) {
   505→        return { success: true, message: "Task not mapped, ignoring delete" }
   506→      }
   507→      // Delete the TrishulHub task
   508→      await db.task.delete({ where: { id: thTaskId } })
   509→      await removeLarkTaskMapping(thTaskId)
   510→
   511→      try {
   512→        const { logAudit } = await import("@/lib/audit-log")
   513→        await logAudit({
   514→          userId: eventData.operator?.open_id || "lark-webhook",
   515→          userName: "Lark Webhook",
   516→          userRole: "SYSTEM",
   517→          department: "TEAM_WORK",
   518→          page: "tasks",
   519→          action: "DELETE",
   520→          entityType: "Task",
   521→          entityId: thTaskId,
   522→          description: `Task deleted via Lark sync`,
   523→          status: "SUCCESS",
   524→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id }),
   525→        })
   526→      } catch { /* audit log is best-effort */ }
   527→
   528→      return { success: true, message: "Task deleted from TrishulHub" }
   529→    }
   530→
   531→    // For create/update/complete, we need the full task data from Lark
   532→    // Add 1s delay to give Lark eventual-consistency time to propagate
   533→    await new Promise((resolve) => setTimeout(resolve, 1000))
   534→
   535→    const larkTask = await getTask(tasklist_id, task_id)
   536→    if (larkTask) {
   537→      console.log(`[lark/sync] Fetched from Lark: title="${larkTask.title}", status="${larkTask.status}", priority="${larkTask.priority}", assignee="${larkTask.assignee?.open_id || 'none'}"`)
   538→    } else {
   539→      console.warn(`[lark/sync] getTask returned null for task ${task_id} in list ${tasklist_id}`)
   540→    }
   541→
   542→    // Fallback: if getTask returns null (e.g. completed tasks return 99404 on free tier),
   543→    // retry once after a short delay, then inspect event body for status clues.
   544→    if (!larkTask) {
   545→      if (thTaskId) {
   546→        // Retry getTask once after 2s — Lark eventual consistency may need more time
   547→        console.warn(`[lark/sync] getTask returned null for mapped task ${thTaskId}, retrying...`)
   548→        await new Promise((resolve) => setTimeout(resolve, 2000))
   549→        const retryTask = await getTask(tasklist_id, task_id)
   550→
   551→        if (retryTask) {
   552→          // Retry succeeded — process normally via the update path below
   553→          // Rewrite larkTask so the code below handles it
   554→          Object.assign(eventData, { _retryTask: retryTask })
   555→          // Fall through to the main update logic — we'll read from retryTask
   556→          const title = retryTask.title
   557→          const description = retryTask.description
   558→          const status = STATUS_FROM_LARK[retryTask.status] || "TODO"
   559→          const priority = PRIORITY_FROM_LARK[retryTask.priority] || "MEDIUM"
   560→
   561→          let assignedTo: string | undefined
   562→          if (retryTask.assignee?.open_id) {
   563→            assignedTo = await getUserIdByLarkOpenId(retryTask.assignee.open_id) || undefined
   564→          }
   565→
   566→          const deadline = retryTask.due?.timestamp ? new Date(parseInt(retryTask.due.timestamp)) : null
   567→
   568→          const updateData: Record<string, unknown> = {}
   569→          if (title) updateData.title = title
   570→          if (description !== undefined) updateData.description = description
   571→          updateData.status = status
   572→          updateData.priority = priority
   573→          if (assignedTo !== undefined) updateData.assignedTo = assignedTo
   574→          if (deadline !== null) updateData.deadline = deadline
   575→          if (status === "DONE") updateData.completedAt = new Date()
   576→
   577→          await db.task.update({ where: { id: thTaskId }, data: updateData })
   578→          markSyncGuard(thTaskId, "FROM_LARK")
   579→
   580→          const { logAudit } = await import("@/lib/audit-log")
   581→          await logAudit({
   582→            userId: eventData.operator?.open_id || "lark-webhook",
   583→            userName: "Lark Webhook",
   584→            userRole: "SYSTEM",
   585→            department: "TEAM_WORK",
   586→            page: "tasks",
   587→            action: "STATUS_CHANGE",
   588→            entityType: "Task",
   589→            entityId: thTaskId,
   590→            description: `Task status changed to "${status}" via Lark sync (retried)`,
   591→            newValue: status,
   592→            status: "SUCCESS",
   593→            metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, larkStatus: retryTask.status }),
   594→          })
   595→
   596→          await logSync({
   597→            direction: "FROM_LARK",
   598→            action: "STATUS_CHANGE",
   599→            status: "SUCCESS",
   600→            taskId: thTaskId,
   601→            larkTaskId: task_id,
   602→            larkTaskListId: tasklist_id,
   603→            metadata: { inferredStatus: status, eventType, retried: true },
   604→          })
   605→          return { success: true, message: `Task ${thTaskId} updated to ${status} (retried getTask)` }
   606→        }
   607→
   608→        // Both getTask calls failed — inspect event body for status change clues.
   609→        const eventBody = (eventData as Record<string, unknown>)
   610→        const changedFields = eventBody.changed_fields as string[] | undefined
   611→
   612→        // Determine if this is likely a completion:
   613→        // 1. Event type contains done/complete/finish keywords
   614→        // 2. OR changed_fields includes "status" (something changed, likely to done since getTask fails)
   615→        const likelyCompleted =
   616→          eventType.includes("done") || eventType.includes("complete") || eventType.includes("finish") ||
   617→          (Array.isArray(changedFields) && changedFields.includes("status"))
   618→
   619→        const newStatus = likelyCompleted ? "DONE" : "IN_PROGRESS";
   620→        const fallbackData: Record<string, unknown> = { status: newStatus };
   621→        if (newStatus === "DONE") fallbackData.completedAt = new Date();
   622→
   623→        const existingTask = await db.task.findUnique({ where: { id: thTaskId }, select: { status: true } });
   624→        console.warn(`[lark/sync] getTask returned null twice for mapped task ${thTaskId}. Event: ${eventType}. Assuming status="${newStatus}" (was "${existingTask?.status || 'unknown'}")`);
   625→
   626→        await db.task.update({ where: { id: thTaskId }, data: fallbackData })
   627→        markSyncGuard(thTaskId, "FROM_LARK")
   628→
   629→        const { logAudit } = await import("@/lib/audit-log")
   630→        await logAudit({
   631→          userId: eventData.operator?.open_id || "lark-webhook",
   632→          userName: "Lark Webhook",
   633→          userRole: "SYSTEM",
   634→          department: "TEAM_WORK",
   635→          page: "tasks",
   636→          action: "STATUS_CHANGE",
   637→          entityType: "Task",
   638→          entityId: thTaskId,
   639→          description: `Task status changed to "${newStatus}" via Lark sync (inferred — getTask failed)`,
   640→          newValue: newStatus,
   641→          status: "SUCCESS",
   642→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, inferred: true, previousStatus: existingTask?.status }),
   643→        })
   644→
   645→        await logSync({
   646→          direction: "FROM_LARK",
   647→          action: "STATUS_CHANGE",
   648→          status: "SUCCESS",
   649→          taskId: thTaskId,
   650→          larkTaskId: task_id,
   651→          larkTaskListId: tasklist_id,
   652→          metadata: { inferredStatus: newStatus, eventType, retried: true },
   653→        })
   654→        return { success: true, message: `Task ${thTaskId} updated to ${newStatus} (getTask null, inferred from event + changed_fields)` }
   655→      }
   656→      return { success: false, message: "Failed to fetch task from Lark and no existing mapping" }
   657→    }
   658→
   659→    // Circular sync guard: skip if we just synced TO_LARK for this task
   660→    if (thTaskId && isSyncGuarded(thTaskId, "TO_LARK")) {
   661→      console.log(`[lark/sync] Skipping FROM_LARK for ${thTaskId} — circular guard active`)
   662→      return { success: true, message: "Circular guard: skipped" }
   663→    }
   664→
   665→    // Map Lark task data to TrishulHub format
   666→    const title = larkTask.title
   667→    const description = larkTask.description
   668→    const status = STATUS_FROM_LARK[larkTask.status] || "TODO"
   669→    console.log(`[lark/sync] Status mapping: lark="${larkTask.status}" → th="${status}"`)
   670→    const priority = PRIORITY_FROM_LARK[larkTask.priority] || "MEDIUM"
   671→
   672→    // Resolve assignee
   673→    let assignedTo: string | undefined
   674→    if (larkTask.assignee?.open_id) {
   675→      assignedTo = await getUserIdByLarkOpenId(larkTask.assignee.open_id) || undefined
   676→    }
   677→
   678→    // Resolve due date
   679→    const deadline = larkTask.due?.timestamp ? new Date(parseInt(larkTask.due.timestamp)) : null
   680→
   681→    if (thTaskId) {
   682→      // UPDATE existing task
   683→      const updateData: Record<string, unknown> = {}
   684→      if (title) updateData.title = title
   685→      if (description !== undefined) updateData.description = description
   686→      updateData.status = status
   687→      updateData.priority = priority
   688→      if (assignedTo !== undefined) updateData.assignedTo = assignedTo
   689→      if (deadline !== null) updateData.deadline = deadline
   690→
   691→      // Set completedAt if status changed to DONE
   692→      if (status === "DONE") {
   693→        updateData.completedAt = new Date()
   694→      }
   695→
   696→      await db.task.update({ where: { id: thTaskId }, data: updateData })
   697→      console.log(`[lark/sync] Updated TH task ${thTaskId}: status=${updateData.status}, priority=${updateData.priority}, assignedTo=${updateData.assignedTo || 'unchanged'}`)
   698→
   699→      // Mark circular guard so our update handler doesn't echo back
   700→      markSyncGuard(thTaskId, "FROM_LARK")
   701→
   702→      // Log to both LarkSyncLog and AuditLog
   703→      await logSync({
   704→        direction: "FROM_LARK",
   705→        action: status === "DONE" ? "STATUS_CHANGE" : "UPDATE",
   706→        status: "SUCCESS",
   707→        taskId: thTaskId,
   708→        larkTaskId: task_id,
   709→        larkTaskListId: tasklist_id,
   710→      })
   711→
   712→      try {
   713→        const { logAudit } = await import("@/lib/audit-log")
   714→        await logAudit({
   715→          userId: eventData.operator?.open_id || "lark-webhook",
   716→          userName: "Lark Webhook",
   717→          userRole: "SYSTEM",
   718→          department: "TEAM_WORK",
   719→          page: "tasks",
   720→          action: status === "DONE" ? "STATUS_CHANGE" : "UPDATE",
   721→          entityType: "Task",
   722→          entityId: thTaskId,
   723→          description: `Task ${status === "DONE" ? "marked as DONE" : "updated"} via Lark sync`,
   724→          newValue: status,
   725→          status: "SUCCESS",
   726→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, larkStatus: larkTask.status }),
   727→        })
   728→      } catch { /* audit log is best-effort */ }
   729→
   730→      return { success: true, message: `Task ${thTaskId} updated from Lark` }
   731→    } else {
   732→      // CREATE new task (originated in Lark)
   733→      const newTask = await db.task.create({
   734→        data: {
   735→          title,
   736→          description: description || null,
   737→          status,
   738→          priority,
   739→          assignedTo: assignedTo || null,
   740→          deadline,
   741→          assigneeType: "HUMAN",
   742→          category: "GENERAL",
   743→          projectId: await findProjectByTaskList(tasklist_id),
   744→        },
   745→      })
   746→
   747→      // Save mapping
   748→      await saveLarkTaskMapping(newTask.id, task_id, tasklist_id)
   749→
   750→      // Mark guard for the newly created task
   751→      markSyncGuard(newTask.id, "FROM_LARK")
   752→
   753→      try {
   754→        const { logAudit } = await import("@/lib/audit-log")
   755→        await logAudit({
   756→          userId: eventData.operator?.open_id || "lark-webhook",
   757→          userName: "Lark Webhook",
   758→          userRole: "SYSTEM",
   759→          department: "TEAM_WORK",
   760→          page: "tasks",
   761→          action: "CREATE",
   762→          entityType: "Task",
   763→          entityId: newTask.id,
   764→          description: `Task created from Lark: "${title}"`,
   765→          newValue: status,
   766→          status: "SUCCESS",
   767→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id }),
   768→        })
   769→      } catch { /* audit log is best-effort */ }
   770→
   771→      return { success: true, message: `New task ${newTask.id} created from Lark` }
   772→    }
   773→  } catch (err) {
   774→    const msg = err instanceof Error ? err.message : String(err)
   775→    console.error("[lark/sync] Webhook handling error:", msg)
   776→    return { success: false, message: msg }
   777→  }
   778→}

   784→ * Called fire-and-forget from the project members POST route.
   785→ */
   786→export async function addProjectMemberToLarkTaskList(
   787→  projectId: string,
   788→  userId: string
   789→): Promise<void> {
   790→  const config = await getLarkConfig()
   791→  if (!config?.enabled) return
   792→

   794→    // Find the tasklist mapping for this project
   795→    const { getAppSetting } = await import("@/lib/db")
   796→    const raw = await getAppSetting(`lark_tasklist_${projectId}`)
   797→    if (!raw) return
   798→
   799→    const parsed = JSON.parse(raw)
   800→    const tasklistId = parsed.tasklistId
   801→    if (!tasklistId) return
   802→
   803→    // Get user's Lark open_id
   804→    const openId = await getUserLarkOpenId(userId)
   805→    if (!openId) return
   806→
   807→    // Add as editor to the task list
   808→    await addTaskListMember(tasklistId, [{ id: openId, role: "editor" }])
   809→    console.log(`[lark/sync] Added user ${userId} (${openId}) to tasklist ${tasklistId}`)
   810→  } catch (err) {
   811→    console.error("[lark/sync] addProjectMemberToLarkTaskList failed:", err instanceof Error ? err.message : err)
   812→  }
   813→}

   816→ * Remove a TrishulHub project member from the corresponding Lark task list.
   817→ * Called fire-and-forget from the project members DELETE route.
   818→ */
   819→export async function removeProjectMemberFromLarkTaskList(
   820→  projectId: string,
   821→  userId: string
   822→): Promise<void> {
   823→  const config = await getLarkConfig()
   824→  if (!config?.enabled) return
   825→
   826→  try {
   827→    const { getAppSetting, delAppSetting } = await import("@/lib/db")
   828→    const raw = await getAppSetting(`lark_tasklist_${projectId}`)
   829→    if (!raw) return
   830→
   831→    const parsed = JSON.parse(raw)
   832→    const tasklistId = parsed.tasklistId
   833→    if (!tasklistId) return
   834→
   835→    // Get user's Lark open_id
   836→    const openId = await getUserLarkOpenId(userId)
   837→    if (!openId) return
   838→
   839→    // Remove from task list
   840→    await removeTaskListMember(tasklistId, [{ id: openId }])
   841→    console.log(`[lark/sync] Removed user ${userId} (${openId}) from tasklist ${tasklistId}`)
   842→
   843→    // Cleanup: check if project has no members left — if so, delete the tasklist
   844→    const memberCount = await db.projectMember.count({ where: { projectId } })
   845→    if (memberCount === 0) {
   846→      const { deleteTaskList } = await import("./client")
   847→      await deleteTaskList(tasklistId)
   848→      await delAppSetting(`lark_tasklist_${projectId}`)
   849→      console.log(`[lark/sync] Deleted empty tasklist ${tasklistId} for project ${projectId}`)
   850→    }
   851→  } catch (err) {
   852→    console.error("[lark/sync] removeProjectMemberFromLarkTaskList failed:", err instanceof Error ? err.message : err)
   853→  }
   854→}

   856→/**
   857→ * Find a project ID by looking up the tasklist mapping.
   858→ */
   859→async function findProjectByTaskList(tasklistId: string): Promise<string | null> {
   860→  try {
   861→    const rows = await db.$queryRawUnsafe<Array<{ key: string; value: string }>>(
   862→      'SELECT "key", "value" FROM "AppSetting" WHERE "key" LIKE ?',
   863→      'lark_tasklist_%'
   864→    )
   865→
   866→    for (const row of rows) {
   867→      try {
   868→        const parsed = JSON.parse(row.value)
   869→        if (parsed.tasklistId === tasklistId) {
   870→          const projectId = row.key.replace("lark_tasklist_", "")
   871→          if (projectId && projectId !== "__default__") {
   872→            const project = await db.project.findUnique({ where: { id: projectId }, select: { id: true } })
   873→            return project?.id || null
   874→          }
   875→        }
   876→      } catch {
   877→        continue
   878→      }
   879→    }
   880→  } catch {
   881→    // Ignore
   882→  }
   883→
   884→  return null
