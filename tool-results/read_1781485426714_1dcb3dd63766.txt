    22→    22→// ━━ CIRCULAR SYNC GUARD ━━
    23→    23→
    24→    24→/** In-memory map: "taskId:direction" → timestamp. Prevents TH→Lark→TH→Lark loops.
    25→    25→ *  Cooldown: 10 seconds.
    26→    26→ */
    27→    27→const _syncGuard = new Map<string, number>()
    28→    28→const SYNC_GUARD_COOLDOWN_MS = 10_000
    29→    29→
    30→    30→function markSyncGuard(taskId: string, direction: string): void {
    31→    31→  _syncGuard.set(`${taskId}:${direction}`, Date.now())
    32→    32→}
    33→    33→
    34→    34→function isSyncGuarded(taskId: string, direction: string): boolean {
    35→    35→  const ts = _syncGuard.get(`${taskId}:${direction}`)
    36→    36→  if (!ts) return false
    37→    37→  if (Date.now() - ts > SYNC_GUARD_COOLDOWN_MS) {
    38→    38→    _syncGuard.delete(`${taskId}:${direction}`)
    39→    39→    return false
    40→    40→  }
    41→    41→  return true
    42→    42→}
    43→    43→
    44→    44→// ━━ WEBHOOK EVENT DEDUPLICATION ━━
    45→    45→
    46→    46→/** In-memory set of processed event_ids. Prevents duplicate processing from Lark retries.
    47→    47→ *  Window: 30 seconds.
    48→    48→ */
    49→    49→const _processedEvents = new Map<string, number>()
    50→    50→const DEDUP_WINDOW_MS = 30_000
    51→    51→
    52→    52→function isEventDuplicate(eventId: string): boolean {
    53→    53→  const ts = _processedEvents.get(eventId)
    54→    54→  if (!ts) return false
    55→    55→  if (Date.now() - ts > DEDUP_WINDOW_MS) {
    56→    56→    _processedEvents.delete(eventId)
    57→    57→    return false
    58→    58→  }
    59→    59→  return true
    60→    60→}
    61→    61→
    62→    62→function markEventProcessed(eventId: string): void {
    63→    63→  _processedEvents.set(eventId, Date.now())
    64→    64→}
    65→    65→
    66→    66→// Periodic cleanup of stale entries (every 60s)
    67→    67→if (typeof setInterval !== "undefined") {
    68→    68→  setInterval(() => {
    69→    69→    const now = Date.now()
    70→    70→    for (const [key, ts] of _syncGuard) {
    71→    71→      if (now - ts > SYNC_GUARD_COOLDOWN_MS) _syncGuard.delete(key)
    72→    72→    }
    73→    73→    for (const [key, ts] of _processedEvents) {
    74→    74→      if (now - ts > DEDUP_WINDOW_MS) _processedEvents.delete(key)
    75→    75→    }
    76→    76→  }, 60_000)
    77→    77→}

   114→   114→/**
   115→   115→ * Get the Lark open_id for a TrishulHub user.
   116→   116→ * Checks LarkUserMapping table first, then tries email lookup.
   117→   117→ */
   118→   118→export async function getUserLarkOpenId(userId: string): Promise<string | null> {
   119→   119→  try {
   120→   120→    // Check mapping table
   121→   121→    const rows = await db.$queryRawUnsafe<Array<{ larkOpenId: string }>>(
   122→   122→      'SELECT "larkOpenId" FROM "LarkUserMapping" WHERE "userId" = ?',
   123→   123→      userId
   124→   124→    )
   125→   125→    if (rows.length > 0 && rows[0].larkOpenId) {
   126→   126→      return rows[0].larkOpenId
   127→   127→    }
   128→   128→
   129→   129→    // Try email lookup
   130→   130→    const user = await db.user.findUnique({ where: { id: userId }, select: { email: true } })
   131→   131→    if (user?.email) {
   132→   132→      const larkUser = await lookupUserByEmail(user.email)
   133→   133→      if (larkUser?.open_id) {
   134→   134→        // Auto-save the mapping
   135→   135→        await db.$executeRawUnsafe(
   136→   136→          'INSERT OR IGNORE INTO "LarkUserMapping" ("id", "userId", "larkOpenId", "larkName", "larkEmail", "matchedBy", "createdAt") VALUES (?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)',
   137→   137→          `map_${userId}_${Date.now()}`,
   138→   138→          userId,
   139→   139→          larkUser.open_id,
   140→   140→          larkUser.name || "",
   141→   141→          larkUser.email || "",
   142→   142→          "email_auto"
   143→   143→        )
   144→   144→        return larkUser.open_id
   145→   145→      }
   146→   146→    }
   147→   147→
   148→   148→    return null
   149→   149→  } catch (err) {
   150→   150→    console.error("[lark/sync] Failed to get user Lark ID:", err)
   151→   151→    return null
   152→   152→  }
   153→   153→}

   230→   230→/**
   231→   231→ * Sync a task creation from TrishulHub to Lark.
   232→   232→ * Call this AFTER the task is saved in the DB.
   233→   233→ */
   234→   234→export async function syncTaskToLark(
   235→   235→  taskId: string,
   236→   236→  data: {
   237→   237→    title: string
   238→   238→    description?: string
   239→   239→    status: string
   240→   240→    priority: string
   241→   241→    assignedTo?: string
   242→   242→    projectId?: string
   243→   243→    deadline?: string | Date
   244→   244→  },
   245→   245→  userId: string,
   246→   246→  userName?: string
   247→   247→): Promise<void> {
   248→   248→  const config = await getLarkConfig()
   249→   249→  if (!config?.enabled) return
   250→   250→
   251→   251→  try {
   252→   252→    // Get or create the per-user task list for the project
   253→   253→    let tasklistId: string | undefined
   254→   254→
   255→   255→    if (data.projectId) {
   256→   256→      const project = await db.project.findUnique({ where: { id: data.projectId }, select: { name: true } })
   257→   257→      if (project) {
   258→   258→        // Use per-user task list: "ProjectName — UserName"
   259→   259→        const creatorName = userName || (await db.user.findUnique({ where: { id: userId }, select: { name: true } }))?.name || userId
   260→   260→        let result = await getOrCreateUserProjectTaskList(data.projectId, project.name, userId, creatorName)
   261→   261→        if (!result) {
   262→   262→          // Retry once — transient API errors can cause first attempt to fail
   263→   263→          console.warn(`[lark/sync] Retrying per-user task list creation for project ${data.projectId} (${project.name}) user ${creatorName}`)
   264→   264→          result = await getOrCreateUserProjectTaskList(data.projectId, project.name, userId, creatorName)
   265→   265→        }
   266→   266→        if (result) {
   267→   267→          tasklistId = result.tasklistId
   268→   268→          if (result.created) {
   269→   269→            console.log(`[lark/sync] Created new per-user Lark task list: ${project.name} — ${creatorName}`)
   270→   270→          }
   271→   271→        } else {
   272→   272→          console.error(`[lark/sync] FAILED to get/create per-user task list for project ${data.projectId} user ${userId} — aborting sync`)
   273→   273→          await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Failed to create per-user task list for "${project.name}"` })
   274→   274→          return
   275→   275→        }
   276→   276→      } else {
   277→   277→        console.error(`[lark/sync] Project ${data.projectId} not found in DB — aborting sync`)
   278→   278→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Project ${data.projectId} not found in database` })
   279→   279→        return
   280→   280→      }
   281→   281→    }
   282→   282→
   283→   283→    if (!tasklistId) {
   284→   284→      // No project — use a default "TrishulHub Tasks" list
   285→   285→      const creatorName = userName || (await db.user.findUnique({ where: { id: userId }, select: { name: true } }))?.name || userId
   286→   286→      const defaultResult = await getOrCreateUserProjectTaskList("__default__", "TrishulHub Tasks", userId, creatorName)
   287→   287→      if (!defaultResult) {
   288→   288→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Failed to get/create default task list" })
   289→   289→        return
   290→   290→      }
   291→   291→      tasklistId = defaultResult.tasklistId
   292→   292→    }
   293→   293→
   294→   294→    // Resolve assignee's Lark open_id
   295→   295→    let assigneeOpenId: string | undefined
   296→   296→    if (data.assignedTo) {
   297→   297→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   298→   298→    }
   299→   299→
   300→   300→    // Convert deadline to unix ms
   301→   301→    let dueTimestamp: number | undefined
   302→   302→    if (data.deadline) {
   303→   303→      dueTimestamp = new Date(data.deadline).getTime()
   304→   304→    }
   305→   305→
   306→   306→    // Create task — bare creation only (no members, no status, no priority on create API)
   307→   307→    const larkTask = await createTask(tasklistId, {
   308→   308→      title: data.title,
   309→   309→      description: data.description,
   310→   310→      dueTimestamp,
   311→   311→    })
   312→   312→
   313→   313→    if (larkTask) {
   314→   314→      await saveLarkTaskMapping(taskId, larkTask.task_id, tasklistId)
   315→   315→
   316→   316→      // Set description, due, status & priority via separate update
   317→   317→      // (create API only accepts tasklist_id + summary)
   318→   318→      await updateTask(larkTask.task_id, {
   319→   319→        description: data.description,
   320→   320→        status: data.status,
   321→   321→        priority: data.priority,
   322→   322→        dueTimestamp,
   323→   323→      })
   324→   324→
   325→   325→      // Assign member via add_members endpoint (create API doesn't support members)
   326→   326→      if (assigneeOpenId) {
   327→   327→        await addTaskMember(larkTask.task_id, [
   328→   328→          { id: assigneeOpenId, role: "assignee" },
   329→   329→        ])
   330→   330→      }
   331→   331→
   332→   332→      // Add assignee as tasklist member (editor role) so the tasklist
   333→   333→      // appears in the user's Lark Task Center
   334→   334→      if (assigneeOpenId) {
   335→   335→        await addTaskListMember(tasklistId, [
   336→   336→          { id: assigneeOpenId, role: "editor" },
   337→   337→        ])
   338→   338→      }
   339→   339→
   340→   340→      // Mark circular guard to prevent webhook from echoing back
   341→   341→      markSyncGuard(taskId, "TO_LARK")
   342→   342→
   343→   343→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "SUCCESS", taskId, larkTaskId: larkTask.task_id, larkTaskListId: tasklistId, projectId: data.projectId, userId })
   344→   344→    } else {
   345→   345→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Lark API returned null task" })
   346→   346→    }
   347→   347→  } catch (err) {
   348→   348→    const msg = err instanceof Error ? err.message : String(err)
   349→   349→    await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: msg })
   350→   350→  }
   351→   351→}

   353→   353→/**
   354→   354→ * Sync a task update from TrishulHub to Lark.
   355→   355→ */
   356→   356→export async function syncTaskUpdateToLark(
   357→   357→  taskId: string,
   358→   358→  data: {
   359→   359→    title?: string
   360→   360→    description?: string
   361→   361→    status?: string
   362→   362→    priority?: string
   363→   363→    assignedTo?: string
   364→   364→    deadline?: string | Date | null
   365→   365→  },
   366→   366→  userId: string
   367→   367→): Promise<void> {
   368→   368→  const config = await getLarkConfig()
   369→   369→  if (!config?.enabled) return
   370→   370→
   371→   371→  // Circular sync guard: skip if we just synced FROM Lark for this task
   372→   372→  if (isSyncGuarded(taskId, "FROM_LARK")) {
   373→   373→    console.log(`[lark/sync] Skipping TO_LARK update for ${taskId} — circular guard active`)
   374→   374→    return
   375→   375→  }
   376→   376→
   377→   377→  try {
   378→   378→    const mapping = await getLarkTaskMapping(taskId)
   379→   379→    if (!mapping) {
   380→   380→      // No mapping exists — try full sync (task may have been created before Lark was enabled)
   381→   381→      const task = await db.task.findUnique({ where: { id: taskId } })
   382→   382→      if (task) {
   383→   383→        await syncTaskToLark(taskId, {
   384→   384→          title: task.title,
   385→   385→          description: task.description || undefined,
   386→   386→          status: task.status,
   387→   387→          priority: task.priority,
   388→   388→          assignedTo: task.assignedTo || undefined,
   389→   389→          projectId: task.projectId || undefined,
   390→   390→          deadline: task.deadline || undefined,
   391→   391→        }, userId)
   392→   392→      }
   393→   393→      return
   394→   394→    }
   395→   395→
   396→   396→    // Resolve assignee's Lark open_id
   397→   397→    let assigneeOpenId: string | undefined
   398→   398→    if (data.assignedTo) {
   399→   399→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   400→   400→    }
   401→   401→
   402→   402→    // Convert deadline to unix ms
   403→   403→    let dueTimestamp: number | undefined
   404→   404→    if (data.deadline) {
   405→   405→      dueTimestamp = new Date(data.deadline).getTime()
   406→   406→    }
   407→   407→
   408→   408→    // Update task — NO extra, NO assignee in body (use member APIs instead)
   409→   409→    const larkTask = await updateTask(mapping.larkTaskId, {
   410→   410→      title: data.title,
   411→   411→      description: data.description,
   412→   412→      status: data.status,
   413→   413→      priority: data.priority,
   414→   414→      dueTimestamp,
   415→   415→    })
   416→   416→
   417→   417→    // Handle assignee change via member API
   418→   418→    if (data.assignedTo !== undefined && assigneeOpenId) {
   419→   419→      await addTaskMember(mapping.larkTaskId, [
   420→   420→        { id: assigneeOpenId, role: "assignee" },
   421→   421→      ])
   422→   422→    }
   423→   423→
   424→   424→    if (larkTask) {
   425→   425→      // Mark circular guard
   426→   426→      markSyncGuard(taskId, "TO_LARK")
   427→   427→
   428→   428→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   429→   429→    } else {
   430→   430→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark API returned null" })
   431→   431→    }
   432→   432→  } catch (err) {
   433→   433→    const msg = err instanceof Error ? err.message : String(err)
   434→   434→    await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, userId, error: msg })
   435→   435→  }
   436→   436→}

   439→   439→ * Sync a task deletion from TrishulHub to Lark.
   440→   440→ */
   441→   441→export async function syncTaskDeleteToLark(taskId: string, userId: string): Promise<void> {
   442→   442→  const config = await getLarkConfig()
   443→   443→  if (!config?.enabled) return
   444→   444→
   445→   445→  try {
   446→   446→    const mapping = await getLarkTaskMapping(taskId)
   447→   447→    if (!mapping) {
   448→   448→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "SKIPPED", taskId, userId, error: "No Lark mapping found" })
   449→   449→      return
   450→   450→    }
   451→   451→
   452→   452→    const success = await larkDeleteTask(mapping.larkTaskListId, mapping.larkTaskId)
   453→   453→    await removeLarkTaskMapping(taskId)
   454→   454→
   455→   455→    if (success) {
   456→   456→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   457→   457→    } else {
   458→   458→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark delete API failed" })

   528→   528→      } catch { /* audit log is best-effort */ }

   595→   595→            metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, larkStatus: retryTask.status }),

   676→   676→    if (larkTask.assignee?.open_id) {

   678→   678→    }
   679→   679→

   681→   681→    const deadline = larkTask.due?.timestamp ? new Date(parseInt(larkTask.due.timestamp)) : null

   684→   684→      // UPDATE existing task
   685→   685→      const updateData: Record<string, unknown> = {}
   686→   686→      if (title) updateData.title = title
   687→   687→      if (description !== undefined) updateData.description = description

   695→   695→        updateData.completedAt = new Date()
   696→   696→      }
   697→   697→
   698→   698→      await db.task.update({ where: { id: thTaskId }, data: updateData })
   699→   699→      console.log(`[lark/sync] Updated TH task ${thTaskId}: status=${updateData.status}, priority=${updateData.priority}, assignedTo=${updateData.assignedTo || 'unchanged'}`)

   731→   731→
   732→   732→      return { success: true, message: `Task ${thTaskId} updated from Lark` }
   733→   733→    } else {
   734→   734→      // CREATE new task (originated in Lark)
   735→   735→      const newTask = await db.task.create({

   877→   877→        }
