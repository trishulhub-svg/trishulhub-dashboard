   205→function formatRelativeTime(dateStr: string): string {
   206→  if (!dateStr) return "unknown";
   207→  const date = new Date(dateStr);
   208→  if (isNaN(date.getTime())) return "unknown";
   209→  const now = new Date();
   210→  const diffMs = now.getTime() - date.getTime();
   211→  const diffMin = Math.floor(diffMs / 60000);
   212→  if (diffMin < 1) return "just now";
   213→  if (diffMin < 60) return `${diffMin}m ago`;
   214→  const diffHr = Math.floor(diffMin / 60);
   215→  if (diffHr < 24) return `${diffHr}h ago`;
   216→  const diffDay = Math.floor(diffHr / 24);
   217→  if (diffDay < 7) return `${diffDay}d ago`;
   218→  return safeDateStr(date);
   219→}

   972→// ─── Bridge: Listens for logout event to auto-minimize floating boards ──
   973→function LogoutBridge() {
   974→  const { signalLogout } = useFloatingBoards();
   975→  useEffect(() => {
   976→    const handler = () => signalLogout();
   977→    window.addEventListener("trishulhub:logout", handler);
   978→    return () => window.removeEventListener("trishulhub:logout", handler);
   979→  }, [signalLogout]);
   980→  return null;
   981→}
