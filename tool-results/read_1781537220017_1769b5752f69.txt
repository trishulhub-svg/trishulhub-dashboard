    22→    22→// ━━ CIRCULAR SYNC GUARD ━━
    23→    23→
    24→    24→/** In-memory map: "taskId:direction" → timestamp. Prevents TH→Lark→TH→Lark loops.
    25→    25→ *  Cooldown: 10 seconds.
    26→    26→ */
    27→    27→const _syncGuard = new Map<string, number>()
    28→    28→const SYNC_GUARD_COOLDOWN_MS = 10_000
    29→    29→
    30→    30→function markSyncGuard(taskId: string, direction: string): void {
    31→    31→  _syncGuard.set(`${taskId}:${direction}`, Date.now())
    32→    32→}
    33→    33→
    34→    34→function isSyncGuarded(taskId: string, direction: string): boolean {
    35→    35→  const ts = _syncGuard.get(`${taskId}:${direction}`)
    36→    36→  if (!ts) return false
    37→    37→  if (Date.now() - ts > SYNC_GUARD_COOLDOWN_MS) {
    38→    38→    _syncGuard.delete(`${taskId}:${direction}`)
    39→    39→    return false
    40→    40→  }
    41→    41→  return true
    42→    42→}
    43→    43→
    44→    44→// ━━ WEBHOOK EVENT DEDUPLICATION ━━
    45→    45→
    46→    46→/** In-memory set of processed event_ids. Prevents duplicate processing from Lark retries.
    47→    47→ *  Window: 30 seconds.
    48→    48→ */
    49→    49→const _processedEvents = new Map<string, number>()
    50→    50→const DEDUP_WINDOW_MS = 30_000
    51→    51→
    52→    52→function isEventDuplicate(eventId: string): boolean {
    53→    53→  const ts = _processedEvents.get(eventId)
    54→    54→  if (!ts) return false
    55→    55→  if (Date.now() - ts > DEDUP_WINDOW_MS) {
    56→    56→    _processedEvents.delete(eventId)
    57→    57→    return false
    58→    58→  }
    59→    59→  return true
    60→    60→}
    61→    61→
    62→    62→function markEventProcessed(eventId: string): void {
    63→    63→  _processedEvents.set(eventId, Date.now())
    64→    64→}
    65→    65→
    66→    66→// Periodic cleanup of stale entries (every 60s)
    67→    67→if (typeof setInterval !== "undefined") {
    68→    68→  setInterval(() => {
    69→    69→    const now = Date.now()
    70→    70→    for (const [key, ts] of _syncGuard) {
    71→    71→      if (now - ts > SYNC_GUARD_COOLDOWN_MS) _syncGuard.delete(key)
    72→    72→    }
    73→    73→    for (const [key, ts] of _processedEvents) {
    74→    74→      if (now - ts > DEDUP_WINDOW_MS) _processedEvents.delete(key)
    75→    75→    }
    76→    76→  }, 60_000)
    77→    77→}

   118→   118→export async function getUserLarkOpenId(userId: string): Promise<string | null> {
   119→   119→  try {
   120→   120→    // Check mapping table
   121→   121→    const rows = await db.$queryRawUnsafe<Array<{ larkOpenId: string }>>(
   122→   122→      'SELECT "larkOpenId" FROM "LarkUserMapping" WHERE "userId" = ?',

   126→   126→      return rows[0].larkOpenId

   129→   129→    // Try email lookup
   130→   130→    const user = await db.user.findUnique({ where: { id: userId }, select: { email: true } })
   131→   131→    if (user?.email) {
   132→   132→      const larkUser = await lookupUserByEmail(user.email)
   133→   133→      if (larkUser?.open_id) {
   134→   134→        // Auto-save the mapping
   135→   135→        await db.$executeRawUnsafe(
   136→   136→          'INSERT OR IGNORE INTO "LarkUserMapping" ("id", "userId", "larkOpenId", "larkName", "larkEmail", "matchedBy", "createdAt") VALUES (?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)',
   137→   137→          `map_${userId}_${Date.now()}`,
   138→   138→          userId,
   139→   139→          larkUser.open_id,
   140→   140→          larkUser.name || "",
   141→   141→          larkUser.email || "",
   142→   142→          "email_auto"
   143→   143→        )
   144→   144→        return larkUser.open_id
   145→   145→      }
   146→   146→    }
   147→   147→
   148→   148→    return null
   149→   149→  } catch (err) {
   150→   150→    console.error("[lark/sync] Failed to get user Lark ID:", err)

   230→   230→/**
   231→   231→ * Sync a task creation from TrishulHub to Lark.
   232→   232→ * Call this AFTER the task is saved in the DB.
   233→   233→ */
   234→   234→export async function syncTaskToLark(
   235→   235→  taskId: string,
   236→   236→  data: {
   237→   237→    title: string
   238→   238→    description?: string
   239→   239→    status: string
   240→   240→    priority: string
   241→   241→    assignedTo?: string
   242→   242→    projectId?: string
   243→   243→    deadline?: string | Date
   244→   244→  },
   245→   245→  userId: string,
   246→   246→  userName?: string
   247→   247→): Promise<void> {
   248→   248→  const config = await getLarkConfig()
   249→   249→  if (!config?.enabled) {
   250→   250→    console.warn(`[lark/sync] syncTaskToLark SKIPPED for ${taskId} — Lark sync is disabled`)
   251→   251→    await logSync({ direction: "TO_LARK", action: "CREATE", status: "SKIPPED", taskId, userId, error: "Lark sync is disabled in Access Hub settings" })
   252→   252→    return
   253→   253→  }
   254→   254→
   255→   255→  try {
   256→   256→    // Get or create the per-user task list for the project
   257→   257→    let tasklistId: string | undefined
   258→   258→
   259→   259→    if (data.projectId) {
   260→   260→      const project = await db.project.findUnique({ where: { id: data.projectId }, select: { name: true } })
   261→   261→      if (project) {
   262→   262→        // Use per-user task list: "ProjectName — UserName"
   263→   263→        const creatorName = userName || (await db.user.findUnique({ where: { id: userId }, select: { name: true } }))?.name || userId
   264→   264→        let result = await getOrCreateUserProjectTaskList(data.projectId, project.name, userId, creatorName)
   265→   265→        if (!result) {
   266→   266→          // Retry once — transient API errors can cause first attempt to fail
   267→   267→          console.warn(`[lark/sync] Retrying per-user task list creation for project ${data.projectId} (${project.name}) user ${creatorName}`)
   268→   268→          result = await getOrCreateUserProjectTaskList(data.projectId, project.name, userId, creatorName)
   269→   269→        }
   270→   270→        if (result) {
   271→   271→          tasklistId = result.tasklistId
   272→   272→          if (result.created) {
   273→   273→            console.log(`[lark/sync] Created new per-user Lark task list: ${project.name} — ${creatorName}`)
   274→   274→          }
   275→   275→        } else {
   276→   276→          console.error(`[lark/sync] FAILED to get/create per-user task list for project ${data.projectId} user ${userId} — aborting sync`)
   277→   277→          await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Failed to create per-user task list for "${project.name}"` })
   278→   278→          return
   279→   279→        }
   280→   280→      } else {
   281→   281→        console.error(`[lark/sync] Project ${data.projectId} not found in DB — aborting sync`)
   282→   282→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Project ${data.projectId} not found in database` })
   283→   283→        return
   284→   284→      }
   285→   285→    }
   286→   286→
   287→   287→    if (!tasklistId) {
   288→   288→      // No project — use a default "TrishulHub Tasks" list
   289→   289→      const creatorName = userName || (await db.user.findUnique({ where: { id: userId }, select: { name: true } }))?.name || userId
   290→   290→      const defaultResult = await getOrCreateUserProjectTaskList("__default__", "TrishulHub Tasks", userId, creatorName)
   291→   291→      if (!defaultResult) {
   292→   292→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Failed to get/create default task list" })
   293→   293→        return
   294→   294→      }
   295→   295→      tasklistId = defaultResult.tasklistId
   296→   296→    }
   297→   297→
   298→   298→    // Resolve assignee's Lark open_id
   299→   299→    let assigneeOpenId: string | undefined
   300→   300→    if (data.assignedTo) {
   301→   301→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   302→   302→    }
   303→   303→
   304→   304→    // Convert deadline to unix ms
   305→   305→    let dueTimestamp: number | undefined
   306→   306→    if (data.deadline) {
   307→   307→      dueTimestamp = new Date(data.deadline).getTime()
   308→   308→    }
   309→   309→
   310→   310→    // Create task — bare creation only (no members, no status, no priority on create API)
   311→   311→    const larkTask = await createTask(tasklistId, {
   312→   312→      title: data.title,
   313→   313→      description: data.description,
   314→   314→      dueTimestamp,
   315→   315→    })
   316→   316→
   317→   317→    if (larkTask) {
   318→   318→      await saveLarkTaskMapping(taskId, larkTask.task_id, tasklistId)
   319→   319→
   320→   320→      // Set description, due, status & priority via separate update
   321→   321→      // (create API only accepts tasklist_id + summary)
   322→   322→      await updateTask(larkTask.task_id, {
   323→   323→        description: data.description,
   324→   324→        status: data.status,
   325→   325→        priority: data.priority,
   326→   326→        dueTimestamp,
   327→   327→      })
   328→   328→
   329→   329→      // Assign member via add_members endpoint (create API doesn't support members)
   330→   330→      if (assigneeOpenId) {
   331→   331→        await addTaskMember(larkTask.task_id, [
   332→   332→          { id: assigneeOpenId, role: "assignee" },
   333→   333→        ])
   334→   334→      }
   335→   335→
   336→   336→      // Add assignee as tasklist member (editor role) so the tasklist
   337→   337→      // appears in the user's Lark Task Center
   338→   338→      if (assigneeOpenId) {
   339→   339→        await addTaskListMember(tasklistId, [
   340→   340→          { id: assigneeOpenId, role: "editor" },
   341→   341→        ])
   342→   342→      }
   343→   343→
   344→   344→      // Mark circular guard to prevent webhook from echoing back
   345→   345→      markSyncGuard(taskId, "TO_LARK")
   346→   346→
   347→   347→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "SUCCESS", taskId, larkTaskId: larkTask.task_id, larkTaskListId: tasklistId, projectId: data.projectId, userId })
   348→   348→    } else {
   349→   349→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Lark API returned null task" })
   350→   350→    }
   351→   351→  } catch (err) {
   352→   352→    const msg = err instanceof Error ? err.message : String(err)
   353→   353→    await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: msg })
   354→   354→  }
   355→   355→}

   357→   357→/**
   358→   358→ * Sync a task update from TrishulHub to Lark.
   359→   359→ */
   360→   360→export async function syncTaskUpdateToLark(
   361→   361→  taskId: string,
   362→   362→  data: {
   363→   363→    title?: string
   364→   364→    description?: string
   365→   365→    status?: string
   366→   366→    priority?: string
   367→   367→    assignedTo?: string
   368→   368→    deadline?: string | Date | null
   369→   369→  },
   370→   370→  userId: string
   371→   371→): Promise<void> {
   372→   372→  const config = await getLarkConfig()
   373→   373→  if (!config?.enabled) return
   374→   374→
   375→   375→  // Circular sync guard: skip if we just synced FROM Lark for this task
   376→   376→  if (isSyncGuarded(taskId, "FROM_LARK")) {
   377→   377→    console.log(`[lark/sync] Skipping TO_LARK update for ${taskId} — circular guard active`)
   378→   378→    return
   379→   379→  }
   380→   380→
   381→   381→  try {
   382→   382→    const mapping = await getLarkTaskMapping(taskId)
   383→   383→    if (!mapping) {
   384→   384→      // No mapping exists — try full sync (task may have been created before Lark was enabled)
   385→   385→      const task = await db.task.findUnique({ where: { id: taskId } })
   386→   386→      if (task) {
   387→   387→        await syncTaskToLark(taskId, {
   388→   388→          title: task.title,
   389→   389→          description: task.description || undefined,
   390→   390→          status: task.status,
   391→   391→          priority: task.priority,
   392→   392→          assignedTo: task.assignedTo || undefined,
   393→   393→          projectId: task.projectId || undefined,
   394→   394→          deadline: task.deadline || undefined,
   395→   395→        }, userId)
   396→   396→      }
   397→   397→      return
   398→   398→    }
   399→   399→
   400→   400→    // Resolve assignee's Lark open_id
   401→   401→    let assigneeOpenId: string | undefined
   402→   402→    if (data.assignedTo) {
   403→   403→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   404→   404→    }
   405→   405→
   406→   406→    // Convert deadline to unix ms
   407→   407→    let dueTimestamp: number | undefined
   408→   408→    if (data.deadline) {
   409→   409→      dueTimestamp = new Date(data.deadline).getTime()
   410→   410→    }
   411→   411→
   412→   412→    // Update task — NO extra, NO assignee in body (use member APIs instead)
   413→   413→    const larkTask = await updateTask(mapping.larkTaskId, {
   414→   414→      title: data.title,
   415→   415→      description: data.description,
   416→   416→      status: data.status,
   417→   417→      priority: data.priority,
   418→   418→      dueTimestamp,
   419→   419→    })
   420→   420→
   421→   421→    // Handle assignee change via member API
   422→   422→    if (data.assignedTo !== undefined && assigneeOpenId) {
   423→   423→      await addTaskMember(mapping.larkTaskId, [
   424→   424→        { id: assigneeOpenId, role: "assignee" },
   425→   425→      ])
   426→   426→    }
   427→   427→
   428→   428→    if (larkTask) {
   429→   429→      // Mark circular guard
   430→   430→      markSyncGuard(taskId, "TO_LARK")
   431→   431→
   432→   432→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   433→   433→    } else {
   434→   434→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark API returned null" })
   435→   435→    }
   436→   436→  } catch (err) {
   437→   437→    const msg = err instanceof Error ? err.message : String(err)
   438→   438→    await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, userId, error: msg })
   439→   439→  }
   440→   440→}

   442→   442→/**
   443→   443→ * Sync a task deletion from TrishulHub to Lark.
   444→   444→ */
   445→   445→export async function syncTaskDeleteToLark(taskId: string, userId: string): Promise<void> {
   446→   446→  const config = await getLarkConfig()
   447→   447→  if (!config?.enabled) return
   448→   448→
   449→   449→  try {
   450→   450→    const mapping = await getLarkTaskMapping(taskId)
   451→   451→    if (!mapping) {
   452→   452→      await logSync({ direction: "TO_LARK", action: "DELETE", status: "SKIPPED", taskId, userId, error: "No Lark mapping found" })
   453→   453→      return
   454→   454→    }
   455→   455→
   456→   456→    const success = await larkDeleteTask(mapping.larkTaskListId, mapping.larkTaskId)
   457→   457→    await removeLarkTaskMapping(taskId)
   458→   458→
   459→   459→    if (success) {

   865→   865→async function findProjectByTaskList(tasklistId: string): Promise<string | null> {
   866→   866→  try {
