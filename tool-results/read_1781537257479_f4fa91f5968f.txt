   515→      }
   516→    }
   517→    data.assignedTo = newAssignee
   518→  }
   519→  if (body.assigneeType !== undefined) {
   520→    if (!["HUMAN", "AI"].includes(String(body.assigneeType))) {
   521→      return NextResponse.json({ error: "Invalid assigneeType. Must be HUMAN or AI" }, { status: 400 })
   522→    }
   523→    data.assigneeType = String(body.assigneeType)
   524→  }
   525→
   526→  // C21: Deadline validation
   527→  if (body.deadline !== undefined) {
   528→    if (body.deadline) {
   529→      const d = new Date(String(body.deadline))
   530→      if (isNaN(d.getTime())) {
   531→        return NextResponse.json({ error: "Invalid deadline date" }, { status: 400 })
   532→      }
   533→    }
   534→    data.deadline = body.deadline ? new Date(String(body.deadline)) : null
   535→  }
   536→
   537→  if (body.category !== undefined) {
   538→    if (!VALID_TASK_CATEGORIES.includes(String(body.category))) {
   539→      return NextResponse.json({ error: `Invalid category. Must be one of: ${VALID_TASK_CATEGORIES.join(", ")}` }, { status: 400 })
   540→    }
   541→    data.category = String(body.category)
   542→  }
   543→
   544→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   545→  // APPROVAL FLOW — status change logic
   546→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   547→  if (body.status !== undefined) {
   548→    if (!VALID_TASK_STATUSES.includes(String(body.status))) {
   549→      return NextResponse.json({ error: `Invalid status. Must be one of: ${VALID_TASK_STATUSES.join(", ")}` }, { status: 400 })
   550→    }
   551→
   552→    const newStatus = body.status as string
   553→    const currentStatus = existingTask.status
   554→
   555→    // W1: Status transition validation (non-admin only)
   556→    if (newStatus !== currentStatus && !isAdmin(userRole)) {
   557→      const allowed = VALID_TRANSITIONS[currentStatus] || []
   558→      if (!allowed.includes(newStatus)) {
   559→        return NextResponse.json({ error: `Invalid status transition: ${currentStatus} → ${newStatus}. Allowed: ${allowed.length > 0 ? allowed.join(", ") : "none (terminal state)"}` }, { status: 409 })
   560→      }
   561→    }
   562→
   563→    // ── CASE 1: User is trying to mark task as DONE ──
   564→    if (newStatus === "DONE") {
   565→      // If task is already AWAITING_APPROVAL, this is an approval action
   566→      if (currentStatus === "AWAITING_APPROVAL") {
   567→        // Only admin/superadmin can approve
   568→        if (!isAdmin(userRole)) {
   569→          return NextResponse.json({ error: "Forbidden: Only admin or superadmin can approve tasks" }, { status: 403 })
   570→        }
   571→
   572→        // Self-approval prevention: ADMIN cannot approve tasks assigned to themselves
   573→        if (userRole === "ADMIN" && existingTask.assignedTo === userId) {
   574→          return NextResponse.json({ error: "Forbidden: You cannot approve your own task. Only superadmin can approve your tasks." }, { status: 403 })
   575→        }
   576→
   577→        // APPROVE — set status, approvedBy, approvedAt, completedAt
   578→        data.status = "DONE"
   579→        data.approvedBy = userId
   580→        data.approvedAt = new Date()
   581→        data.completedAt = new Date()
   582→
   583→      } else {
   584→        // Task is NOT currently AWAITING_APPROVAL — user wants to "complete" it
   585→        // SUPERADMIN can directly mark as DONE (no approval needed for their own actions)
   586→        if (userRole === "SUPER_ADMIN") {
   587→          data.status = "DONE"
   588→          data.approvedBy = userId
   589→          data.approvedAt = new Date()
   590→          data.completedAt = new Date()
   591→        } else {
   592→          // All other users (DEVELOPER, ADMIN) → goes to AWAITING_APPROVAL
   593→          data.status = "AWAITING_APPROVAL"
   594→        }
   595→      }
   596→    } else {
   597→      // Regular status change (not DONE)
   598→      data.status = newStatus
   599→    }
   600→  }
   601→
   602→  // Only admins can change projectId on a task
   603→  if (body.projectId !== undefined) {
   604→    if (!isAdmin(userRole)) {
   605→      return NextResponse.json({ error: "Forbidden: Only admins can move tasks between projects" }, { status: 403 })
   606→    }
   607→    data.projectId = String(body.projectId)
   608→  }
   609→
   610→  // Check if data object is empty — nothing to update
   611→  if (Object.keys(data).length === 0) {
   612→    return NextResponse.json({ error: "No fields to update" }, { status: 400 })
   613→  }
   614→
   615→  // Check if assignee is on approved leave during the task period
   616→  const assignedUserId = typeof data.assignedTo === "string" ? data.assignedTo : null
   617→  const effectiveDeadline = data.deadline instanceof Date ? data.deadline : (existingTask?.deadline || null)
   618→
   619→  if (assignedUserId && effectiveDeadline) {
   620→    const assigneeLeave = await checkAssigneeLeave(db, assignedUserId, effectiveDeadline)
   621→    if (assigneeLeave) {
   622→      return NextResponse.json({
   623→        error: `Cannot assign task: ${assigneeLeave.name} is on ${assigneeLeave.leaveType.replace("_", " ").toLowerCase()} leave from ${new Date(assigneeLeave.startDate).toLocaleDateString()} to ${new Date(assigneeLeave.endDate).toLocaleDateString()}`,
   624→      }, { status: 400 })
   625→    }
   626→  }
   627→
   628→  // Update the task
   629→  const updatedTask = await db.task.update({ where: { id }, data })
   630→
   631→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   632→  // NOTIFICATIONS
   633→  // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   634→
   635→  // Reassignment notification — when assignedTo changed to a different user
   636→  if (assignedUserId && existingTask.assignedTo !== assignedUserId) {
   637→    await sendNotification(
   638→      assignedUserId,
   639→      "Task Reassigned to You",
   640→      `Task "${existingTask.title}" has been reassigned to you`,
   641→      "TASK",
   642→      "/dashboard/tasks"
   643→    )
   644→  }
   645→
   646→  const finalStatus = data.status as string | undefined
   647→
   648→  // I8: Task sent for approval → notify all admin/superadmin (batch createMany)
   649→  // Fetch admins AND assignee in a single combined query to avoid separate DB lookup
   650→  if (finalStatus === "AWAITING_APPROVAL") {
   651→    const assigneeId = existingTask.assignedTo
   652→    // Batch fetch admins AND the assignee in one query
   653→    const allUsers = await db.user.findMany({
   654→      where: {
   655→        OR: [
   656→          { role: { in: ["SUPER_ADMIN", "ADMIN"] }, isActive: true },
   657→          ...(assigneeId ? [{ id: assigneeId }] : []),
   658→        ]
   659→      },
   660→      select: { id: true, name: true, role: true }
   661→    })
   662→    // Extract assignee name from the combined result
   663→    const assigneeName = assigneeId
   664→      ? (allUsers.find(u => u.id === assigneeId)?.name || "Someone")
   665→      : "Someone"
   666→    // Filter to just admins for notifications
   667→    const adminUsers = allUsers.filter(u => u.role === "SUPER_ADMIN" || u.role === "ADMIN")
   668→
   669→    if (adminUsers.length > 0) {
   670→      await db.notification.createMany({
   671→        data: adminUsers.map(admin => ({
   672→          userId: admin.id,
   673→          title: `Task approval needed: ${existingTask.title}`,
   674→          message: `${assigneeName} submitted "${existingTask.title}" for review`,
   675→          type: "TASK",
   676→          link: `/dashboard/todos`,
   677→          isRead: false,
   678→        }))
   679→      })
   680→    }
   681→  }
   682→
   683→  // Task approved → notify the assignee
   684→  if (finalStatus === "DONE" && data.approvedBy && existingTask.assignedTo && existingTask.assignedTo !== userId) {
   685→    const taskLink = `/dashboard/projects/todos`
   686→    await sendNotification(
   687→      existingTask.assignedTo,
   688→      "Task Approved",
   689→      `Your task "${existingTask.title}" has been approved by ${userName}.`,
   690→      "SUCCESS",
   691→      taskLink
   692→    )
   693→  }
   694→
   695→  // Task rejected (sent back) → notify the assignee
   696→  if (finalStatus && finalStatus !== "AWAITING_APPROVAL" && finalStatus !== "DONE" && existingTask.status === "AWAITING_APPROVAL") {
   697→    if (existingTask.assignedTo) {
   698→      const taskLink = `/dashboard/projects/todos`
   699→      await sendNotification(
   700→        existingTask.assignedTo,
   701→        "Task Revision Needed",
   702→        `Your task "${existingTask.title}" was sent back by ${userName}. Status: ${finalStatus.replace("_", " ")}.`,
   703→        "WARNING",
   704→        taskLink
   705→      )
   706→    }
   707→  }
   708→
   709→  // Background: sync tasks to Git (fire-and-forget) — only when task has a project
   710→  if (updatedTask.projectId) {
   711→    syncTasksToGit().catch(() => {})
   712→  }
   713→  // Background: sync task update to Lark (fire-and-forget)
   714→  const larkUpdateData: { title?: string; description?: string; status?: string; priority?: string; assignedTo?: string; deadline?: string | Date | null } = {}
   715→  if (data.title) larkUpdateData.title = String(data.title)
   716→  if (data.description !== undefined) larkUpdateData.description = (data.description as string | null) ?? undefined
   717→  if (data.status) larkUpdateData.status = String(data.status)
   718→  if (data.priority) larkUpdateData.priority = String(data.priority)
   719→  if (data.assignedTo !== undefined) larkUpdateData.assignedTo = (data.assignedTo as string) || undefined
   720→  if (data.deadline !== undefined) larkUpdateData.deadline = data.deadline as Date | null | undefined
   721→  if (Object.keys(larkUpdateData).length > 0) {
   722→    syncTaskUpdateToLark(id, larkUpdateData, userId).catch((err) => console.error("[tasks] Lark update sync error:", err))
   723→  }
   724→  return NextResponse.json(serializeTask(updatedTask))
   725→  } catch (error: unknown) {
   726→    console.error("[tasks] PATCH error:", error instanceof Error ? error.message : String(error))
   727→    return NextResponse.json({ error: "An error occurred" }, { status: 500 })
   728→  }
   729→}

   731→export async function DELETE(req: NextRequest) {
   732→  try {
   733→  await ensureAllTables()
   734→
   735→  const session = await getServerSession(authOptions)
   736→  if (!session?.user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
   737→
   738→  // Rate limit
   739→  const rl = rateLimit(`tasks-write-${session.user.id}`, RATE_LIMITS.crmWrite.limit, RATE_LIMITS.crmWrite.windowMs)
   740→  if (!rl.success) {
   741→    return NextResponse.json({ error: "Rate limit exceeded" }, { status: 429 })
   742→  }
   743→
   744→  const userRole = session.user.role
   745→  const userId = session.user.id
   746→
   747→  // I7: TODO: Move to /api/tasks/[id]/route.ts for proper REST
   748→  // Currently using query params for task ID instead of URL path
   749→  const { searchParams } = new URL(req.url)
   750→  const id = searchParams.get("id")
   751→
   752→  if (!id) return NextResponse.json({ error: "Task ID is required" }, { status: 400 })
   753→
   754→  // Verify task exists
   755→  const existingTask = await db.task.findUnique({ where: { id } })
   756→  if (!existingTask) return NextResponse.json({ error: "Task not found" }, { status: 404 })
   757→
   758→  // Developers can only delete tasks in their assigned projects, or standalone tasks they created/are assigned to
   759→  if (!isAdmin(userRole)) {
   760→    if (existingTask.projectId) {
   761→      // Project task — check membership
   762→      const membership = await db.projectMember.findFirst({
   763→        where: { userId, projectId: existingTask.projectId }
   764→      })
   765→      if (!membership && existingTask.createdBy !== userId && existingTask.assignedTo !== userId) {
   766→        return NextResponse.json({ error: "Forbidden: You can only delete tasks in your assigned projects or tasks assigned/created by you" }, { status: 403 })
   767→      }
   768→    } else {
   769→      // Standalone task — must be creator or assignee
   770→      if (existingTask.createdBy !== userId && existingTask.assignedTo !== userId) {
   771→        return NextResponse.json({ error: "Forbidden: You can only delete tasks assigned or created by you" }, { status: 403 })
   772→      }
   773→    }
   774→  }
   775→
   776→  await db.task.delete({ where: { id } })
   777→  // Background: sync tasks to Git (fire-and-forget) — only when task had a project
   778→  if (existingTask.projectId) {
   779→    syncTasksToGit().catch(() => {})
   780→  }
   781→  // Background: sync task deletion to Lark (fire-and-forget)
   782→  syncTaskDeleteToLark(id, userId).catch((err) => console.error("[tasks] Lark delete sync error:", err))
   783→  return NextResponse.json({ success: true })
   784→  } catch (error: unknown) {
   785→    console.error("[tasks] DELETE error:", error instanceof Error ? error.message : String(error))
   786→    return NextResponse.json({ error: "An error occurred" }, { status: 500 })
   787→  }
   788→}
