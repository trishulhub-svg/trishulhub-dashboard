   230→/**
   231→ * Sync a task creation from TrishulHub to Lark.
   232→ * Call this AFTER the task is saved in the DB.
   233→ */
   234→export async function syncTaskToLark(
   235→  taskId: string,
   236→  data: {
   237→    title: string
   238→    description?: string
   239→    status: string
   240→    priority: string
   241→    assignedTo?: string
   242→    projectId?: string
   243→    deadline?: string | Date
   244→  },
   245→  userId: string,
   246→  userName?: string
   247→): Promise<void> {
   248→  const config = await getLarkConfig()
   249→  if (!config?.enabled) {
   250→    console.warn(`[lark/sync] syncTaskToLark SKIPPED for ${taskId} — Lark sync is disabled`)
   251→    await logSync({ direction: "TO_LARK", action: "CREATE", status: "SKIPPED", taskId, userId, error: "Lark sync is disabled in Access Hub settings" })
   252→    return
   253→  }
   254→
   255→  try {
   256→    // Get or create the per-user task list for the project
   257→    let tasklistId: string | undefined
   258→
   259→    if (data.projectId) {
   260→      const project = await db.project.findUnique({ where: { id: data.projectId }, select: { name: true } })
   261→      if (project) {
   262→        // Use per-user task list: "ProjectName — UserName"
   263→        const creatorName = userName || (await db.user.findUnique({ where: { id: userId }, select: { name: true } }))?.name || userId
   264→        let result = await getOrCreateUserProjectTaskList(data.projectId, project.name, userId, creatorName)
   265→        if (!result) {
   266→          // Retry once — transient API errors can cause first attempt to fail
   267→          console.warn(`[lark/sync] Retrying per-user task list creation for project ${data.projectId} (${project.name}) user ${creatorName}`)
   268→          result = await getOrCreateUserProjectTaskList(data.projectId, project.name, userId, creatorName)
   269→        }
   270→        if (result) {
   271→          tasklistId = result.tasklistId
   272→          if (result.created) {
   273→            console.log(`[lark/sync] Created new per-user Lark task list: ${project.name} — ${creatorName}`)
   274→          }
   275→        } else {
   276→          console.error(`[lark/sync] FAILED to get/create per-user task list for project ${data.projectId} user ${userId} — aborting sync`)
   277→          await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Failed to create per-user task list for "${project.name}"` })
   278→          return
   279→        }
   280→      } else {
   281→        console.error(`[lark/sync] Project ${data.projectId} not found in DB — aborting sync`)
   282→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: `Project ${data.projectId} not found in database` })
   283→        return
   284→      }
   285→    }
   286→
   287→    if (!tasklistId) {
   288→      // No project — use a default "TrishulHub Tasks" list
   289→      const creatorName = userName || (await db.user.findUnique({ where: { id: userId }, select: { name: true } }))?.name || userId
   290→      const defaultResult = await getOrCreateUserProjectTaskList("__default__", "TrishulHub Tasks", userId, creatorName)
   291→      if (!defaultResult) {
   292→        await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Failed to get/create default task list" })
   293→        return
   294→      }
   295→      tasklistId = defaultResult.tasklistId
   296→    }
   297→
   298→    // Resolve assignee's Lark open_id
   299→    let assigneeOpenId: string | undefined
   300→    if (data.assignedTo) {
   301→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   302→    }
   303→
   304→    // Convert deadline to unix ms
   305→    let dueTimestamp: number | undefined
   306→    if (data.deadline) {
   307→      dueTimestamp = new Date(data.deadline).getTime()
   308→    }
   309→
   310→    // Create task — bare creation only (no members, no status, no priority on create API)
   311→    const larkTask = await createTask(tasklistId, {
   312→      title: data.title,
   313→      description: data.description,
   314→      dueTimestamp,
   315→    })
   316→
   317→    if (larkTask) {
   318→      await saveLarkTaskMapping(taskId, larkTask.task_id, tasklistId)
   319→
   320→      // Set description, due, status & priority via separate update
   321→      // (create API only accepts tasklist_id + summary)
   322→      await updateTask(larkTask.task_id, {
   323→        description: data.description,
   324→        status: data.status,
   325→        priority: data.priority,
   326→        dueTimestamp,
   327→      })
   328→
   329→      // Assign member via add_members endpoint (create API doesn't support members)
   330→      if (assigneeOpenId) {
   331→        await addTaskMember(larkTask.task_id, [
   332→          { id: assigneeOpenId, role: "assignee" },
   333→        ])
   334→      }
   335→
   336→      // Add assignee as tasklist member (editor role) so the tasklist
   337→      // appears in the user's Lark Task Center
   338→      if (assigneeOpenId) {
   339→        await addTaskListMember(tasklistId, [
   340→          { id: assigneeOpenId, role: "editor" },
   341→        ])
   342→      }
   343→
   344→      // Mark circular guard to prevent webhook from echoing back
   345→      markSyncGuard(taskId, "TO_LARK")
   346→
   347→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "SUCCESS", taskId, larkTaskId: larkTask.task_id, larkTaskListId: tasklistId, projectId: data.projectId, userId })
   348→    } else {
   349→      await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: "Lark API returned null task" })
   350→    }
   351→  } catch (err) {
   352→    const msg = err instanceof Error ? err.message : String(err)
   353→    await logSync({ direction: "TO_LARK", action: "CREATE", status: "FAILED", taskId, userId, error: msg })
   354→  }
   355→}

   360→export async function syncTaskUpdateToLark(
   361→  taskId: string,

   364→    description?: string

   366→    priority?: string
   367→    assignedTo?: string

   373→  if (!config?.enabled) return

   379→  }

   392→          assignedTo: task.assignedTo || undefined,
   393→          projectId: task.projectId || undefined,
   394→          deadline: task.deadline || undefined,
   395→        }, userId)
   396→      }
   397→      return
   398→    }
   399→
   400→    // Resolve assignee's Lark open_id
   401→    let assigneeOpenId: string | undefined
   402→    if (data.assignedTo) {
   403→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   404→    }
   405→
   406→    // Convert deadline to unix ms
   407→    let dueTimestamp: number | undefined
   408→    if (data.deadline) {
   409→      dueTimestamp = new Date(data.deadline).getTime()
   410→    }
   411→
   412→    // Update task — NO extra, NO assignee in body (use member APIs instead)
   413→    const larkTask = await updateTask(mapping.larkTaskId, {
   414→      title: data.title,
   415→      description: data.description,
   416→      status: data.status,
   417→      priority: data.priority,
   418→      dueTimestamp,
   419→    })
   420→
   421→    // Handle assignee change via member API
   422→    if (data.assignedTo !== undefined && assigneeOpenId) {
   423→      await addTaskMember(mapping.larkTaskId, [
   424→        { id: assigneeOpenId, role: "assignee" },
   425→      ])
   426→    }
   427→
   428→    if (larkTask) {
   429→      // Mark circular guard
   430→      markSyncGuard(taskId, "TO_LARK")
   431→
   432→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   433→    } else {
   434→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark API returned null" })
   435→    }
   436→  } catch (err) {
   437→    const msg = err instanceof Error ? err.message : String(err)
   438→    await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, userId, error: msg })
   439→  }
   440→}

   472→/**
   473→ * Handle incoming Lark webhook events.
   474→ * @param eventId — Lark event_id for deduplication
   475→ * @param eventType — Lark event type
   476→ * @param eventData — Parsed event payload
   477→ */
   478→export async function handleLarkWebhookEvent(
   479→  eventId: string,
   480→  eventType: string,
   481→  eventData: {
   482→    task_id: string
   483→    tasklist_id: string
   484→    operator?: { open_id: string }
   485→    [key: string]: unknown
   486→  }
   487→): Promise<{ success: boolean; message: string }> {
   488→  const config = await getLarkConfig()
   489→  if (!config?.enabled) {
   490→    return { success: false, message: "Lark sync is disabled" }
   491→  }
   492→
   493→  // ── Dedup check ──
   494→  if (isEventDuplicate(eventId)) {
   495→    console.log(`[lark/sync] Dedup: skipping duplicate event ${eventId}`)
   496→    return { success: true, message: "Duplicate event, skipped" }
   497→  }
   498→  markEventProcessed(eventId)
   499→
   500→  const { task_id, tasklist_id } = eventData
   501→  console.log(`[lark/sync] Processing event: ${eventType} (id: ${eventId}) for task: ${task_id}`)
   502→  console.log(`[lark/sync] Event details: tasklist_id=${tasklist_id}, operator=${eventData.operator?.open_id || 'none'}`)
   503→
   504→  try {
   505→    // Find existing mapping
   506→    const thTaskId = await getTaskIdByLarkId(task_id)
   507→    console.log(`[lark/sync] Mapping lookup: larkTaskId=${task_id} → thTaskId=${thTaskId || 'NOT FOUND'}`)
   508→
   509→    if (eventType === "task.task.deleted") {
   510→      if (!thTaskId) {
   511→        return { success: true, message: "Task not mapped, ignoring delete" }
   512→      }
   513→      // Delete the TrishulHub task
   514→      await db.task.delete({ where: { id: thTaskId } })
   515→      await removeLarkTaskMapping(thTaskId)
   516→
   517→      try {
   518→        const { logAudit } = await import("@/lib/audit-log")
   519→        await logAudit({
   520→          userId: eventData.operator?.open_id || "lark-webhook",
   521→          userName: "Lark Webhook",
   522→          userRole: "SYSTEM",
   523→          department: "TEAM_WORK",
   524→          page: "tasks",
   525→          action: "DELETE",
   526→          entityType: "Task",
   527→          entityId: thTaskId,
   528→          description: `Task deleted via Lark sync`,
   529→          status: "SUCCESS",
   530→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id }),
   531→        })
   532→      } catch { /* audit log is best-effort */ }
   533→
   534→      return { success: true, message: "Task deleted from TrishulHub" }
   535→    }
   536→
   537→    // For create/update/complete, we need the full task data from Lark
   538→    // Add 1s delay to give Lark eventual-consistency time to propagate
   539→    await new Promise((resolve) => setTimeout(resolve, 1000))
   540→
   541→    const larkTask = await getTask(tasklist_id, task_id)
   542→    if (larkTask) {
   543→      console.log(`[lark/sync] Fetched from Lark: title="${larkTask.title}", status="${larkTask.status}", priority="${larkTask.priority}", assignee="${larkTask.assignee?.open_id || 'none'}"`)
   544→    } else {
   545→      console.warn(`[lark/sync] getTask returned null for task ${task_id} in list ${tasklist_id}`)
   546→    }
   547→
   548→    // Fallback: if getTask returns null (e.g. completed tasks return 99404 on free tier),
   549→    // retry once after a short delay, then inspect event body for status clues.
   550→    if (!larkTask) {
   551→      if (thTaskId) {
   552→        // Retry getTask once after 2s — Lark eventual consistency may need more time
   553→        console.warn(`[lark/sync] getTask returned null for mapped task ${thTaskId}, retrying...`)
   554→        await new Promise((resolve) => setTimeout(resolve, 2000))
   555→        const retryTask = await getTask(tasklist_id, task_id)
   556→
   557→        if (retryTask) {
   558→          // Retry succeeded — process normally via the update path below
   559→          // Rewrite larkTask so the code below handles it
   560→          Object.assign(eventData, { _retryTask: retryTask })
   561→          // Fall through to the main update logic — we'll read from retryTask
   562→          const title = retryTask.title
   563→          const description = retryTask.description
   564→          const status = STATUS_FROM_LARK[retryTask.status] || "TODO"
   565→          const priority = PRIORITY_FROM_LARK[retryTask.priority] || "MEDIUM"
   566→
   567→          let assignedTo: string | undefined
   568→          if (retryTask.assignee?.open_id) {
   569→            assignedTo = await getUserIdByLarkOpenId(retryTask.assignee.open_id) || undefined
   570→          }
   571→
   572→          const deadline = retryTask.due?.timestamp ? new Date(parseInt(retryTask.due.timestamp)) : null
   573→
   574→          const updateData: Record<string, unknown> = {}
   575→          if (title) updateData.title = title
   576→          if (description !== undefined) updateData.description = description
   577→          updateData.status = status
   578→          updateData.priority = priority
   579→          if (assignedTo !== undefined) updateData.assignedTo = assignedTo
   580→          if (deadline !== null) updateData.deadline = deadline
   581→          if (status === "DONE") updateData.completedAt = new Date()
   582→
   583→          await db.task.update({ where: { id: thTaskId }, data: updateData })
   584→          markSyncGuard(thTaskId, "FROM_LARK")
   585→
   586→          const { logAudit } = await import("@/lib/audit-log")
   587→          await logAudit({
   588→            userId: eventData.operator?.open_id || "lark-webhook",
   589→            userName: "Lark Webhook",
   590→            userRole: "SYSTEM",
   591→            department: "TEAM_WORK",
   592→            page: "tasks",
   593→            action: "STATUS_CHANGE",
   594→            entityType: "Task",
   595→            entityId: thTaskId,
   596→            description: `Task status changed to "${status}" via Lark sync (retried)`,
   597→            newValue: status,
   598→            status: "SUCCESS",
   599→            metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, larkStatus: retryTask.status }),
   600→          })
   601→
   602→          await logSync({
   603→            direction: "FROM_LARK",
   604→            action: "STATUS_CHANGE",
   605→            status: "SUCCESS",
   606→            taskId: thTaskId,
   607→            larkTaskId: task_id,
   608→            larkTaskListId: tasklist_id,
   609→            metadata: { inferredStatus: status, eventType, retried: true },
   610→          })
   611→          return { success: true, message: `Task ${thTaskId} updated to ${status} (retried getTask)` }
   612→        }
   613→
   614→        // Both getTask calls failed — inspect event body for status change clues.
   615→        const eventBody = (eventData as Record<string, unknown>)
   616→        const changedFields = eventBody.changed_fields as string[] | undefined
   617→
   618→        // Determine if this is likely a completion:
   619→        // 1. Event type contains done/complete/finish keywords
   620→        // 2. OR changed_fields includes "status" (something changed, likely to done since getTask fails)
   621→        const likelyCompleted =
   622→          eventType.includes("done") || eventType.includes("complete") || eventType.includes("finish") ||
   623→          (Array.isArray(changedFields) && changedFields.includes("status"))
   624→
   625→        const newStatus = likelyCompleted ? "DONE" : "IN_PROGRESS";
   626→        const fallbackData: Record<string, unknown> = { status: newStatus };
   627→        if (newStatus === "DONE") fallbackData.completedAt = new Date();
   628→
   629→        const existingTask = await db.task.findUnique({ where: { id: thTaskId }, select: { status: true } });
   630→        console.warn(`[lark/sync] getTask returned null twice for mapped task ${thTaskId}. Event: ${eventType}. Assuming status="${newStatus}" (was "${existingTask?.status || 'unknown'}")`);
   631→
   632→        await db.task.update({ where: { id: thTaskId }, data: fallbackData })
   633→        markSyncGuard(thTaskId, "FROM_LARK")
   634→
   635→        const { logAudit } = await import("@/lib/audit-log")
   636→        await logAudit({
   637→          userId: eventData.operator?.open_id || "lark-webhook",
   638→          userName: "Lark Webhook",
   639→          userRole: "SYSTEM",
   640→          department: "TEAM_WORK",
   641→          page: "tasks",
   642→          action: "STATUS_CHANGE",
   643→          entityType: "Task",
   644→          entityId: thTaskId,
   645→          description: `Task status changed to "${newStatus}" via Lark sync (inferred — getTask failed)`,
   646→          newValue: newStatus,
   647→          status: "SUCCESS",
   648→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, inferred: true, previousStatus: existingTask?.status }),
   649→        })
   650→
   651→        await logSync({
   652→          direction: "FROM_LARK",
   653→          action: "STATUS_CHANGE",
   654→          status: "SUCCESS",
   655→          taskId: thTaskId,
   656→          larkTaskId: task_id,
   657→          larkTaskListId: tasklist_id,
   658→          metadata: { inferredStatus: newStatus, eventType, retried: true },
   659→        })
   660→        return { success: true, message: `Task ${thTaskId} updated to ${newStatus} (getTask null, inferred from event + changed_fields)` }
   661→      }
   662→      return { success: false, message: "Failed to fetch task from Lark and no existing mapping" }
   663→    }
   664→
   665→    // Circular sync guard: skip if we just synced TO_LARK for this task
   666→    if (thTaskId && isSyncGuarded(thTaskId, "TO_LARK")) {
   667→      console.log(`[lark/sync] Skipping FROM_LARK for ${thTaskId} — circular guard active`)
   668→      return { success: true, message: "Circular guard: skipped" }
   669→    }
   670→
   671→    // Map Lark task data to TrishulHub format
   672→    const title = larkTask.title
   673→    const description = larkTask.description
   674→    const status = STATUS_FROM_LARK[larkTask.status] || "TODO"
   675→    console.log(`[lark/sync] Status mapping: lark="${larkTask.status}" → th="${status}"`)
   676→    const priority = PRIORITY_FROM_LARK[larkTask.priority] || "MEDIUM"
   677→
   678→    // Resolve assignee
   679→    let assignedTo: string | undefined
   680→    if (larkTask.assignee?.open_id) {
   681→      assignedTo = await getUserIdByLarkOpenId(larkTask.assignee.open_id) || undefined
   682→    }
   683→
   684→    // Resolve due date
   685→    const deadline = larkTask.due?.timestamp ? new Date(parseInt(larkTask.due.timestamp)) : null
   686→
   687→    if (thTaskId) {
   688→      // UPDATE existing task
   689→      const updateData: Record<string, unknown> = {}
   690→      if (title) updateData.title = title
   691→      if (description !== undefined) updateData.description = description
   692→      updateData.status = status
   693→      updateData.priority = priority
   694→      if (assignedTo !== undefined) updateData.assignedTo = assignedTo
   695→      if (deadline !== null) updateData.deadline = deadline
   696→
   697→      // Set completedAt if status changed to DONE
   698→      if (status === "DONE") {
   699→        updateData.completedAt = new Date()
   700→      }
   701→
   702→      await db.task.update({ where: { id: thTaskId }, data: updateData })
   703→      console.log(`[lark/sync] Updated TH task ${thTaskId}: status=${updateData.status}, priority=${updateData.priority}, assignedTo=${updateData.assignedTo || 'unchanged'}`)
   704→
   705→      // Mark circular guard so our update handler doesn't echo back
   706→      markSyncGuard(thTaskId, "FROM_LARK")
   707→
   708→      // Log to both LarkSyncLog and AuditLog
   709→      await logSync({
   710→        direction: "FROM_LARK",
   711→        action: status === "DONE" ? "STATUS_CHANGE" : "UPDATE",
   712→        status: "SUCCESS",
   713→        taskId: thTaskId,
   714→        larkTaskId: task_id,
   715→        larkTaskListId: tasklist_id,
   716→      })
   717→
   718→      try {
   719→        const { logAudit } = await import("@/lib/audit-log")
   720→        await logAudit({
   721→          userId: eventData.operator?.open_id || "lark-webhook",
   722→          userName: "Lark Webhook",
   723→          userRole: "SYSTEM",
   724→          department: "TEAM_WORK",
   725→          page: "tasks",
   726→          action: status === "DONE" ? "STATUS_CHANGE" : "UPDATE",
   727→          entityType: "Task",
   728→          entityId: thTaskId,
   729→          description: `Task ${status === "DONE" ? "marked as DONE" : "updated"} via Lark sync`,
   730→          newValue: status,
   731→          status: "SUCCESS",
   732→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, larkStatus: larkTask.status }),
   733→        })
   734→      } catch { /* audit log is best-effort */ }
   735→
   736→      return { success: true, message: `Task ${thTaskId} updated from Lark` }
   737→    } else {
   738→      // CREATE new task (originated in Lark)
   739→      const newTask = await db.task.create({
   740→        data: {
   741→          title,
   742→          description: description || null,
   743→          status,
   744→          priority,
   745→          assignedTo: assignedTo || null,
   746→          deadline,
   747→          assigneeType: "HUMAN",
   748→          category: "GENERAL",
   749→          projectId: await findProjectByTaskList(tasklist_id),
   750→        },
   751→      })
   752→
   753→      // Save mapping
   754→      await saveLarkTaskMapping(newTask.id, task_id, tasklist_id)
   755→
   756→      // Mark guard for the newly created task
   757→      markSyncGuard(newTask.id, "FROM_LARK")
   758→
   759→      try {
   760→        const { logAudit } = await import("@/lib/audit-log")
   761→        await logAudit({
   762→          userId: eventData.operator?.open_id || "lark-webhook",
   763→          userName: "Lark Webhook",
   764→          userRole: "SYSTEM",
   765→          department: "TEAM_WORK",
   766→          page: "tasks",
   767→          action: "CREATE",
   768→          entityType: "Task",
   769→          entityId: newTask.id,
   770→          description: `Task created from Lark: "${title}"`,
   771→          newValue: status,
   772→          status: "SUCCESS",
   773→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id }),
   774→        })
   775→      } catch { /* audit log is best-effort */ }
   776→
   777→      return { success: true, message: `New task ${newTask.id} created from Lark` }
   778→    }
   779→  } catch (err) {
   780→    const msg = err instanceof Error ? err.message : String(err)
   781→    console.error("[lark/sync] Webhook handling error:", msg)
   782→    return { success: false, message: msg }
   783→  }
   784→}

   821→/**
   822→ * Remove a TrishulHub project member from the corresponding Lark task list.
   823→ * Called fire-and-forget from the project members DELETE route.
   824→ */
   825→export async function removeProjectMemberFromLarkTaskList(
   826→  projectId: string,
   827→  userId: string
   828→): Promise<void> {
   829→  const config = await getLarkConfig()
   830→  if (!config?.enabled) return
   831→
   832→  try {
   833→    const { getAppSetting, delAppSetting } = await import("@/lib/db")
   834→    const raw = await getAppSetting(`lark_tasklist_${projectId}`)
   835→    if (!raw) return
   836→
   837→    const parsed = JSON.parse(raw)
   838→    const tasklistId = parsed.tasklistId
   839→    if (!tasklistId) return
   840→
   841→    // Get user's Lark open_id
   842→    const openId = await getUserLarkOpenId(userId)
   843→    if (!openId) return
   844→
   845→    // Remove from task list
   846→    await removeTaskListMember(tasklistId, [{ id: openId }])
   847→    console.log(`[lark/sync] Removed user ${userId} (${openId}) from tasklist ${tasklistId}`)
   848→
   849→    // Cleanup: check if project has no members left — if so, delete the tasklist
   850→    const memberCount = await db.projectMember.count({ where: { projectId } })
   851→    if (memberCount === 0) {
   852→      const { deleteTaskList } = await import("./client")
   853→      await deleteTaskList(tasklistId)
   854→      await delAppSetting(`lark_tasklist_${projectId}`)
   855→      console.log(`[lark/sync] Deleted empty tasklist ${tasklistId} for project ${projectId}`)
   856→    }
   857→  } catch (err) {
   858→    console.error("[lark/sync] removeProjectMemberFromLarkTaskList failed:", err instanceof Error ? err.message : err)
   859→  }
   860→}

   862→/**
   863→ * Find a project ID by looking up the tasklist mapping.
   864→ */
   865→async function findProjectByTaskList(tasklistId: string): Promise<string | null> {
   866→  try {
   867→    const rows = await db.$queryRawUnsafe<Array<{ key: string; value: string }>>(
   868→      'SELECT "key", "value" FROM "AppSetting" WHERE "key" LIKE ?',
   869→      'lark_tasklist_%'
   870→    )
   871→
   872→    for (const row of rows) {
   873→      try {
   874→        const parsed = JSON.parse(row.value)
   875→        if (parsed.tasklistId === tasklistId) {
   876→          const projectId = row.key.replace("lark_tasklist_", "")
   877→          if (projectId && projectId !== "__default__") {
   878→            const project = await db.project.findUnique({ where: { id: projectId }, select: { id: true } })
   879→            return project?.id || null
   880→          }
   881→        }
   882→      } catch {
   883→        continue
   884→      }
   885→    }
   886→  } catch {
   887→    // Ignore
   888→  }
   889→
   890→  return null
   891→}
