   228→   397→      return
   229→   398→    }
   230→   399→
   231→   400→    // Resolve assignee's Lark open_id

   233→   402→    if (data.assignedTo) {
   234→   403→      assigneeOpenId = await getUserLarkOpenId(data.assignedTo) || undefined
   235→   404→    }
   236→   405→

   240→   409→      dueTimestamp = new Date(data.deadline).getTime()
   241→   410→    }
   242→   411→
   243→   412→    // Update task — NO extra, NO assignee in body (use member APIs instead)
   244→   413→    const larkTask = await updateTask(mapping.larkTaskId, {
   245→   414→      title: data.title,
   246→   415→      description: data.description,

   255→   424→        { id: assigneeOpenId, role: "assignee" },
   256→   425→      ])
   257→   426→    }
   258→   427→

   260→   429→      // Mark circular guard
   261→   430→      markSyncGuard(taskId, "TO_LARK")
   262→   431→
   263→   432→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "SUCCESS", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId })
   264→   433→    } else {
   265→   434→      await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, larkTaskId: mapping.larkTaskId, larkTaskListId: mapping.larkTaskListId, userId, error: "Lark API returned null" })
   266→   435→    }
   267→   436→  } catch (err) {
   268→   437→    const msg = err instanceof Error ? err.message : String(err)
   269→   438→    await logSync({ direction: "TO_LARK", action: "UPDATE", status: "FAILED", taskId, userId, error: msg })
   270→   439→  }

   303→   472→/**

   310→   479→  eventId: string,
   311→   480→  eventType: string,
   312→   481→  eventData: {

   314→   483→    tasklist_id: string

   321→   490→    return { success: false, message: "Lark sync is disabled" }

   362→   531→        })

   365→   534→      return { success: true, message: "Task deleted from TrishulHub" }
   366→   535→    }
   367→   536→

   371→   540→

   384→   553→        console.warn(`[lark/sync] getTask returned null for mapped task ${thTaskId}, retrying...`)
   385→   554→        await new Promise((resolve) => setTimeout(resolve, 2000))

   387→   556→

   413→   582→

   424→   593→            action: "STATUS_CHANGE",
   425→   594→            entityType: "Task",
   426→   595→            entityId: thTaskId,
   427→   596→            description: `Task status changed to "${status}" via Lark sync (retried)`,
   428→   597→            newValue: status,
   429→   598→            status: "SUCCESS",
   430→   599→            metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, larkStatus: retryTask.status }),
   431→   600→          })
   432→   601→
   433→   602→          await logSync({
   434→   603→            direction: "FROM_LARK",
   435→   604→            action: "STATUS_CHANGE",
   436→   605→            status: "SUCCESS",
   437→   606→            taskId: thTaskId,
   438→   607→            larkTaskId: task_id,
   439→   608→            larkTaskListId: tasklist_id,
   440→   609→            metadata: { inferredStatus: status, eventType, retried: true },
   441→   610→          })
   442→   611→          return { success: true, message: `Task ${thTaskId} updated to ${status} (retried getTask)` }
   443→   612→        }
   444→   613→
   445→   614→        // Both getTask calls failed — inspect event body for status change clues.
   446→   615→        const eventBody = (eventData as Record<string, unknown>)
   447→   616→        const changedFields = eventBody.changed_fields as string[] | undefined
   448→   617→
   449→   618→        // Determine if this is likely a completion:
   450→   619→        // 1. Event type contains done/complete/finish keywords
   451→   620→        // 2. OR changed_fields includes "status" (something changed, likely to done since getTask fails)
   452→   621→        const likelyCompleted =
   453→   622→          eventType.includes("done") || eventType.includes("complete") || eventType.includes("finish") ||
   454→   623→          (Array.isArray(changedFields) && changedFields.includes("status"))
   455→   624→
   456→   625→        const newStatus = likelyCompleted ? "DONE" : "IN_PROGRESS";
   457→   626→        const fallbackData: Record<string, unknown> = { status: newStatus };
   458→   627→        if (newStatus === "DONE") fallbackData.completedAt = new Date();
   459→   628→
   460→   629→        const existingTask = await db.task.findUnique({ where: { id: thTaskId }, select: { status: true } });
   461→   630→        console.warn(`[lark/sync] getTask returned null twice for mapped task ${thTaskId}. Event: ${eventType}. Assuming status="${newStatus}" (was "${existingTask?.status || 'unknown'}")`);
   462→   631→
   463→   632→        await db.task.update({ where: { id: thTaskId }, data: fallbackData })
   464→   633→        markSyncGuard(thTaskId, "FROM_LARK")
   465→   634→
   466→   635→        const { logAudit } = await import("@/lib/audit-log")
   467→   636→        await logAudit({
   468→   637→          userId: eventData.operator?.open_id || "lark-webhook",
   469→   638→          userName: "Lark Webhook",
   470→   639→          userRole: "SYSTEM",
   471→   640→          department: "TEAM_WORK",
   472→   641→          page: "tasks",
   473→   642→          action: "STATUS_CHANGE",
   474→   643→          entityType: "Task",
   475→   644→          entityId: thTaskId,
   476→   645→          description: `Task status changed to "${newStatus}" via Lark sync (inferred — getTask failed)`,
   477→   646→          newValue: newStatus,
   478→   647→          status: "SUCCESS",
   479→   648→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, inferred: true, previousStatus: existingTask?.status }),
   480→   649→        })
   481→   650→
   482→   651→        await logSync({
   483→   652→          direction: "FROM_LARK",
   484→   653→          action: "STATUS_CHANGE",
   485→   654→          status: "SUCCESS",
   486→   655→          taskId: thTaskId,
   487→   656→          larkTaskId: task_id,
   488→   657→          larkTaskListId: tasklist_id,
   489→   658→          metadata: { inferredStatus: newStatus, eventType, retried: true },
   490→   659→        })
   491→   660→        return { success: true, message: `Task ${thTaskId} updated to ${newStatus} (getTask null, inferred from event + changed_fields)` }
   492→   661→      }
   493→   662→      return { success: false, message: "Failed to fetch task from Lark and no existing mapping" }
   494→   663→    }
   495→   664→
   496→   665→    // Circular sync guard: skip if we just synced TO_LARK for this task
   497→   666→    if (thTaskId && isSyncGuarded(thTaskId, "TO_LARK")) {
   498→   667→      console.log(`[lark/sync] Skipping FROM_LARK for ${thTaskId} — circular guard active`)
   499→   668→      return { success: true, message: "Circular guard: skipped" }
   500→   669→    }
   501→   670→
   502→   671→    // Map Lark task data to TrishulHub format
   503→   672→    const title = larkTask.title
   504→   673→    const description = larkTask.description
   505→   674→    const status = STATUS_FROM_LARK[larkTask.status] || "TODO"
   506→   675→    console.log(`[lark/sync] Status mapping: lark="${larkTask.status}" → th="${status}"`)
   507→   676→    const priority = PRIORITY_FROM_LARK[larkTask.priority] || "MEDIUM"
   508→   677→
   509→   678→    // Resolve assignee
   510→   679→    let assignedTo: string | undefined
   511→   680→    if (larkTask.assignee?.open_id) {
   512→   681→      assignedTo = await getUserIdByLarkOpenId(larkTask.assignee.open_id) || undefined
   513→   682→    }
   514→   683→
   515→   684→    // Resolve due date
   516→   685→    const deadline = larkTask.due?.timestamp ? new Date(parseInt(larkTask.due.timestamp)) : null
   517→   686→
   518→   687→    if (thTaskId) {
   519→   688→      // UPDATE existing task
   520→   689→      const updateData: Record<string, unknown> = {}
   521→   690→      if (title) updateData.title = title
   522→   691→      if (description !== undefined) updateData.description = description
   523→   692→      updateData.status = status
   524→   693→      updateData.priority = priority
   525→   694→      if (assignedTo !== undefined) updateData.assignedTo = assignedTo
   526→   695→      if (deadline !== null) updateData.deadline = deadline
   527→   696→
   528→   697→      // Set completedAt if status changed to DONE
   529→   698→      if (status === "DONE") {
   530→   699→        updateData.completedAt = new Date()
   531→   700→      }
   532→   701→
   533→   702→      await db.task.update({ where: { id: thTaskId }, data: updateData })
   534→   703→      console.log(`[lark/sync] Updated TH task ${thTaskId}: status=${updateData.status}, priority=${updateData.priority}, assignedTo=${updateData.assignedTo || 'unchanged'}`)
   535→   704→
   536→   705→      // Mark circular guard so our update handler doesn't echo back
   537→   706→      markSyncGuard(thTaskId, "FROM_LARK")
   538→   707→
   539→   708→      // Log to both LarkSyncLog and AuditLog
   540→   709→      await logSync({
   541→   710→        direction: "FROM_LARK",
   542→   711→        action: status === "DONE" ? "STATUS_CHANGE" : "UPDATE",
   543→   712→        status: "SUCCESS",
   544→   713→        taskId: thTaskId,
   545→   714→        larkTaskId: task_id,
   546→   715→        larkTaskListId: tasklist_id,
   547→   716→      })
   548→   717→
   549→   718→      try {
   550→   719→        const { logAudit } = await import("@/lib/audit-log")
   551→   720→        await logAudit({
   552→   721→          userId: eventData.operator?.open_id || "lark-webhook",
   553→   722→          userName: "Lark Webhook",
   554→   723→          userRole: "SYSTEM",
   555→   724→          department: "TEAM_WORK",
   556→   725→          page: "tasks",
   557→   726→          action: status === "DONE" ? "STATUS_CHANGE" : "UPDATE",
   558→   727→          entityType: "Task",
   559→   728→          entityId: thTaskId,
   560→   729→          description: `Task ${status === "DONE" ? "marked as DONE" : "updated"} via Lark sync`,
   561→   730→          newValue: status,
   562→   731→          status: "SUCCESS",
   563→   732→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id, larkStatus: larkTask.status }),
   564→   733→        })
   565→   734→      } catch { /* audit log is best-effort */ }
   566→   735→
   567→   736→      return { success: true, message: `Task ${thTaskId} updated from Lark` }
   568→   737→    } else {
   569→   738→      // CREATE new task (originated in Lark)
   570→   739→      const newTask = await db.task.create({
   571→   740→        data: {
   572→   741→          title,
   573→   742→          description: description || null,
   574→   743→          status,
   575→   744→          priority,
   576→   745→          assignedTo: assignedTo || null,
   577→   746→          deadline,
   578→   747→          assigneeType: "HUMAN",
   579→   748→          category: "GENERAL",
   580→   749→          projectId: await findProjectByTaskList(tasklist_id),
   581→   750→        },
   582→   751→      })
   583→   752→
   584→   753→      // Save mapping
   585→   754→      await saveLarkTaskMapping(newTask.id, task_id, tasklist_id)
   586→   755→
   587→   756→      // Mark guard for the newly created task
   588→   757→      markSyncGuard(newTask.id, "FROM_LARK")
   589→   758→
   590→   759→      try {
   591→   760→        const { logAudit } = await import("@/lib/audit-log")
   592→   761→        await logAudit({
   593→   762→          userId: eventData.operator?.open_id || "lark-webhook",
   594→   763→          userName: "Lark Webhook",
   595→   764→          userRole: "SYSTEM",
   596→   765→          department: "TEAM_WORK",
   597→   766→          page: "tasks",
   598→   767→          action: "CREATE",
   599→   768→          entityType: "Task",
   600→   769→          entityId: newTask.id,
   601→   770→          description: `Task created from Lark: "${title}"`,
   602→   771→          newValue: status,
   603→   772→          status: "SUCCESS",
   604→   773→          metadata: JSON.stringify({ source: "lark_webhook", eventType, larkTaskId: task_id }),
   605→   774→        })
   606→   775→      } catch { /* audit log is best-effort */ }
   607→   776→
   608→   777→      return { success: true, message: `New task ${newTask.id} created from Lark` }
   609→   778→    }
   610→   779→  } catch (err) {
   611→   780→    const msg = err instanceof Error ? err.message : String(err)
   612→   781→    console.error("[lark/sync] Webhook handling error:", msg)
   613→   782→    return { success: false, message: msg }
   614→   783→  }

   706→   875→        if (parsed.tasklistId === tasklistId) {
   707→   876→          const projectId = row.key.replace("lark_tasklist_", "")

   709→   878→            const project = await db.project.findUnique({ where: { id: projectId }, select: { id: true } })
   710→   879→            return project?.id || null
   711→   880→          }
   712→   881→        }
   713→   882→      } catch {
   714→   883→        continue
   715→   884→      }
   716→   885→    }
   717→   886→  } catch {
   718→   887→    // Ignore
