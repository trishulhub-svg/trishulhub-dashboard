   149→function formatDateOnly(d: Date): string {
   150→  const year = d.getFullYear();
   151→  const month = String(d.getMonth() + 1).padStart(2, "0");
   152→  const day = String(d.getDate()).padStart(2, "0");
   153→  return `${year}-${month}-${day}`;
   154→}
   155→
   156→function getWeekStart(date: Date): Date {
   157→  const d = new Date(date);
   158→  const day = d.getDay();
   159→  d.setDate(d.getDate() - day);
   160→  d.setHours(0, 0, 0, 0);
   161→  return d;
   162→}
   163→
   164→function getWeekDates(startDate: Date): Date[] {
   165→  const dates: Date[] = [];
   166→  for (let i = 0; i < 7; i++) {
   167→    const d = new Date(startDate);
   168→    d.setDate(startDate.getDate() + i);
   169→    dates.push(d);
   170→  }
   171→  return dates;
   172→}
   173→
   174→function getUserInitials(name: string): string {
   175→  return name
   176→    .split(" ")
   177→    .map((n) => n[0])
   178→    .join("")
   179→    .toUpperCase()
   180→    .slice(0, 2);
   181→}
   182→
   183→function timeToMinutes(time: string): number {
   184→  const [h, m] = time.split(":").map(Number);
   185→  return h * 60 + m;
   186→}
