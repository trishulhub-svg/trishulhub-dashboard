   157→function formatDateOnly(d: Date): string {
   158→  const year = d.getFullYear();
   159→  const month = String(d.getMonth() + 1).padStart(2, "0");
   160→  const day = String(d.getDate()).padStart(2, "0");
   161→  return `${year}-${month}-${day}`;
   162→}
   163→
   164→function getWeekStart(date: Date): Date {
   165→  const d = new Date(date);
   166→  const day = d.getDay();
   167→  d.setDate(d.getDate() - day);
   168→  d.setHours(0, 0, 0, 0);
   169→  return d;
   170→}
   171→
   172→function getWeekDates(startDate: Date): Date[] {
   173→  const dates: Date[] = [];
   174→  for (let i = 0; i < 7; i++) {
   175→    const d = new Date(startDate);
   176→    d.setDate(startDate.getDate() + i);
   177→    dates.push(d);
   178→  }
   179→  return dates;
   180→}
   181→
   182→function getUserInitials(name: string): string {
   183→  return name
   184→    .split(" ")
   185→    .map((n) => n[0])
   186→    .join("")
   187→    .toUpperCase()
   188→    .slice(0, 2);
   189→}
   190→
   191→function timeToMinutes(time: string): number {
   192→  const [h, m] = time.split(":").map(Number);
   193→  return h * 60 + m;
   194→}
