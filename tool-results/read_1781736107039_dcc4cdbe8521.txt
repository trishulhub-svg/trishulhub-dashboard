   158→   158→function formatDateOnly(d: Date): string {
   159→   159→  const year = d.getFullYear();
   160→   160→  const month = String(d.getMonth() + 1).padStart(2, "0");
   161→   161→  const day = String(d.getDate()).padStart(2, "0");
   162→   162→  return `${year}-${month}-${day}`;
   163→   163→}
   164→   164→
   165→   165→function getWeekStart(date: Date): Date {
   166→   166→  const d = new Date(date);
   167→   167→  const day = d.getDay();
   168→   168→  d.setDate(d.getDate() - day);
   169→   169→  d.setHours(0, 0, 0, 0);
   170→   170→  return d;
   171→   171→}
   172→   172→
   173→   173→function getWeekDates(startDate: Date): Date[] {
   174→   174→  const dates: Date[] = [];
   175→   175→  for (let i = 0; i < 7; i++) {
   176→   176→    const d = new Date(startDate);
   177→   177→    d.setDate(startDate.getDate() + i);
   178→   178→    dates.push(d);
   179→   179→  }
   180→   180→  return dates;
   181→   181→}

   183→   183→function getUserInitials(name: string): string {
   184→   184→  return name
   185→   185→    .split(" ")
   186→   186→    .map((n) => n[0])
   187→   187→    .join("")
   188→   188→    .toUpperCase()
   189→   189→    .slice(0, 2);
   190→   190→}
   191→   191→
   192→   192→function timeToMinutes(time: string): number {
   193→   193→  const [h, m] = time.split(":").map(Number);
   194→   194→  return h * 60 + m;
   195→   195→}
