   855→   855→  const formatRelativeTime = (dateStr: string | null) => {
   856→   856→    if (!dateStr) return null;
   857→   857→    const date = new Date(dateStr);
   858→   858→    const now = new Date();
   859→   859→    const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
   860→   860→    if (seconds < 60) return "Just now";
   861→   861→    if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`;
   862→   862→    if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`;
